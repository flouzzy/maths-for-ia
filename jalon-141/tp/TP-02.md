# TP 2 : Simulation de la Convergence Uniforme

Nous allons générer plusieurs hypothèses et vérifier que l'écart entre le risque théorique et empirique de la pire hypothèse diminue.

```python
import random

def true_distribution():
    # p(y=1 | x>0.5) = 1, p(y=-1 | x<=0.5) = 1
    x = random.uniform(0, 1)
    y = 1 if x > 0.5 else -1
    return x, y

def generate_sample(n):
    return [true_distribution() for _ in range(n)]

def true_risk(threshold):
    # L'erreur théorique si on coupe à threshold au lieu de 0.5
    if threshold < 0.5:
        # Erreur pour x in [threshold, 0.5]: prédit 1, est -1
        return (0.5 - threshold) / 1.0
    else:
        # Erreur pour x in [0.5, threshold]: prédit -1, est 1
        return (threshold - 0.5) / 1.0

def tp2_convergence():
    thresholds = [0.1, 0.3, 0.5, 0.7, 0.9] # L'espace H restreint
    ns = [10, 100, 1000, 10000]

    for n in ns:
        sample = generate_sample(n)
        data = [s[0] for s in sample]
        labels = [s[1] for s in sample]

        max_diff = 0
        for t in thresholds:
            r_emp = sum(1 for i in range(n) if (1 if data[i]>=t else -1) != labels[i]) / n
            r_true = true_risk(t)
            diff = abs(r_emp - r_true)
            if diff > max_diff:
                max_diff = diff

        # Le max diff doit se concentrer vers 0 avec n grand
        # Pour n=10000, la diff est de l'ordre de 0.01
        assert n < 10000 or max_diff < 0.05, f"Convergence échouée: max_diff = {max_diff}"
    print("Validation TP2 passée.")

tp2_convergence()
```
