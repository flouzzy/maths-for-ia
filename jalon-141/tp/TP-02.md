# TP 2 : Simulation de la déviation de Glivenko-Cantelli
**Objectif :** Estimer par simulation la déviation uniforme maximale entre $F_n(t)$ et $F(t)$ pour vérifier la convergence asymptotique.

**Code Python :**
```python
import math
import random

def uniform_cdf(t, a=0.0, b=1.0):
    \"\"\"Fonction de répartition d'une loi uniforme sur [a,b].\"\"\"
    if t < a: return 0.0
    if t > b: return 1.0
    return (t - a) / (b - a)

def empirical_cdf(sample, t):
    return sum(1 for x in sample if x <= t) / float(len(sample))

def compute_max_deviation(sample, eval_points):
    \"\"\"
    Calcule le supremum de l'écart sur un ensemble fini de points d'évaluation.
    \"\"\"
    max_dev = 0.0
    for t in eval_points:
        dev = abs(empirical_cdf(sample, t) - uniform_cdf(t))
        if dev > max_dev:
            max_dev = dev
    return max_dev

# Test de validation mathématique
def test_glivenko_cantelli_convergence():
    random.seed(42) # Fixer la graine pour reproductibilité
    eval_points = [x/100.0 for x in range(101)] # Points dans [0,1]

    # Tailles d'échantillon croissantes
    n_small = 10
    n_large = 1000

    sample_small = [random.uniform(0, 1) for _ in range(n_small)]
    sample_large = [random.uniform(0, 1) for _ in range(n_large)]

    dev_small = compute_max_deviation(sample_small, eval_points)
    dev_large = compute_max_deviation(sample_large, eval_points)

    print(f"Déviation max (n={n_small}): {dev_small:.4f}")
    print(f"Déviation max (n={n_large}): {dev_large:.4f}")

    # La théorie prédit que la déviation diminue avec n
    assert dev_large < dev_small, "La déviation n'a pas diminué en augmentant la taille de l'échantillon."
    print("La convergence de Glivenko-Cantelli est validée empiriquement.")

if __name__ == "__main__":
    test_glivenko_cantelli_convergence()
```
