# TP 5 : Non-uniformité de la déviation pour VC = infinie
**Objectif :** Simuler le cas dégénéré où la dimension VC est infinie (classe de toutes les parties finies), en construisant la pire fonction empirique.

**Code Python :**
```python
import random

def risk_real(E_h):
    \"\"\"
    Risque réel pour une fonction caractéristique d'un ensemble fini E_h
    sous une distribution uniforme continue.
    La probabilité continue d'un ensemble fini est toujours 0.
    \"\"\"
    return 0.0

def risk_empirical(E_h, sample):
    \"\"\"
    Risque empirique sur un échantillon pour la fonction caractéristique de E_h.
    \"\"\"
    # Compte les éléments de l'échantillon qui appartiennent à E_h
    count = sum(1 for x in sample if x in E_h)
    return count / float(len(sample))

# Test de validation mathématique
def test_infinite_vc_non_convergence():
    random.seed(123)

    n_sample = 500
    # Génération d'un échantillon selon la loi uniforme continue sur [0, 1]
    sample = [random.uniform(0, 1) for _ in range(n_sample)]

    # L'astuce démoniaque : on choisit pour H l'ensemble exact de l'échantillon empirique
    # Puisque H est un ensemble fini, son indicatrice appartient à la classe de VC infinie
    worst_case_E_h = set(sample)

    r_real = risk_real(worst_case_E_h)
    r_emp = risk_empirical(worst_case_E_h, sample)

    print(f"Risque réel R(h) : {r_real:.2f}")
    print(f"Risque empirique R_n(h) : {r_emp:.2f}")
    print(f"Déviation maximale : {abs(r_emp - r_real):.2f}")

    assert r_real == 0.0, "Le risque réel devrait être strictement 0."
    assert r_emp == 1.0, "Le risque empirique devrait être strictement 1."
    assert abs(r_emp - r_real) == 1.0, "La déviation ne tend pas vers 0, invalidant Glivenko-Cantelli."

    print("La non-convergence pour une classe de VC infinie est validée mathématiquement.")

if __name__ == "__main__":
    test_infinite_vc_non_convergence()
```
