# TP 5 : Complexité de Rademacher Empirique

Calcul "From scratch" de la borne de Rademacher pour un petit ensemble de données.

```python
import random

def rademacher_complexity(data, hypotheses, num_simulations=500):
    '''
    Calcule la complexité de Rademacher empirique.
    hypothèses: list de listes de prédictions sur les données
    '''
    n = len(data)
    total_sup = 0

    for _ in range(num_simulations):
        # Tirage des variables de Rademacher sigma in {-1, 1}
        sigmas = [random.choice([-1, 1]) for _ in range(n)]

        max_correlation = 0
        for h_preds in hypotheses:
            correlation = sum(sigmas[i] * h_preds[i] for i in range(n)) / n
            if correlation > max_correlation:
                max_correlation = correlation

        total_sup += max_correlation

    return total_sup / num_simulations

# Test mathématique
# Si nous avons 1 seul point, et 2 hypothèses (prédit 1, ou prédit -1)
data_test = [1.0]
hyps = [[1], [-1]]
rad_comp = rademacher_complexity(data_test, hyps, 1000)
# Sigma est +1 ou -1. Sup(sigma*1, sigma*(-1)) est toujours 1. E[1] = 1.
assert abs(rad_comp - 1.0) < 0.05, f"La complexité attendue est 1. Obtenu: {rad_comp}"
print("Validation TP5 passée.")
```
