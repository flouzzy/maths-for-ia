# TP 1 : Calcul du Risque Empirique "From Scratch"

Ce TP construit en Python pur le cœur de la minimisation de la perte, sans `numpy`.

```python
def predict(x, threshold, sign):
    '''Prédit +1 ou -1 en fonction du seuil (decision stump).'''
    if sign == 1:
        return 1 if x >= threshold else -1
    else:
        return 1 if x < threshold else -1

def empirical_risk_01(data, labels, threshold, sign):
    '''
    Calcule le risque empirique 0-1.
    data : list of floats
    labels : list of 1 or -1
    '''
    n = len(data)
    errors = 0
    for i in range(n):
        y_pred = predict(data[i], threshold, sign)
        if y_pred != labels[i]:
            errors += 1
    return errors / n

# TEST MATHEMATIQUE RIGOUREUX
data_test = [0.1, 0.4, 0.5, 0.8]
labels_test = [-1, -1, 1, 1]
# Seuil optimal entre 0.4 et 0.5
risk = empirical_risk_01(data_test, labels_test, 0.45, 1)
assert risk == 0.0, "Le risque devrait être 0 pour une séparation parfaite."

risk_bad = empirical_risk_01(data_test, labels_test, 0.9, 1)
assert risk_bad == 0.5, "Deux erreurs attendues sur 4 points = 0.5"
print("Validation TP1 passée.")
```
