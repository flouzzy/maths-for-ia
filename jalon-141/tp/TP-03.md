# TP 3 : Calcul de la fonction de croissance d'une classe d'intervalles
**Objectif :** Implémenter le calcul du nombre d'étiquetages possibles (fonction de croissance) pour la classe des intervalles gauches $(-\infty, a]$.

**Code Python :**
```python
def dichotomize_points(points, threshold):
    \"\"\"
    Applique la fonction indicatrice h(x) = I_{x <= threshold} à un ensemble de points.
    Retourne le tuple d'étiquettes pour pouvoir l'ajouter à un set.
    \"\"\"
    return tuple(1 if x <= threshold else 0 for x in points)

def growth_function_left_rays(points):
    \"\"\"
    Calcule le nombre d'étiquetages distincts réalisables par la classe des demi-droites gauches.
    \"\"\"
    # Trier les points pour balayer les seuils pertinents
    sorted_points = sorted(points)
    n = len(sorted_points)

    # Les seuils pertinents sont: avant le 1er point, et juste après chaque point
    thresholds = [sorted_points[0] - 1.0] # Un seuil qui étiquette tout à 0
    for p in sorted_points:
        thresholds.append(p) # Le seuil exactement sur le point étiquette à 1 ce point et les précédents

    unique_labelings = set()
    for t in thresholds:
        labeling = dichotomize_points(points, t)
        unique_labelings.add(labeling)

    return len(unique_labelings)

# Test de validation mathématique
def test_growth_function():
    # Test avec 1 point
    assert growth_function_left_rays([2.5]) == 2, "Erreur avec n=1"

    # Test avec 2 points
    assert growth_function_left_rays([1.0, 3.0]) == 3, "Erreur avec n=2. Devrait être n+1=3."

    # Test avec 5 points
    assert growth_function_left_rays([1, 2, 3, 4, 5]) == 6, "Erreur avec n=5. Devrait être n+1=6."

    print("Le calcul de la fonction de croissance Pi(n) = n+1 est validé mathématiquement.")

if __name__ == "__main__":
    test_growth_function()
```
