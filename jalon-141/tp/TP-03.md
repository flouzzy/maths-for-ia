# TP 3 : Calculateur de la Fonction de Croissance de Sauer

Implémentation de la borne polynomiale de Vapnik-Chervonenkis.

```python
def binom(n, k):
    if k < 0 or k > n:
        return 0
    if k == 0 or k == n:
        return 1
    num = 1
    den = 1
    for i in range(1, k + 1):
        num *= (n - i + 1)
        den *= i
    return num // den

def sauer_shelah_bound(n, d):
    '''
    Calcule la somme des coefficients binomiaux jusqu'à d.
    Borne le nombre de dichotomies.
    '''
    total = 0
    for i in range(d + 1):
        total += binom(n, i)
    return total

# TEST MATHEMATIQUE RIGOUREUX
val = sauer_shelah_bound(10, 3)
assert val == 176, "Erreur de calcul combinatoire : attendu 176."

val_shatter = sauer_shelah_bound(3, 3)
assert val_shatter == 8, "n=d doit donner 2^d = 8."
```
