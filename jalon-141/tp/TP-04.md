# TP 4 : Validation du lemme de Sauer-Shelah
**Objectif :** Vérifier numériquement l'inégalité de la borne de Sauer-Shelah $\Pi_{\mathcal{H}}(n) \le \sum_{i=0}^d \binom{n}{i}$.

**Code Python :**
```python
import math

def binomial_coefficient(n, k):
    \"\"\"Calcule le coefficient binomial 'n parmi k'.\"\"\"
    if k < 0 or k > n:
        return 0
    if k == 0 or k == n:
        return 1
    # Formule itérative pour éviter les débordements avec math.factorial
    k = min(k, n - k)
    c = 1
    for i in range(k):
        c = c * (n - i) // (i + 1)
    return c

def sauer_shelah_bound(n, d):
    \"\"\"
    Calcule la borne de Sauer-Shelah : somme des (n parmi i) pour i de 0 à d.
    \"\"\"
    return sum(binomial_coefficient(n, i) for i in range(d + 1))

# Test de validation mathématique
def test_sauer_shelah():
    # Pour la classe des demi-droites (d=1)
    d = 1
    for n in range(1, 10):
        # La fonction de croissance exacte est n+1
        exact_growth = n + 1
        bound = sauer_shelah_bound(n, d)
        # Pour d=1, la borne est exacte : n parmi 0 + n parmi 1 = 1 + n
        assert bound == exact_growth, f"Échec pour d=1, n={n}. Borne: {bound}, Exact: {exact_growth}"

    # Pour la classe des intervalles (d=2)
    # La fonction de croissance exacte pour les intervalles est (n(n+1)/2) + 1
    d = 2
    for n in range(2, 10):
        exact_growth = (n * (n + 1)) // 2 + 1
        bound = sauer_shelah_bound(n, d)
        # Pour d=2, la borne est aussi exacte : (n parmi 0) + (n parmi 1) + (n parmi 2) = 1 + n + n(n-1)/2 = 1 + n(n+1)/2
        assert bound == exact_growth, f"Échec pour d=2, n={n}. Borne: {bound}, Exact: {exact_growth}"

    print("La borne de Sauer-Shelah est vérifiée et exacte pour d=1 et d=2.")

if __name__ == "__main__":
    test_sauer_shelah()
```
