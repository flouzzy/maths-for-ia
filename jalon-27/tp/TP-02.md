# TP 02 : Vérification de l'orthogonalité d'une matrice

## Objectif Mathématique
Implémenter une vérification stricte $A^T A = I_n$ sans utiliser Numpy.

## Code Python Pur

```python
def transpose(A):
    rows, cols = len(A), len(A[0])
    return [[A[i][j] for i in range(rows)] for j in range(cols)]

def multiply(A, B):
    rows_A, cols_A = len(A), len(A[0])
    cols_B = len(B[0])
    result = [[0]*cols_B for _ in range(rows_A)]
    for i in range(rows_A):
        for j in range(cols_B):
            result[i][j] = sum(A[i][k]*B[k][j] for k in range(cols_A))
    return result

def is_orthogonal(A):
    rows, cols = len(A), len(A[0])
    if rows != cols:
        return False
    AtA = multiply(transpose(A), A)
    for i in range(rows):
        for j in range(cols):
            expected = 1.0 if i == j else 0.0
            if abs(AtA[i][j] - expected) > 1e-9:
                return False
    return True

# Validation
import math
theta = math.pi / 4
c, s = math.cos(theta), math.sin(theta)
Rot = [[c, -s], [s, c]]
assert is_orthogonal(Rot), "Matrice de rotation est orthogonale"
print("TP 02 validé.")
```
