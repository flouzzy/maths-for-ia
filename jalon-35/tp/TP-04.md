# TP 4 : Calcul de la distance à un compact

```python
import math

# K est un cercle de rayon 1, centre 0. Compact (ferme borne)
def distance_to_circle(x, y):
    # La projection orthogonale est triviale
    r = math.sqrt(x**2 + y**2)
    if r == 0:
        return 1.0 # Au centre
    proj_x = x / r
    proj_y = y / r
    return math.sqrt((x - proj_x)**2 + (y - proj_y)**2)

# Point exterieur
pt = (2.0, 0.0)
dist = distance_to_circle(pt[0], pt[1])

print(f"Distance au compact: {dist}")
assert abs(dist - 1.0) < 1e-9

# L'infimum est atteint pour le point (1,0) du compact
```
