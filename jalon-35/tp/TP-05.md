# TP 5 : Optimisation sur un ensemble compact

```python
import numpy as np

# Théorème des bornes atteintes
# On cherche a maximiser f(x, y) = x^2 + 2y sur le carre ferme [-1, 1]x[-1, 1] (compact)

def f(x, y):
    return x**2 + 2*y

# Discrétisation du compact
x_vals = np.linspace(-1, 1, 100)
y_vals = np.linspace(-1, 1, 100)

max_val = -float('inf')
best_x = 0
best_y = 0

for x in x_vals:
    for y in y_vals:
        val = f(x, y)
        if val > max_val:
            max_val = val
            best_x = x
            best_y = y

print(f"Maximum empirique: {max_val} atteint en ({best_x}, {best_y})")
# Le maximum theorique est en (+/-1, 1) ou f = 1 + 2 = 3
assert abs(max_val - 3.0) < 1e-2
```
