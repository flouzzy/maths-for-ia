# TP 1 : Vérification numérique de l'ouverture d'une boule

```python
import math
import random

def norm(x, y):
    return math.sqrt(x**2 + y**2)

# Centre et rayon de la boule ouverte U = B((0,0), 1)
center = (0.0, 0.0)
R = 1.0

# Une suite convergeant vers x_lim = (0.5, 0.5) in U
# x_n = (0.5 + 1/(n+1), 0.5 - 1/(n+1))
x_lim = (0.5, 0.5)

assert norm(x_lim[0] - center[0], x_lim[1] - center[1]) < R, "La limite doit etre dans l'ouvert"

def suite(n):
    return (0.5 + 1.0/(n+1), 0.5 - 1.0/(n+1))

# Trouver empiriquement le rang N tel que x_n in U
N = -1
for n in range(100):
    point = suite(n)
    dist_to_center = norm(point[0] - center[0], point[1] - center[1])
    if dist_to_center < R:
        # A partir de ce n, on est dans l'ouvert (on verifie empiriquement pour les suivants)
        is_inside = True
        for k in range(n, n+50):
            pt = suite(k)
            if norm(pt[0] - center[0], pt[1] - center[1]) >= R:
                is_inside = False
                break
        if is_inside:
            N = n
            break

print(f"Rang de capture N = {N}")
assert N != -1, "La suite doit entrer dans l'ouvert"
```
