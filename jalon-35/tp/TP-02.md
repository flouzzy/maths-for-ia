# TP 2 : Test d'un ensemble fermé par convergence d'une suite

```python
# F = [0, 1] dans R (fermé)
# U = [0, 1[ dans R (non fermé)

def suite_fermee(n):
    return 1 - 1.0 / (n + 1)  # Converge vers 1 in F

def suite_non_fermee(n):
    return 1 - 1.0 / (n + 1)  # Converge vers 1 NOT IN U

lim_f = 1.0
lim_u = 1.0

# L'ensemble F contient-il la limite ?
assert 0.0 <= lim_f <= 1.0, "La limite est bien dans le fermé F"

# L'ensemble U contient-il la limite ?
is_in_u = (0.0 <= lim_u < 1.0)
assert not is_in_u, "La limite s'échappe de l'ensemble U, prouvant qu'il n'est pas fermé"

print("Vérification de la fermeture séquentielle réussie.")
```
