# TP 05 : Convergence en Loi par Fonction de Répartition Empirique (Théorème Central Limite)

Ce TP est une implémentation primitive et autonome du TCL. Nous simulons la somme de variables uniformes centrées réduites pour prouver que leur somme cumulée s'approche de la Fonction de Répartition de la Gaussienne (Convergence en Distribution/Loi).

```python
import random
import math

def uniforme_centree_reduite():
    # L'uniforme sur [0,1] a une moyenne 1/2 et variance 1/12
    # X_c = (U - 1/2) / (1 / sqrt(12)) = sqrt(12) * (U - 0.5)
    u = random.random()
    return math.sqrt(12.0) * (u - 0.5)

def echantillon_TCL(n):
    # Somme normalisee de N variables
    somme = sum(uniforme_centree_reduite() for _ in range(n))
    return somme / math.sqrt(n)

def fonction_repartition_empirique(t, n, n_simulations):
    # Calcule P(Z_n <= t) empiriquement
    succes = 0
    for _ in range(n_simulations):
        z = echantillon_TCL(n)
        if z <= t:
            succes += 1
    return succes / n_simulations

def approx_F_gaussienne(t):
    # Approximation de la fonction d'erreur (Erf) pour la repartition exacte N(0,1)
    return 0.5 * (1 + math.erf(t / math.sqrt(2)))

# Validation Analytique de la Convergence en Distribution
N_termes_somme = 50
simulations_mc = 2000
seuil_test = 1.0 # On teste la queue de distribution à Z <= 1

proba_empirique = fonction_repartition_empirique(seuil_test, N_termes_somme, simulations_mc)
proba_theorique = approx_F_gaussienne(seuil_test)

erreur_absolue = abs(proba_empirique - proba_theorique)

print(f"Probabilité empirique P(Z <= 1) : {proba_empirique:.4f}")
print(f"Probabilité théorique gaussienne  : {proba_theorique:.4f}")
print(f"Erreur d'approximation TCL        : {erreur_absolue:.4f}")

# La convergence en loi stipule que l'écart doit être asymptotiquement nul
assert erreur_absolue < 0.05, "Le TCL doit garantir une approximation gaussienne avec moins de 5% d'erreur avec 2000 echantillons"
```
