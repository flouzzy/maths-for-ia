# TP 1 : Illustration empirique de la convergence en probabilité vs p.s.

Dans ce TP, nous allons simuler la suite "masse glissante" vue en exercice : $X_n(\omega) = n \mathbb{1}_{]0, 1/n]}(\omega)$.
L'objectif est d'observer numériquement la convergence ponctuelle pour un $\omega$ fixé, et le comportement de l'espérance pour la non-convergence $L^1$.

```python
import random

def simulate_masse_glissante(omega, n_max):
    # Renvoie la suite X_1(omega), ..., X_n_max(omega)
    sequence = []
    for n in range(1, n_max + 1):
        if 0 < omega <= 1.0 / n:
            sequence.append(n)
        else:
            sequence.append(0)
    return sequence

# 1. Vérification de la convergence presque sûre (ponctuelle)
omega_test = 0.05
n_max = 50
seq = simulate_masse_glissante(omega_test, n_max)

print(f"Trajectoire pour omega = {omega_test} :")
print(seq)

# On vérifie mathématiquement que la suite finit par stationner à 0
# Dès que n > 1 / omega, soit n > 20, la suite vaut 0.
assert seq[20] == 0, "Erreur: La suite ne stationne pas à 0 comme prévu."
assert seq[-1] == 0, "Erreur: La limite finale n'est pas 0."
print("Validation mathématique : La trajectoire converge bien vers 0.")

# 2. Vérification de la non-convergence L1 (Espérance)
# On approche l'espérance par méthode de Monte-Carlo pour un n fixé
def esperance_empirique(n, nb_simulations=100000):
    somme = 0.0
    for _ in range(nb_simulations):
        omega = random.uniform(0, 1) # Loi uniforme sur [0, 1]
        val = n if omega <= 1.0/n else 0
        somme += val
    return somme / nb_simulations

e_10 = esperance_empirique(10)
e_100 = esperance_empirique(100)

print(f"Espérance empirique E[X_10] = {e_10:.4f}")
print(f"Espérance empirique E[X_100] = {e_100:.4f}")

# Mathématiquement, E[X_n] = 1 pour tout n.
assert 0.8 < e_10 < 1.2, "Erreur : L'espérance devrait être proche de 1."
assert 0.8 < e_100 < 1.2, "Erreur : L'espérance devrait être proche de 1."
print("Validation mathématique : L'espérance reste constante autour de 1, pas de convergence L1.")
```
