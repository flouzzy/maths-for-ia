# TP 3 : Calcul Numérique de l'Erreur $L^2$

Objectif : Approximer numériquement l'espérance quadratique pour illustrer le cas d'une suite qui converge en probabilité mais diverge en norme $L^2$.

```python
def density_Xn(omega, n):
    """
    Variable aléatoire X_n(omega) = sqrt(n) * 1_]0, 1/n](omega)
    """
    if 0 < omega <= 1.0 / n:
        return n**0.5
    return 0.0

def compute_L2_error(n, num_samples=100000):
    """
    Calcule par intégration numérique (Monte Carlo) E[|X_n - 0|^2]
    """
    import random

    sum_sq = 0.0
    for _ in range(num_samples):
        omega = random.uniform(0, 1)
        val = density_Xn(omega, n)
        sum_sq += val ** 2

    return sum_sq / num_samples

if __name__ == "__main__":
    # La théorie dit que l'erreur L2 vaut exactement 1 pour tout n
    err_10 = compute_L2_error(10)
    err_100 = compute_L2_error(100)

    assert 0.8 < err_10 < 1.2, "L'erreur L2 théorique de 1 n'est pas vérifiée"
    assert 0.8 < err_100 < 1.2, "L'erreur L2 théorique de 1 n'est pas vérifiée"
    print("Convergence L2 réfutée, erreur maintenue à 1.0.")

```
