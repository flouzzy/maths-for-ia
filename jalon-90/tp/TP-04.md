# TP 4 : Illustration du Théorème Central Limite (Convergence en Loi)

Objectif : Simuler des sommes de variables uniformes et comparer numériquement leur fonction de répartition empirique avec celle d'une Gaussienne pour valider la convergence en loi.

```python
import random

def generate_sum_uniforms(n):
    """
    Génère Z_n = sqrt(3/n) * sum(U_i) où U_i ~ Uniform(-1, 1).
    """
    s = sum(random.uniform(-1, 1) for _ in range(n))
    return (3.0 / n)**0.5 * s

def empirical_cdf(data, x):
    """
    Calcule la proportion des valeurs de data <= x.
    """
    count = sum(1 for d in data if d <= x)
    return count / len(data)

def normal_cdf_approx(x):
    """
    Approximation numérique grossière de l'intégrale de Gauss (sans scipy).
    """
    import math
    if x < -5: return 0.0
    if x > 5: return 1.0

    steps = 1000
    start = -5.0
    dx = (x - start) / steps
    integral = 0.0
    for i in range(steps):
        t = start + i * dx
        val = (1.0 / math.sqrt(2 * math.pi)) * math.exp(-0.5 * t * t)
        integral += val * dx
    return integral

if __name__ == "__main__":
    samples = [generate_sum_uniforms(50) for _ in range(5000)]

    # Testons le point x = 0 (la médiane doit être environ 0.5)
    emp_prob_0 = empirical_cdf(samples, 0.0)
    theo_prob_0 = normal_cdf_approx(0.0)

    assert abs(emp_prob_0 - theo_prob_0) < 0.05, "L'approximation par la loi normale au point 0 est mauvaise."
    print("La convergence en loi vers la normale est numériquement validée.")

```
