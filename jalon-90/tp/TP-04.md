# TP 4 : Illustration de la bosse glissante (Convergence en proba, divergence L1)

Ce TP simule l'exemple du cours où une suite converge en probabilité vers 0 mais ne converge pas en espérance.

```python
import random

def simule_bosse_glissante(n):
    # X_n prend la valeur n avec proba 1/n, sinon 0
    u = random.random()
    if u < 1.0 / n:
        return n
    else:
        return 0

def evaluation_bosse(n, n_simulations):
    valeurs = [simule_bosse_glissante(n) for _ in range(n_simulations)]

    # Convergence en proba : Prob(X_n > epsilon)
    epsilon = 0.5
    proba_emp = sum(1 for v in valeurs if v > epsilon) / n_simulations

    # Convergence L1 : Esperance(|X_n|)
    esperance_emp = sum(valeurs) / n_simulations

    return proba_emp, esperance_emp

p_10, e_10 = evaluation_bosse(10, 5000)
p_100, e_100 = evaluation_bosse(100, 5000)
p_1000, e_1000 = evaluation_bosse(1000, 5000)

# La proba tend vers 0
assert p_10 > p_100 > p_1000 or p_1000 < 0.01
# L'espérance reste autour de 1 (elle ne tend pas vers 0)
assert abs(e_10 - 1.0) < 0.5
assert abs(e_100 - 1.0) < 0.5
assert abs(e_1000 - 1.0) < 0.5

print("TP4: Bosse glissante validée.")
```
