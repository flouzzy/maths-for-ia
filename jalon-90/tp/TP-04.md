# TP 04 : Divergence du Maximum de Variables Gaussiennes Indépendantes

Démontrons par des méthodes de Monte Carlo (sans bibliothèques externes) que le maximum d'une suite i.i.d. de lois normales standard n'admet aucune borne stochastique : il converge en probabilité vers l'infini.

```python
import random
import math

def Box_Muller():
    # Generation de deux gaussiennes i.i.d N(0,1) via Box-Muller pur
    U1 = random.random()
    U2 = random.random()
    # Eviter le log(0)
    while U1 == 0.0:
        U1 = random.random()

    R = math.sqrt(-2.0 * math.log(U1))
    Theta = 2.0 * math.pi * U2
    Z0 = R * math.cos(Theta)
    return Z0

def echantillonner_maximum_gaussien(n):
    # Retourne le max d'un echantillon de taille n
    maximum = -1e9
    for _ in range(n):
        z = Box_Muller()
        if z > maximum:
            maximum = z
    return maximum

def esperance_empirique_maximum(n, n_simulations):
    somme_max = 0.0
    for _ in range(n_simulations):
        somme_max += echantillonner_maximum_gaussien(n)
    return somme_max / n_simulations

# Exécution du test d'explosion asymptotique
n_simul = 500

max_N10 = esperance_empirique_maximum(10, n_simul)
max_N100 = esperance_empirique_maximum(100, n_simul)
max_N1000 = esperance_empirique_maximum(1000, n_simul)

print("Espérance du Maximum (N=10)   :", max_N10)
print("Espérance du Maximum (N=100)  :", max_N100)
print("Espérance du Maximum (N=1000) :", max_N1000)

# Validation stricte des propriétés probabilistes de l'extrêmum
assert max_N1000 > max_N100 > max_N10, "L'espérance du maximum doit croître strictement avec N"
assert max_N1000 > 2.5, "Le maximum pour N=1000 doit significativement s'éloigner du centre"
```
