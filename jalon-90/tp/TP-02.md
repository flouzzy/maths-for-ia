# TP 2 : Le Théorème Central Limite (Convergence en Loi)

Nous allons illustrer la convergence en loi en simulant la somme de variables aléatoires uniformes, qui doit converger vers une loi Normale.
Nous utilisons un estimateur empirique de la fonction de répartition et vérifions l'écart maximal.

```python
import random
import math

def generate_uniform():
    # Variable uniforme sur [-0.5, 0.5] de moyenne 0 et variance 1/12
    return random.random() - 0.5

def generate_sum(n):
    return sum(generate_uniform() for _ in range(n))

n_vars = 12 # Classiquement utilisé pour approcher une loi normale car var(somme) = 12 * (1/12) = 1
nb_samples = 10000
samples = [generate_sum(n_vars) for _ in range(nb_samples)]
samples.sort()

# Fonction de répartition empirique (ECDF)
def ecdf(x, sorted_samples):
    # Recherche binaire manuelle (from scratch) pour compter les éléments <= x
    left, right = 0, len(sorted_samples) - 1
    count = 0
    while left <= right:
        mid = (left + right) // 2
        if sorted_samples[mid] <= x:
            count = mid + 1
            left = mid + 1
        else:
            right = mid - 1
    return count / len(sorted_samples)

# Fonction de répartition de la loi normale standard (approximation numérique)
def cdf_normale_standard(x):
    # Utilisation de la fonction d'erreur (erf)
    return (1.0 + math.erf(x / math.sqrt(2.0))) / 2.0

# Comparaison des deux fonctions de répartition
points_test = [-2.0, -1.0, 0.0, 1.0, 2.0]
max_error = 0.0

print("Comparaison de la fonction de répartition (ECDF vs Normale):")
for pt in points_test:
    emp = ecdf(pt, samples)
    theo = cdf_normale_standard(pt)
    err = abs(emp - theo)
    max_error = max(max_error, err)
    print(f"x = {pt:>4} | ECDF = {emp:.4f} | Theo = {theo:.4f} | Err = {err:.4f}")

# La théorie de Kolmogorov-Smirnov assure que l'erreur doit être faible
assert max_error < 0.05, f"Erreur maximale trop grande ({max_error}), la convergence en loi n'est pas observée."
print(f"Validation mathématique : L'erreur maximale de la fonction de répartition est très faible ({max_error:.4f}), confirmant la convergence en loi.")
```
