# TP 2 : Convergence du maximum d'une loi uniforme

Ce TP simule l'exercice sur le maximum de variables uniformes et valide sa convergence en probabilité vers 1.

```python
import random

def max_uniforme(n):
    max_val = 0.0
    for _ in range(n):
        val = random.random()
        if val > max_val:
            max_val = val
    return max_val

def proba_erreur_max(n_simulations, n, epsilon):
    erreurs = 0
    for _ in range(n_simulations):
        m = max_uniforme(n)
        # La limite théorique est 1. On compte les cas où |m - 1| > epsilon
        if abs(m - 1.0) > epsilon:
            erreurs += 1
    return erreurs / n_simulations

# Validation :
epsilon = 0.05
p_10 = proba_erreur_max(1000, 10, epsilon)
p_50 = proba_erreur_max(1000, 50, epsilon)
p_200 = proba_erreur_max(1000, 200, epsilon)

assert p_10 > p_50, "La variance diminue avec n."
assert p_50 > p_200, "La variance diminue avec n."
assert p_200 < 0.01, "La probabilité d'erreur doit être négligeable pour n=200."
print("TP2: Convergence du maximum validée.")
```
