# TP 2 : La machine à écrire glissante en Python

Objectif : Coder l'exemple pathologique d'une suite qui converge en probabilité mais pas presque sûrement, et vérifier par calcul numérique les probabilités.

```python
def get_interval(n):
    """
    Retourne les bornes de l'intervalle I_n pour la variable aléatoire X_n.
    Pour n = 2^k + j, I_n = [j/2^k, (j+1)/2^k].
    """
    if n == 1:
        return 0.0, 1.0

    k = 0
    temp_n = n
    while temp_n > 1:
        temp_n //= 2
        k += 1

    j = n - (2**k)
    length = 1.0 / (2**k)
    return j * length, (j + 1) * length

def evaluate_Xn(n, omega):
    start, end = get_interval(n)
    if start <= omega <= end:
        return 1.0
    return 0.0

if __name__ == "__main__":
    # Test de convergence en probabilité : la mesure de l'intervalle tend vers 0
    # La mesure (probabilité) est la différence entre fin et début
    for n in [10, 100, 1000]:
        s, e = get_interval(n)
        prob = e - s
        assert prob <= 2.0 / n, "La probabilité de l'erreur ne tend pas vers 0 !"

    # Test de non-convergence presque sûre pour un omega fixé
    omega_test = 0.42
    values = [evaluate_Xn(n, omega_test) for n in range(1, 100)]
    assert 1.0 in values and 0.0 in values, "La suite ne devrait pas converger pour un omega donné"
    print("La machine à écrire glisse correctement !")

```
