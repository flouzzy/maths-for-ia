```python
# tp_05_lemme_fatou.py
# Le Lemme de Fatou énonce une inégalité stricte lorsque la masse 's'échappe' à l'infini.
# Nous simulons une bosse coulissante : f_n(x) = n sur [n, n + 1/n], 0 ailleurs.

def integrale_bosse(n):
    # La bosse est un rectangle de hauteur n et de largeur 1/n.
    # Son intégrale vaut n * (1/n) = 1.0, indépendamment de n.
    return 1.0

def limite_inferieure_bosse_en_x(x):
    # Pour tout point fixe x, la bosse glisse vers l'infini.
    # Il existe un rang N pour lequel la bosse dépasse définitivement x.
    # Donc f_n(x) = 0 pour tout n > N. La limite (et donc la liminf) est 0.
    return 0.0

def integrale_limite_inferieure():
    # L'intégrale de la fonction constante nulle est 0.0
    return 0.0

def test_lemme_fatou():
    # Calcul des limites

    # 1. liminf de l'intégrale (limite de la suite constante 1.0)
    suite_integrales = [integrale_bosse(n) for n in range(1, 100)]
    liminf_integrales = suite_integrales[-1] # Vaut exactement 1.0

    # 2. Intégrale de la liminf
    int_de_la_liminf = integrale_limite_inferieure() # Vaut exactement 0.0

    # Application et validation du Lemme de Fatou :
    # \int \liminf f_n \leq \liminf \int f_n

    assert int_de_la_liminf <= liminf_integrales, "Le Lemme de Fatou mathématique s'est effondré."

    # Vérification de l'inégalité stricte (fuite de masse à l'infini)
    assert int_de_la_liminf < liminf_integrales, "L'exemple devrait provoquer une inégalité stricte."

    print(f"TP 05 Validé : Int(\liminf f) = {int_de_la_liminf} < \liminf(Int(f)) = {liminf_integrales}")

if __name__ == "__main__":
    test_lemme_fatou()
```
