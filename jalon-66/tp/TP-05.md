# TP 5 : Application IA (Divergence KL discrète)
L'intégrale de Lebesgue avec des mesures discrètes permet de calculer l'entropie relative.

```python
import math

def divergence_kl_discrete(p, q, domaine):
    # Calcule D_KL(P || Q) = Somme P(x) log(P(x)/Q(x))
    kl = 0.0
    for x in domaine:
        px = p.get(x, 0.0)
        qx = q.get(x, 0.0)
        if px > 0:
            if qx == 0:
                return float('inf') # Divergence infinie
            kl += px * math.log(px / qx)
    return kl

# Validation
if __name__ == "__main__":
    domaine = [0, 1, 2]

    # Distribution P
    P = {0: 0.2, 1: 0.5, 2: 0.3}

    # Distribution Q
    Q = {0: 0.1, 1: 0.6, 2: 0.3}

    # Distribution uniforme U
    U = {0: 1/3, 1: 1/3, 2: 1/3}

    kl_pq = divergence_kl_discrete(P, Q, domaine)
    kl_pu = divergence_kl_discrete(P, U, domaine)

    print(f"D_KL(P||Q) = {kl_pq:.4f}")
    print(f"D_KL(P||Uniform) = {kl_pu:.4f}")

    assert kl_pq >= 0, "Divergence KL doit etre positive"
    assert divergence_kl_discrete(P, P, domaine) == 0.0, "D_KL(P||P) doit valoir 0"
```
