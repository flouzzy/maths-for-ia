# TP 4 : Linéarité de l'intégrale pour des fonctions simples

Ce TP implémente en Python pur les concepts de base vus dans le cours sur l'intégrale de Lebesgue pour les fonctions simples et positives.

```python
def add_simple_functions(intervals1, values1, intervals2, values2):
    """
    Raffinement de partitions pour sommer deux fonctions simples.
    Pour simplifier, on suppose que les intervalles sont disjoints en interne
    et que l'union des intervalles couvre un meme segment global.
    Cette fonction cree les intersections.
    """
    result_intervals = []
    result_values = []

    for (a1, b1), v1 in zip(intervals1, values1):
        for (a2, b2), v2 in zip(intervals2, values2):
            start = max(a1, a2)
            end = min(b1, b2)
            if start < end:
                result_intervals.append((start, end))
                result_values.append(v1 + v2)

    return result_intervals, result_values

def calculate_integral(intervals, values):
    return sum(v * (b - a) for (a, b), v in zip(intervals, values))

def test_linearity():
    int1 = [(0, 2), (2, 4)]
    val1 = [1.0, 3.0]

    int2 = [(0, 1), (1, 4)]
    val2 = [2.0, 0.5]

    # Sum functions
    int_sum, val_sum = add_simple_functions(int1, val1, int2, val2)
    int_expected = [(0, 1), (1, 2), (2, 4)]
    val_expected = [1.0+2.0, 1.0+0.5, 3.0+0.5]

    assert int_sum == int_expected
    assert val_sum == [3.0, 1.5, 3.5]

    i1 = calculate_integral(int1, val1) # 1*2 + 3*2 = 8
    i2 = calculate_integral(int2, val2) # 2*1 + 0.5*3 = 2 + 1.5 = 3.5
    i_sum = calculate_integral(int_sum, val_sum) # 3*1 + 1.5*1 + 3.5*2 = 3 + 1.5 + 7 = 11.5

    assert abs(i_sum - (i1 + i2)) < 1e-9
    assert abs(i_sum - 11.5) < 1e-9

    print("TP 04 passed.")

if __name__ == "__main__":
    test_linearity()
```
