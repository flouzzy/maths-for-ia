## TP 4 : Simulation de la mesure extérieure

**Objectif :** Coder une approche naïve de la mesure extérieure de Lebesgue pour un ensemble de points.

```python
def outer_measure_approx(points, epsilon):
    intervals = []
    for p in points:
        intervals.append([p - epsilon/2, p + epsilon/2])

    intervals.sort(key=lambda x: x[0])
    merged = []

    for interval in intervals:
        if not merged:
            merged.append(interval)
        else:
            last = merged[-1]
            if interval[0] <= last[1]:
                last[1] = max(last[1], interval[1])
            else:
                merged.append(interval)

    measure = 0
    for interval in merged:
        measure += (interval[1] - interval[0])

    return measure

if __name__ == "__main__":
    points = [0.1, 0.15, 0.4, 0.42, 0.8]
    eps = 0.1
    m = outer_measure_approx(points, eps)
    print("Outer measure approx:", m)
    assert m <= len(points) * eps
```
