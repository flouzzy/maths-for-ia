### Intégration d'une fonction avec support de mesure nulle

Implémentation du théorème montrant que si une fonction n'est strictement positive que sur un ensemble de mesure nulle, alors son intégrale est nulle.

```python
def integrale_support_nul(valeurs_x, fonction_f, oracle_mesure):
    integrale = 0.0

    for x in valeurs_x:
        val = fonction_f(x)
        # La fonction oracle_mesure retourne la mesure du singleton {x}
        mesure_x = oracle_mesure(x)

        # Ajout a l'integrale globale
        integrale = integrale + (val * mesure_x)

    return integrale

# L'espace est {0, 1, 2, 3, 4, 5}
# L'oracle de mesure donne un poids 0.5 a '0' et '1', et un poids STRICTEMENT 0 aux autres.
def mon_oracle(x):
    if x == 0 or x == 1:
        return 0.5
    return 0.0

# La fonction vaut 1000 sur '4' et '5' (qui sont de mesure nulle)
def ma_fonction(x):
    if x >= 4:
        return 1000.0
    return 0.0

res_integrale = integrale_support_nul([0, 1, 2, 3, 4, 5], ma_fonction, mon_oracle)

# L'integrale doit etre rigoureusement nulle car la fonction s'annule sur les ensembles de mesure non nulle
assert res_integrale == 0.0, "Theoreme de la mesure nulle invalide"
print("Validation TP4 reussie : Integrale =", res_integrale)
```
