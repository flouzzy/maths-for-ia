# TP 4 : Simulation de la convergence monotone croissante (Beppo-Levi)

```python
# tp_04_beppo_levi.py
# Vérification du passage à la limite sous l'intégrale pour une suite croissante.

def eval_f_n(n, xs):
    # Suite croissante de fonctions : f_n(x) = (1 - x^n) * x^2 sur [0,1]
    return [(1.0 - (x ** n)) * (x ** 2) for x in xs]

def eval_f_limite(xs):
    # Limite simple : f(x) = x^2 (sauf en 1, mais de mesure nulle)
    return [x ** 2 for x in xs]

def integrale_riemann_approche(valeurs, pas):
    # Approximation par rectangles pour évaluer l'intégrale sur une grille fine
    return sum(valeurs) * pas

def test_convergence_monotone():
    pas = 0.001
    grille_x = [i * pas for i in range(int(1/pas))] # Intervalle [0, 1[

    integrales_fn = []

    # Calcul itératif des intégrales de la suite
    for n in range(1, 20):
        valeurs_n = eval_f_n(n, grille_x)
        int_n = integrale_riemann_approche(valeurs_n, pas)
        integrales_fn.append(int_n)

        # Validation de croissance de la suite des intégrales
        if n > 1:
            assert int_n >= integrales_fn[-2], "Erreur: Les intégrales doivent croître par le théorème."

    # Calcul de l'intégrale de la limite
    valeurs_lim = eval_f_limite(grille_x)
    integrale_lim_theorique = integrale_riemann_approche(valeurs_lim, pas)

    # Validation du théorème de Beppo-Levi : la limite des intégrales est l'intégrale de la limite
    ecart_beppo_levi = abs(integrale_lim_theorique - integrales_fn[-1])

    # La convergence est garantie
    assert ecart_beppo_levi < 0.05, f"Violation de Beppo-Levi, écart final = {ecart_beppo_levi}"

    print("TP 04 Validé : Le théorème de convergence monotone est empiriquement vérifié.")

if __name__ == "__main__":
    test_convergence_monotone()

```
