# TP 4 : Simulation de l'inégalité de Markov
Vérification empirique de l'inégalité de Markov sur une distribution.

```python
import random

def generer_echantillon_positif(n):
    # Genere un echantillon selon une loi exponentielle de parametre lambda=1
    return [random.expovariate(1.0) for _ in range(n)]

def verifier_markov(echantillon, t):
    n = len(echantillon)
    esperance_empirique = sum(echantillon) / n

    # Mesure empirique de {x | f(x) >= t}
    count_sup_t = sum(1 for x in echantillon if x >= t)
    prob_empirique = count_sup_t / n

    borne_markov = esperance_empirique / t
    return prob_empirique, borne_markov

# Validation
if __name__ == "__main__":
    random.seed(42)
    echantillon = generer_echantillon_positif(10000)

    t_values = [2.0, 3.0, 5.0]
    for t in t_values:
        p, borne = verifier_markov(echantillon, t)
        print(f"t={t}: P(X >= t) = {p:.4f} <= Borne Markov = {borne:.4f}")
        assert p <= borne, f"Inégalité de Markov violée pour t={t}"
```
