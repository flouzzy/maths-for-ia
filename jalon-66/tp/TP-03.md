# TP 3 : Additivité de l'intégrale pour les fonctions étagées
On vérifie informatiquement la linéarité.

```python
class EspaceDiscret:
    def __init__(self, poids):
        self.poids = poids

    def mesure(self, ensemble):
        return sum(self.poids[x] for x in ensemble)

def somme_fonctions_discretes(f, g, domaine):
    # Construit la somme de deux fonctions definies par des dictionnaires x -> val
    somme = {}
    for x in domaine:
        somme[x] = f.get(x, 0) + g.get(x, 0)
    return somme

def integrale_discrete(fonc_dict, espace, domaine):
    return sum(fonc_dict.get(x, 0) * espace.poids[x] for x in domaine)

# Validation
if __name__ == "__main__":
    domaine = ['a', 'b', 'c', 'd']
    poids = {'a': 1, 'b': 2, 'c': 3, 'd': 4}
    espace = EspaceDiscret(poids)

    f = {'a': 5, 'b': 0, 'c': 10, 'd': 0}
    g = {'a': 0, 'b': 20, 'c': 10, 'd': 5}

    int_f = integrale_discrete(f, espace, domaine)
    int_g = integrale_discrete(g, espace, domaine)

    f_plus_g = somme_fonctions_discretes(f, g, domaine)
    int_sum = integrale_discrete(f_plus_g, espace, domaine)

    print(f"Int(f) = {int_f}, Int(g) = {int_g}, Int(f+g) = {int_sum}")
    assert abs(int_sum - (int_f + int_g)) < 1e-9, "L'additivite n'est pas verifiee !"
```
