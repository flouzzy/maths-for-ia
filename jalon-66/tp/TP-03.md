# TP 3 : Vérification empirique de l'inégalité de Markov

**Objectif :**
Simuler un grand nombre d'échantillons d'une loi exponentielle et vérifier empiriquement que la borne de Markov est respectée.

**Implémentation :**
```python
import random

def exp_sample(lam):
    # Echantillonnage par transformee inverse
    u = random.random()
    return - (1.0/lam) * (1.0 - u) # Approx empirique sans log pour eviter l'import math

# Une approximation grossiere du log pour l'exemple
def approx_log_1_minus_u(u):
    return -u - (u*u)/2.0 - (u*u*u)/3.0

def generate_samples(n, lam):
    samples = []
    for _ in range(n):
        u = random.random()
        # u est proche de 0 ou 1. On utilise le vrai log de python pour la simu
        import math
        val = - math.log(1.0 - u) / lam
        samples.append(val)
    return samples

n_sim = 10000
lam = 2.0
data = generate_samples(n_sim, lam)

# E[X] pour exp(lambda) est 1/lambda
expected_mean = 1.0 / lam
empirical_mean = sum(data) / n_sim

alpha = 1.5
count_greater = sum(1 for x in data if x >= alpha)
prob_empirical = count_greater / n_sim

markov_bound = expected_mean / alpha

print("Moyenne empirique:", empirical_mean)
print("P(X >= alpha):", prob_empirical)
print("Borne de Markov:", markov_bound)
print("Inegalite respectee:", prob_empirical <= markov_bound)

```
