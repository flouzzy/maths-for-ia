## TP 3 : Intégrale de Lebesgue d'une Fonction Simple

Maintenant que nous avons notre espace et nos fonctions simples, nous pouvons coder l'intégrale de Lebesgue pour ces fonctions de base.

```python
class EspaceMesureFini:
    def __init__(self, omega):
        self.omega = set(omega)

    def mesure_comptage(self, A):
        return len(set(A).intersection(self.omega))

def integrer_fonction_simple(s, espace, mesure_func):
    integrale_totale = 0
    for valeur, points in s.composantes.items():
        mesure_ensemble = mesure_func(points)
        integrale_totale += valeur * mesure_ensemble
    return integrale_totale

class FonctionSimple:
    def __init__(self):
        self.composantes = {}
    def ajouter_palier(self, valeur, points):
        self.composantes[valeur] = set(points)

# Application
X = EspaceMesureFini([1, 2, 3, 4, 5, 6, 7])
f = FonctionSimple()
f.ajouter_palier(2, [1, 2])       # Valeur 2 sur 2 points
f.ajouter_palier(4, [3, 4, 5])    # Valeur 4 sur 3 points

# Integrale attendue : 2*2 + 4*3 = 4 + 12 = 16
resultat = integrer_fonction_simple(f, X, X.mesure_comptage)
assert resultat == 16, "Erreur de calcul integral"
print(f"TP3: Integrale calculee avec succes = {resultat}")
```
