# TP 3 : Vérification empirique de l'inégalité de Markov

```python
# tp_03_inegalite_markov.py
# L'inégalité de Markov borne la probabilité des valeurs extrêmes par l'espérance.

def integrale_empirique_discrete(valeurs):
    # Dans un contexte uniforme discret, l'intégrale (espérance) est la moyenne
    return sum(valeurs) / len(valeurs)

def mesure_queue_distribution(valeurs, a):
    # Compte la proportion (mesure) des éléments dépassant le seuil 'a'
    elements_extremes = [x for x in valeurs if x >= a]
    return len(elements_extremes) / len(valeurs)

def test_markov():
    # Une distribution arbitraire de valeurs positives (loi exponentielle simulée)
    donnees = [0.1, 0.5, 1.2, 0.8, 3.4, 0.2, 1.5, 2.1, 0.4, 5.0]

    # Validation stricte de positivité (Hypothèse de Markov)
    for x in donnees:
        assert x >= 0, "L'inégalité de Markov nécessite une fonction positive."

    esperance = integrale_empirique_discrete(donnees)

    # Test pour plusieurs seuils a > 0
    seuils = [1.0, 2.0, 4.0, 10.0]

    for a in seuils:
        mesure_reelle = mesure_queue_distribution(donnees, a)
        borne_markov = esperance / a

        # Validation mathématique de l'inégalité
        assert mesure_reelle <= borne_markov, f"Violation de Markov au seuil {a} : {mesure_reelle} > {borne_markov}"

    print("TP 03 Validé : L'inégalité de Markov est formellement respectée par le jeu de données.")

if __name__ == "__main__":
    test_markov()

```
