# TP 3 : Intégrale de Lebesgue par approximation sur un domaine discret

Ce TP implémente en Python pur les concepts de base vus dans le cours sur l'intégrale de Lebesgue pour les fonctions simples et positives.

```python
def lebesgue_integral_approx_discrete(domain, f, measure, n):
    """
    Calcule l'integrale de Lebesgue approchee de la fonction f sur un domaine discret.
    L'approche se fait par le supremum de fonctions simples bornees par f,
    ou on utilise le decoupage dyadique d'ordre n.
    domain: list of discrete elements.
    f: dict mapping elements to positive floats.
    measure: dict mapping elements to positive floats (the measure).
    n: depth of dyadic approximation.
    """
    integral = 0.0
    for x in domain:
        fx = f[x]
        assert fx >= 0, "Function must be positive."
        if fx >= n:
            val = float(n)
        else:
            k = int(fx * (2**n))
            val = k / (2**n)
        integral += val * measure[x]
    return integral

def test_discrete_lebesgue_integral():
    domain = ['a', 'b', 'c']
    f = {'a': 1.25, 'b': 0.75, 'c': 3.5}
    measure = {'a': 2.0, 'b': 1.0, 'c': 0.5}

    # Avec n=1 (grid 0, 0.5, threshold 1)
    # f('a') = 1.25 >= 1 -> val = 1.0
    # f('b') = 0.75 -> k=1 -> val = 0.5
    # f('c') = 3.5 >= 1 -> val = 1.0
    # Int: 1.0 * 2.0 + 0.5 * 1.0 + 1.0 * 0.5 = 2.0 + 0.5 + 0.5 = 3.0
    res_n1 = lebesgue_integral_approx_discrete(domain, f, measure, 1)
    assert abs(res_n1 - 3.0) < 1e-9

    # Avec n=2 (grid 0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 1.75, threshold 2)
    # f('a') = 1.25 -> k=5 -> val = 1.25
    # f('b') = 0.75 -> k=3 -> val = 0.75
    # f('c') = 3.5 >= 2 -> val = 2.0
    # Int: 1.25 * 2.0 + 0.75 * 1.0 + 2.0 * 0.5 = 2.5 + 0.75 + 1.0 = 4.25
    res_n2 = lebesgue_integral_approx_discrete(domain, f, measure, 2)
    assert abs(res_n2 - 4.25) < 1e-9

    print("TP 03 passed.")

if __name__ == "__main__":
    test_discrete_lebesgue_integral()
```
