# TP 1 : Approximation de l'aire sous une parabole par fonctions étagées

```python
# tp_01_approximation_parabole.py
# Cet algorithme implémente la construction explicite de l'approximation d'une fonction mesurable
# par une suite de fonctions étagées croissantes, selon la définition de Lebesgue.

def f(x):
    # La fonction mesurable à intégrer : f(x) = x^2
    return x * x

def s_n(x, n):
    # Implémente l'approximation s_n(x) en découpant l'axe des ordonnées [0, n] en 2^n pas
    val = f(x)
    if val >= n:
        return float(n)

    # Résolution de l'inégalité : k/(2^n) <= f(x) < (k+1)/(2^n)
    pas = 2**n
    k = int(val * pas)
    return k / pas

def test_approximation():
    x_test = 1.5
    f_val = f(x_test) # 2.25

    suite_approximations = []

    for n in range(1, 10):
        val_etagee = s_n(x_test, n)
        suite_approximations.append(val_etagee)

        # Validation mathématique 1 : L'approximation est minorante
        assert val_etagee <= f_val, f"Erreur: s_n({x_test}) = {val_etagee} dépasse f({x_test}) = {f_val}"

        # Validation mathématique 2 : La suite des approximations est croissante
        if n > 1:
            assert val_etagee >= suite_approximations[-2], "Erreur: La suite n'est pas monotone croissante."

    # Validation mathématique 3 : Convergence asymptotique
    ecart = abs(f_val - suite_approximations[-1])
    assert ecart < 1e-2, f"Erreur: Mauvaise convergence, écart final de {ecart}"

    print("TP 01 Validé : La construction théorique de s_n est correcte en pratique.")

if __name__ == "__main__":
    test_approximation()

```
