# TP 1 : Représentation et calcul de fonctions étagées
On modélise l'intégrale d'une fonction étagée sur un espace discret fini.

```python
class EspaceMesure:
    def __init__(self, ensemble, poids):
        # ensemble: list of elements
        # poids: dict mapping elements to their measure
        self.ensemble = ensemble
        self.poids = poids

    def mesure(self, sous_ensemble):
        return sum(self.poids[x] for x in sous_ensemble if x in self.poids)

class FonctionEtagee:
    def __init__(self, valeurs_dict):
        # valeurs_dict: dict mapping values (a_i) to list of elements (A_i)
        self.valeurs = valeurs_dict

    def integrale(self, espace):
        resultat = 0.0
        for val, sous_ensemble in self.valeurs.items():
            resultat += val * espace.mesure(sous_ensemble)
        return resultat

# Validation
if __name__ == "__main__":
    X = [1, 2, 3, 4, 5]
    poids = {1: 0.1, 2: 0.2, 3: 0.1, 4: 0.5, 5: 0.1}
    espace = EspaceMesure(X, poids)

    # f(x) = 10 sur {1, 2}, 20 sur {3}, 0 ailleurs
    f = FonctionEtagee({10: [1, 2], 20: [3]})

    # Integrale = 10*(0.1+0.2) + 20*(0.1) = 3 + 2 = 5.0
    val_int = f.integrale(espace)
    print(f"Integrale calculee : {val_int}")
    assert abs(val_int - 5.0) < 1e-6, "Erreur de calcul d'integrale"
```
