# TP 2 : Approximation d'une fonction continue par une suite de fonctions étagées

Ce TP implémente la construction théorique du théorème d'approximation : tronquer la fonction f(x) = x^2 et la discrétiser sur l'axe des ordonnées pour construire s_n <= f.

```python
def f(x):
    return x * x

def s_n(x, n):
    # Truncate at height n
    if f(x) >= n:
        return n

    # Partition [0, n) into intervals of size 1/2^n
    # Find k such that k/2^n <= f(x) < (k+1)/2^n
    step = 1.0 / (2**n)
    val = f(x)
    k = int(val / step)
    return k * step

if __name__ == "__main__":
    x = 1.5 # f(x) = 2.25

    # n=1: max height 1. f(x) = 2.25 >= 1, so s_1(x) = 1
    assert s_n(x, 1) == 1.0

    # n=2: max height 2. f(x) = 2.25 >= 2, so s_2(x) = 2
    assert s_n(x, 2) == 2.0

    # n=3: max height 3. step = 1/8 = 0.125
    # val = 2.25. k = 2.25 / 0.125 = 18.
    # s_3(x) = 18 * 0.125 = 2.25
    assert abs(s_n(x, 3) - 2.25) < 1e-6

    # Ensure monotonicity for x = 0.8: f(x) = 0.64
    s1 = s_n(0.8, 1) # step = 0.5. k = int(0.64/0.5) = 1. s1 = 0.5
    s2 = s_n(0.8, 2) # step = 0.25. k = int(0.64/0.25) = 2. s2 = 0.5
    s3 = s_n(0.8, 3) # step = 0.125. k = int(0.64/0.125) = 5. s3 = 0.625
    assert s1 <= s2 <= s3 <= f(0.8)

    print("TP2 OK")
```
