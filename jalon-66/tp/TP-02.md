# TP 2 : Calcul analytique strict de l'intégrale d'une fonction étagée

```python
# tp_02_integrale_etagee.py
# Implémentation du calcul formel de l'intégrale d'une fonction étagée sur une partition.

def mesure_lebesgue_intervalle(a, b):
    # La mesure de Lebesgue standard sur un intervalle réel
    if a >= b:
        return 0.0
    return b - a

def integrale_etagee(valeurs_alpha, partition_intervalles):
    # Somme canonique : \sum \alpha_i \mu(A_i)
    assert len(valeurs_alpha) == len(partition_intervalles), "La forme canonique exige un alignement parfait."

    integrale = 0.0
    for alpha, (borne_inf, borne_sup) in zip(valeurs_alpha, partition_intervalles):
        assert alpha >= 0, "La théorie initiale de Lebesgue s'applique aux fonctions étagées POSITIVES."
        integrale += alpha * mesure_lebesgue_intervalle(borne_inf, borne_sup)

    return integrale

def test_integrale():
    # Définition de la fonction : s(x) = 3 sur [0, 2] et 5 sur [4, 5]
    alphas = [3.0, 5.0]
    intervalles = [(0.0, 2.0), (4.0, 5.0)]

    # Calcul manuel attendu : 3 * (2 - 0) + 5 * (5 - 4) = 6 + 5 = 11
    valeur_calculee = integrale_etagee(alphas, intervalles)
    valeur_theorique = 11.0

    # Validation mathématique
    assert abs(valeur_calculee - valeur_theorique) < 1e-9, f"Erreur de calcul : trouvé {valeur_calculee}, attendu {valeur_theorique}"
    print(f"TP 02 Validé : L'intégrale de Lebesgue de la fonction étagée vaut bien {valeur_calculee}.")

if __name__ == "__main__":
    test_integrale()
```
