## TP 2 : Construction de Fonctions Simples (Étagées)

Nous allons représenter informatiquement une fonction simple. Mathématiquement, c'est une somme pondérée de fonctions indicatrices d'ensembles disjoints.

```python
class FonctionSimple:
    def __init__(self):
        # Stockage sous la forme d'un dictionnaire {valeur: ensemble_de_points}
        self.composantes = {}

    def ajouter_palier(self, valeur, points):
        assert valeur >= 0, "Pour l'instant, on s'interesse aux fonctions positives"
        points_set = set(points)
        # Verification que l'ensemble est disjoint des precedents (partitionnement)
        for val_existante, pts_existants in self.composantes.items():
            assert len(points_set.intersection(pts_existants)) == 0, "Les ensembles doivent etre disjoints"

        self.composantes[valeur] = points_set

    def evaluer(self, x):
        for valeur, points in self.composantes.items():
            if x in points:
                return valeur
        return 0 # 0 par defaut si en dehors du support

s = FonctionSimple()
s.ajouter_palier(5, [1, 2])
s.ajouter_palier(3, [3, 4, 5])

assert s.evaluer(1) == 5
assert s.evaluer(4) == 3
assert s.evaluer(6) == 0
print("TP2: Tests passes")
```
