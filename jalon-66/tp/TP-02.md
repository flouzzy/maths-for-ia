# TP 2 : Approximation d'une fonction mesurable par des étagées
On approche une fonction continue sur [0, 1] par des fonctions étagées, illustrant le passage au supremum.

```python
def mesure_lebesgue_intervalle(a, b):
    return max(0.0, b - a)

def approche_etagee_inferieure(f, a, b, n_steps):
    # Approche la fonction f par une fonction constante par morceaux sur des sous-intervalles
    step = (b - a) / n_steps
    integrale = 0.0

    for i in range(n_steps):
        x_start = a + i * step
        x_end = a + (i + 1) * step
        # Pour une vraie borne inf sur l'intervalle on prendrait le min.
        # Ici on simplifie en prenant la valeur a gauche (valable si f est croissante)
        val_inf = f(x_start)
        integrale += val_inf * mesure_lebesgue_intervalle(x_start, x_end)

    return integrale

# Validation
if __name__ == "__main__":
    # f(x) = x**2, integrale sur [0,1] vaut 1/3
    f = lambda x: x**2

    # Avec n=100
    int_approx = approche_etagee_inferieure(f, 0.0, 1.0, 100)
    print(f"Integrale approchee (n=100) : {int_approx}")
    assert int_approx < 1/3, "L'approche inferieure doit minorer l'integrale exacte"

    # Avec n=1000, la precision doit etre meilleure
    int_approx_1000 = approche_etagee_inferieure(f, 0.0, 1.0, 1000)
    print(f"Integrale approchee (n=1000) : {int_approx_1000}")
    assert int_approx_1000 > int_approx, "La suite d'approximations doit etre croissante"
```
