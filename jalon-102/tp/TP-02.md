## TP-02 : Fuite à l'infini dans $\ell^2$
Implémentation de l'exemple du cours : la base canonique $(e_n)$ dans un espace de grande dimension (simulant $\ell^2$) pour observer la convergence faible vers 0 sans convergence forte.

```python
import numpy as np

def tp02_fuite_infini():
    dim = 1000 # Approximation de la dimension infinie

    # Définition de e_n
    def get_en(n):
        v = np.zeros(dim)
        v[n] = 1.0
        return v

    # Vecteur test y dans l^2
    y = np.array([1.0 / (k + 1) for k in range(dim)])
    assert np.sum(y**2) < np.inf, "Le vecteur test n'est pas dans l^2"

    # Test de convergence faible
    for n in [10, 100, 500, 900]:
        e_n = get_en(n)
        # Forme linéaire L_y(x) = <y, x>
        L_en = np.dot(y, e_n)
        # La valeur doit tendre vers 0
        if n == 900:
            assert abs(L_en) < 1e-2, "La convergence faible vers 0 a échoué"

    # Test de non-convergence forte
    e_10 = get_en(10)
    e_20 = get_en(20)
    dist = np.linalg.norm(e_10 - e_20)
    assert abs(dist - np.sqrt(2)) < 1e-5, "La distance entre e_n et e_m n'est pas sqrt(2)"

    print("TP-02 Validé: La suite (e_n) converge faiblement vers 0, mais pas fortement.")

if __name__ == "__main__":
    tp02_fuite_infini()
```
