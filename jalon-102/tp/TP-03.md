## TP-03 : Semi-continuité inférieure de la norme
Illustration numérique de la semi-continuité inférieure faible de la norme, $\|x\| \le \liminf \|x_n\|$, avec un exemple de flot.

```python
import numpy as np

def tp03_sci_norme():
    dim = 200

    # x_n = e_0 + e_n
    # converge faiblement vers e_0
    e_0 = np.zeros(dim)
    e_0[0] = 1.0

    def get_xn(n):
        v = np.copy(e_0)
        if n < dim and n > 0:
            v[n] = 1.0
        return v

    # Vecteur test pour la convergence faible
    y = np.array([1.0/(k+1)**2 for k in range(dim)])

    L_target = np.dot(y, e_0)

    # La suite converge faiblement vers e_0
    L_xn = np.dot(y, get_xn(150))
    assert abs(L_xn - L_target) < 1e-2, "La convergence faible a échoué"

    # Norme de la limite
    norm_limit = np.linalg.norm(e_0)
    assert norm_limit == 1.0

    # Norme des termes de la suite
    norm_xn = np.linalg.norm(get_xn(150))
    assert abs(norm_xn - np.sqrt(2)) < 1e-5

    # On vérifie la SCI: ||x|| <= liminf ||x_n||
    # 1.0 <= sqrt(2)
    assert norm_limit <= norm_xn, "La norme n'est pas semi-continue inférieurement"

    print("TP-03 Validé: La limite faible a une norme strictement inférieure à la limite des normes.")

if __name__ == "__main__":
    tp03_sci_norme()
```
