# TP 5 : Algorithme de Newton en 2D pour l'optimisation locale

L'algorithme de Newton utilise l'inverse de la matrice hessienne pour s'affranchir du conditionnement et atteindre l'optimum en une seule étape pour une fonction quadratique.

```python
# tp5_newton_algorithm.py

def invert_2d(matrix):
    """Inverse une matrice 2x2. Lève une erreur si singulière."""
    det = matrix[0][0]*matrix[1][1] - matrix[0][1]*matrix[1][0]
    assert det != 0, "Matrice singulière, non inversible"

    return [
        [matrix[1][1] / det, -matrix[0][1] / det],
        [-matrix[1][0] / det, matrix[0][0] / det]
    ]

def multiply_matrix_vector(matrix, vec):
    return [
        matrix[0][0] * vec[0] + matrix[0][1] * vec[1],
        matrix[1][0] * vec[0] + matrix[1][1] * vec[1]
    ]

def newton_step(x, y, grad_f, H_f):
    grad = grad_f(x, y)
    H = H_f(x, y)
    H_inv = invert_2d(H)

    # Delta = H_inv * grad
    delta = multiply_matrix_vector(H_inv, grad)

    return x - delta[0], y - delta[1]

if __name__ == "__main__":
    # Fonction f(x, y) = 3(x-2)^2 + 5(y+4)^2
    def grad_f(x, y):
        return [6*(x-2), 10*(y+4)]

    def H_f(x, y):
        return [[6.0, 0.0], [0.0, 10.0]]

    # Test
    start_x, start_y = -10.0, 20.0
    opt_x, opt_y = newton_step(start_x, start_y, grad_f, H_f)

    # L'algorithme de Newton converge en UNE seule itération pour une paraboloïde !
    assert abs(opt_x - 2.0) < 1e-7, "La méthode de Newton n'a pas convergé en un pas sur x"
    assert abs(opt_y - (-4.0)) < 1e-7, "La méthode de Newton n'a pas convergé en un pas sur y"

    print("TP 5 validé : Convergence instantanée de la méthode de Newton vérifiée.")
```
