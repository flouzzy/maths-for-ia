# TP 4 : Mise en évidence du conditionnement géométrique

On simule l'impact d'une matrice hessienne mal conditionnée sur la descente de gradient classique.

```python
# tp4_conditionnement.py

def descend(lr, H_diag):
    # H_diag = [H_xx, H_yy]. On simule la trajectoire pour f = 0.5 * (H_xx*x^2 + H_yy*y^2)
    # L'optimum est en (0,0)
    x, y = 1.0, 1.0
    history = []
    for _ in range(20):
        # Le gradient analytique est Hx
        gx, gy = H_diag[0] * x, H_diag[1] * y
        x = x - lr * gx
        y = y - lr * gy
        history.append((x, y))
    return x, y

if __name__ == "__main__":
    # Un ravin très étroit : conditionnement élevé
    H_bad = [1.0, 100.0]

    # Si le learning rate est réglé pour le petit axe (lr=0.5), la fonction explose sur l'axe y
    x_bad, y_bad = descend(lr=0.5, H_diag=H_bad)

    assert abs(y_bad) > 100, "Le système devrait diverger sur l'axe y à cause du conditionnement"
    assert abs(x_bad) < 1, "Le système devrait converger sur l'axe x"

    # Si on abaisse le learning rate pour éviter la divergence sur y
    x_safe, y_safe = descend(lr=0.015, H_diag=H_bad)

    assert abs(y_safe) < 1, "Le système devrait converger prudemment sur y"
    assert abs(x_safe - 1.0) < 0.3, "La convergence sur x est par conséquent très lente (conditionnement)"

    print("TP 4 validé : Phénomènes de conditionnement et instabilité reproduits.")
```
