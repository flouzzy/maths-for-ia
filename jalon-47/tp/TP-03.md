# TP 3 : Recherche de points critiques par descente de gradient et analyse

On cherche le minimum d'une fonction et on vérifie a posteriori qu'il s'agit bien d'un minimum en examinant la Hessienne approximée au point d'arrêt.

```python
# tp3_gradient_descent.py

def gradient_descent(grad_f, start_x, start_y, lr=0.01, epochs=1000):
    x, y = start_x, start_y
    for _ in range(epochs):
        gx, gy = grad_f(x, y)
        x = x - lr * gx
        y = y - lr * gy
    return x, y

def determinant_2d(matrix):
    return matrix[0][0]*matrix[1][1] - matrix[0][1]*matrix[1][0]

def trace_2d(matrix):
    return matrix[0][0] + matrix[1][1]

if __name__ == "__main__":
    def grad_f(x, y):
        return [2*x - 2, 8*y - 8]

    x_opt, y_opt = gradient_descent(grad_f, 0.0, 0.0)

    assert abs(x_opt - 1.0) < 1e-4, "Optimum x non trouvé"
    assert abs(y_opt - 1.0) < 1e-4, "Optimum y non trouvé"

    # La matrice hessienne analytique pour cette fonction quadratique f(x,y)=x^2-2x+4y^2-8y est H=[[2,0],[0,8]]
    H_opt = [[2, 0], [0, 8]]
    det = determinant_2d(H_opt)
    tr = trace_2d(H_opt)

    assert det > 0 and tr > 0, "Les critères de minimum local strict ne sont pas respectés"

    print("TP 3 validé : Descente de gradient convergente et analyse spectrale validée.")
```
