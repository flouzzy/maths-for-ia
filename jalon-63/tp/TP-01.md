# Modélisation ensembliste élémentaire

## Code Source Python

```python
def is_algebra(X, subsets):
    """
    Verifies if a collection of subsets forms an algebra over X.
    - X must be in subsets
    - If A is in subsets, X \ A must be in subsets
    - If A and B are in subsets, A union B must be in subsets
    """
    # Condition 1: X is in the algebra
    if not any(set(s) == set(X) for s in subsets):
        return False

    for A in subsets:
        # Condition 2: Complement is in algebra
        comp_A = set(X) - set(A)
        if not any(set(s) == comp_A for s in subsets):
            return False

        # Condition 3: Union is in algebra
        for B in subsets:
            union_AB = set(A) | set(B)
            if not any(set(s) == union_AB for s in subsets):
                return False

    return True

# Validation
X = [1, 2, 3]
trivial_algebra = [[], [1, 2, 3]]
power_set = [[], [1], [2], [3], [1, 2], [1, 3], [2, 3], [1, 2, 3]]
invalid_algebra = [[], [1, 2, 3], [1]]

assert is_algebra(X, trivial_algebra) == True
assert is_algebra(X, power_set) == True
assert is_algebra(X, invalid_algebra) == False
print("TP-01 Validated")
```
