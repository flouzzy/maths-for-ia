# Mesure de comptage et monotonie

## Code Source Python

```python
def counting_measure(subset):
    """
    Calculates the counting measure for a finite subset.
    """
    return len(subset)

def validate_monotonicity(A, B):
    """
    If A is a subset of B, then measure(A) <= measure(B).
    """
    is_subset = set(A).issubset(set(B))
    if not is_subset:
        raise ValueError("A is not a subset of B")

    return counting_measure(A) <= counting_measure(B)

def validate_boole_inequality(subsets):
    """
    Validates the sub-additivity inequality: measure(union(A_i)) <= sum(measure(A_i))
    """
    union_all = set()
    sum_measures = 0

    for subset in subsets:
        union_all = union_all | set(subset)
        sum_measures += counting_measure(subset)

    return counting_measure(list(union_all)) <= sum_measures

# Validation
A = [1, 2]
B = [1, 2, 3, 4]
subsets = [[1, 2, 3], [3, 4, 5], [5, 6]]

assert validate_monotonicity(A, B) == True
assert validate_boole_inequality(subsets) == True
print("TP-03 Validated")
```
