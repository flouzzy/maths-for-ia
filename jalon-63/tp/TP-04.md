# Approximation de l'ensemble de Cantor

## Code Source Python

```python
def cantor_intervals(n_iterations):
    """
    Generates the intervals of the Cantor set at iteration n.
    Returns a list of tuples (start, end).
    """
    intervals = [(0.0, 1.0)]
    for _ in range(n_iterations):
        next_intervals = []
        for start, end in intervals:
            length = end - start
            third = length / 3.0
            next_intervals.append((start, start + third))
            next_intervals.append((end - third, end))
        intervals = next_intervals
    return intervals

def lebesgue_measure_intervals(intervals):
    """
    Calculates the standard length measure of a union of disjoint intervals.
    """
    total_length = 0.0
    for start, end in intervals:
        total_length += (end - start)
    return total_length

# Validation
# Iteration 0: measure is 1
assert abs(lebesgue_measure_intervals(cantor_intervals(0)) - 1.0) < 1e-9
# Iteration 1: measure is 2/3
assert abs(lebesgue_measure_intervals(cantor_intervals(1)) - 2.0/3.0) < 1e-9
# Iteration 5: measure is (2/3)^5 = 32/243 ~ 0.131687
assert abs(lebesgue_measure_intervals(cantor_intervals(5)) - (2.0/3.0)**5) < 1e-9
print("TP-04 Validated")
```
