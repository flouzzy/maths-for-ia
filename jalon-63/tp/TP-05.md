# Théorème de continuité croissante (Simulation finie)

## Code Source Python

```python
def measure_custom(subset):
    """
    A custom weighted measure on integers.
    Weight of an integer n is 1/(2^n).
    """
    total = 0.0
    for n in subset:
        if n >= 0:
            total += 1.0 / (2.0**n)
    return total

def simulate_increasing_continuity(max_N):
    """
    Simulates continuity from below for the sequence of sets A_N = {0, 1, ..., N}.
    The limit set A is the set of all non-negative integers.
    The theoretical measure of A is sum(1/2^n) from 0 to infinity, which is 2.
    """
    A_N = []
    measure_N = 0.0

    for i in range(max_N + 1):
        A_N.append(i)
        measure_N = measure_custom(A_N)

    return measure_N

# Validation
# The sum from n=0 to 10 of 1/2^n is 2 - 1/2^10 = 1.9990234375
result = simulate_increasing_continuity(10)
expected = 2.0 - (1.0 / (2.0**10))
assert abs(result - expected) < 1e-9

# As N grows large, it approaches 2.0
result_large = simulate_increasing_continuity(50)
assert abs(result_large - 2.0) < 1e-10

print("TP-05 Validated")
```
