# Calculateur de mesure de Dirac

## Code Source Python

```python
def dirac_measure(point_a, subset):
    """
    Calculates the Dirac measure of a subset relative to point_a.
    Returns 1 if point_a is in the subset, 0 otherwise.
    """
    if point_a in subset:
        return 1
    return 0

def validate_sigma_additivity_dirac(point_a, disjoint_subsets):
    """
    Validates sigma-additivity for Dirac measure on a finite family of disjoint subsets.
    """
    union_all = set()
    sum_measures = 0

    for subset in disjoint_subsets:
        union_all = union_all | set(subset)
        sum_measures += dirac_measure(point_a, subset)

    measure_of_union = dirac_measure(point_a, list(union_all))
    return measure_of_union == sum_measures

# Validation
point = 42
subsets = [[1, 2, 3], [4, 5, 42], [100, 200]]

assert dirac_measure(42, [1, 2, 3]) == 0
assert dirac_measure(42, [4, 5, 42]) == 1
assert validate_sigma_additivity_dirac(point, subsets) == True
print("TP-02 Validated")
```
