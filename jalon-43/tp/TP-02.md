# TP 2 : Calcul numérique de l'exponentielle de matrice et résolution de systèmes

Ce travail pratique vise à implémenter "from scratch" l'exponentielle de matrice et résoudre un système différentiel de dimension 2, uniquement en Python pur sans numpy.

```python
# Implémentation en Python pur pour l'exponentielle de matrice et le solveur de système
import math

def mat_mult(A, B):
    # Multiplication de deux matrices 2x2
    return [
        [A[0][0]*B[0][0] + A[0][1]*B[1][0], A[0][0]*B[0][1] + A[0][1]*B[1][1]],
        [A[1][0]*B[0][0] + A[1][1]*B[1][0], A[1][0]*B[0][1] + A[1][1]*B[1][1]]
    ]

def mat_add(A, B):
    # Addition de deux matrices 2x2
    return [
        [A[0][0]+B[0][0], A[0][1]+B[0][1]],
        [A[1][0]+B[1][0], A[1][1]+B[1][1]]
    ]

def mat_scale(scalar, A):
    # Multiplication scalaire
    return [
        [scalar * A[0][0], scalar * A[0][1]],
        [scalar * A[1][0], scalar * A[1][1]]
    ]

def exp_matrix_2x2(A, num_terms=30):
    # Calcul de l'exponentielle par troncature de la serie de Taylor
    I = [[1.0, 0.0], [0.0, 1.0]]
    result = I
    current_term = I
    for k in range(1, num_terms):
        # A_pow_k / k! = (A_pow_k_minus_1 / (k-1)!) * (A / k)
        current_term = mat_mult(current_term, mat_scale(1.0/k, A))
        result = mat_add(result, current_term)
    return result

def solve_system(A, Y0, t):
    # Solution du probleme Y'(t) = A Y(t) via l'exponentielle
    At = mat_scale(t, A)
    exp_At = exp_matrix_2x2(At)
    # Calcul exp(At) * Y0
    return [
        exp_At[0][0]*Y0[0] + exp_At[0][1]*Y0[1],
        exp_At[1][0]*Y0[0] + exp_At[1][1]*Y0[1]
    ]

if __name__ == '__main__':
    # Test avec la matrice A du TP
    A = [[2.0, 1.0], [0.0, 2.0]]
    Y0 = [1.0, 1.0]
    t = 1.0

    Y_t = solve_system(A, Y0, t)

    # Valeurs analytiques
    exact_y1 = math.exp(2.0 * t) + t * math.exp(2.0 * t)
    exact_y2 = math.exp(2.0 * t)

    print(f"Numerical result at t={t}: {Y_t}")
    print(f"Exact     result at t={t}: [{exact_y1}, {exact_y2}]")

    # Validation
    assert abs(Y_t[0] - exact_y1) < 1e-4, "Erreur sur y1"
    assert abs(Y_t[1] - exact_y2) < 1e-4, "Erreur sur y2"
    print("TP 2 : Tests unitaires de validation mathematique passes avec succes.")
```
