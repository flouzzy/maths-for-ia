# TP 4 : Fonctionnelle sous-linéaire sur des polynômes

Manipulons une fonctionnelle sous-linéaire sur un espace de polynômes de degré 2.

```python
class PolynomeDeg2:
    def __init__(self, c0, c1, c2):
        self.c0 = c0
        self.c1 = c1
        self.c2 = c2

    def evaluer(self, x):
        return self.c0 + self.c1 * x + self.c2 * x**2

def p_norme_sup(poly):
    # Approximation grossiere de la norme infinie sur [0, 1] par echantillonnage
    n_samples = 100
    max_val = 0
    for i in range(n_samples + 1):
        x = i / n_samples
        val = abs(poly.evaluer(x))
        if val > max_val:
            max_val = val
    return max_val

def tp_poly_sous_lineaire():
    poly1 = PolynomeDeg2(1.0, -2.0, 1.0) # (1-x)^2
    poly2 = PolynomeDeg2(0.0, 1.0, 0.0)  # x

    poly_sum = PolynomeDeg2(1.0, -1.0, 1.0) # somme

    p1 = p_norme_sup(poly1)
    p2 = p_norme_sup(poly2)
    p_sum = p_norme_sup(poly_sum)

    assert p_sum <= p1 + p2 + 1e-6, "Sous-additivite violee sur les polynomes."

    lmbda = 3.0
    poly1_scaled = PolynomeDeg2(lmbda*1.0, lmbda*-2.0, lmbda*1.0)
    p1_scaled = p_norme_sup(poly1_scaled)

    assert abs(p1_scaled - lmbda * p1) < 1e-6, "Homogeneite violee sur les polynomes."

    print("TP4: Fonctionnelle sous-lineaire polynomiale validee.")

if __name__ == '__main__':
    tp_poly_sous_lineaire()
```
