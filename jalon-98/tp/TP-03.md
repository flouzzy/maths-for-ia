# TP 3 : Séparation point-convexe (Hyperplan d'appui)

Ce script calcule numériquement l'hyperplan qui sépare le cercle unité d'un point exterieur.

```python
import math

def tp_hyperplan():
    # Point exterieur
    x0, y0 = 3.0, 4.0

    # Norme euclidienne (distance à l'origine)
    norme = math.sqrt(x0**2 + y0**2)

    # Forme lineaire f(x,y) = ax + by telle que f(x0,y0) = norme et ||f|| = 1
    # On sait que le gradient de la norme au point (x0,y0) donne la direction de f
    a = x0 / norme
    b = y0 / norme

    def f(x, y):
        return a * x + b * y

    # Verification que ||f|| = 1
    assert abs(math.sqrt(a**2 + b**2) - 1.0) < 1e-9, "La norme de f n'est pas 1"

    # Verification que f(x0, y0) = ||(x0, y0)||
    assert abs(f(x0, y0) - norme) < 1e-9, "L'evaluation n'est pas correcte"

    # L'hyperplan est H = { (x,y) : f(x,y) = 1 }
    # Pour tout point de la boule B = { (x,y) : x**2 + y**2 <= 1 }, on doit avoir f(x,y) <= 1

    for angle_deg in range(360):
        angle = math.radians(angle_deg)
        x_c = math.cos(angle)
        y_c = math.sin(angle)
        assert f(x_c, y_c) <= 1.0 + 1e-9, "Le point du cercle depasse l'hyperplan !"

    print("TP3: Hyperplan d'appui construit et valide.")

if __name__ == '__main__':
    tp_hyperplan()
```
