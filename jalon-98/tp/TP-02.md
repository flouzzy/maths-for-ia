# TP 2 : Construction d'une jauge de Minkowski

Dans ce TP, on construit numériquement la jauge d'un convexe C (le carre unite centré en l'origine).

```python
def jauge_minkowski(x, y):
    # C = { (x, y) : max(|x|, |y|) <= 1 }
    # p(x,y) = inf { L > 0 | (x/L, y/L) in C }
    # (x/L, y/L) in C <=> max(|x/L|, |y/L|) <= 1 <=> L >= max(|x|, |y|)
    # Donc inf L = max(|x|, |y|)
    return max(abs(x), abs(y))

def tp_jauge():
    # Validation de l'homogeneite
    x, y = 3.0, -4.0
    lmbda = 2.5

    val1 = jauge_minkowski(lmbda * x, lmbda * y)
    val2 = lmbda * jauge_minkowski(x, y)

    assert abs(val1 - val2) < 1e-9, "L'homogeneite n'est pas respectee."

    # Validation de la sous-additivite
    x2, y2 = -1.0, 5.0

    val_sum = jauge_minkowski(x + x2, y + y2)
    sum_vals = jauge_minkowski(x, y) + jauge_minkowski(x2, y2)

    assert val_sum <= sum_vals + 1e-9, "La sous-additivite n'est pas respectee."
    print("TP2: Jauge valide.")

if __name__ == '__main__':
    tp_jauge()
```
