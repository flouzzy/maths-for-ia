# TP 4 : Approximation canonique d'une fonction positive (Théorème 4)

Ce TP propose de manipuler programmatiquement des fonctions mesurables, des indicatrices et l'approximation étagée sans utiliser de frameworks avancés.

```python
# Implémentation du théorème d'approximation s_n -> f
class CanonicalApproximation:
    def __init__(self, f, n):
        self.f = f
        self.n = n

    def __call__(self, x):
        val = self.f(x)
        if val >= self.n:
            return self.n

        # Trouver k tel que k/2^n <= f(x) < (k+1)/2^n
        for k in range(0, self.n * (2**self.n)):
            if (k / (2**self.n)) <= val < ((k + 1) / (2**self.n)):
                return k / (2**self.n)
        return 0

def test_approximation():
    f = lambda x: x**2
    x0 = 1.5 # f(1.5) = 2.25

    s1 = CanonicalApproximation(f, 1) # k/2, val >= 1
    s2 = CanonicalApproximation(f, 2) # k/4, val >= 2
    s3 = CanonicalApproximation(f, 3) # k/8, val >= 3

    # 2.25 >= 1 donc s1(1.5) = 1
    assert s1(x0) == 1
    # 2.25 >= 2 donc s2(1.5) = 2
    assert s2(x0) == 2
    # 2.25 n'est pas >= 3. 2.25 * 8 = 18. Donc k=18
    # 18/8 = 2.25. k/2^n <= val < (k+1)/2^n -> 18/8 <= 2.25 < 19/8 (2.375)
    assert s3(x0) == 18 / 8
    print("TP 4: Approximation canonique OK")

if __name__ == "__main__":
    test_approximation()
```
