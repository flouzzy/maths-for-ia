# TP 4 : Validation Empirique des Bornes de Généralisation de Bousquet-Elisseeff

## Objectif
Ce TP a pour but de vérifier numériquement la validité de la borne de généralisation de Bousquet-Elisseeff sur un problème de régression synthétique. Nous allons comparer le risque réel d'un estimateur stable avec son risque empirique augmenté de la pénalité de stabilité.

---

## Code de Simulation et de Validation

```python
import numpy as np

def train_ridge(X, y, lmbda):
    n, d = X.shape
    A = np.dot(X.T, X) + n * lmbda * np.eye(d)
    b = np.dot(X.T, y)
    return np.linalg.solve(A, b)

# Fixer la graine
np.random.seed(42)

# Paramètres du problème
n = 150        # Taille de l'échantillon d'apprentissage
d = 4          # Dimension
lmbda = 0.5    # Forte régularisation pour assurer la stabilité
R = 1.0        # ||x||_2 <= R
Y_max = 2.0    # |y| <= Y_max

# Distribution théorique D : x uniforme dans la boule, y = <w*, x> + bruit
w_star = np.random.normal(size=d)
w_star /= np.linalg.norm(w_star)

def sample_data(num_samples):
    X = np.random.normal(size=(num_samples, d))
    X /= np.linalg.norm(X, axis=1, keepdims=True)
    X *= np.random.uniform(0, R, size=(num_samples, 1))
    
    # y = <w*, x> + bruit
    y = np.dot(X, w_star) + np.random.normal(0, 0.1, size=num_samples)
    y = np.clip(y, -Y_max, Y_max)
    return X, y

# 1. Tirer l'échantillon d'apprentissage S et entraîner le modèle
X_train, y_train = sample_data(n)
w_S = train_ridge(X_train, y_train, lmbda)

# 2. Risque empirique ( hinge loss modifiée ou quadratique )
# Utilisons la perte quadratique bornée par M = (R * ||w|| + Y_max)^2
w_norm = np.linalg.norm(w_S)
M = (R * w_norm + Y_max) ** 2
empirical_loss = np.mean((np.dot(X_train, w_S) - y_train) ** 2)

# 3. Risque réel estimé sur un très grand échantillon de test
X_test, y_test = sample_data(100000)
real_loss = np.mean((np.dot(X_test, w_S) - y_test) ** 2)

# 4. Calculer la constante de stabilité uniforme théorique
# beta = 4 * R^2 * Y_max^2 / (lambda * n)
beta = (4 * (R**2) * (Y_max**2)) / (lmbda * n)

# 5. Évaluer la borne de généralisation de Bousquet-Elisseeff pour delta = 0.05
delta = 0.05
generalization_gap = real_loss - empirical_loss
theoretical_bound = 2 * beta + (4 * n * beta + M) * np.sqrt(np.log(1/delta) / (2 * n))

print("--- EXHAUSTIVITÉ DES RISQUES ---")
print(f"Risque Empirique R_n(w_S)      : {empirical_loss:.6f}")
print(f"Risque Réel R(w_S)              : {real_loss:.6f}")
print(f"Écart de généralisation réel    : {generalization_gap:.6f}")
print(f"Borne théorique de stabilité    : {theoretical_bound:.6f}")

# Assertion de validation mathématique
assert generalization_gap <= theoretical_bound, "Erreur: L'écart réel dépasse la borne théorique !"
print("\n=> L'inégalité de généralisation de Bousquet-Elisseeff est respectée.")
```
