# TP 1 : Simulation de la Stabilité Algorithmique de la Régression Ridge

## Objectif
Ce TP vise à implémenter la régression Ridge à partir de zéro (from scratch) en Python avec NumPy et à vérifier empiriquement la borne de stabilité uniforme théorique $\beta \le \frac{4 R^2 Y_{\max}^2}{\lambda n}$.

---

## Code de Simulation et de Validation

Copiez et exécutez le script suivant dans votre environnement Python :

```python
import numpy as np

def train_ridge(X, y, lmbda):
    """
    Entraîne un modèle de régression Ridge à partir de zéro.
    w = (X^T X + n * lmbda * I)^-1 X^T y
    """
    n, d = X.shape
    A = np.dot(X.T, X) + n * lmbda * np.eye(d)
    b = np.dot(X.T, y)
    w = np.linalg.solve(A, b)
    return w

def loss_ridge(w, x, y):
    """
    Calcule la perte quadratique pour un exemple unique.
    """
    return (np.dot(w, x) - y) ** 2

# Fixer la graine aléatoire pour la reproductibilité
np.random.seed(42)

# Paramètres
n = 100        # Taille de l'échantillon
d = 5          # Dimension des caractéristiques
lmbda = 0.1    # Paramètre de régularisation lambda
R = 1.0        # Borne sur les entrées ||x||_2 <= R
Y_max = 2.0    # Borne sur les cibles |y| <= Y_max

# 1. Générer l'échantillon S
X = np.random.normal(size=(n, d))
# Normaliser chaque point pour s'assurer que ||x||_2 <= R
X_norms = np.linalg.norm(X, axis=1, keepdims=True)
X = R * X / np.maximum(X_norms, 1.0)
y = np.random.uniform(-Y_max, Y_max, size=n)

# 2. Entraîner le modèle original sur S
w_S = train_ridge(X, y, lmbda)

# 3. Perturber le premier point d'entraînement (i = 0)
x_prime_0 = np.random.normal(size=d)
x_prime_0 = R * x_prime_0 / np.maximum(np.linalg.norm(x_prime_0), 1.0)
y_prime_0 = np.random.uniform(-Y_max, Y_max)

X_mod = X.copy()
X_mod[0] = x_prime_0
y_mod = y.copy()
y_mod[0] = y_prime_0

# 4. Entraîner le modèle perturbé sur S_mod
w_S_mod = train_ridge(X_mod, y_mod, lmbda)

# 5. Calculer la distance réelle entre w_S et w_S_mod
w_diff_norm = np.linalg.norm(w_S - w_S_mod)
w_diff_bound = (2 * R * Y_max) / (lmbda * n)

print("--- VALIDATION DES POIDS RIDGE ---")
print(f"Distance réelle ||w_S - w_S_mod||_2 : {w_diff_norm:.6f}")
print(f"Borne théorique sur la distance     : {w_diff_bound:.6f}")

# Assertion de validation mathématique
assert w_diff_norm <= w_diff_bound, "Erreur: La distance réelle dépasse la borne théorique !"
print("=> La borne sur les poids est validée par l'assertion.")

# 6. Vérifier la stabilité uniforme sur la perte (sur un grand ensemble de points de test)
num_test = 10000
X_test = np.random.normal(size=(num_test, d))
X_test_norms = np.linalg.norm(X_test, axis=1, keepdims=True)
X_test = R * X_test / np.maximum(X_test_norms, 1.0)
y_test = np.random.uniform(-Y_max, Y_max, size=num_test)

losses_S = np.array([loss_ridge(w_S, X_test[j], y_test[j]) for j in range(num_test)])
losses_S_mod = np.array([loss_ridge(w_S_mod, X_test[j], y_test[j]) for j in range(num_test)])

max_loss_diff = np.max(np.abs(losses_S - losses_S_mod))
theoretical_beta = (4 * (R**2) * (Y_max**2)) / (lmbda * n) * (1 + (R**2)/lmbda) # borne lâche combinée

print("\n--- VALIDATION DE LA STABILITÉ DE LA PERTE ---")
print(f"Différence maximale de perte observée : {max_loss_diff:.6f}")
print(f"Borne théorique de stabilité uniforme : {theoretical_beta:.6f}")

# Assertion de validation sur la perte
assert max_loss_diff <= theoretical_beta, "Erreur: La différence maximale de perte dépasse la borne théorique !"
print("=> La stabilité uniforme de la perte est validée par l'assertion.")
```
