# TP 1 : Calcul d'énergie et identité de Parseval (Créneau)

**Objectif :** Implémenter numériquement le calcul des coefficients de Fourier et vérifier formellement l'identité de Parseval.

```python
import math

def f_signal(t):
    t_mod = t % (2 * math.pi)
    return 1.0 if t_mod < math.pi else -1.0

def compute_fourier_coefficients(f, n_max, num_points=2000):
    a, b = [], []
    a0 = 0.0
    dt = (2 * math.pi) / num_points

    for i in range(num_points):
        a0 += f(i * dt) * dt
    a0 /= math.pi

    for n in range(1, n_max + 1):
        an_val, bn_val = 0.0, 0.0
        for i in range(num_points):
            t = i * dt
            an_val += f(t) * math.cos(n * t) * dt
            bn_val += f(t) * math.sin(n * t) * dt
        a.append(an_val / math.pi)
        b.append(bn_val / math.pi)

    return a0, a, b

def calculate_energies(f, a0, a, b, num_points=2000):
    dt = (2 * math.pi) / num_points
    signal_energy = sum((f(i * dt) ** 2) * dt for i in range(num_points)) / (2 * math.pi)

    parseval_energy = (a0 ** 2) / 4 + sum(0.5 * (an**2 + bn**2) for an, bn in zip(a, b))
    return signal_energy, parseval_energy

if __name__ == "__main__":
    n_max = 50
    a0, a, b = compute_fourier_coefficients(f_signal, n_max)
    E_sig, E_pars = calculate_energies(f_signal, a0, a, b)

    assert abs(E_sig - E_pars) < 0.1, "L'identité de Parseval n'est pas vérifiée."
    print("TP1: Identité de Parseval validée pour le créneau.")
```
