# TP 2 : Filtrage passe-bas et conservation d'énergie

**Objectif :** Mesurer le pourcentage d'énergie perdue lorsqu'on tronque une série de Fourier, illustrant la convergence $L^2$.

```python
import math

def f_signal(t):
    # Signal triangulaire
    t = t % (2*math.pi)
    return t if t < math.pi else 2*math.pi - t

def exact_energy():
    # Intégrale exacte de t^2 sur [0, pi] normalisée
    return (math.pi**2) / 3

def compute_bn(n):
    # a_n car la fonction est paire, mais avec un décalage
    # L'énergie des harmoniques impairs vaut 4/(pi * n^2)
    if n % 2 == 0: return 0
    return -4 / (math.pi * n**2)

if __name__ == "__main__":
    E_exact = exact_energy()
    a0 = math.pi # Moyenne exacte
    E_pars_0 = (a0**2) / 4

    total_energy = E_pars_0
    for n in range(1, 10):
        an = compute_bn(n)
        total_energy += 0.5 * (an**2)
        ratio = total_energy / E_exact
        print(f"Harmoniques jusqu'a {n}: {ratio:.4f} de l'energie totale.")

    assert ratio > 0.99, "La convergence L2 devrait depasser 99% avec 10 harmoniques."
    print("TP2: Analyse du filtrage validee.")
```
