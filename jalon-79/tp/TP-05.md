# TP 5 : Parseval et Bruit Blanc

**Objectif :** Vérifier que l'énergie d'un bruit blanc est répartie uniformément sur toutes ses fréquences.

```python
import random
import math

def generate_white_noise(num_points):
    return [random.uniform(-1, 1) for _ in range(num_points)]

def compute_energy_time(signal):
    return sum(x**2 for x in signal) / len(signal)

def compute_fourier_amplitudes(signal):
    N = len(signal)
    amplitudes_sq = []

    # On simule une DFT basique (sans utiliser numpy/fft)
    for k in range(N // 2):
        real, imag = 0.0, 0.0
        for n in range(N):
            angle = 2 * math.pi * k * n / N
            real += signal[n] * math.cos(angle)
            imag -= signal[n] * math.sin(angle)

        # Parseval normalise
        if k == 0:
            amplitudes_sq.append((real/N)**2)
        else:
            amplitudes_sq.append(2 * ((real/N)**2 + (imag/N)**2))

    return amplitudes_sq

if __name__ == "__main__":
    N = 256
    noise = generate_white_noise(N)

    E_time = compute_energy_time(noise)
    amps = compute_fourier_amplitudes(noise)
    E_freq = sum(amps)

    print(f"Energie temporelle (Variance): {E_time:.4f}")
    print(f"Energie frequentielle: {E_freq:.4f}")

    assert abs(E_time - E_freq) < 0.05, "Le theoreme de Parseval (cas discret) n'est pas verifie."
    print("TP5: Repartition de l'energie du bruit validee par Parseval.")
```
