## TP 5 : Continuité du produit de convolution via Plancherel

**Objectif :** Constater numériquement que le produit de convolution de deux signaux L2 produit une fonction continue, sans hautes fréquences dominantes.

```python
import math

def test_convolution_continuity():
    a, b = -5.0, 5.0
    N = 800
    dt = (b - a) / N

    def sig1(t):
        return 1.0 if abs(t) < 1.5 else 0.0
    def sig2(t):
        return 1.0 if abs(t) < 1.0 else 0.0

    def convolve(x):
        res = 0.0
        for i in range(N):
            tau = a + i * dt
            # g(x - tau)
            shift = x - tau
            if abs(shift) < 1.0:
                res += sig1(tau) * 1.0 * dt
        return res

    # Evaluate convolution at two close points
    x1 = 0.5
    x2 = 0.501

    val1 = convolve(x1)
    val2 = convolve(x2)

    diff = abs(val1 - val2)
    print("Conv(0.5):", val1)
    print("Conv(0.501):", val2)
    print("Difference:", diff)

    assert diff < 0.01, "Convolution does not appear continuous"
    print("Continuity of L2 convolution verified.")

if __name__ == "__main__":
    test_convolution_continuity()
```
