# TP 5 : Calcul de la norme spectrale d'un noyau de convolution

## Description
Estimation de la norme L2 (borne de Lipschitz) d'un opérateur de convolution 1D en trouvant le maximum du module de sa DFT.

## Code Python
```python
def compute_dft(signal):
    import cmath
    import math
    N = len(signal)
    dft = []
    for k in range(N):
        val = 0.0 + 0.0j
        for n in range(N):
            angle = -2j * cmath.pi * k * n / N
            val += signal[n] * cmath.exp(angle)
        dft.append(val)
    return dft

def pad_signal(kernel, target_length):
    padded = list(kernel)
    while len(padded) < target_length:
        padded.append(0.0)
    return padded

# Noyau de convolution simple
kernel = [0.25, 0.5, 0.25]
N = 64
padded_kernel = pad_signal(kernel, N)

dft_kernel = compute_dft(padded_kernel)

# Trouver l'amplitude maximale de la DFT (Norme spectrale)
spectral_norm = 0.0
for val in dft_kernel:
    amplitude = abs(val)
    if amplitude > spectral_norm:
        spectral_norm = amplitude

# Pour ce noyau specifique (filtre moyenneur), la somme des poids est 1.0,
# qui est atteint a la frequence 0.
assert abs(spectral_norm - 1.0) < 1e-10, "La norme spectrale calculee est incorrecte"

print(f"TP5: Norme spectrale evaluee avec succes: {spectral_norm}")

```
