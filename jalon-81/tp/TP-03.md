# TP 3 : Produit de convolution discret et théorème spectral

## Description
Implémentation d'une convolution discrète 1D et vérification que la transformée de Fourier de la convolution est le produit des transformées.

## Code Python
```python
def compute_dft(signal):
    import cmath
    import math
    N = len(signal)
    dft = []
    for k in range(N):
        val = 0.0 + 0.0j
        for n in range(N):
            angle = -2j * cmath.pi * k * n / N
            val += signal[n] * cmath.exp(angle)
        dft.append(val)
    return dft

def circular_convolution(x, h):
    N = len(x)
    y = [0] * N
    for n in range(N):
        for k in range(N):
            y[n] += x[k] * h[(n - k) % N]
    return y

signal_1 = [1.0, 2.0, 0.0, -1.0]
signal_2 = [0.5, 0.0, -0.5, 1.0]

conv_result = circular_convolution(signal_1, signal_2)
dft_conv = compute_dft(conv_result)

dft_1 = compute_dft(signal_1)
dft_2 = compute_dft(signal_2)

# Multiply DFTs pointwise
dft_product = [dft_1[i] * dft_2[i] for i in range(len(signal_1))]

for i in range(len(dft_conv)):
    assert abs(dft_conv[i] - dft_product[i]) < 1e-10, "La convolution spectrale ne correspond pas"

print("TP3: Theoreme de convolution verifie avec succes.")

```
