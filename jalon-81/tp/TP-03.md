# TP 3 : Prolongement d'opérateur de L1 vers L2 via densité


Nous simulons ici le passage à la limite de l'espace L1 vers L2. Un signal sin(x)/x (sinc) n'est pas dans L1 mais dans L2. Nous l'approchons par une suite de troncatures temporelles, qui, elles, sont dans L1, et nous vérifions que leurs transformées de Fourier convergent (suite de Cauchy).

```python
import cmath
import math

def dft_matrix(N):
    W = [[0]*N for _ in range(N)]
    for k in range(N):
        for n in range(N):
            angle = -2 * math.pi * k * n / N
            W[k][n] = cmath.exp(complex(0, angle))
    return W

def mat_vec_mult(M, v):
    return [sum(M[i][j]*v[j] for j in range(len(v))) for i in range(len(M))]

def sinc(x):
    if x == 0:
        return 1.0
    return math.sin(x)/x

N = 64
t_values = [0.5 * (n - N//2) for n in range(N)]

# Sinc non tronquee (ou tronquee tres loin)
f_complet = [sinc(t) for t in t_values]

# Approximations tronquees
def f_tronquee(M):
    return [sinc(t_values[i]) if (N//2 - M) <= i <= (N//2 + M) else 0 for i in range(N)]

W = dft_matrix(N)

f_small = f_tronquee(10)
f_med = f_tronquee(20)

hat_small = mat_vec_mult(W, f_small)
hat_med = mat_vec_mult(W, f_med)
hat_complet = mat_vec_mult(W, f_complet)

# Calcul des normes L2 discretes des differences
def dist(v1, v2):
    return math.sqrt(sum(abs(v1[i] - v2[i])**2 for i in range(len(v1))))

dist_small = dist(hat_complet, hat_small)
dist_med = dist(hat_complet, hat_med)

print(f"Distance pour petite troncature : {dist_small}")
print(f"Distance pour moyenne troncature : {dist_med}")

assert dist_med < dist_small, "La suite ne semble pas converger dans L2 !"
```
