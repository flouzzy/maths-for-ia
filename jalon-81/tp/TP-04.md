# TP 4 : Principe d'incertitude dans l'espace discret


Nous calculons la dispersion temporelle (variance temporelle) et la dispersion fréquentielle d'un signal, pour vérifier l'inégalité de Heisenberg-Gabor.

```python
import cmath
import math

def dft(x):
    N = len(x)
    return [sum(x[n] * cmath.exp(complex(0, -2 * math.pi * k * n / N)) for n in range(N)) for k in range(N)]

N = 32
# Creation d'une gaussienne tres etroite en temps
sigma_time = 1.0
signal_time = [math.exp(- ((n - N//2)**2) / (2 * sigma_time**2)) for n in range(N)]
norm_time = math.sqrt(sum(x**2 for x in signal_time))
signal_time = [x / norm_time for x in signal_time] # normalisation L2 = 1

spectre = dft(signal_time)
norm_freq = sum(abs(x)**2 for x in spectre)

# La variance (dispersion) calcule l'etalement de l'energie
var_time = sum(((n - N//2)**2) * (signal_time[n]**2) for n in range(N))

var_freq = 0.0
for k in range(N):
    freq_idx = k if k < N//2 else k - N
    var_freq += (freq_idx**2) * (abs(spectre[k])**2)
var_freq = var_freq / norm_freq # division par la norme au carre (energie)

print(f"Dispersion temporelle : {var_time:.4f}")
print(f"Dispersion frequentielle : {var_freq:.4f}")
print(f"Produit des dispersions : {var_time * var_freq:.4f}")

assert var_time * var_freq > 0.1, "Violation du theoreme d'incertitude : le produit est trop faible."
```
