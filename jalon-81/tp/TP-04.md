# TP 4 : Filtre passe-bas idéal (Opérateur de projection PW)

## Description
Simulation de l'opérateur de projection dans l'espace des fréquences en annulant les hautes fréquences et reconstruction par DFT inverse.

## Code Python
```python
def compute_dft(signal):
    import cmath
    N = len(signal)
    dft = []
    for k in range(N):
        val = 0.0 + 0.0j
        for n in range(N):
            angle = -2j * cmath.pi * k * n / N
            val += signal[n] * cmath.exp(angle)
        dft.append(val)
    return dft

def compute_idft(dft_signal):
    import cmath
    N = len(dft_signal)
    signal = []
    for n in range(N):
        val = 0.0 + 0.0j
        for k in range(N):
            angle = 2j * cmath.pi * k * n / N
            val += dft_signal[k] * cmath.exp(angle)
        signal.append(val.real / N)
    return signal

# Signal bruite (basse frequence + haute frequence)
import math
N = 32
signal = [math.sin(2 * math.pi * 2 * n / N) + 0.5 * math.cos(2 * math.pi * 12 * n / N) for n in range(N)]

dft_sig = compute_dft(signal)

# Appliquer le projecteur spectral P_W (coupure des hautes frequences, k > 4 et k < 28)
cutoff = 4
for k in range(N):
    if k > cutoff and k < N - cutoff:
        dft_sig[k] = 0.0 + 0.0j

filtered_signal = compute_idft(dft_sig)

# Le signal filtre devrait ressembler a une sinusoide pure de frequence 2
for n in range(N):
    expected_val = math.sin(2 * math.pi * 2 * n / N)
    assert abs(filtered_signal[n] - expected_val) < 1e-10, "Le filtre passe-bas a echoue"

print("TP4: Filtrage passe-bas spectral reussi.")
```
