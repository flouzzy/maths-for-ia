# TP 04 : Intégration numérique par importance

Calcul approché de l'intégrale de g(x)*exp(-x) sur [0, infini[

```python
import random
import math

def random_exponential(lambd):
    # Genere une exponentielle de parametre lambda par methode de la transformee inverse.
    u = random.random()
    return -math.log(1 - u) / lambd

def compute_integral(n_samples):
    # On veut integrer g(x)e^{-x} = (x^2)e^{-x}.
    # La valeur theorique est Gamma(3) = 2.
    total = 0
    for _ in range(n_samples):
        x = random_exponential(1.0)
        # g(x) = x^2
        total += x**2

    return total / n_samples

def test_importance_sampling():
    n = 50000
    val = compute_integral(n)
    print(f"Valeur estimee de l'integrale : {val}")

    assert abs(val - 2.0) < 0.1, "L'integrale evaluee par importance sampling diverge de 2.0"
    print("TP04 Valide: L'integrateur de Monte-Carlo converge vers Gamma(3)=2.")

if __name__ == '__main__':
    test_importance_sampling()
```
