# TP 01 : Convergence empirique du jet de dé

L'objectif de ce TP est de vérifier informatiquement la Loi Forte des Grands Nombres sur une distribution simple.

```python
import random

def simulate_dice_lfgn(n_throws):
    # Simule n lancers de de et retourne l'evolution de la moyenne.
    total = 0
    means = []
    for i in range(1, n_throws + 1):
        throw = random.randint(1, 6)
        total += throw
        means.append(total / i)
    return means

def test_lfgn():
    n = 50000
    means = simulate_dice_lfgn(n)
    final_mean = means[-1]

    # The theoretical expectation of a fair 6-sided die is 3.5
    expected = 3.5

    print(f"Moyenne apres {n} lancers : {final_mean}")
    assert abs(final_mean - expected) < 0.05, "La LFGN semble violee, ecart trop grand."
    print("TP01 Valide: La moyenne empirique converge bien vers l'esperance.")

if __name__ == '__main__':
    test_lfgn()
```
