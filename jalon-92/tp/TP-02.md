# TP 02 : Calcul de pi par Monte-Carlo

Nous allons estimer la valeur de pi en utilisant l'aire du quart de disque unité.

```python
import random
import math

def estimate_pi(n_points):
    # Estime pi en jetant n_points aleatoirement dans le carre [0,1]^2.
    points_in_circle = 0
    for _ in range(n_points):
        x = random.random()
        y = random.random()
        if x**2 + y**2 <= 1.0:
            points_in_circle += 1

    # Pi/4 is the expected ratio
    return 4 * points_in_circle / n_points

def test_monte_carlo():
    n = 100000
    pi_estimate = estimate_pi(n)
    print(f"Estimation de Pi avec {n} points : {pi_estimate}")

    # Assert correctness within a statistical margin
    assert abs(pi_estimate - math.pi) < 0.05, "L'estimation de pi est mauvaise."
    print("TP02 Valide: L'estimateur de Monte-Carlo converge vers la valeur exacte.")

if __name__ == '__main__':
    test_monte_carlo()
```
