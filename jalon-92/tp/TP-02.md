# TP 2 : Intégration de Monte-Carlo pour estimer Pi

## Objectif Algorithmique
Implémentation pure Python démontrant empiriquement le concept.

## Code Python

```python
def generer_uniforme(seed, n):
    '''Génère n nombres pseudo-aléatoires uniformes dans [0, 1].'''
    x = seed
    nombres = []
    for _ in range(n):
        x = (1664525 * x + 1013904223) % (2**32)
        nombres.append(x / (2**32))
    return nombres

def estimer_pi(n):
    '''Estime Pi en calculant l'aire d'un quart de disque unité.'''
    points_x = generer_uniforme(12345, n)
    points_y = generer_uniforme(67890, n)

    points_dans_disque = 0
    for i in range(n):
        x = points_x[i]
        y = points_y[i]
        # Condition d'appartenance au disque de rayon 1
        if x*x + y*y <= 1.0:
            points_dans_disque += 1

    # La probabilité est (Pi/4) / 1
    proportion = points_dans_disque / n
    pi_estime = proportion * 4.0
    return pi_estime

if __name__ == '__main__':
    n = 200000
    pi_est = estimer_pi(n)
    vrai_pi = 3.14159265

    ecart = abs(pi_est - vrai_pi)
    print('Pi estimé :', pi_est)
    assert ecart < 0.02, 'L\'estimation de Monte-Carlo diverge trop.'
    print('TP2 validé : L\'intégration de Monte-Carlo via LFGN converge.')
```
