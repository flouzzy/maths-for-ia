# TP 5 : Convergence de la matrice de covariance empirique en dimension 2

## Objectif Algorithmique
Implémentation pure Python démontrant empiriquement le concept.

## Code Python

```python
def gen_uniforme(seed, n):
    x = seed
    res = []
    for _ in range(n):
        x = (1664525 * x + 1013904223) % (2**32)
        res.append(x / (2**32))
    return res

if __name__ == '__main__':
    n = 100000
    X = gen_uniforme(123, n)
    Y = gen_uniforme(456, n)

    # On centre les variables (Espérance théo = 0.5)
    # Covariance théorique entre X et Y indépendants = 0
    # Variance théorique d'une uniforme sur [0,1] = 1/12 = 0.08333...

    somme_X2 = 0
    somme_Y2 = 0
    somme_XY = 0

    for i in range(n):
        x_c = X[i] - 0.5
        y_c = Y[i] - 0.5
        somme_X2 += x_c * x_c
        somme_Y2 += y_c * y_c
        somme_XY += x_c * y_c

    var_X = somme_X2 / n
    var_Y = somme_Y2 / n
    cov_XY = somme_XY / n

    print('Variance empirique X :', var_X)
    print('Variance empirique Y :', var_Y)
    print('Covariance empirique XY :', cov_XY)

    assert abs(var_X - (1/12)) < 0.01, 'Variance X incorrecte'
    assert abs(var_Y - (1/12)) < 0.01, 'Variance Y incorrecte'
    assert abs(cov_XY - 0) < 0.01, 'Covariance XY incorrecte'
    print('TP5 validé : La matrice de covariance empirique converge vers la théorique (LFGN multi-dimensionnelle).')
```
