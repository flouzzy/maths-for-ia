# TP 05 : Vérification de la convergence de la variance

Nous vérifions que la variance empirique converge bien vers la variance théorique.

```python
import random

def compute_empirical_variance(n_samples):
    # Loi uniforme continue sur [0, 1].
    # Esperance theorique : 1/2
    # Variance theorique : 1/12 ~= 0.083333
    samples = [random.random() for _ in range(n_samples)]
    mean = sum(samples) / n_samples

    var_sum = 0
    for x in samples:
        var_sum += (x - mean)**2

    empirical_var = var_sum / n_samples
    return empirical_var

def test_empirical_variance():
    n = 20000
    var = compute_empirical_variance(n)
    expected_var = 1/12

    print(f"Variance empirique : {var}")
    print(f"Variance theorique : {expected_var}")

    assert abs(var - expected_var) < 0.01, "La variance empirique ne converge pas correctement."
    print("TP05 Valide: L'estimateur de la variance est consistant.")

if __name__ == '__main__':
    test_empirical_variance()
```
