# TP 3 : Le critère de Sylvester

## Objectif Mathématique
Déterminer si une forme quadratique est définie positive en appliquant le théorème (critère) de Sylvester, qui nécessite de vérifier la stricte positivité de tous les mineurs principaux dominants.

## Code Python (From Scratch)

```python
def sous_matrice_principale(A, k):
    # Extrait la sous-matrice de taille kxk (les k premières lignes/colonnes)
    return [[A[i][j] for j in range(k)] for i in range(k)]

def determinant_recursif(M):
    n = len(M)
    if n == 1:
        return M[0][0]
    if n == 2:
        return M[0][0]*M[1][1] - M[0][1]*M[1][0]

    det = 0
    for j in range(n):
        # Développement par rapport à la première ligne
        sous_M = [[M[i][col] for col in range(n) if col != j] for i in range(1, n)]
        sign = (-1) ** j
        det += sign * M[0][j] * determinant_recursif(sous_M)
    return det

def est_definie_positive(A):
    n = len(A)
    for k in range(1, n + 1):
        M_k = sous_matrice_principale(A, k)
        det_k = determinant_recursif(M_k)
        if det_k <= 0:
            return False
    return True

if __name__ == '__main__':
    # Matrice définie positive (mineurs: 2, 2*3-1=5, det > 0)
    A_dp = [
        [2, 1, 0],
        [1, 3, 1],
        [0, 1, 4]
    ]

    # Matrice avec trace > 0 mais non définie positive (mineurs: 1, 1*1-3*3 = -8)
    A_non_dp = [
        [1, 3, 0],
        [3, 1, 0],
        [0, 0, 5]
    ]

    assert est_definie_positive(A_dp) == True
    assert est_definie_positive(A_non_dp) == False

    print("TP3 validé : Le critère de Sylvester identifie correctement la signature.")
```
