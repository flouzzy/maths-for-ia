## TP 4 : Algorithme des K-Plus Proches Voisins (KNN) pur Python

Implémentation naïve de KNN pour illustrer l'importance du choix de la distance dans un processus de décision.

```python
def distance_euclidienne(u, v):
    return sum((x-y)**2 for x,y in zip(u,v))**0.5

class KNN:
    def __init__(self, k, dist_func=distance_euclidienne):
        self.k = k
        self.dist_func = dist_func
        self.data = []
        self.labels = []

    def fit(self, X, y):
        self.data = X
        self.labels = y

    def predict(self, x_new):
        # Calculer toutes les distances
        distances = [(self.dist_func(x_new, x_train), label)
                     for x_train, label in zip(self.data, self.labels)]

        # Trier par distance
        distances.sort(key=lambda item: item[0])

        # Prendre les k plus proches
        k_nearest = distances[:self.k]

        # Vote majoritaire (naif)
        votes = {}
        for _, label in k_nearest:
            votes[label] = votes.get(label, 0) + 1

        return max(votes.items(), key=lambda item: item[1])[0]

if __name__ == "__main__":
    X_train = [[1, 1], [1, 2], [5, 5], [6, 5]]
    y_train = ["Classe_A", "Classe_A", "Classe_B", "Classe_B"]

    knn = KNN(k=3)
    knn.fit(X_train, y_train)

    x_test = [2, 2]
    pred = knn.predict(x_test)
    print(f"Prediction pour {x_test} : {pred}")
    assert pred == "Classe_A"
```
