## TP 5 : Distance de Hausdorff entre deux ensembles discrets

Calcul de la distance de Hausdorff entre deux ensembles de points dans un plan 2D.

```python
def dist_euclidienne(p1, p2):
    return ((p1[0]-p2[0])**2 + (p1[1]-p2[1])**2)**0.5

def dist_point_ensemble(point, ensemble):
    '''Calcule d(x, A) = inf {d(x,a) pour a dans A}'''
    return min(dist_euclidienne(point, a) for a in ensemble)

def hausdorff(A, B):
    '''Distance de Hausdorff entre A et B.'''
    max_A_to_B = max(dist_point_ensemble(a, B) for a in A)
    max_B_to_A = max(dist_point_ensemble(b, A) for b in B)
    return max(max_A_to_B, max_B_to_A)

if __name__ == "__main__":
    # Carre de cote 1
    ensemble_A = [(0,0), (1,0), (0,1), (1,1)]
    # Carre decale de (0.5, 0.5)
    ensemble_B = [(0.5,0.5), (1.5,0.5), (0.5,1.5), (1.5,1.5)]

    dH = hausdorff(ensemble_A, ensemble_B)
    print(f"Distance de Hausdorff entre A et B : {dH}")

    # La distance de A a lui-meme doit etre 0
    assert hausdorff(ensemble_A, ensemble_A) == 0.0
```
