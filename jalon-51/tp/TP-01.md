### TP 1 : Implémentation des Distances Classiques

Ce TP propose d'implémenter les distances euclidienne, de Manhattan et de Minkowski en Python pur. Ces fonctions sont la base de l'espace métrique usuel.

```python
def distance_minkowski(p1, p2, p):
    # p1 et p2 sont des listes de coordonnees (vecteurs de dimension n)
    assert len(p1) == len(p2), "Les points doivent etre de meme dimension"
    assert p >= 1, "Le parametre p de Minkowski doit etre >= 1 pour respecter l'inegalite triangulaire"

    somme = 0
    for i in range(len(p1)):
        somme += abs(p1[i] - p2[i]) ** p

    return somme ** (1 / p)

def distance_euclidienne(p1, p2):
    return distance_minkowski(p1, p2, 2)

def distance_manhattan(p1, p2):
    return distance_minkowski(p1, p2, 1)

# Tests de validation
pt_A = [0, 0]
pt_B = [3, 4]

dist_eucl = distance_euclidienne(pt_A, pt_B)
assert abs(dist_eucl - 5.0) < 1e-9, "Erreur sur la distance euclidienne"

dist_manh = distance_manhattan(pt_A, pt_B)
assert abs(dist_manh - 7.0) < 1e-9, "Erreur sur la distance de Manhattan"

print("Toutes les distances classiques sont valides.")
```
