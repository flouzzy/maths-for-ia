## TP 1 : Implémentation des distances usuelles (L1, L2, Linf)

Ce TP implémente les trois distances fondamentales sur des vecteurs de dimension n et vérifie empiriquement l'inégalité triangulaire.

```python
import math

def distance_l1(u, v):
    '''Distance de Manhattan.'''
    assert len(u) == len(v), "Les vecteurs doivent avoir la meme dimension."
    return sum(abs(x - y) for x, y in zip(u, v))

def distance_l2(u, v):
    '''Distance Euclidienne canonique.'''
    assert len(u) == len(v)
    return math.sqrt(sum((x - y)**2 for x, y in zip(u, v)))

def distance_linf(u, v):
    '''Distance de Tchebychev (Maximum).'''
    assert len(u) == len(v)
    return max(abs(x - y) for x, y in zip(u, v))

if __name__ == "__main__":
    A = [1.0, 0.0, -2.0]
    B = [4.0, 4.0, 2.0]
    C = [0.0, 5.0, 1.0]

    # Test L2
    d_AB = distance_l2(A, B)
    d_AC = distance_l2(A, C)
    d_CB = distance_l2(C, B)

    # Inegalite triangulaire AB <= AC + CB
    print("Verification Inegalite Triangulaire L2 :")
    print("d(A,B) =", d_AB)
    print("d(A,C) + d(C,B) =", d_AC + d_CB)
    assert d_AB <= d_AC + d_CB + 1e-9
```
