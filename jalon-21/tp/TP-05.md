# TP 5 : Point fixe de Picard (Equa. Diff.)
**Objectif :** Résoudre $y' = y$ itérativement et vérifier la convergence uniforme algorithmique de la suite de fonctions.

**Implémentation :**
```python
def int_trapez(f, a, b, steps=100):
    h = (b - a) / steps
    s = 0.5 * (f(a) + f(b))
    for i in range(1, steps):
        s += f(a + i * h)
    return s * h

def iter_picard(y_n_valeurs, t_valeurs):
    y_n1_valeurs = []
    for t in t_valeurs:
        # Intégrer y_n de 0 à t. C'est fait numériquement.
        # Pour une implémentation exacte des itérés discrets :
        idx = int(t * 100) # approximation simple pour 100 pts par unité
        if idx == 0:
            y_n1_valeurs.append(1.0)
        else:
            integral = sum(y_n_valeurs[:idx]) * 0.01 # somme de Riemann
            y_n1_valeurs.append(1.0 + integral)
    return y_n1_valeurs

t_grille = [i*0.01 for i in range(101)] # [0, 1]
y_init = [1.0] * len(t_grille)

y_courant = y_init
import math
for n in range(1, 10):
    y_courant = iter_picard(y_courant, t_grille)

max_err = max(abs(y_courant[i] - math.exp(t_grille[i])) for i in range(len(t_grille)))
print(f"Erreur après 9 itérations: {max_err}")
assert max_err < 0.05, "La convergence est très rapide sur le compact."
```
