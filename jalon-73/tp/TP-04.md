# TP 4 : Démonstration de la non-convergence en norme L-infinity

Nous allons coder la suite de fonctions du cours : $f_n(x) = n x e^{-nx}$ sur un domaine fini $[0, 10]$ et mesurer ses normes L1 et Linf pour illustrer numériquement les théorèmes.

```python
import math

def f_n(n, x):
    return n * x * math.exp(-n * x)

def compute_norms(n, a, b, num_steps=1000):
    step_size = (b - a) / num_steps
    l1_sum = 0.0
    linf_max = 0.0

    for i in range(num_steps):
        x = a + i * step_size
        val = abs(f_n(n, x))
        l1_sum += val * step_size
        if val > linf_max:
            linf_max = val

    return l1_sum, linf_max

# Tests and Validation
l1_n10, linf_n10 = compute_norms(10, 0.0, 10.0)
l1_n100, linf_n100 = compute_norms(100, 0.0, 10.0)

# As n increases, L1 norm should go to 0
assert l1_n100 < l1_n10, "L1 norm is not strictly decreasing!"
assert l1_n100 < 0.1, "L1 norm should be close to 0"

# However, Linf norm should stay around 1/e (approx 0.367)
target_linf = math.exp(-1)
assert abs(linf_n10 - target_linf) < 1e-2, "Linf not preserved"
assert abs(linf_n100 - target_linf) < 1e-2, "Linf not preserved"

print("TP 4 Validation Passed!")
```
