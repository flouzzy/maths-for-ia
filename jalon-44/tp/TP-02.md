# TP 2 : Simulation numérique de limites directionnelles

Nous allons étudier empiriquement la fonction f(x,y) = (x*y) / (x^2 + y^2).
On sait analytiquement que sa limite n'existe pas en l'origine. Le but de ce script est d'approcher l'origine par différentes droites y = m*x pour observer numériquement la non-unicité de la limite.

```python
def f(x, y):
    """
    Evalue la fonction f(x,y) = xy / (x^2 + y^2).
    """
    if x == 0 and y == 0:
        return None  # Non definie en l'origine
    return (x * y) / (x**2 + y**2)

def directional_limit(m, num_steps=5):
    """
    Approche l'origine selon la droite y = mx avec des valeurs de x de plus en plus petites.
    """
    print(f"Approche selon la droite y = {m}x :")
    x = 1.0
    for _ in range(num_steps):
        x = x / 10.0
        y = m * x
        value = f(x, y)
        print(f"  x={x:.5e}, y={y:.5e} -> f(x,y) = {value:.4f}")
    return value

# Validation mathematique
def test_directional_limits():
    # Selon la theorie analytique :
    # f(x, mx) = x(mx) / (x^2 + (mx)^2) = m*x^2 / (x^2(1+m^2)) = m / (1+m^2)

    # Pour y = 0x (m=0), la limite theorique est 0 / (1+0) = 0
    limit_m0 = directional_limit(0)
    assert abs(limit_m0 - 0.0) < 1e-6, "Echec pour m=0"

    # Pour y = 1x (m=1), la limite theorique est 1 / (1+1) = 0.5
    limit_m1 = directional_limit(1)
    assert abs(limit_m1 - 0.5) < 1e-6, "Echec pour m=1"

    # Pour y = -1x (m=-1), la limite theorique est -1 / (1+1) = -0.5
    limit_m_minus1 = directional_limit(-1)
    assert abs(limit_m_minus1 - (-0.5)) < 1e-6, "Echec pour m=-1"

    print("TP 2 valide : Les limites directionnelles numeriques correspondent aux predictions analytiques.")

if __name__ == "__main__":
    test_directional_limits()
```
