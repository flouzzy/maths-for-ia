# TP 3 : Coordonnées polaires et continuité

Nous cherchons à implémenter une conversion en coordonnées polaires pour vérifier numériquement la limite de la fonction g(x,y) = (x^3 + y^3) / (x^2 + y^2).
On sait que cette fonction converge vers 0 en l'origine.

```python
import math

def g(x, y):
    """Evalue g(x,y) = (x^3 + y^3) / (x^2 + y^2)."""
    if x == 0 and y == 0:
        return 0.0 # Prolongement par continuite
    return (x**3 + y**3) / (x**2 + y**2)

def to_polar(x, y):
    """Convertit les coordonnees cartesiennes (x, y) en coordonnees polaires (r, theta)."""
    r = math.sqrt(x**2 + y**2)
    theta = math.atan2(y, x)
    return r, theta

def from_polar(r, theta):
    """Convertit les coordonnees polaires en cartesiennes."""
    x = r * math.cos(theta)
    y = r * math.sin(theta)
    return x, y

def test_polar_continuity():
    # Test de conversion
    x_test, y_test = 3.0, 4.0
    r, theta = to_polar(x_test, y_test)
    assert abs(r - 5.0) < 1e-6, "Erreur dans le calcul du rayon polaire."

    # Test de la limite lorsque r tend vers 0 independamment de theta
    angles = [0.0, math.pi/4, math.pi/2, math.pi, 2*math.pi]
    radii = [1.0, 0.1, 0.01, 0.001, 0.0001]

    for theta in angles:
        for r in radii:
            x, y = from_polar(r, theta)
            val = g(x, y)
            # La theorie dit que |g(x,y)| <= 2r.
            # Verifions que la valeur est bornee analytiquement
            assert abs(val) <= 2.0 * r + 1e-9, f"Echec de la majoration theorique pour r={r}, theta={theta}"

    # Test specifique au voisinage immediat
    x_small, y_small = from_polar(1e-5, math.pi/3)
    assert abs(g(x_small, y_small)) < 1e-4, "La fonction ne semble pas converger vers 0."

    print("TP 3 valide : La decroissance de la fonction est bien bornee par le rayon polaire, confirmant la continuite a l'origine.")

if __name__ == "__main__":
    test_polar_continuity()
```
