# TP 4 : Boules Ouvertes et Inclusions

Dans ce TP, nous allons simuler informatiquement la définition de la topologie de R^n.
Nous allons implémenter la notion de "Boule Ouverte" et vérifier si certains points appartiennent à ces boules.

```python
def distance(p1, p2):
    """Calcule la distance euclidienne entre deux points de dimension n."""
    assert len(p1) == len(p2), "Les points doivent avoir la meme dimension."
    return sum((a - b) ** 2 for a, b in zip(p1, p2)) ** 0.5

class OpenBall:
    def __init__(self, center, radius):
        """
        Initialise une boule ouverte.
        Args:
            center (tuple ou list): Le point central.
            radius (float): Le rayon strictement positif.
        """
        assert radius > 0, "Le rayon d'une boule ouverte doit etre strictement positif."
        self.center = center
        self.radius = radius

    def contains(self, point):
        """Verifie si un point appartient a l'interieur strict de la boule."""
        return distance(self.center, point) < self.radius

# Validation mathematique
def test_open_balls():
    # Creation d'une boule en 2D centree en (1, 2) de rayon 2.0
    B1 = OpenBall((1, 2), 2.0)

    # Le centre doit toujours appartenir a la boule ouverte
    assert B1.contains((1, 2)) == True, "Echec : Le centre n'est pas dans la boule."

    # Un point proche doit appartenir
    assert B1.contains((2, 2)) == True, "Echec : Le point (2,2) (distance 1) devrait etre dedans."

    # Un point sur la frontiere exacte (distance=2) NE DOIT PAS appartenir (c'est une boule OUVERTE)
    assert B1.contains((3, 2)) == False, "Echec topologique : La frontiere est incluse alors que la boule est ouverte."

    # Un point lointain
    assert B1.contains((10, 10)) == False, "Echec : Point trop eloigne inclus par erreur."

    # Test en dimension superieure (R^4)
    B_R4 = OpenBall((0, 0, 0, 0), 1.0)
    assert B_R4.contains((0.5, 0.5, 0.5, 0.5)) == True, "Echec : (0.5, 0.5, 0.5, 0.5) a une distance de sqrt(4*0.25)=1. Ce point est sur la frontiere."
    # Correction : distance de (0.5, 0.5, 0.5, 0.5) a (0,0,0,0) = sqrt(0.25+0.25+0.25+0.25) = sqrt(1.0) = 1.0
    # C'est la frontiere, donc CA NE DOIT PAS Y ETRE.
    assert B_R4.contains((0.5, 0.5, 0.5, 0.5)) == False, "Echec de raisonnement sur R^4."

    assert B_R4.contains((0.1, 0.1, 0.1, 0.1)) == True, "Echec d'inclusion R^4 valide."

    print("TP 4 valide : La semantique topologique de la boule ouverte est preservee (frontieres exclues).")

if __name__ == "__main__":
    test_open_balls()
```
