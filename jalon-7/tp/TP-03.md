## Objectif

Ce Travail Pratique vise à solidifier votre compréhension des espaces vectoriels abstraits en vous plongeant dans l'implémentation algorithmique de concepts fondamentaux de l'algèbre linéaire appliqués aux polynômes. Vous développerez des outils pour :

1.  **Représenter des polynômes** comme des vecteurs de coordonnées dans une base standard.
2.  **Construire une matrice de coordonnées** à partir d'une famille de polynômes.
3.  **Implémenter l'algorithme d'élimination de Gauss** pour transformer une matrice en sa forme échelonnée par lignes (REF).
4.  **Déterminer si une famille de polynômes est linéairement indépendante**.
5.  **Extraire une base** pour le sous-espace vectoriel engendré par une famille de polynômes et en calculer la dimension.
6.  **Vérifier si une famille de polynômes forme une base** pour un espace vectoriel donné (par exemple, P_n, l'espace des polynômes de degré au plus n).

Ce TP met l'accent sur une implémentation "from scratch", sans l'aide de bibliothèques d'algèbre linéaire de haut niveau, afin de maîtriser les mécanismes sous-jacents.

import math

# Constante pour la comparaison des nombres flottants afin de gérer les imprécisions
EPSILON = 1e-9

def poly_to_vector_coords(poly_coeffs, max_degree):
    """
    Convertit une liste de coefficients de polynôme en un vecteur de coordonnées
    de taille fixe, en complétant avec des zéros si nécessaire.
    Un polynôme a_0 + a_1*x + ... + a_k*x^k est représenté par [a_0, a_1, ..., a_k].
    L'espace P_max_degree est l'espace des polynômes de degré au plus max_degree,
    sa dimension est max_degree + 1.

    Ex: poly_coeffs = [1, 2, 3] pour 1 + 2x + 3x^2
    max_degree = 3 (pour P_3) -> [1.0, 2.0, 3.0, 0.0]
    """
    if not isinstance(poly_coeffs, list):
        raise TypeError("Les coefficients du polynôme doivent être une liste.")

    # La dimension de l'espace P_max_degree est max_degree + 1
    target_dim = max_degree + 1

    if len(poly_coeffs) > target_dim:
        raise ValueError(f"Le polynôme a un degré trop élevé pour l'espace P_{max_degree}.")

    vector_coords = poly_coeffs + [0.0] * (target_dim - len(poly_coeffs))

    return [float(c) for c in vector_coords]

def create_matrix_from_polynomials(polynomials, max_degree):
    """
    Crée une matrice où chaque ligne représente les coordonnées d'un polynôme
    dans la base standard {1, x, x^2, ..., x^max_degree}.
    """
    if not isinstance(polynomials, list):
        raise TypeError("La liste de polynômes doit être une liste de listes de coefficients.")
    if not all(isinstance(p, list) for p in polynomials):
        raise TypeError("Chaque polynôme doit être représenté par une liste de coefficients.")

    matrix = []
    for poly_coeffs in polynomials:
        matrix.append(poly_to_vector_coords(poly_coeffs, max_degree))
    return matrix

def _swap_rows(matrix, r1, r2):
    """Échange deux lignes dans une matrice."""
    matrix[r1], matrix[r2] = matrix[r2], matrix[r1]

def _scale_row(matrix, r, scalar):
    """Multiplie une ligne par un scalaire."""
    if abs(scalar) < EPSILON:
        raise ValueError("Impossible de multiplier une ligne par zéro (ou presque zéro).")
    matrix[r] = [x * scalar for x in matrix[r]]

def _add_rows(matrix, target_row, source_row, scalar):
    """Ajoute un multiple d'une ligne source à une ligne cible."""
    matrix[target_row] = [matrix[target_row][i] + scalar * matrix[source_row][i]
                          for i in range(len(matrix[target_row]))]

def gaussian_elimination(matrix):
    """
    Applique l'élimination de Gauss pour transformer une matrice en sa forme
    échelonnée par lignes (REF).
    Retourne une nouvelle matrice (pour ne pas modifier l'originale).
    """
    if not matrix:
        return []

    num_rows = len(matrix)
    num_cols = len(matrix[0])

    # Créer une copie pour ne pas modifier la matrice originale
    ref_matrix = [row[:] for row in matrix]

    pivot_row = 0
    for col in range(num_cols):
        if pivot_row >= num_rows:
            break

        # Trouver une ligne non nulle pour le pivot dans la colonne actuelle
        # On cherche la ligne avec la plus grande valeur absolue pour le pivot
        # afin d'améliorer la stabilité numérique.
        max_val_row = pivot_row
        for i in range(pivot_row + 1, num_rows):
            if abs(ref_matrix[i][col]) > abs(ref_matrix[max_val_row][col]):
                max_val_row = i

        if abs(ref_matrix[max_val_row][col]) < EPSILON:
            # La colonne est entièrement nulle (ou presque) sous le pivot_row,
            # passer à la colonne suivante.
            continue

        # Échanger la ligne avec le pivot le plus grand vers la position du pivot_row
        _swap_rows(ref_matrix, pivot_row, max_val_row)

        # Normaliser la ligne pivot (rendre le pivot égal à 1)
        pivot_value = ref_matrix[pivot_row][col]
        _scale_row(ref_matrix, pivot_row, 1.0 / pivot_value)

        # Éliminer les éléments sous le pivot
        for i in range(pivot_row + 1, num_rows):
            factor = ref_matrix[i][col]
            _add_rows(ref_matrix, i, pivot_row, -factor)

        pivot_row += 1

    return ref_matrix

def get_rank(ref_matrix):
    """
    Calcule le rang d'une matrice en forme échelonnée par lignes (REF).
    Le rang est le nombre de lignes non nulles.
    """
    if not ref_matrix:
        return 0

    rank = 0
    for row in ref_matrix:
        # Une ligne est non nulle si au moins un de ses éléments est non nul (au-delà de EPSILON)
        if any(abs(x) > EPSILON for x in row):
            rank += 1
    return rank

def is_linearly_independent(polynomials, max_degree):
    """
    Détermine si une famille de polynômes est linéairement indépendante.
    Une famille de k vecteurs est LI si le rang de la matrice formée par leurs
    coordonnées (en tant que lignes) est égal à k (le nombre de polynômes).
    """
    if not polynomials:
        return True # La famille vide est LI par convention

    matrix = create_matrix_from_polynomials(polynomials, max_degree)
    ref_matrix = gaussian_elimination(matrix)

    return get_rank(ref_matrix) == len(polynomials)

def find_basis_and_dimension(polynomials, max_degree):
    """
    Trouve une base pour le sous-espace engendré par les polynômes donnés
    et retourne cette base (sous forme de vecteurs de coordonnées issus de la REF)
    ainsi que sa dimension.
    Les lignes non nulles de la matrice en forme échelonnée par lignes (REF)
    forment une base pour l'espace des lignes (l'espace engendré par les polynômes).
    Ces vecteurs de base ne sont pas nécessairement les polynômes originaux,
    mais des combinaisons linéaires d'entre eux.
    """
    if not polynomials:
        return [], 0

    matrix = create_matrix_from_polynomials(polynomials, max_degree)
    ref_matrix = gaussian_elimination(matrix)

    basis_vectors_coords = []
    for row in ref_matrix:
        if any(abs(x) > EPSILON for x in row):
            basis_vectors_coords.append(row)

    dimension = len(basis_vectors_coords)

    return basis_vectors_coords, dimension

def is_spanning_set(polynomials, max_degree, target_dimension):
    """
    Détermine si une famille de polynômes engendre un espace de dimension `target_dimension`.
    Cela signifie que la dimension du sous-espace engendré par ces polynômes doit être
    égale à `target_dimension`.
    """
    _, dimension = find_basis_and_dimension(polynomials, max_degree)
    return dimension == target_dimension

def is_basis(polynomials, max_degree, target_dimension):
    """
    Détermine si une famille de polynômes forme une base pour un espace
    de dimension `target_dimension`.
    Une famille de vecteurs est une base si elle est linéairement indépendante ET
    si le nombre de vecteurs est égal à la dimension de l'espace.
    (Ces deux conditions impliquent qu'elle engendre l'espace).
    """
    # Pour être une base, le nombre de vecteurs doit être égal à la dimension de l'espace.
    if len(polynomials) != target_dimension:
        return False

    # Et ils doivent être linéairement indépendants.
    return is_linearly_independent(polynomials, max_degree)
