## Implémentation en Python

```python
import math

# Constante pour la comparaison de nombres flottants (tolérance d'erreur)
EPSILON = 1e-9

# --- Fonctions utilitaires pour les polynômes ---

def pad_polynomial(p, target_degree):
    """
    Complète un polynôme avec des zéros pour atteindre un degré cible.
    Un polynôme est représenté par une liste de coefficients [a0, a1, ..., an].
    Ex: pad_polynomial([1, 2], 2) -> [1.0, 2.0, 0.0] (pour 1 + 2x dans P_2)
    """
    if len(p) > target_degree + 1:
        raise ValueError(f"Le polynôme {p} a un degré supérieur à {target_degree}")

    # Assurer que tous les coefficients sont des flottants pour éviter les problèmes de type
    p_float = [float(coeff) for coeff in p]
    return p_float + [0.0] * (target_degree + 1 - len(p_float))

def add_polynomials(p1, p2):
    """
    Additionne deux polynômes.
    Les polynômes sont représentés par des listes de coefficients [a0, a1, ..., an].
    """
    max_len = max(len(p1), len(p2))
    p1_padded = pad_polynomial(p1, max_len - 1)
    p2_padded = pad_polynomial(p2, max_len - 1)
    return [c1 + c2 for c1, c2 in zip(p1_padded, p2_padded)]

def scalar_multiply_polynomial(scalar, p):
    """
    Multiplie un polynôme par un scalaire.
    """
    return [float(scalar) * float(c) for c in p]

# --- Fonctions utilitaires pour les matrices (pour l'élimination de Gauss) ---

def polynomial_list_to_matrix(polynomials, max_degree):
    """
    Convertit une liste de polynômes en une matrice de leurs coefficients.
    Chaque polynôme devient une ligne de la matrice.
    Les polynômes sont d'abord complétés au max_degree.
    """
    matrix = []
    for p in polynomials:
        matrix.append(pad_polynomial(p, max_degree))
    return matrix

def copy_matrix(matrix):
    """Crée une copie profonde d'une matrice."""
    return [row[:] for row in matrix]

def swap_rows(matrix, r1, r2):
    """Échange deux lignes dans une matrice."""
    matrix[r1], matrix[r2] = matrix[r2], matrix[r1]

def gaussian_elimination(matrix):
    """
    Applique l'élimination de Gauss à une matrice pour la mettre sous forme échelonnée.
    Modifie la matrice en place.
    """
    num_rows = len(matrix)
    if num_rows == 0:
        return matrix
    num_cols = len(matrix[0])

    lead_row = 0
    for col in range(num_cols):
        if lead_row >= num_rows:
            break

        # Trouver une ligne avec un pivot non nul dans la colonne actuelle
        pivot_row = lead_row
        while pivot_row < num_rows and abs(matrix[pivot_row][col]) < EPSILON:
            pivot_row += 1

        if pivot_row < num_rows:
            # Échanger la ligne pivot avec la ligne principale
            swap_rows(matrix, lead_row, pivot_row)

            # Normaliser la ligne pivot (rendre le pivot égal à 1)
            pivot_value = matrix[lead_row][col]
            if abs(pivot_value) > EPSILON: # Éviter division par zéro si pivot est très proche de zéro
                for j in range(col, num_cols):
                    matrix[lead_row][j] /= pivot_value

                # Éliminer les autres éléments de la colonne
                for i in range(num_rows):
                    if i != lead_row:
                        factor = matrix[i][col]
                        for j in range(col, num_cols):
                            matrix[i][j] -= factor * matrix[lead_row][j]
            lead_row += 1
    return matrix

def get_rank(matrix):
    """
    Calcule le rang d'une matrice après l'élimination de Gauss.
    Le rang est le nombre de lignes non nulles.
    """
    rank = 0
    for row in matrix:
        # Une ligne est non nulle si au moins un de ses éléments est non nul (au-delà de la tolérance)
        if any(abs(x) > EPSILON for x in row):
            rank += 1
    return rank

# --- Fonctions pour l'analyse des propriétés d'espaces vectoriels ---

def is_linearly_independent(polynomials, max_degree):
    """
    Vérifie si une famille de polynômes est linéairement indépendante dans P_max_degree.
    """
    if not polynomials:
        return True # La famille vide est LI

    # Convertir les polynômes en une matrice de coefficients
    matrix = polynomial_list_to_matrix(polynomials, max_degree)

    # Appliquer l'élimination de Gauss pour trouver le rang
    echelon_form_matrix = copy_matrix(matrix) # Travailler sur une copie
    gaussian_elimination(echelon_form_matrix)

    # Une famille est LI si le rang de la matrice des coefficients est égal au nombre de vecteurs
    return get_rank(echelon_form_matrix) == len(polynomials)

def is_spanning_set(polynomials, max_degree):
    """
    Vérifie si une famille de polynômes est une famille génératrice pour P_max_degree.
    """
    # La dimension de P_max_degree est max_degree + 1
    dimension_space = max_degree + 1

    if not polynomials:
        # La famille vide ne peut générer que l'espace vectoriel nul (dimension 0)
        return dimension_space == 0

    # Convertir les polynômes en une matrice de coefficients
    matrix = polynomial_list_to_matrix(polynomials, max_degree)

    # Appliquer l'élimination de Gauss pour trouver le rang
    echelon_form_matrix = copy_matrix(matrix)
    gaussian_elimination(echelon_form_matrix)

    # Une famille est génératrice si le rang de la matrice des coefficients est égal à la dimension de l'espace
    # (Cela signifie que les vecteurs génèrent un sous-espace de même dimension que l'espace entier)
    return get_rank(echelon_form_matrix) == dimension_space

def is_basis(polynomials, max_degree):
    """
    Vérifie si une famille de polynômes forme une base pour P_max_degree.
    """
    dimension_space = max_degree + 1

    # Une base doit avoir exactement 'dimension_space' vecteurs
    if len(polynomials) != dimension_space:
        return False

    # Une base doit être linéairement indépendante (ce qui implique qu'elle est génératrice si la taille est bonne)
    return is_linearly_independent(polynomials, max_degree)

# Matrice 3x3 avec dépendance

assert get_rank(m4) == 1, "Test rang m4 échoué"

# Matrice avec des zéros au début
m5 = [[0.0, 1.0, 2.0], [1.0, 2.0, 3.0], [0.0, 0.0, 1.0]]

# Dimension de P_1 est 2.

assert is_linearly_independent([p1_std_0, p1_std_1], max_deg_1), "Test LI P1 base standard échoué"

p1_alt_0 = [1.0] # 1

# Test 5: Indépendance Linéaire dans P_2 (polynômes de degré <= 2)

p2_alt_0 = [1.0]
