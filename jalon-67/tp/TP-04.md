# Objectif : Simuler g_n = inf_{k>=n} f_k et vérifier que la suite (g_n)
# est bien croissante même si (f_n) ne l'est pas.

import random

# f_k va osciller, ce ne sera pas monotone.
def f_k(x, k):
    return (1.0 + math.sin(k * x)) / x if x > 0 else 0

def g_n(x, n, K_max):
    # K_max est une borne arbitraire pour estimer l'infimum
    valeurs = []
    for k in range(n, K_max):
        valeurs.append(f_k(x, k))
    min_val = valeurs[0]
    for v in valeurs:
        if v < min_val:
            min_val = v
    return min_val

print("Verification de la croissance de l'infimum g_n :")
x_test = 2.5
K_max = 50
g_vals = []
for n in range(1, 10):
    val = g_n(x_test, n, K_max)
    g_vals.append(round(val, 4))
    print("g_n pour n =", n, "vaut", round(val, 4))

is_croissant = True
for i in range(len(g_vals) - 1):
    if g_vals[i] > g_vals[i+1]:
        is_croissant = False

print("La suite (g_n) est-elle croissante ? :", is_croissant)
