# TP 04 : Espérance limite en Machine Learning

## Objectif
Dans un cadre d'apprentissage, on simule un risque empirique (erreur de classification) calculé sur un dataset qui s'agrandit. La suite de fonctions correspond au risque maximum mesuré.

## Implémentation

```python
import random

def loss_function(prediction, truth):
    return (prediction - truth)**2

def expected_risk(N_samples, n_models):
    # On simule la fonction f_n(x) = max_{1 <= i <= n} L_i(x)
    # Cette suite est croissante par definition du max.

    dataset = [(random.uniform(0, 1), random.uniform(0, 1)) for _ in range(N_samples)]

    def evaluate_fn(model_id, x):
        # fonction arbitraire positive
        pred = (x[0] * model_id) % 1.0
        return loss_function(pred, x[1])

    def fn_max(n, x):
        return max(evaluate_fn(i, x) for i in range(1, n+1))

    # Integrale empirique (Esperance)
    def expected_fn(n):
        return sum(fn_max(n, x) for x in dataset) / N_samples

    return expected_fn

def test_risk_convergence():
    random.seed(42)
    risk_evaluator = expected_risk(1000, 50)

    risks = []
    for n in range(1, 20):
        risks.append(risk_evaluator(n))

    print("Evolution du risque pire-cas (croissant) :", risks[:5])

    # Verification de la croissance requise par Beppo Levi
    for i in range(len(risks)-1):
        assert risks[i] <= risks[i+1], "Le risque max ne peut pas decroitre"

if __name__ == "__main__":
    test_risk_convergence()
```
