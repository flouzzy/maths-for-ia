# TP 4 : Divergence de Kullback-Leibler et limites

### Objectif
Évaluer numériquement la stabilité de la divergence KL sous une approximation monotone de la distribution cible, en s'appuyant sur le TCM.

### Implémentation Python pur

```python
import math

# P is the true distribution (e.g., standard normal)
def P_pdf(x):
    return (1.0 / math.sqrt(2 * math.pi)) * math.exp(-0.5 * x**2)

# Q_n is a sequence of unnormalized distributions approaching P from below
def Q_n_unnormalized(x, n):
    # Approximation of exp(-x^2/2) using (1 - x^2/(2n))^n for |x| < sqrt(2n)
    if abs(x) < math.sqrt(2 * n):
        return (1.0 / math.sqrt(2 * math.pi)) * (1 - (x**2) / (2 * n))**n
    return 1e-10 # to avoid log(0)

def test_kl_convergence_tcm():
    # KL(P || Q) = Integral P(x) log(P(x)/Q(x)) dx

    # We evaluate for increasing n
    a, b = -5.0, 5.0
    steps = 5000
    step_size = (b - a) / steps

    kl_values = []
    for n in [1, 5, 20, 100, 500]:
        kl = 0.0
        for i in range(steps):
            x = a + (i + 0.5) * step_size
            px = P_pdf(x)
            qx = Q_n_unnormalized(x, n)
            # -log(Q_n) decreases as Q_n increases towards P
            kl += px * math.log(px / qx) * step_size
        kl_values.append(kl)

    print("Suite des divergences KL (P || Q_n) :", kl_values)

    # As n -> infinity, Q_n -> P, so KL -> 0
    assert kl_values[-1] < 0.05
    # Since Q_n approaches P from below, log(1/Q_n) drops monotonically,
    # ensuring monotonic convergence of the KL integral by Beppo-Levi.
    assert kl_values[0] > kl_values[-1]
    print("TP4: Convergence monotone de la divergence KL validee.")

test_kl_convergence_tcm()

```
