# TP 4 : Espérance par la formule de la queue

Illustration de la formule $E[X] = \sum P(X \ge n)$ justifiée par Beppo Levi (Exercice 10).

\begin{lstlisting}[language=Python]
# TP4: Esperance par la somme des probabilites de queue (cas fini)

# Supposons un de non truque a 6 faces
probs = [0, 1.0/6, 1.0/6, 1.0/6, 1.0/6, 1.0/6, 1.0/6] # X = 1, 2, ..., 6

# Calcul classique : somme(x * p(x))
esperance_classique = 0.0
for x in range(1, 7):
    esperance_classique += x * probs[x]

# Calcul par queue : somme(P(X >= n))
esperance_queue = 0.0
for n in range(1, 7):
    # P(X >= n) = sum(probs[x] pour x >= n)
    prob_queue = 0.0
    for x in range(n, 7):
        prob_queue += probs[x]
    esperance_queue += prob_queue

print("Classique:", esperance_classique)
print("Queue:", esperance_queue)

# Les deux methodes doivent etre egales a 3.5
assert abs(esperance_classique - esperance_queue) < 0.0001
assert abs(esperance_classique - 3.5) < 0.0001
\end{lstlisting}
