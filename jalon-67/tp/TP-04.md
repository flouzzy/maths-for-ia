
# TP 4 : Détection empirique de la monotonie

Dans les environnements d'optimisation (comme la montée en gradient sur des MDP), on a souvent besoin de s'assurer qu'un processus empirique est monotone. Ce TP implémente une fonction Python qui audite la monotonicité croissante d'une suite de listes de valeurs.

```python
# Validation pure Python, sans modules externes

def verify_monotonicity(sequence_of_functions, domain):
    # sequence_of_functions est une liste de dict {x: f_n(x)}
    n_length = len(sequence_of_functions)

    for i in range(n_length - 1):
        fn = sequence_of_functions[i]
        fnp1 = sequence_of_functions[i+1]

        for x in domain:
            if fn[x] > fnp1[x] + 1e-9: # tolerance float
                return False
    return True

def run_tp4():
    domain = [0.1, 0.5, 1.0, 2.0]

    # 1. Suite croissante e^{-x/n}
    import math
    seq_croissante = []
    for n in range(1, 10):
        fn = {x: math.exp(-x / n) for x in domain}
        seq_croissante.append(fn)

    assert verify_monotonicity(seq_croissante, domain) == True, "La suite devrait etre detectee croissante"

    # 2. Suite oscillante sin(nx)
    seq_non_croissante = []
    for n in range(1, 10):
        fn = {x: math.sin(n * x) for x in domain}
        seq_non_croissante.append(fn)

    assert verify_monotonicity(seq_non_croissante, domain) == False, "La suite devrait etre detectee NON croissante"

    print("TP 4 : Auditeur de monotonie valide.")

if __name__ == "__main__":
    run_tp4()
```
