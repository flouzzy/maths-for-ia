# TP 04 : Interversion pour une fonction analytique négative


Ce TP valide numériquement le calcul vu en exercice pour $\int_0^1 \frac{-\ln(x)}{1-x} dx$.

```python
import math

def integrate(func, a, b, n_steps=1000):
    step = (b - a) / n_steps
    # On evite 0 pour contourner le log(0)
    return sum(func(a + (i + 0.5) * step) * step for i in range(n_steps))

def func_target(x):
    return -math.log(x) / (1 - x)

def serie_partial(n):
    # Somme des integrales interverties (1/(k+1)^2)
    return sum(1 / (k + 1)**2 for k in range(n))

def test_tcm_tp04():
    a, b = 0.0, 1.0

    int_numeric = integrate(func_target, a, b, 5000)

    # Valeur theorique donnee par le TCM et l'IPP : pi^2 / 6
    valeur_tcm = serie_partial(100)
    valeur_theo = math.pi**2 / 6

    assert abs(int_numeric - valeur_theo) < 5e-2, "Integration numerique defaillante"
    assert abs(valeur_tcm - valeur_theo) < 1e-2, "Somme de la serie defaillante"

    print(f"TP 04: Validation Integrale -ln(x)/(1-x). Num={int_numeric:.4f}, Serie={valeur_tcm:.4f}, Theo={valeur_theo:.4f}")

if __name__ == '__main__':
    test_tcm_tp04()
```
