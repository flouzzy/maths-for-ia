# TP 4 : L'escalier de Lebesgue (Approximation de fonctions mesurables)

**Objectif :** Coder l'approximation canonique par le bas d'une fonction continue à l'aide de fonctions simples étagées.

```python
def f_target(x):
    # Fonction que l'on veut integrer : f(x) = x^3
    return x ** 3

def lebesgue_staircase(n, x, f):
    # Approximation par fonction etagee
    # f_n(x) = k/2^n  si k/2^n <= f(x) < (k+1)/2^n
    val = f(x)
    if val >= n:
        return float(n)

    # Trouver le plancher discret
    resolution = 2 ** n
    k = int(val * resolution)
    return float(k) / resolution

def test_lebesgue_staircase():
    x_test = 0.8
    # 1. Verification de la croissance de l'approximation
    val_n1 = lebesgue_staircase(1, x_test, f_target)
    val_n2 = lebesgue_staircase(2, x_test, f_target)
    val_n3 = lebesgue_staircase(3, x_test, f_target)

    assert val_n1 <= val_n2 <= val_n3, "L'escalier n'est pas croissant"

    # 2. Convergence point par point
    val_target = f_target(x_test)
    val_n10 = lebesgue_staircase(10, x_test, f_target)

    assert abs(val_n10 - val_target) < (1.0 / (2**10)), "L'escalier ne converge pas vers la fonction cible"
    print("TP4: Construction des fonctions simples etagees validee.")

test_lebesgue_staircase()
```
