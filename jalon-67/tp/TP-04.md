# TP 4 : Esperance infinie et series
**Objectif :** Calcul de l'espérance par troncation pour une variable de Pareto.
```python
def density_pareto(x, alpha=1.5):
    # Densite x**(-alpha) sur [1, infini[
    if x < 1:
        return 0
    return alpha * (x ** -(alpha + 1))

def expect_tronc(max_x, N_samples=10000):
    dx = (max_x - 1.0) / N_samples
    esp = 0
    for i in range(N_samples):
        x = 1.0 + i * dx
        esp += x * density_pareto(x) * dx
    return esp

if __name__ == "__main__":
    # L'esperance theorique est 1.5 / (1.5 - 1) = 3
    e_10 = expect_tronc(10)
    e_100 = expect_tronc(100)
    e_1000 = expect_tronc(1000)

    assert e_10 < e_100 < e_1000, "La suite des esperances tronquees est croissante"
    assert abs(e_1000 - 3.0) < 0.1, "L'esperance tronquee approche l'esperance vraie"
    print("TP4 reussi: Application probabiliste valide.")
```
