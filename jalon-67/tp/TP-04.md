# TP 4 : Simulation du processus de Poisson (Application IA/Proba)

```python
import random

def tp_4_poisson(lambda_param, intervalles):
    # En theorie des probas, le TCM garantit que l'esperance d'une somme infinie
    # de variables aleatoires positives est la somme des esperances.

    print("Simulation : Nombre moyen d'evenements par intervalle dt")
    esperance_somme = 0.0
    dt = 1.0 / intervalles

    # Calcul theorique par additivite de l'esperance (via TCM)
    for i in range(intervalles):
        esperance_locale = lambda_param * dt
        esperance_somme += esperance_locale

    print("Esperance totale (somme des petites esperances) =", esperance_somme)
    print("Parametre Lambda theorique global =", lambda_param)
    print("Le TCM permet de passer a la limite avec un nombre infini de petits intervalles.")

if __name__ == "__main__":
    tp_4_poisson(5.0, 1000)
```
