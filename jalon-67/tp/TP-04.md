## TP 4 : Calcul probabiliste d'espérance de variables croissantes

Objectif : Simuler des variables aléatoires (montée d'eau, maximum successif) pour valider empiriquement le TCM appliqué aux probabilités.

\begin{lstlisting}
# Variables aleatoires croissantes et Esperance

import random

def simulate_expectation():
    num_simulations = 10000

    # On tire une suite infinie de nombres dans [0, 1]
    # On etudie le maximum courant : M_n = max(U_1, ..., U_n)
    # Clairement M_n <= M_n+1, donc c'est une suite croissante de VA.
    # lim M_n = 1 presque surement.

    n_values = [1, 2, 5, 10, 50, 100]
    esperances = {n: 0.0 for n in n_values}

    for _ in range(num_simulations):
        courant_max = 0.0
        for i in range(1, 101):
            u = random.random()
            if u > courant_max:
                courant_max = u
            if i in esperances:
                esperances[i] += courant_max

    print("Esperance empirique de M_n :")
    for n in n_values:
        empirique = esperances[n] / num_simulations
        theorique = n / (n + 1) # P(M_n <= x) = x^n -> E(M_n) = n/(n+1)
        print(f"n={n:3} : Empirique = {empirique:.4f}, Theorique = {theorique:.4f}")
        assert abs(empirique - theorique) < 1e-2

    print("\nLa suite des esperances converge bien vers 1 = Esperance(lim M_n).")

if __name__ == "__main__":
    random.seed(42)
    simulate_expectation()
\end{lstlisting}
