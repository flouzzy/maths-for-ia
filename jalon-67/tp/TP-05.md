# TP 05 : Approximation par fonctions étagées

## Objectif
Le theoreme de convergence monotone est utilisé pour prouver que toute fonction mesurable positive peut être approchée par une suite croissante de fonctions étagées. Ce TP implémente cette construction fondamentale.

## Implémentation

```python
import math

def f_cible(x):
    # Fonction à approcher (doit être >= 0)
    return math.sin(x)**2 * x

def s_n(x, n):
    # Definition standard des fonctions etagees s_n de la theorie de Lebesgue
    max_val = n
    if f_cible(x) >= max_val:
        return max_val

    # Decoupage de [0, n] en n*2^n intervalles de taille 1/2^n
    for k in range(n * 2**n):
        seuil_bas = k / (2**n)
        seuil_haut = (k + 1) / (2**n)
        if seuil_bas <= f_cible(x) < seuil_haut:
            return seuil_bas
    return 0.0

def test_approximation_etagee():
    x_test = 2.5 # Point de test
    vrai_valeur = f_cible(x_test)

    valeurs_sn = []
    for n in range(1, 8):
        valeurs_sn.append(s_n(x_test, n))

    print(f"Approximations successives en x={x_test} pour f(x)={vrai_valeur:.4f}")
    for n, val in enumerate(valeurs_sn, 1):
        print(f"n={n}: {val:.4f}")

    # Validations Beppo Levi
    # 1. Croissance
    for i in range(len(valeurs_sn)-1):
        assert valeurs_sn[i] <= valeurs_sn[i+1] + 1e-9, "La construction de Lebesgue echoue (non-croissante)"

    # 2. Convergence
    assert abs(valeurs_sn[-1] - vrai_valeur) < 0.05, "La limite de Lebesgue n'est pas atteinte"

if __name__ == "__main__":
    test_approximation_etagee()
```
