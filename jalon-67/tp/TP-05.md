
# TP 5 : Comportement de perte avec régularisation décroissante

On simule l'exemple de l'IA vu en cours : $\lim \int L(x) e^{-\epsilon|x|} = \int L(x)$. On intègre $L(x) = x^2$ sur $[0, 5]$.

```python
# Validation pure Python, sans modules externes

def L(x):
    return x * x

def reg_func(epsilon, x):
    import math
    return L(x) * math.exp(-epsilon * x)

def integrate(func, eps_arg, a, b, steps=1000):
    w = (b - a) / steps
    area = 0.0
    for i in range(steps):
        x = a + i * w
        area += func(eps_arg, x) * w
    return area

def run_tp5():
    # Integral non régularisée L(x) sur [0, 5]
    # Int_0^5 x^2 dx = [x^3/3] = 125/3 = 41.666...
    val_true = integrate(lambda e, x: L(x), 0, 0, 5, 2000)

    epsilons = [1.0, 0.5, 0.1, 0.01, 0.001]
    results = []

    for eps in epsilons:
        val = integrate(reg_func, eps, 0, 5, 2000)
        results.append(val)

    # La suite des résultats doit s'approcher de val_true
    # Et comme eps décroît, e^{-eps x} croît, donc l'intégrale croît
    for i in range(len(results) - 1):
        assert results[i] < results[i+1], "La suite des integrales régularisées doit croître"

    # Le dernier élément doit être très proche de la vérité
    diff = abs(results[-1] - val_true)
    assert diff < 1.0, f"La limite n'est pas atteinte. diff={diff}"

    print("TP 5 : TCM verifie l'asymptotique de la regularisation de perte.")

if __name__ == "__main__":
    run_tp5()
```
