# TP 5 : Sommation terme à terme d'une série de fonctions exponentielles

```python
def tp_5_serie_exponentielle(n_max):
    # On integre f(x) = exp(x) sur [0, 1] en l'ecrivant comme une somme infinie
    # f(x) = sum(x^k / k!)
    # Le corollaire du TCM justifie l'interversion sum / int

    import math

    integrale_exacte = math.exp(1.0) - 1.0

    somme_integrales = 0.0
    print("Verification de l'interversion Serie-Integrale pour exp(x)")
    print("Valeur theorique :", round(integrale_exacte, 6))

    for k in range(n_max):
        # Integrale de x^k / k! sur [0, 1] est 1 / ((k+1) * k!) = 1 / (k+1)!
        integrale_terme = 1.0 / math.factorial(k + 1)
        somme_integrales += integrale_terme
        if k < 5 or k == n_max - 1:
            print("Terme", k, ": somme partielle =", round(somme_integrales, 6))

    print("Ecart avec theorie :", abs(integrale_exacte - somme_integrales))

if __name__ == "__main__":
    tp_5_serie_exponentielle(10)
```
