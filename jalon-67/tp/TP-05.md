# TP 5 : Fonctions etagées et approx
**Objectif :** Approcher l'intégrale d'une fonction continue par des fonctions étagées.
```python
def f_cible(x):
    return x * x

def etagee(n, x):
    # Approximation par le plancher
    step = 1.0 / n
    k = int(x / step)
    return f_cible(k * step)

def int_etagee(n):
    step = 1.0 / n
    total = 0
    for k in range(n):
        total += f_cible(k * step) * step
    return total

if __name__ == "__main__":
    val = int_etagee(100)
    # L'integrale analytique de x*x sur [0, 1] est 1/3
    assert abs(val - 0.333) < 0.05
    assert int_etagee(10) < int_etagee(100), "La suite des integrales de fonctions etagees doit croitre"
    print("TP5 reussi: Integration de Lebesgue par etagement validee.")
```
