# TP 5 : Simulation d'espérance d'un temps d'arrêt

```python
import random

# X ~ Geom(p)
p = 0.3
def get_X():
    count = 1
    while random.random() > p:
        count += 1
    return count

N_sims = 10000
xs = [get_X() for _ in range(N_sims)]

# Espérance empirique
mean_X = sum(xs)/N_sims

# Somme des P(X >= n)
sum_probs = 0
for n in range(1, 30):
    prob = sum(1 for x in xs if x >= n) / N_sims
    sum_probs += prob

assert abs(mean_X - sum_probs) < 0.1
print('TP 5 Validé')
```
