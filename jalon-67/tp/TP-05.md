# TP 5 : Vérification de la mesure de comptage

### Objectif
Utiliser l'ordinateur pour vérifier l'inversion d'une double série infinie via le TCM, en validant l'égalité numérique des deux ordres de sommation.

### Implémentation Python pur

```python
def a_ij(i, j):
    # A positive series term: 1 / (2^i * 3^j)
    return 1.0 / ((2 ** i) * (3 ** j))

def test_double_sum_tcm():
    limit_inf = 50 # enough to capture the convergence

    # 1. Sum over i first, then j
    sum_i_first = 0.0
    for j in range(limit_inf):
        inner_sum = 0.0
        for i in range(1, limit_inf): # start at 1 to match a specific series, or 0
            inner_sum += a_ij(i, j)
        sum_i_first += inner_sum

    # 2. Sum over j first, then i
    sum_j_first = 0.0
    for i in range(1, limit_inf):
        inner_sum = 0.0
        for j in range(limit_inf):
            inner_sum += a_ij(i, j)
        sum_j_first += inner_sum

    # Analytical theoretical value:
    # sum_{i=1} (1/2)^i = 1
    # sum_{j=0} (1/3)^j = 1 / (1 - 1/3) = 3/2 = 1.5
    # Total = 1 * 1.5 = 1.5

    print(f"Somme (ordre 1) : {sum_i_first:.5f}")
    print(f"Somme (ordre 2) : {sum_j_first:.5f}")

    assert abs(sum_i_first - 1.5) < 1e-4
    assert abs(sum_i_first - sum_j_first) < 1e-8
    print("TP5: Inversion de la mesure de comptage validee par TCM.")

test_double_sum_tcm()

```
