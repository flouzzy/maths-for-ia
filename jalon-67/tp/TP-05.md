## TP 5 : Algorithme de seuillage et approximation simple

Objectif : Écrire un algorithme qui construit explicitement la suite de fonctions étagées croissantes qui converge vers une fonction positive mesurable (construction canonique de Lebesgue).

\begin{lstlisting}
# Approximation d'une fonction f par des fonctions simples croissantes
# C'est la base de la definition de l'integrale de Lebesgue

def f(x):
    # Fonction a integrer : x^2
    return x * x

def s_n(x, n):
    # Construction canonique de Lebesgue
    # On decoupe [0, n] en 2^n.n morceaux
    if f(x) >= n:
        return n

    k = int(f(x) * (2 ** n))
    return k / (2 ** n)

def verify_approximation():
    x_test = 1.5 # f(x) = 2.25
    print(f"Valeur cible f({x_test}) = {f(x_test)}")

    for n in range(1, 10):
        val_sn = s_n(x_test, n)
        print(f"n={n}: s_n({x_test}) = {val_sn}")
        # On verifie la croissance
        if n > 1:
            val_prev = s_n(x_test, n-1)
            assert val_sn >= val_prev

    # Calcul de l'integrale par les s_n (somme de Riemann modifiee)
    # L'integrale de Lebesgue approche l'integrale de Riemann ici.
    print("\nVerification de la croissance p.p. reussie.")

if __name__ == "__main__":
    verify_approximation()
\end{lstlisting}
