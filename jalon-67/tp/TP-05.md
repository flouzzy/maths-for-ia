## TP 5 : Fatou vs Beppo Levi en Python

Simulation pour observer l'inégalité stricte dans le lemme de Fatou sur un espace d'états fini.

```python
def tp5_fatou_simulation():
    # Espace d'etats = {1, 2}, mesure uniforme
    # Suite f_n alterne entre [2, 0] et [0, 2]
    # L'integrale vaut 1 (car (2+0)/2 = 1)
    # liminf f_n vaut [0, 0], integrale vaut 0.

    f = []
    for n in range(10):
        if n % 2 == 0:
            f.append([2, 0])
        else:
            f.append([0, 2])

    int_fn = []
    for fn in f:
        val = 0.5 * fn[0] + 0.5 * fn[1]
        int_fn.append(val)

    liminf_fn = [0, 0] # min entre elements pairs et impairs

    int_liminf = 0.5 * liminf_fn[0] + 0.5 * liminf_fn[1]

    liminf_of_int = int_fn[0] # La suite int_fn est constante egale a 1

    print("Integrale de la liminf :", int_liminf)
    print("Liminf des integrales :", liminf_of_int)

    assert int_liminf < liminf_of_int

tp5_fatou_simulation()
```
