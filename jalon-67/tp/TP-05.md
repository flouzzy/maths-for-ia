## TP 5 : Espérance et Somme Infinie en IA

```python
# Objectif : Illustrer le TCM dans un contexte statistique (IA).
# Calcul de l'espérance d'une somme infinie de variables de Poisson.
# On approche l'espérance de la somme par la somme des espérances.

import random

# Echantillonnage d'une loi de Poisson de paramètre lambda
def sample_poisson(lam):
    L = math.exp(-lam)
    p = 1.0
    k = 0
    while p > L:
        k += 1
        u = random.random()
        p *= u
    return k - 1

# On va considérer une somme de variables de Poisson avec lambda_n = 1/2^n
# La somme theorique des lambda_n pour n>=1 est 1.
# Donc l'espérance de la somme est 1.

n_simulations = 5000
N_max = 15

print("Simulation Espérance de somme infinie :")
somme_des_esperances = 0.0

# Calcul de la somme des espérances empiriques
for n in range(1, N_max + 1):
    lam_n = 1.0 / (2.0 ** n)
    somme_empirique_n = 0.0
    for _ in range(n_simulations):
        somme_empirique_n += sample_poisson(lam_n)
    esperance_n = somme_empirique_n / n_simulations
    somme_des_esperances += esperance_n

print("Somme des esperances partielles (empirique) =", round(somme_des_esperances, 4))

# Calcul de l'espérance de la somme totale
somme_totale_echantillons = 0.0
for _ in range(n_simulations):
    somme_individu = 0.0
    for n in range(1, N_max + 1):
        lam_n = 1.0 / (2.0 ** n)
        somme_individu += sample_poisson(lam_n)
    somme_totale_echantillons += somme_individu

esperance_globale = somme_totale_echantillons / n_simulations
print("Esperance globale de la somme (empirique) =", round(esperance_globale, 4))
print("Valeur theorique attendue = 1.0")
```
