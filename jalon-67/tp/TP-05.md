# TP 5 : Le Lemme de Fatou par Beppo Levi

## Objectif
Implémentation d'une suite (inf) pour valider l'inégalité de Fatou.

## Implémentation en Python

```python
def simulate_tp_5():
    # TP 5: Inegalite de Fatou
    # f_n(x) = x^2 si n pair, x^3 si n impair sur [0, 1]

    num_points = 500
    dx = 1.0 / num_points

    def f(n, x):
        if n % 2 == 0:
            return x**2
        else:
            return x**3

    # liminf f_n(x) = min(x^2, x^3) = x^3 sur [0,1]
    # Integrale de la liminf : integrale de x^3 = 1/4 = 0.25

    # liminf des integrales:
    # int f_n pour pair = 1/3 = 0.333
    # int f_n pour impair = 1/4 = 0.25
    # liminf des integrales = min(1/3, 1/4) = 0.25

    # Fatou: int liminf f_n <= liminf int f_n

    integral_liminf_fn = 0.0
    for i in range(num_points):
        x = i * dx
        val = min(x**2, x**3) # la liminf exacte ici
        integral_liminf_fn += val * dx

    assert abs(integral_liminf_fn - 0.25) < 0.01
    # On valide que 0.25 <= 0.25
    return True

if __name__ == '__main__':
    simulate_tp_5()
    print('TP 5 valide.')
```
