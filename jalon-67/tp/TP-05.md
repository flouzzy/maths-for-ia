# TP 5 : Supremum partiel comme suite croissante

**Objectif :** Programmer la création de la suite g_n = sup(f_k) et vérifier numériquement qu'elle devient croissante.

## Code Python (Pure Python, Zéro Framework)

```python
import random

# Generons une suite de fonctions "bruitées" f_n(x) = (x^2) * random(0, 1) + 1/n
# Elle n'est PAS croissante.
random.seed(42)

def f_n(x, n):
    return (x ** 2) * random.random() + 1.0 / n

def integrate_rectangle(f_vals, a, b, num_steps):
    step = (b - a) / num_steps
    return sum(f_vals) * step

if __name__ == "__main__":
    num_steps = 100
    x_vals = [i / float(num_steps) for i in range(num_steps)]

    N = 10
    # Calcul de f_n pour tous les x
    f_matrix = [[f_n(x, n) for x in x_vals] for n in range(1, N + 1)]

    # Verifions que f n'est pas croissante
    f1_val = f_matrix[0][50]
    f2_val = f_matrix[1][50]
    print(f"Non monotonie: f_1(0.5)={f1_val:.3f}, f_2(0.5)={f2_val:.3f}")

    # Construction de g_n(x) = sup_{1..n} f_k(x)
    g_matrix = []
    for n in range(1, N + 1):
        g_n = []
        for i_x in range(num_steps):
            sup_val = max(f_matrix[k][i_x] for k in range(n))
            g_n.append(sup_val)
        g_matrix.append(g_n)

    # Verifions que g_n EST croissante
    g1_val = g_matrix[0][50]
    g2_val = g_matrix[1][50]
    print(f"Monotonie recuperee: g_1(0.5)={g1_val:.3f}, g_2(0.5)={g2_val:.3f}")

    assert g2_val >= g1_val

```
