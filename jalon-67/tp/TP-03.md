# TP 3 : Simulation Numérique du Théorème de Beppo Levi

Ce TP illustre la convergence monotone en calculant numériquement l'intégrale d'une suite croissante de fonctions.

```python
# TP 3 - Implementation from scratch
import math

def f_n(x, n):
    # Suite croissante de fonctions: f_n(x) = x * (1 - 1/n) (variante pour le TP 3)
    return (x**3) * (1.0 - 1.0/n)

def riemann_sum(f, n, a, b, num_steps):
    step = (b - a) / num_steps
    total = 0.0
    for j in range(num_steps):
        x = a + j * step
        total += f(x, n) * step
    return total

a = 0.0
b = 1.0
num_steps = 1000

print("--- Simulation for n values ---")
for n in [2, 10, 100, 1000]:
    integral_n = riemann_sum(f_n, n, a, b, num_steps)
    print(f"n={n}, Integral = {integral_n:.5f}")

# Expected limit for f(x) = x**3 on [0, 1] is 1/(3+1)
expected_limit = 1.0 / (3 + 1.0)
print(f"Expected limit (n -> infinity): {expected_limit:.5f}")
```
