
# TP 3 : Corollaire de sommation sur une mesure discrète

En Machine Learning, calculer une espérance sur un domaine fini ou dénombrable s'appuie sur le corollaire du TCM. Ici, on vérifie l'interversion de deux boucles (la somme infinie et l'espérance).

```python
# Validation pure Python, sans modules externes

def prob_n(n):
    # Distribution géométrique p(n) = (1/2)^{n+1}
    return (0.5) ** (n + 1)

def term_k(k, n):
    # Une variable aléatoire X_k(n) qui dépend du tirage n
    # On simule sum_{k} E[X_k] = E[sum_{k} X_k]
    # Posons X_k(n) = (n / (n + 1)) ** k
    return (float(n) / float(n + 1)) ** k

def run_tp3():
    # Côté gauche : sum_k ( E_n [ X_k(n) ] )
    left_side = 0.0
    for k in range(1, 50): # troncature
        # Esperance E_n [ X_k(n) ]
        esp_xk = 0.0
        for n in range(50):
            esp_xk += prob_n(n) * term_k(k, n)
        left_side += esp_xk

    # Côté droit : E_n [ sum_k X_k(n) ]
    right_side = 0.0
    for n in range(50):
        # somme interieure
        sum_x = 0.0
        for k in range(1, 50):
            sum_x += term_k(k, n)
        # Esperance
        right_side += prob_n(n) * sum_x

    diff = abs(left_side - right_side)
    assert diff < 1e-5, f"L'interversion est invalide! diff={diff}"
    print(f"TP 3 : Interversion validee sur distribution discrete (diff = {diff:.8f}).")

if __name__ == "__main__":
    run_tp3()
```
