# TP 03 : Le Lemme de Fatou en Python

Le code Python pure suivant démontre algorithmiquement le concept étudié.

```python
import math

# Suite oscillante: f_n(x) = 1 + sin(n * x)
# Liminf est 0
def f_oscillating(x, n):
    return 1 + math.sin(n * x)

def integrate_riemann(func, n, a, b, num_steps=1000):
    step_size = (b - a) / num_steps
    total = 0.0
    for i in range(num_steps):
        x_mid = a + (i + 0.5) * step_size
        total += func(x_mid, n) * step_size
    return total

# Calcul de \int f_n sur [0, 2pi]
n_vals = [10, 20, 30]
integrals = [integrate_riemann(f_oscillating, n, 0, 2 * math.pi) for n in n_vals]

# int(1 + sin(nx)) sur un multiple de 2pi/n est 2pi.
# Donc la liminf de l'intégrale est 2pi.
liminf_int = min(integrals)

# L'intégrale de la liminf (qui est 0 ponctuellement)
int_liminf = 0.0

assert int_liminf <= liminf_int
print("TP 3 Validé : l'inégalité de Fatou est respectée :", int_liminf, "<=", liminf_int)
```
