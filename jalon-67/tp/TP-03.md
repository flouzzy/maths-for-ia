# TP 3 : Approximation numérique d'une intégrale par fonctions étagées

```python
def tp_3_fonctions_etagees(resolution):
    # On approche f(x) = x^2 sur [0, 1] par des fonctions etagees croissantes f_n
    # f_n(x) = (floor(n * x) / n)^2

    def integrale_f_n(n):
        somme = 0.0
        largeur_palier = 1.0 / n
        for k in range(n):
            valeur = (k / n) ** 2
            somme += valeur * largeur_palier
        return somme

    print("Convergence des integrales de fonctions etagees vers l'integrale de x^2 (1/3)")
    for n in [10, 50, 100, 500]:
        approx = integrale_f_n(n)
        erreur = abs((1.0/3.0) - approx)
        print("n =", n, "Approx =", round(approx, 5), "Erreur =", round(erreur, 5))

if __name__ == "__main__":
    tp_3_fonctions_etagees(100)
```
