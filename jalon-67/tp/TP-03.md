# TP 3 : Intégration terme à terme d'une série entière

**Objectif :** Utiliser Python pour valider l'interversion d'une série entière infinie en exploitant le corollaire du TCM.

```python
def term_u(n, t):
    # u_n(t) = t^n
    return t ** n

def integrate_term(n, x):
    # int_0^x t^n dt = (x^(n+1))/(n+1)
    return (x ** (n + 1)) / (n + 1)

def test_series_integration():
    x = 0.5 # t dans [0, 0.5], ce qui rend les termes positifs

    # Calcul direct de l'integrale de la somme : int_0^x (1/(1-t)) dt = -ln(1-x)
    import math
    exact_integral_of_sum = -math.log(1 - x)

    # Calcul de la somme des integrales : sum_0^N (x^(n+1))/(n+1)
    sum_of_integrals = 0.0
    N = 100
    for n in range(N):
        sum_of_integrals += integrate_term(n, x)

    assert abs(sum_of_integrals - exact_integral_of_sum) < 1e-10, "L'interversion serie-integrale a echoue numériquement."
    print("TP3: Interversion Somme-Integrale pour série positive validee.")

test_series_integration()
```
