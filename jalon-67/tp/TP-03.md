# TP 03 : Contre-exemple sans positivité

## Objectif
Construire une suite de fonctions (f_n) qui converge simplement vers 0, mais dont la limite des intégrales n'est pas nulle, mettant en evidence que le theoreme est faux sans positivite ou monotonie.

## Implémentation

def f_n(x, n):
    # La fameuse fonction "bosse glissante"
    if 0 <= x <= 1/n:
        return n**2 * x  # monte
    elif 1/n < x <= 2/n:
        return n - n**2 * (x - 1/n) # descend
    return 0.0

def test_bosse_glissante():
    # Pour un x fixe positif, f_n(x) finit toujours par valoir 0
    # On teste pour x = 0.5
    x_test = 0.5
    valeurs_x = []
    for n in range(1, 10):
        valeurs_x.append(f_n(x_test, n))

    print("Evolution ponctuelle en x=0.5 :", valeurs_x)
    assert valeurs_x[-1] == 0, "La suite ne converge pas vers 0 ponctuellement"

    # Aire = (2/n * n) / 2 = 1. L'integrale vaut toujours 1.

        return 1.0

    limite_integrales = 1.0
    integrale_limite = 0.0 # car la limite ponctuelle est la fonction nulle

    assert limite_integrales != integrale_limite, "Les limites devraient differer"
    print("Contre-exemple valide : interversion interdite.")
