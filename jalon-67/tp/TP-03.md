# TP 3 : Interversion série-intégrale numériquement

**Objectif :** Calculer de deux manières l'intégrale d'une série entière convergente pour illustrer le corollaire.

## Code Python (Pure Python, Zéro Framework)

```python
def serie_terme_n(x, n):
    # u_n(x) = x^n
    return x ** n

def integrate_terme(n, a, b, num_steps):
    step = (b - a) / num_steps
    total = 0.0
    for i in range(num_steps):
        x = a + i * step
        total += serie_terme_n(x, n) * step
    return total

if __name__ == "__main__":
    num_steps = 1000
    n_max = 50

    # Somme des integrales
    somme_des_integrales = 0.0
    for n in range(1, n_max): # n=1 a n_max
        somme_des_integrales += integrate_terme(n, 0, 1, num_steps)

    # La valeur exacte de somme(1/(n+1)) n'est pas bornee (diverge),
    # mais calculons jusqu'a n_max pour voir la correspondance.
    print("Somme des integrales jusqu'a n=50:", somme_des_integrales)

    # Integrale de la somme (tronquee)
    integrale_de_la_somme = 0.0
    step = 1.0 / num_steps
    for i in range(num_steps):
        x = i * step
        somme_partielle = sum(serie_terme_n(x, n) for n in range(1, n_max))
        integrale_de_la_somme += somme_partielle * step

    print("Integrale de la somme tronquee:", integrale_de_la_somme)

    # Assertion d'egalite (aux erreurs numeriques pres)
    assert abs(somme_des_integrales - integrale_de_la_somme) < 0.1

```
