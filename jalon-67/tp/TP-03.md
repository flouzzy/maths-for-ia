## TP 3 : Interversion somme-intégrale

```python
# Objectif : Vérifier numériquement le corollaire du TCM
# sur l'interversion somme et intégrale.
# int_0^1 (-ln(x)/(1-x)) dx = sum_1^infty 1/n^2 = pi^2 / 6

import math

def terme_n(x, n):
    if x == 0 or x == 1:
        return 0.0
    return - (x ** n) * math.log(x)

def fonction_limite(x):
    if x == 0 or x == 1:
        return 0.0
    return - math.log(x) / (1 - x)

def integrate(func, a, b, steps, n_val=None):
    width = (b - a) / steps
    total = 0.0
    for i in range(steps):
        x = a + i * width + (width / 2.0)  # point milieu pour eviter les bornes
        if n_val is not None:
            total += func(x, n_val) * width
        else:
            total += func(x) * width
    return total

a, b = 0.0, 1.0
steps = 2000

integrale_totale = integrate(fonction_limite, a, b, steps)
print("Integrale globale directe = ", round(integrale_totale, 5))

somme_integrales = 0.0
for n in range(20):
    I_n = integrate(terme_n, a, b, steps, n)
    somme_integrales += I_n
    if n % 5 == 0 or n == 19:
        print("Somme partielle k=0 a", n, "des integrales =", round(somme_integrales, 5))

pi_carre_sur_6 = (math.pi ** 2) / 6.0
print("Valeur theorique pi^2/6 = ", round(pi_carre_sur_6, 5))
```
