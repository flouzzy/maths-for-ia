# TP 3 : Application IA - Sommation d'Espérances

### Objectif
Utiliser le corollaire du TCM sur les séries pour calculer l'espérance marginale dans un modèle de mélange infini (type GMM).

### Implémentation Python pur

```python
import math

def poisson_prob(k, lam):
    return (lam**k * math.exp(-lam)) / math.factorial(k)

def gaussian_density(x, mu, sigma):
    return (1.0 / (sigma * math.sqrt(2 * math.pi))) * math.exp(-0.5 * ((x - mu) / sigma) ** 2)

def test_infinite_mixture_expectation():
    # We have an infinite mixture of Gaussians.
    # The weight of the k-th component is P(K=k) for K ~ Poisson(lam=2.0)
    # The k-th component is Normal(mu=k, sigma=1.0)
    # We want to find the overall expected value E[X].
    # By TCM (Fubini/Beppo-Levi for series), E[X] = sum( P(K=k) * E[X|K=k] )

    lam = 2.0

    # 1. Theoretical calculation using Beppo-Levi
    expected_value_tcm = 0.0
    for k in range(50): # up to infinity approximation
        weight = poisson_prob(k, lam)
        mean_k = float(k)
        expected_value_tcm += weight * mean_k

    # The sum of k * P(K=k) is exactly the mean of Poisson, which is lam.
    assert abs(expected_value_tcm - lam) < 1e-5

    # 2. Numerical integration of the mixture density over x
    def mixture_pdf(x):
        density = 0.0
        for k in range(30):
            density += poisson_prob(k, lam) * gaussian_density(x, float(k), 1.0)
        return density

    # E[X] = Integral( x * pdf(x) dx )
    def expected_integrand(x):
        return x * mixture_pdf(x)

    # Integrate over [-10, 20] which covers most of the mass
    a, b = -10.0, 20.0
    steps = 10000
    step_size = (b - a) / steps
    numerical_expected_value = 0.0
    for i in range(steps):
        x = a + (i + 0.5) * step_size
        numerical_expected_value += expected_integrand(x) * step_size

    print(f"Esperance theorique (TCM): {expected_value_tcm:.5f}")
    print(f"Esperance empirique (Integral): {numerical_expected_value:.5f}")
    assert abs(expected_value_tcm - numerical_expected_value) < 0.01
    print("TP3: Interversion Somme-Integrale validee pour un modele IA.")

test_infinite_mixture_expectation()

```
