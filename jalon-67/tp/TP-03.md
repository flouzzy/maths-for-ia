## TP 3 : Interversion série-intégrale numériquement

Objectif : Coder le calcul d'une intégrale par le développement en série entière de son intégrande, justifié par le corollaire du TCM.

\begin{lstlisting}
# Interversion d'une serie de fonctions et de l'integrale
# Integrale de -ln(x)/(1-x) = somme(1/n**2)

import math

def integrate_midpoint(f, a, b, n=1000):
    h = (b - a) / n
    total = 0.0
    for i in range(n):
        x = a + (i + 0.5) * h
        total += f(x) * h
    return total

def test_series_integral():
    # 1. Calcul de l'integrale directement
    # On evite 0 et 1 car il y a des singularites
    def integrand(x):
        return -math.log(x) / (1 - x)

    integral_val = integrate_midpoint(integrand, 1e-5, 1-1e-5, 10000)
    print(f"Valeur de l'integrale par quadrature : {integral_val:.5f}")

    # 2. Calcul par sommation terme a terme de la serie
    # Serie de Taylor : 1/(1-x) = sum(x**k)
    # L'integrale de -x**k * ln(x) est 1/(k+1)**2

    somme = 0.0
    n_terms = 1000
    for k in range(n_terms):
        somme += 1.0 / ((k + 1) ** 2)

    print(f"Valeur de la somme infinie (N={n_terms}) : {somme:.5f}")

    # 3. Valeur theorique zeta(2) = pi**2 / 6
    theorique = (math.pi ** 2) / 6
    print(f"Valeur theorique (zeta(2)) : {theorique:.5f}")

    assert abs(integral_val - theorique) < 1e-2
    assert abs(somme - theorique) < 1e-3

if __name__ == "__main__":
    test_series_integral()
\end{lstlisting}
