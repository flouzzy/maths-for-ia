# TP 3 : Beppo Levi et Séries de probabilités

## Objectif
Calcul d'espérance sur une somme infinie de variables de Poisson.

## Implémentation en Python

```python
def simulate_tp_3():
    # TP 3: Esperance d'une somme infinie
    # E[sum(X_i)] = sum(E[X_i]) par Beppo Levi (termes positifs)

    import random

    # Simuler des variables de Poisson de param lambda_i = (1/2)^i
    n_terms = 10
    n_samples = 10000

    def simulate_poisson(lam):
        # Simulation basique d'une loi de Poisson
        import math
        L = math.exp(-lam)
        p = 1.0
        k = 0
        while p > L:
            k += 1
            p *= random.random()
        return k - 1

    total_sum_samples = []
    for _ in range(n_samples):
        sample_sum = 0
        for i in range(1, n_terms + 1):
            lam = (0.5) ** i
            sample_sum += simulate_poisson(lam)
        total_sum_samples.append(sample_sum)

    empirical_expectation = sum(total_sum_samples) / n_samples

    # Valeur theorique sum((1/2)^i) = 1
    theoretical_expectation = sum([(0.5)**i for i in range(1, n_terms + 1)])

    assert abs(empirical_expectation - theoretical_expectation) < 0.1
    return True

if __name__ == '__main__':
    simulate_tp_3()
    print('TP 3 valide.')
```
