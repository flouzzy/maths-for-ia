# TP 3 : Interversion serie-integrale
**Objectif :** Simuler la sommation terme à terme de fonctions positives.
```python
def u_n(n, x):
    # Terme d'une somme infinie: x**n
    return x**n

def sum_integral(N_max, N_samples=1000):
    dx = 1.0 / N_samples
    total_integral = 0
    # Limite des sommes partielles
    for i in range(N_samples):
        x = i * dx
        val_x = sum(u_n(n, x) for n in range(1, N_max))
        total_integral += val_x * dx
    return total_integral

if __name__ == "__main__":
    # La somme diverge vers l'infini lentement (serie harmonique)
    val_10 = sum_integral(10)
    val_100 = sum_integral(100)

    assert val_10 < val_100, "L'integrale doit croitre"
    assert val_100 > 2.0, "La serie doit diverger vers l'infini"
    print("TP3 reussi: Interversion calculee.")
```
