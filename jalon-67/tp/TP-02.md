# TP 2 : Le contre-exemple de la "bosse glissante" non croissante

```python
def evaluer_bosse(x, n):
    # f_n(x) = n si 0 < x < 1/n, 0 sinon
    if 0 < x and x < 1.0/n:
        return n
    return 0

def tp_2_bosse_glissante(n):
    print("Analyse de f_" + str(n))

    # Verification de la decroissance ponctuelle
    x_test = 0.5
    f_1 = evaluer_bosse(x_test, 1)
    f_2 = evaluer_bosse(x_test, 2)
    print("Pour x =", x_test, ", f_1(x) =", f_1, "et f_2(x) =", f_2)
    if f_1 > f_2:
        print("La suite n'est pas croissante ! Le TCM ne s'applique pas.")

    # L'integrale analytique de f_n est la base (1/n) * hauteur (n) = 1
    integrale = 1.0
    print("Integrale theorique de f_" + str(n), "=", integrale)
    print("Limite des integrales = 1")
    print("Integrale de la limite (qui est 0 partout) = 0")
    print("1 != 0 : Le theoreme echoue car l'hypothese de croissance n'est pas respectee.")

if __name__ == "__main__":
    tp_2_bosse_glissante(5)
```
