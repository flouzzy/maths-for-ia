# TP 2 : Série harmonique et convergence à l'infini

Vérifions empiriquement l'exercice 1 : l'intégration terme à terme d'une série divergente.

\begin{lstlisting}[language=Python]
# TP2: Somme infinie de monomes x^n sur [0, 1]

def integrale_x_pow_n(n):
    # Valeur exacte de int(x^n) sur [0, 1]
    return 1.0 / (n + 1.0)

def approx_serie_integrales(k_max):
    s = 0.0
    for n in range(1, k_max + 1):
        s += integrale_x_pow_n(n)
    return s

# On montre la croissance (divergence lente vers +infini)
print("S(10):", approx_serie_integrales(10))
print("S(100):", approx_serie_integrales(100))
print("S(1000):", approx_serie_integrales(1000))

# Validation algorithmique de la divergence : S(1000) doit etre superieur a S(10)
assert approx_serie_integrales(1000) > approx_serie_integrales(10)
\end{lstlisting}
