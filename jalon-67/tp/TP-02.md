## TP 2 : Illustration de la condition de croissance

```python
# Objectif : Montrer ce qui se passe quand la suite de fonctions
# n'est PAS croissante, provoquant l'échec de la convergence des intégrales
# vers l'intégrale de la limite (Perte de masse).
# fn(x) = (1/n) * indicatrice_[0, n](x)
# f_limite(x) = 0.

def f_n(x, n):
    if x >= 0 and x <= n:
        return 1.0 / n
    return 0.0

def f_limite(x):
    return 0.0

def integrate(func, a, b, steps, n_val=None):
    width = (b - a) / steps
    total = 0.0
    for i in range(steps):
        x = a + i * width
        if n_val is not None:
            total += func(x, n_val) * width
        else:
            total += func(x) * width
    return total

steps = 10000
print("Illustration du contre-exemple sans croissance :")
# L'intervalle doit grandir avec n pour capturer toute l'intégrale
for n in [10, 50, 100, 500]:
    integral_n = integrate(f_n, 0.0, n + 1.0, steps, n)
    print("n =", n, "-> integrale =", round(integral_n, 5))

integral_limite = integrate(f_limite, 0.0, 1000.0, steps)
print("Integrale de la limite =", round(integral_limite, 5))
print("Conclusion : La masse 1 s'echappe vers l'infini.")
```
