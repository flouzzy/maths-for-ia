**Objectif :** Montrer numériquement pourquoi le TCM échoue si la suite n'est pas croissante (la bosse glissante).

def sliding_bump(n, x):
    # La bosse est un creneau de largeur 1 centre sur n
    if n <= x < n + 1:
        return 1.0
    return 0.0

def exact_integral_bump(n):
    # L'aire du creneau est toujours 1.0
    return 1.0

def pointwise_limit(x):
    # Pour n'importe quel x fini, si n > x, la bosse est passee, donc f(x) = 0
    return 0.0

def test_sliding_bump():
    # 1. Verifier que la suite n'est pas croissante
    x_test = 2.5
    val_n2 = sliding_bump(2, x_test)  # 1.0
    val_n3 = sliding_bump(3, x_test)  # 0.0
    assert val_n2 > val_n3, "La suite ne devrait pas etre croissante"

    # 2. Verifier l'integrale limite
    # lim_n int f_n
    limit_of_integrals = 1.0 # Toujours 1.0

    # int lim_n f_n
    # f_n -> 0 partout, donc int 0 = 0
    integral_of_limit = 0.0

    assert limit_of_integrals != integral_of_limit
    print("TP2: Contre-exemple de la bosse glissante valide. L'hypothese de croissance est strictement necessaire.")

test_sliding_bump()
