# TP 02 : Intégration terme à terme d'une série

## Objectif
Le corollaire de Beppo Levi stipule que l'on peut intervertir série et intégrale pour des termes positifs. Nous allons le vérifier sur la série geometrique x^n.

## Implémentation

```python
def terme_serie(x, n):
    return x**n

def somme_partielle(x, N):
    return sum(terme_serie(x, n) for n in range(N+1))

def integrate_term_by_term(a, b, N, n_steps=5000):
    dx = (b - a) / n_steps
    somme_des_integrales = 0.0
    for n in range(N+1):
        # Integrale du terme x^n
        integrale_n = 0.0
        for i in range(n_steps):
            x = a + i * dx + dx/2 # milieu
            integrale_n += terme_serie(x, n) * dx
        somme_des_integrales += integrale_n
    return somme_des_integrales

def integrate_sum(a, b, N, n_steps=5000):
    dx = (b - a) / n_steps
    integrale_de_la_somme = 0.0
    for i in range(n_steps):
        x = a + i * dx + dx/2
        integrale_de_la_somme += somme_partielle(x, N) * dx
    return integrale_de_la_somme

def test_interversion():
    a, b = 0, 0.99
    N = 50
    val1 = integrate_term_by_term(a, b, N)
    val2 = integrate_sum(a, b, N)

    print(f"Somme des integrales = {val1}")
    print(f"Integrale de la somme = {val2}")

    assert abs(val1 - val2) < 1e-5, "Echec de l'interversion de Beppo Levi"

if __name__ == "__main__":
    test_interversion()
```
