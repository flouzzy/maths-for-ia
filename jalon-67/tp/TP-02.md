# TP 2 : Contre-exemple au TCM (Suite non croissante)

### Objectif
Démontrer numériquement qu'une suite de fonctions non croissante peut violer l'égalité du Théorème de Convergence Monotone.

### Implémentation Python pur

```python
def f_n_counter(x, n):
    # f_n(x) = (1/n) * indicator(0 <= x <= n)
    return 1.0 / n if 0 <= x <= n else 0.0

def f_lim_counter(x):
    return 0.0

def riemann_sum(func, a, b, num_steps, n_val=None):
    step = (b - a) / num_steps
    total = 0.0
    for i in range(num_steps):
        x = a + (i + 0.5) * step
        if n_val is not None:
            total += func(x, n_val) * step
        else:
            total += func(x) * step
    return total

def test_tcm_failure():
    # Int(f_lim) is obviously 0
    int_limit = riemann_sum(f_lim_counter, 0, 100, 1000)

    # Let's calculate Int(f_n) for various n
    n_values = [10, 50, 100, 500]
    for n in n_values:
        # Evaluate over an interval large enough to capture the support [0, n]
        val = riemann_sum(f_n_counter, 0, n + 10, int(100 * (n + 10)), n_val=n)
        # Theoretically the integral is n * (1/n) = 1
        assert abs(val - 1.0) < 0.01

    print(f"L'integrale de la limite est {int_limit:.5f}")
    print("La limite des integrales est 1.0")
    print("L'egalite est violee car la suite de fonctions decroit. TP2 reussi.")

test_tcm_failure()

```
