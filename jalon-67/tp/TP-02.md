# TP 2 : Mesure empirique du défaut de convergence non monotone

## Objectif
Mettre en évidence le fait que sans monotonie, l'interversion est fausse.

## Implémentation en Python

```python
def simulate_tp_2():
    # TP 2: Contre-exemple si non monotonie
    # f_n(x) = n * x * (1 - x^2)^n sur [0, 1]
    # f_n converge ponctuellement vers 0, donc integrale de la limite est 0
    # Mais limite de l'integrale ne tend pas forcement vers 0 si non-monotone

    n_values = [10, 50, 100, 500]
    num_points = 2000
    dx = 1.0 / num_points

    integrals = []

    for n in n_values:
        integral_fn = 0.0
        for i in range(num_points):
            x = i * dx
            val = n * x * ((1.0 - x * x) ** n)
            integral_fn += val * dx
        integrals.append(integral_fn)

    # Analytiquement l'integrale vaut n / (2*(n+1)) qui tend vers 0.5, pas 0!
    for val in integrals:
        assert abs(val - 0.5) < 0.05

    return True

if __name__ == '__main__':
    simulate_tp_2()
    print('TP 2 valide.')
```
