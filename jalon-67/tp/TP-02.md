# TP 2 : Vérification du contre-exemple de la masse fuyante

**Objectif :** Simuler une suite de fonctions non croissantes où Beppo Levi échoue (masse constante, limite nulle).

## Code Python (Pure Python, Zéro Framework)

```python
def f_n_non_croissante(x, n):
    # f_n = n * 1_{]0, 1/n]}
    if 0 < x <= 1.0 / n:
        return n
    return 0.0

def integrate_rectangle(f, n, a, b, num_steps):
    step = (b - a) / num_steps
    total = 0.0
    for i in range(num_steps):
        x = a + i * step
        total += f(x, n) * step
    return total

if __name__ == "__main__":
    num_steps = 100000

    integrals = []
    for n in range(1, 5):
        val = integrate_rectangle(f_n_non_croissante, n, 0, 1, num_steps)
        integrals.append(val)

    print("Integrales (devraient etre autour de 1.0) :", integrals)

    # Verifions que la limite simple f_n(x) tend vers 0
    x_test = 0.5
    limite_ponctuelle = [f_n_non_croissante(x_test, n) for n in range(1, 10)]
    print("Valeurs de f_n(0.5) :", limite_ponctuelle)

    # Assertion
    assert abs(integrals[0] - 1.0) < 0.01
    assert limite_ponctuelle[-1] == 0

```
