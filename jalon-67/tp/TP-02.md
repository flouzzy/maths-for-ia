## TP 2 : Simulation du phénomène de bosse glissante (Contre-exemple)

Objectif : Montrer algorithmiquement pourquoi une suite qui n'est pas croissante peut faire échouer l'interversion limite/intégrale, illustrant l'importance cruciale de l'hypothèse de croissance dans Beppo Levi.

\begin{lstlisting}
# Contre-exemple : La bosse glissante
# fn(x) = n pour x in ]0, 1/n[ et 0 ailleurs

def integrate_step(f, a, b, num_steps=5000):
    step = (b - a) / num_steps
    area = 0.0
    for i in range(num_steps):
        x = a + (i + 0.5) * step # Point milieu
        area += f(x) * step
    return area

def simulate_sliding_bump():
    def fn(x, n):
        if 0 < x < 1/n:
            return n
        return 0

    print("Calcul des integrales des fn :")
    for n in [1, 2, 5, 10, 100]:
        val = integrate_step(lambda x: fn(x, n), 0, 1)
        print(f"n={n:3} : integrale = {val:.5f} (theorique = 1.0)")
        assert abs(val - 1.0) < 1e-2

    # Calcul de la limite simple
    # Pour tout x > 0, si n > 1/x, fn(x) = 0
    def limit_f(x):
        return 0.0

    val_limite = integrate_step(limit_f, 0, 1)
    print(f"\nIntegrale de la limite = {val_limite:.5f}")

    print("Conclusion: La limite des integrales (1) != l'integrale de la limite (0).")
    print("L'hypothese de croissance de Beppo Levi n'etait pas respectee.")

if __name__ == "__main__":
    simulate_sliding_bump()
\end{lstlisting}
