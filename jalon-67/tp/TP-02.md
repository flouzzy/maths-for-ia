# TP 2 : Echec du TCM sans croissance
**Objectif :** Montrer numériquement la fuite de masse (suite non croissante).
```python
def g_n(n, x):
    # f_n(x) = n * 1_]0, 1/n[ (x). Ne respecte pas la monotonie!
    if 0 < x < 1/n:
        return n
    return 0

def integral_g(n, N_samples=100000):
    dx = 1.0 / N_samples
    total = 0
    for i in range(N_samples):
        x = i * dx
        total += g_n(n, x) * dx
    return total

if __name__ == "__main__":
    val_5 = integral_g(5)
    val_50 = integral_g(50)

    # La limite de l'integrale est 1
    assert abs(val_5 - 1.0) < 0.05
    assert abs(val_50 - 1.0) < 0.05

    # Mais la limite ponctuelle est 0
    assert g_n(100, 0.5) == 0
    print("TP2 reussi: Masse fuyante simulee.")
```
