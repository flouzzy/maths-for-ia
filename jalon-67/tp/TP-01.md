# TP 1 : Validation numérique de l'interversion Somme/Intégrale

## Objectif
Calculer l'intégrale d'une série de fonctions positives.

## Implémentation en Python

```python
def simulate_tp_1():
    # TP 1: Somme infinie vs Integrale
    # On calcule l'integrale de sum(x^n) sur [0, 1-epsilon] (pour eviter la divergence a 1)
    # L'integrale de la somme doit etre egale a la somme des integrales.
    import math

    epsilon = 0.05
    upper_bound = 1.0 - epsilon
    n_terms = 100
    num_points = 1000
    dx = upper_bound / num_points

    # Methode 1: Somme des integrales
    sum_of_integrals = 0.0
    for n in range(1, n_terms + 1):
        # Integrale analytique de x^n est x^(n+1)/(n+1)
        integral_n = (upper_bound**(n+1)) / (n+1)
        sum_of_integrals += integral_n

    # Methode 2: Integrale de la somme numerique
    integral_of_sum = 0.0
    for i in range(num_points):
        x = i * dx
        # Valeur de la somme infinie x/(1-x) au point x
        sum_at_x = x / (1.0 - x)
        integral_of_sum += sum_at_x * dx

    # La fonction analytique exacte de l'integrale de x/(1-x) est -x - ln(1-x)
    exact_val = -upper_bound - math.log(1 - upper_bound)

    assert abs(sum_of_integrals - exact_val) < 0.01
    assert abs(integral_of_sum - exact_val) < 0.05
    return True

if __name__ == '__main__':
    simulate_tp_1()
    print('TP 1 valide.')
```
