## TP 1 : Vérification numérique basique de Beppo Levi

Ce TP simule l'intégrale de la suite de fonctions croissantes $f_n(x) = 1 - (1 - x/n)^n$ sur l'intervalle $[0, 1]$.

```python
def tp1_beppo_levi_simulation():
    steps = 2000
    dx = 1.0 / steps

    # f(x) = 1 - e^(-x) as limit of f_n(x)
    limit_integral = 0
    for s in range(steps):
        x = s * dx

        # Taylor approx to e^-x to avoid importing math inside the strict sandbox,
        # but for pure python we can just implement a simple series or use ** operator carefully.
        # e^-x = 1 - x + x^2/2 - x^3/6
        e_neg_x = 1.0
        term = 1.0
        for i in range(1, 15):
            term = term * (-x) / i
            e_neg_x += term

        val = 1.0 - e_neg_x
        limit_integral += val * dx

    integral_n = []
    n_values = [5, 10, 50, 100, 500]
    for n in n_values:
        current = 0
        for s in range(steps):
            x = s * dx
            val = 1.0 - (1.0 - x/n)**n
            current += val * dx
        integral_n.append(current)

    print("Integrales f_n:", integral_n)
    print("Integrale de la limite f:", limit_integral)
    assert integral_n[-1] <= limit_integral
    assert (limit_integral - integral_n[-1]) < 0.01

tp1_beppo_levi_simulation()
```
