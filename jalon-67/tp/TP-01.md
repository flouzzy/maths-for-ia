# TP 01 : Vérification empirique de Beppo Levi sur une série

Le code Python pure suivant démontre algorithmiquement le concept étudié.

```python
def fn(x, N):
    # f_N(x) = sum_{k=1}^N (x / 2)^k
    result = 0
    for k in range(1, N + 1):
        result += (x / 2.0) ** k
    return result

def integrate_riemann(func, N, a, b, num_steps=1000):
    step_size = (b - a) / num_steps
    total = 0.0
    for i in range(num_steps):
        x_mid = a + (i + 0.5) * step_size
        total += func(x_mid, N) * step_size
    return total

# 1. Vérification de la croissance
for x_test in [0.2, 0.5, 0.9]:
    assert fn(x_test, 5) <= fn(x_test, 6), "La suite doit être croissante !"

# 2. Convergence de l'intégrale
integrals = [integrate_riemann(fn, N, 0, 1) for N in range(1, 15)]

# L'intégrale de la limite (somme de la série géométrique x/2 / (1 - x/2))
def f_lim(x, _):
    return (x / 2.0) / (1 - x / 2.0)

limit_integral = integrate_riemann(f_lim, 0, 0, 1)

print("Intégrale de f_14 :", integrals[-1])
print("Intégrale de la limite :", limit_integral)
assert abs(integrals[-1] - limit_integral) < 0.1, "TCM échoué"
print("TP 1 Validé !")
```
