# TP 1 : Illustration du TCM par la série harmonique

```python
def tp_1_serie_harmonique(max_n):
    # On calcule l'intégrale de f_n(x) = sum(x^k) pour x dans [0, 1]
    # L'intégrale exacte est sum(1/(k+1))

    print("Verification de la divergence de la serie harmonique via TCM")
    somme_integrale = 0.0

    for n in range(1, max_n + 1):
        integrale_terme = 1.0 / n
        somme_integrale += integrale_terme
        if n % 10 == 0:
            print("n =", n, "Integrale de f_n :", round(somme_integrale, 4))

    return somme_integrale

if __name__ == "__main__":
    result = tp_1_serie_harmonique(50)
    print("Apres 50 termes, l'integrale vaut approximativement :", round(result, 4))
    print("La theorie (TCM) predit que cette valeur diverge vers l'infini.")
```
