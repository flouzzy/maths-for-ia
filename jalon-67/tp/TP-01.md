## TP 1 : Vérification numérique naïve du TCM

```python
# Objectif : Vérifier numériquement le Théorème de Convergence Monotone
# sur la suite fn(x) = (1 - x/n)^n * indicatrice_[0, n](x) sur [0, 50].
# On sait que la limite est f(x) = exp(-x).

import math

def f_n(x, n):
    if x >= 0 and x <= n:
        return (1 - x / n) ** n
    return 0.0

def f_limite(x):
    return math.exp(-x)

# Intégration numérique par méthode des rectangles
def integrate(func, a, b, steps, n_val=None):
    width = (b - a) / steps
    total = 0.0
    for i in range(steps):
        x = a + i * width
        if n_val is not None:
            total += func(x, n_val) * width
        else:
            total += func(x) * width
    return total

a, b = 0.0, 20.0
steps = 5000

print("Verification du Theoreme de Convergence Monotone :")
for n in [1, 2, 5, 10, 20, 50, 100]:
    integral_n = integrate(f_n, a, b, steps, n)
    print("n =", n, "-> integrale =", round(integral_n, 5))

integral_limite = integrate(f_limite, a, b, steps)
print("Limite theorique de l'integrale =", round(integral_limite, 5))
```
