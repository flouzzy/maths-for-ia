# TP 1 : Illustration algorithmique de Beppo Levi

Ce TP démontre la convergence monotone en calculant numériquement l'intégrale d'une suite croissante de fonctions.

\begin{lstlisting}[language=Python]
# TP1: Approximation de l'integrale d'une suite croissante
# f_n(x) = (1 - x/n)^n sur [0, n] convergeant vers exp(-x)

def fn(x, n):
    if x >= 0 and x <= n:
        return (1.0 - x/n)**n
    return 0.0

def limite_f(x):
    # Approximation par e^(-x) en developpement limite
    # Implementation sans math.exp pour respect de la consigne (Python pur)
    term = 1.0
    s = 1.0
    for i in range(1, 50):
        term = term * (-x) / i
        s += term
    return s

def integrate_rectangles(func, n_param, a, b, steps=1000):
    width = (b - a) / steps
    area = 0.0
    for i in range(steps):
        x = a + i * width
        area += func(x, n_param) * width
    return area

# Test pour n croissant
b_max = 20.0
for n in [2, 5, 10, 50, 100]:
    integral_n = integrate_rectangles(fn, n, 0.0, b_max, 5000)
    print("n=", n, " Integral =", integral_n)

# L'integrale de exp(-x) de 0 a 20 est environ 1.0
assert abs(integral_n - 1.0) < 0.1
\end{lstlisting}
