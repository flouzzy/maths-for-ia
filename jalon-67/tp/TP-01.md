# TP 1 : Simulation géométrique du Théorème de Convergence Monotone

**Objectif :** Implémenter en Python pur la démonstration visuelle et numérique du TCM. Nous allons générer une suite de fonctions positives et croissantes en évaluant numériquement leurs intégrales.

```python
def function_sequence(n, x):
    # Suite de fonctions qui convergent vers x^2 sur [0,1]
    # f_n(x) = x^2 * (1 - 1/(n+1))
    return (x ** 2) * (1.0 - 1.0 / (n + 1))

def numerical_integration(f, n, a, b, num_steps=1000):
    # Integration de Riemann numerique naive
    step = (b - a) / num_steps
    total_area = 0.0
    for i in range(num_steps):
        x = a + i * step
        total_area += f(n, x) * step
    return total_area

def test_monotone_convergence():
    a, b = 0.0, 1.0

    # 1. Verification de la croissance point par point
    x_test = 0.5
    for n in range(1, 10):
        assert function_sequence(n, x_test) <= function_sequence(n+1, x_test), "Violation de la croissance"
        assert function_sequence(n, x_test) >= 0, "Violation de la positivite"

    # 2. Integration limite
    # L'integrale analytique de la limite (x^2) est 1/3
    expected_limit = 1.0 / 3.0

    # Calcul de la limite des integrales
    integral_n10 = numerical_integration(function_sequence, 10, a, b)
    integral_n100 = numerical_integration(function_sequence, 100, a, b)
    integral_n1000 = numerical_integration(function_sequence, 1000, a, b)

    assert integral_n10 < integral_n100 < integral_n1000
    assert abs(integral_n1000 - expected_limit) < 0.01, "L'interversion n'est pas verifiee numeriquement"
    print("TP1: Convergence Monotone validee avec succes.")

test_monotone_convergence()
```
