# TP 1 : Vérification numérique de la non-interchangeabilité sans croissance

```python
def integrate(f, a, b, n_steps=1000):
    dx = (b-a)/n_steps
    return sum(f(a + i*dx) * dx for i in range(n_steps))

# Suite fn = n * Indicator(0, 1/n)
def f_n(n, x):
    if 0 < x < 1/n:
        return n
    return 0

res_integrals = []
for n in range(1, 100):
    val = integrate(lambda x: f_n(n, x), 0, 1)
    res_integrals.append(val)

# Vérification que l'intégrale reste à 1 alors que la limite de f_n est 0 p.p.
for val in res_integrals:
    assert abs(val - 1.0) < 0.1
print('TP 1 Validé')
```
