## TP 1 : Vérification numérique de l'intégrale d'une suite croissante

Objectif : Simuler numériquement le calcul d'une limite d'intégrale de fonctions croissantes via la méthode des rectangles, pour observer le théorème de convergence monotone.

\begin{lstlisting}
# Verification numerique du Theoreme de Convergence Monotone
# On integre f_n(x) = (1 - x**2 / n)**n sur [0, sqrt(n)]

def integrate(f, a, b, num_steps=1000):
    step = (b - a) / num_steps
    area = 0.0
    for i in range(num_steps):
        x = a + i * step
        area += f(x) * step
    return area

def simulate_mct():
    import math

    def fn(x, n):
        if x < 0 or x > math.sqrt(n):
            return 0.0
        return (1 - (x*x)/n)**n

    def limit_f(x):
        return math.exp(-x*x)

    n_values = [1, 2, 5, 10, 50, 100]

    print("Integration de f_n :")
    for n in n_values:
        # On passe une fonction a un argument x
        integrand = lambda x: fn(x, n)
        val = integrate(integrand, 0, math.sqrt(n), 10000)
        print(f"n={n:3} : In = {val:.5f}")

    print("\nIntegration de la limite e^(-x^2) sur [0, 10] (approche de l'infini) :")
    limit_val = integrate(limit_f, 0, 10, 10000)
    print(f"I_limit = {limit_val:.5f}")

    # Valeur theorique : sqrt(pi)/2
    theoretical = math.sqrt(math.pi) / 2
    print(f"Valeur theorique = {theoretical:.5f}")

    assert abs(limit_val - theoretical) < 1e-4

if __name__ == "__main__":
    simulate_mct()
\end{lstlisting}
