# TP 1 : Vérification numérique de l'intégration terme à terme

Ce script Python démontre numériquement le processus de convergence monotone.

```python
def func_term(x, n):
    # u_n(x) = x^(2n) - x^(2n+1)
    return (x**(2*n)) - (x**(2*n+1))

def riemann_sum_term(n, steps=10000):
    dx = 1.0 / steps
    total = 0.0
    for i in range(steps):
        x = (i + 0.5) * dx
        total += func_term(x, n) * dx
    return total

def theoretical_term(n):
    return 1.0 / ((2*n+1)*(2*n+2))

total_integral = 0.0
N_max = 50
print("n | Numérique | Théorique")
for n in range(1, N_max+1):
    num = riemann_sum_term(n)
    theo = theoretical_term(n)
    total_integral += theo
    assert abs(num - theo) < 1e-4
    if n <= 5:
        print(f"{n} | {num:.6f} | {theo:.6f}")

print(f"\nSomme totale jusqu'a N={N_max} : {total_integral:.6f}")
import math
print(f"Valeur exacte ln(2) - 0.5 = {math.log(2) - 0.5:.6f}")
```
