# TP 1 : Simulation numérique de Beppo Levi

**Objectif :** Vérifier empiriquement l'égalité de la limite des intégrales et de l'intégrale de la limite pour une suite de fonctions croissantes vers l'infini.

## Code Python (Pure Python, Zéro Framework)

```python
def f_n(x, n):
    return (n * (x ** 0.5)) / (1 + n * (x ** 2))

def f_n_croissante(x, n):
    # Une vraie suite croissante: f_n(x) = (1 - 1/(n+1)) * e^x
    return (1 - 1.0 / (n + 1)) * (2.71828 ** x)

def integrate_rectangle(f, n, a, b, num_steps):
    step = (b - a) / num_steps
    total = 0.0
    for i in range(num_steps):
        x = a + i * step
        total += f(x, n) * step
    return total

if __name__ == "__main__":
    a, b = 0.0, 1.0
    num_steps = 1000

    # Pour la suite croissante
    integrals_croissantes = []
    for n in range(1, 10):
        val = integrate_rectangle(f_n_croissante, n, a, b, num_steps)
        integrals_croissantes.append(val)

    # La limite theorique est int(e^x) = e - 1
    limite_theorique = 2.71828 - 1
    print("Integrales de f_n_croissante:", integrals_croissantes)
    print("Limite theorique:", limite_theorique)

    # Assertion (tolerance)
    assert abs(integrals_croissantes[-1] - limite_theorique) < 0.1

```
