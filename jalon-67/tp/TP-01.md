# TP 1 : Simulation numérique du TCM par sommes de Riemann

### Objectif
Implémenter une vérification empirique du théorème de Beppo Levi en calculant séquentiellement l'aire sous une suite croissante de fonctions vers une limite connue.

### Implémentation Python pur

```python
def f_n(x, n):
    return (1 - (x / n)) ** n if x <= n else 0.0

def f_lim(x):
    import math
    return math.exp(-x)

def riemann_sum(func, a, b, num_steps, n_val=None):
    step = (b - a) / num_steps
    total = 0.0
    for i in range(num_steps):
        x = a + (i + 0.5) * step
        if n_val is not None:
            total += func(x, n_val) * step
        else:
            total += func(x) * step
    return total

def test_tcm():
    b_limit = 50.0  # approximate infinity
    steps = 50000

    # Calculate integral of limit function
    int_limit = riemann_sum(f_lim, 0, b_limit, steps)

    # Calculate limits of integrals
    integrals_n = []
    for n in [1, 5, 10, 50, 100]:
        val = riemann_sum(f_n, 0, b_limit, steps, n_val=n)
        integrals_n.append(val)

    print(f"Int(Limit) = {int_limit:.5f}")
    print(f"Limit(Int_n) pour n=100 = {integrals_n[-1]:.5f}")

    assert abs(int_limit - 1.0) < 0.01
    assert abs(integrals_n[-1] - int_limit) < 0.02
    print("TP1: TCM numerique valide avec succes.")

test_tcm()

```
