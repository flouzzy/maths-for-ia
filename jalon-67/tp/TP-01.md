# TP 01 : Intégration numérique d'une suite de fonctions croissantes

## Objectif
Ce TP propose d'illustrer numériquement le théorème de convergence monotone. Nous allons construire une suite de fonctions f_n(x) qui converge en croissant vers f(x) et vérifier que la limite des intégrales est égale à l'intégrale de la limite.

## Implémentation

```python
import math

def f_n(x, n):
    # Suite de fonctions qui converge vers e^(-x)
    # On utilise l'indicatrice sur [0, n] pour rendre la suite croissante
    if 0 <= x <= n:
        return math.exp(-x) * (1 - (x / n)**2)**n
    return 0.0

def f_limit(x):
    return math.exp(-x)

def integrate_riemann(func, a, b, n_steps=10000):
    dx = (b - a) / n_steps
    total = 0.0
    for i in range(n_steps):
        x = a + i * dx
        total += func(x) * dx
    return total

def test_convergence_monotone():
    # Intégrale de la limite f(x) sur [0, 10] (approximativement +infini)
    integral_f = integrate_riemann(f_limit, 0, 10)

    # Intégrales successives des f_n
    for n in [10, 50, 100, 500]:
        # On définit une fonction partielle pour n fixe
        func_to_integrate = lambda x: f_n(x, n)
        integral_fn = integrate_riemann(func_to_integrate, 0, 10)
        print(f"n={n}, Integrale(f_n) = {integral_fn:.6f}")

    print(f"Integrale(f_limit) = {integral_f:.6f}")

    # Assertions pour valider la theorie
    func_100 = lambda x: f_n(x, 100)
    func_500 = lambda x: f_n(x, 500)
    assert integrate_riemann(func_100, 0, 10) <= integrate_riemann(func_500, 0, 10), "Croissance non respectee"
    assert abs(integrate_riemann(func_500, 0, 10) - integral_f) < 0.05, "La limite n'est pas atteinte"

if __name__ == "__main__":
    test_convergence_monotone()
```
