# TP 1 : Simulation numérique du TCM
**Objectif :** Vérifier numériquement que l'intégrale de la limite d'une suite croissante est égale à la limite des intégrales.
```python
def f_n(n, x):
    # Suite croissante : f_n(x) = (1 - x/n)**n sur [0, 1]
    if x >= n:
        return 0
    return (1 - x/n)**n

def integral_approximation(n, N_samples=10000):
    # Intégrale de Riemann sur [0, 1] par rectangles
    dx = 1.0 / N_samples
    total = 0
    for i in range(N_samples):
        x = i * dx
        total += f_n(n, x) * dx
    return total

if __name__ == "__main__":
    import math
    lim_val = 1 - 1/math.e
    val_10 = integral_approximation(10)
    val_100 = integral_approximation(100)

    assert val_10 < val_100, "La suite d'integrales doit etre croissante"
    assert abs(val_100 - lim_val) < 0.01, "La limite des integrales doit tendre vers l'integrale de la limite"
    print("TP1 reussi: Convergence monotone verifiee.")
```
