\subsection*{TP 2 : Approximation de Dirac par des gaussiennes}

Simulation numérique.

\begin{lstlisting}[language=Python]
import math

def fonction_test(x):
    return math.cos(x) * math.exp(-x*x)

def integrer_simpson(f, a, b, N=2000):
    if N % 2 == 1: N += 1
    h = (b - a) / N
    somme = f(a) + f(b)
    for k in range(1, N):
        xk = a + k * h
        somme += 2 * f(xk) if k % 2 == 0 else 4 * f(xk)
    return (h / 3.0) * somme

def gaussienne(n, x):
    return (n / math.sqrt(math.pi)) * math.exp(-n*n * x*x)

def action_gaussienne(n):
    # L'essentiel de l'integrale de la gaussienne est dans [-3/n, 3/n]
    return integrer_simpson(lambda x: gaussienne(n, x) * fonction_test(x), -3.0/n, 3.0/n, 200)

val_theorique = fonction_test(0.0)
erreur = abs(val_theorique - action_gaussienne(40))
assert erreur < 1e-3, "Convergence non atteinte pour la gaussienne"
print("TP2: Convergence par gaussiennes validee.")
\end{lstlisting}
