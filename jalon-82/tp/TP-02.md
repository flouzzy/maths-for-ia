# TP 2 : Intégration numérique pour l'action d'une distribution

L'action d'une distribution régulière $T_f$ sur une fonction test est une intégrale. Nous allons l'approximer numériquement par la méthode des trapèzes.

```python
import math

def fonction_test(x):
    if abs(x) >= 1.0:
        return 0.0
    return math.exp(-1.0 / (1.0 - x**2))

def fonction_f(x):
    # f(x) = x**2, fonction localement integrable
    return x**2

def action_distribution_reguliere(f, phi, a, b, n_steps=10000):
    # Calcul de l'integrale de f(x)*phi(x) sur [a, b]
    dx = (b - a) / n_steps
    integrale = 0.0
    for i in range(n_steps + 1):
        x = a + i * dx
        valeur = f(x) * phi(x)
        # Poids de la methode des trapezes
        poids = 0.5 if (i == 0 or i == n_steps) else 1.0
        integrale += poids * valeur * dx
    return integrale

def test_integration_action():
    # Comme la fonction test est a support dans [-1, 1], l'integrale se limite a ce domaine
    resultat = action_distribution_reguliere(fonction_f, fonction_test, -1.0, 1.0)
    # L'integrande x**2 * phi(x) est positif, l'integrale doit etre positive
    assert resultat > 0.0

    # Sur un intervalle disjoint du support, l'integrale doit etre nulle
    resultat_vide = action_distribution_reguliere(fonction_f, fonction_test, 2.0, 5.0)
    assert abs(resultat_vide) < 1e-10

    print(f"TP2: Resultat de l'action: {resultat}")
    print("TP2: Validation mathematique reussie.")

if __name__ == "__main__":
    test_integration_action()
```
