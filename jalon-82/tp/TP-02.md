# TP 2 : Approximation numérique du Dirac par une suite gaussienne

**Objectif :**
Nous allons implémenter la suite gaussienne qui converge vers la masse de Dirac et vérifier numériquement que son action sur une fonction test donne bien l'évaluation de la fonction en 0.

**Code Python pur :**
```python
import math

def gaussienne(n, x):
    """Suite de fonctions convergeant vers le Dirac."""
    return (n / math.sqrt(math.pi)) * math.exp(-(n**2) * (x**2))

def fonction_test_quelconque(x):
    """Une fonction test simple (cloche centree en 0.5)."""
    if x <= -0.5 or x >= 1.5:
        return 0.0
    # Cloche translatee pour que f(0) ne soit pas trivial
    return math.exp(-1.0 / (1.0 - (x - 0.5)**2))

def integrale_trapezes(f, a, b, n_steps=10000):
    """Integration numerique de base."""
    h = (b - a) / n_steps
    somme = 0.5 * (f(a) + f(b))
    for i in range(1, n_steps):
        somme += f(a + i * h)
    return somme * h

def action_distribution_reguliere(n):
    """Calcule l'integrale de f_n(x) * test(x)."""
    # Le support de la fonction test est [-0.5, 1.5]
    def integrande(x):
        return gaussienne(n, x) * fonction_test_quelconque(x)
    return integrale_trapezes(integrande, -0.5, 1.5)

def tester_convergence():
    valeur_theorique = fonction_test_quelconque(0.0)

    # On teste pour un n grand
    n = 100
    valeur_approche = action_distribution_reguliere(n)

    erreur = abs(valeur_theorique - valeur_approche)
    assert erreur < 1e-2, f"Erreur trop grande : theorique={valeur_theorique}, numerique={valeur_approche}"

    print(f"TP 2 : Convergence verifiee. Valeur theorique = {valeur_theorique:.5f}, Valeur numerique = {valeur_approche:.5f}")

if __name__ == "__main__":
    tester_convergence()

```
