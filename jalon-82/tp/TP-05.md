# TP 5 : Évaluation numérique de la Valeur Principale de Cauchy

La distribution Valeur Principale $\text{vp}(1/x)$ agit sur une fonction test en évitant le point singulier $0$. Nous allons la coder en retirant symétriquement un petit intervalle $[-\epsilon, \epsilon]$ autour de l'origine.

```python
import math

def fonction_test_vp(x):
    # Support entre -2 et 2
    if abs(x) >= 2.0:
        return 0.0
    # Fonction asymetrique pour que l'integrale ne soit pas trivialement nulle
    return math.exp(-x**2) * math.sin(x)

def calcul_valeur_principale(phi, epsilon, R=2.0, n_steps=5000):
    # Integrale sur [-R, -epsilon] et [epsilon, R]

    def integrale_segment(a, b):
        dx = (b - a) / n_steps
        integ = 0.0
        for i in range(n_steps + 1):
            x = a + i * dx
            valeur = phi(x) / x
            poids = 0.5 if (i == 0 or i == n_steps) else 1.0
            integ += poids * valeur * dx
        return integ

    partie_gauche = integrale_segment(-R, -epsilon)
    partie_droite = integrale_segment(epsilon, R)
    return partie_gauche + partie_droite

def test_valeur_principale():
    # On va calculer la VP pour epsilon = 1e-2 et epsilon = 1e-4
    res_eps_grand = calcul_valeur_principale(fonction_test_vp, 0.01)
    res_eps_petit = calcul_valeur_principale(fonction_test_vp, 0.0001)

    # La theorie dit que la limite existe, donc la difference entre
    # deux petits epsilon doit etre tres faible
    diff = abs(res_eps_grand - res_eps_petit)
    assert diff < 1e-2

    print(f"TP5: VP(eps=1e-2) = {res_eps_grand:.4f}")
    print(f"TP5: VP(eps=1e-4) = {res_eps_petit:.4f}")
    print("TP5: Stabilite de la limite verifiee.")

if __name__ == "__main__":
    test_valeur_principale()
```
