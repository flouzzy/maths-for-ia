# TP 5 : Lemme de Riemann-Lebesgue tempéré (Oscillateur rapide)

**Objectif :**
Vérification numérique que l'intégrale d'une fonction test contre un sinus très oscillant tend bien vers 0.

**Code Python pur :**
```python
import math

def cloche(x):
    if abs(x) >= 1.0: return 0.0
    return math.exp(-1.0 / (1.0 - x**2))

def integrer_oscillateur(lambda_freq, n_steps=50000):
    """
    Integre sin(lambda*x) * cloche(x) sur [-1, 1].
    n_steps doit etre grand pour capturer les oscillations rapides.
    """
    h = 2.0 / n_steps
    somme = 0
    # Integration sur [-1, 1]
    for i in range(n_steps):
        x = -1.0 + i * h + h/2.0
        somme += math.sin(lambda_freq * x) * cloche(x)
    return somme * h

def tester_oscillateur():
    # Pour lambda petit, l'integrale peut etre non nulle (bien que la cloche soit paire ici,
    # donc l'integrale avec sin est techniquement 0, on va decaler la cloche)

    def cloche_decalee(x):
        if x <= 0.0 or x >= 2.0: return 0.0
        return math.exp(-1.0 / (1.0 - (x-1.0)**2))

    def integrer_decale(lambda_freq, n_steps=50000):
        h = 2.0 / n_steps
        somme = 0
        for i in range(n_steps):
            x = 0.0 + i * h + h/2.0
            somme += math.sin(lambda_freq * x) * cloche_decalee(x)
        return somme * h

    val_basse_freq = integrer_decale(1.0)
    val_haute_freq = integrer_decale(100.0)
    val_tres_haute_freq = integrer_decale(1000.0)

    assert abs(val_tres_haute_freq) < abs(val_basse_freq), "L'integrale devrait diminuer a haute frequence"
    assert abs(val_tres_haute_freq) < 1e-2, "L'integrale doit tendre vers 0"

    print(f"TP 5 : Oscillateur valide. Basse freq: {val_basse_freq:.5f}, Haute freq: {val_tres_haute_freq:.5e}")

if __name__ == "__main__":
    tester_oscillateur()

```
