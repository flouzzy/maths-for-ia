\subsection*{TP 3 : Simulation du Peigne de Dirac}

Simulation numérique.

\begin{lstlisting}[language=Python]
import math

def fonction_test(x):
    # Support [-5, 5]
    if x <= -5 or x >= 5: return 0.0
    return math.exp(-x*x / 2.0)

def action_peigne():
    somme = 0.0
    # Le support est inclus dans [-4, 4] pour les entiers
    for k in range(-4, 5):
        somme += fonction_test(k)
    return somme

valeur_peigne = action_peigne()
# Verification manuelle de la symetrie
valeur_attendue = fonction_test(0) + 2*fonction_test(1) + 2*fonction_test(2) + 2*fonction_test(3) + 2*fonction_test(4)
assert abs(valeur_peigne - valeur_attendue) < 1e-6
print(f"TP3: Action du Peigne de Dirac = {valeur_peigne:.4f}")
\end{lstlisting}
