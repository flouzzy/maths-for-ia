# TP 3 : Approximation numérique de la masse de Dirac

Ce TP illustre comment une suite de fonctions (ici, des rectangles d'aire 1 de plus en plus fins) "converge" vers la distribution de Dirac en agissant sur une fonction test.

```python
import math

def fonction_test_phi(x):
    return math.cos(x)

def fonction_rectangle(x, n):
    # Rectangle de largeur 2/n et de hauteur n/2 centre en 0. Aire = 1.
    if abs(x) <= 1.0 / n:
        return n / 2.0
    return 0.0

def action_rectangle_sur_phi(n, phi, steps_per_unit=1000):
    # L'integrale ne se fait que sur le support [-1/n, 1/n]
    a = -1.0 / n
    b = 1.0 / n
    n_steps = max(10, int((b - a) * steps_per_unit))
    dx = (b - a) / n_steps
    integrale = 0.0
    for i in range(n_steps + 1):
        x = a + i * dx
        valeur = fonction_rectangle(x, n) * phi(x)
        poids = 0.5 if (i == 0 or i == n_steps) else 1.0
        integrale += poids * valeur * dx
    return integrale

def test_approximation_dirac():
    cible_dirac = fonction_test_phi(0.0) # Cosinus en 0 vaut 1.0

    # Test avec n = 100
    res_100 = action_rectangle_sur_phi(100, fonction_test_phi)

    # Test avec n = 1000
    res_1000 = action_rectangle_sur_phi(1000, fonction_test_phi)

    # L'erreur absolue doit diminuer quand n augmente
    err_100 = abs(res_100 - cible_dirac)
    err_1000 = abs(res_1000 - cible_dirac)

    assert err_1000 < err_100
    assert err_1000 < 1e-4

    print("TP3: Convergence vers le Dirac verifiee.")

if __name__ == "__main__":
    test_approximation_dirac()
```
