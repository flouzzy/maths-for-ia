# TP 4 : Implémentation du Noyau de Fejér et convergence

Le noyau de Fejér joue un rôle fondamental en analyse de Fourier. Bien qu'il soit très oscillant, son action intégrale sur une fonction test reproduit l'action de la masse de Dirac.

```python
import math

def fonction_test_fejer(x):
    # Support arbitraire entre -pi et pi
    if abs(x) >= math.pi:
        return 0.0
    return 1.0 - (abs(x) / math.pi)**2

def noyau_fejer(x, n):
    if abs(x) < 1e-9:
        return n / (2.0 * math.pi) # Limite en 0
    sin_demi = math.sin(x / 2.0)
    sin_n_demi = math.sin(n * x / 2.0)
    return (1.0 / (2.0 * math.pi * n)) * (sin_n_demi / sin_demi)**2

def action_fejer(n, phi, a=-math.pi, b=math.pi, n_steps=20000):
    dx = (b - a) / n_steps
    integrale = 0.0
    for i in range(n_steps + 1):
        x = a + i * dx
        valeur = noyau_fejer(x, n) * phi(x)
        poids = 0.5 if (i == 0 or i == n_steps) else 1.0
        integrale += poids * valeur * dx
    return integrale

def test_noyau_fejer():
    cible = fonction_test_fejer(0.0) # Vaut 1.0

    # Calcul de l'action avec n=50
    res_n50 = action_fejer(50, fonction_test_fejer)

    # La valeur trouvee doit etre relativement proche de la cible de Dirac (1.0)
    assert abs(res_n50 - cible) < 0.1

    print(f"TP4: Action Fejer n=50 : {res_n50:.4f}, Cible: {cible}")
    print("TP4: Validation mathematique reussie.")

if __name__ == "__main__":
    test_noyau_fejer()
```
