# TP 4 : Valeur principale de Cauchy 1/x

\subsection*{TP 4 : Valeur principale de Cauchy 1/x}

Simulation numérique.

\begin{lstlisting}[language=Python]
import math

def fonction_test(x):
    # Fonction asymetrique pour que l'integrale vp ne soit pas trivialement nulle
    if x <= -2 or x >= 2: return 0.0
    return (x + 1) * math.exp(-x*x)

def integrer_simpson(f, a, b, N=500):
    if N % 2 == 1: N += 1
    h = (b - a) / N
    somme = f(a) + f(b)
    for k in range(1, N):
        xk = a + k * h
        somme += 2 * f(xk) if k % 2 == 0 else 4 * f(xk)
    return (h / 3.0) * somme

def action_vp(epsilon):
    # Integrer sur [-2, -epsilon] et [epsilon, 2]
    int_neg = integrer_simpson(lambda x: fonction_test(x)/x, -2.0, -epsilon)
    int_pos = integrer_simpson(lambda x: fonction_test(x)/x, epsilon, 2.0)
    return int_neg + int_pos

vp_01 = action_vp(0.1)
vp_001 = action_vp(0.01)
vp_0001 = action_vp(0.001)

# L'integrale vp(1/x) sur la partie paire exp(-x*x) est 0.
# Sur la partie impaire x*exp(-x*x), vp(1/x) devient integrale(exp(-x*x))
vp_theorique = integrer_simpson(lambda x: math.exp(-x*x), -2.0, 2.0, 1000)

erreur = abs(vp_0001 - vp_theorique)
assert erreur < 1e-2, "Calcul de VP incorrect"
print(f"TP4: Calcul de VP de 1/x reussi. Erreur : {erreur:.4f}")
\end{lstlisting}
