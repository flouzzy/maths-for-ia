# TP 4 : Évaluation numérique de la valeur principale de Cauchy

**Objectif :**
Nous allons implémenter le calcul tronqué et symétrisé pour évaluer numériquement la distribution valeur principale de 1/x.

**Code Python pur :**
```python
import math

def fonction_test_asymetrique(x):
    """Fonction test valant 1 en 0, pour avoir une derivee non nulle"""
    if abs(x) >= 1.0:
        return 0.0
    # Cloche * (1 + x) pour asymetrie
    return math.exp(-1.0 / (1.0 - x**2)) * (1.0 + x)

def integrale_rectangles(f, a, b, n_steps=5000):
    h = (b - a) / n_steps
    somme = 0
    for i in range(n_steps):
        somme += f(a + i * h + h / 2.0)
    return somme * h

def valeur_principale_numerique(f, R, epsilon):
    """
    Evalue int_{epsilon}^R (f(x) - f(-x))/x dx
    """
    def integrande(x):
        return (f(x) - f(-x)) / x

    return integrale_rectangles(integrande, epsilon, R)

def tester_vp():
    R = 1.0 # Support de notre fonction test

    # On fait tendre epsilon vers 0
    val_1 = valeur_principale_numerique(fonction_test_asymetrique, R, 1e-2)
    val_2 = valeur_principale_numerique(fonction_test_asymetrique, R, 1e-4)
    val_3 = valeur_principale_numerique(fonction_test_asymetrique, R, 1e-6)

    # La limite doit exister, donc les valeurs successives doivent etre proches
    diff = abs(val_3 - val_2)
    assert diff < 1e-2, f"La valeur principale ne semble pas converger. Diff = {diff}"
    print(f"TP 4 : Valeur principale evaluee a env. {val_3:.5f}. Convergence stable.")

if __name__ == "__main__":
    tester_vp()

```
