# TP 1 : Implémentation pure Python du réseau UAT

L'objectif de ce TP est de modéliser mathématiquement une couche cachée de neurones (sigmoïde) et de vérifier numériquement qu'avec suffisamment de neurones, on peut approcher une fonction non linéaire.

## Code Python Rigoureux

```python
import math

def sigmoid(x):
    # Sécurisation numérique contre l'overflow
    if x > 50: return 1.0
    if x < -50: return 0.0
    return 1.0 / (1.0 + math.exp(-x))

class SingleLayerNetwork:
    def __init__(self, w, b, alpha):
        assert len(w) == len(b) == len(alpha), "Les dimensions doivent correspondre"
        self.w = w
        self.b = b
        self.alpha = alpha
        self.N = len(w)

    def forward(self, x):
        result = 0.0
        for i in range(self.N):
            activation = sigmoid(self.w[i] * x + self.b[i])
            result += self.alpha[i] * activation
        return result

def main():
    # Définition manuelle d'une "porte" approximée avec deux sigmoïdes (Exo 3)
    # Fonction porte s'activant entre x=2 et x=4
    k = 20.0 # Raideur
    w = [k, k]
    b = [-2*k, -4*k] # Seuils à 2 et 4
    alpha = [1.0, -1.0] # On ajoute la première, on soustrait la seconde

    net = SingleLayerNetwork(w, b, alpha)

    # Validation mathématique
    tests = [
        (0.0, 0.0),
        (3.0, 1.0),
        (5.0, 0.0)
    ]

    for x, expected in tests:
        val = net.forward(x)
        print(f"f({x}) = {val:.4f} (Attendu: {expected})")
        assert abs(val - expected) < 0.05, f"Erreur d'approximation en {x}"

    print("TP 1 : Approximation de la fonction porte validée rigoureusement.")

if __name__ == "__main__":
    main()
```
