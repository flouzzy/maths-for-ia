## TP 1 : Implémentation from scratch du neurone artificiel

```python
# Simulation du théorème d'approximation : Le neurone de base
import math
import random

def sigmoid(x):
    # No LaTeX symbols used, pure Python math
    return 1.0 / (1.0 + math.exp(-x))

def neuron_forward(x_vector, w_vector, b):
    # Dot product
    z = sum(x * w for x, w in zip(x_vector, w_vector)) + b
    return sigmoid(z)

# Test simple
x_test = [0.5, -1.2]
w_test = [0.1, 0.8]
b_test = -0.2
output = neuron_forward(x_test, w_test, b_test)
print("Output of single neuron:", output)
assert 0 <= output <= 1

```
