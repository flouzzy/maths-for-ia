# TP 2 : Construction de la matrice d'activation et résolution par moindres carrés

Pour approcher une fonction complexe, nous générons aléatoirement les poids de la couche cachée, puis nous trouvons les poids de sortie $\alpha$ optimaux mathématiquement via une projection orthogonale (moindres carrés).

## Code Python Rigoureux

```python
import math
import random

def relu(x):
    return max(0.0, x)

def dot_product(v1, v2):
    return sum(x*y for x, y in zip(v1, v2))

def transpose(mat):
    return [[mat[j][i] for j in range(len(mat))] for i in range(len(mat[0]))]

def mat_mult(A, B):
    B_T = transpose(B)
    return [[dot_product(row_A, row_B) for row_B in B_T] for row_A in A]

# Solver très basique par descente de gradient sur les poids de sortie (pas d'inversion directe numpy)
def solve_least_squares_gd(H, Y, epochs=1000, lr=0.01):
    N = len(H[0])
    num_samples = len(Y)
    alpha = [0.0] * N

    for _ in range(epochs):
        for i in range(num_samples):
            pred = dot_product(H[i], alpha)
            error = pred - Y[i]
            for j in range(N):
                alpha[j] -= lr * error * H[i][j] / num_samples
    return alpha

def main():
    random.seed(42)
    def target(x):
        return x**2

    domain = [i/10.0 for i in range(-10, 11)]
    num_neurons = 20

    # Couche cachée fixe (Random Features)
    W = [random.uniform(-5, 5) for _ in range(num_neurons)]
    B = [random.uniform(-5, 5) for _ in range(num_neurons)]

    # Construction de H
    H = []
    Y = []
    for x in domain:
        row = [relu(W[j] * x + B[j]) for j in range(num_neurons)]
        H.append(row)
        Y.append(target(x))

    alpha = solve_least_squares_gd(H, Y)

    # Vérification
    x_test = 0.5
    h_test = [relu(W[j] * x_test + B[j]) for j in range(num_neurons)]
    pred = dot_product(h_test, alpha)

    print(f"Prédiction pour x=0.5 : {pred:.4f} (Vrai = 0.25)")
    assert 0.0 <= pred <= 0.5, "L'approximation a divergé de manière inacceptable."
    print("TP 2 Validé.")

if __name__ == "__main__":
    main()
```
