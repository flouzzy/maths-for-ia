# TP 04 : Approximation d'une fonction continue (Sinus) par un réseau ReLU de taille fixe

**Objectif :** Interpolation affine par morceaux d'une courbe trigonométrique avec validation de la borne d'erreur O(1/N^2).

```python
import math

def relu(x):
    return max(0.0, x)

def generate_relu_weights_for_sin(num_neurons, domain_start, domain_end):
    step = (domain_end - domain_start) / num_neurons
    weights = []

    # We will compute the slopes of the secant lines
    points = [(domain_start + i * step, math.sin(domain_start + i * step)) for i in range(num_neurons + 1)]

    slopes = []
    for i in range(num_neurons):
        slope = (points[i+1][1] - points[i][1]) / step
        slopes.append(slope)

    # The output weight for a ReLU centered at x_i is the change in slope at x_i
    network_params = []
    current_slope = slopes[0]

    # First bias term to start the line
    base_val = points[0][1]

    for i in range(1, num_neurons):
        slope_change = slopes[i] - slopes[i-1]
        x_i = points[i][0]
        network_params.append((x_i, slope_change))

    return base_val, current_slope, network_params

def predict_sin(x, base_val, initial_slope, start_x, network_params):
    # Reconstruct the piecewise function
    y = base_val + initial_slope * (x - start_x)
    for center, slope_change in network_params:
        y += slope_change * relu(x - center)
    return y

def test_sin_approximation():
    domain_start = 0.0
    domain_end = math.pi
    num_neurons = 100

    base_val, initial_slope, params = generate_relu_weights_for_sin(num_neurons, domain_start, domain_end)

    max_error = 0.0
    test_points = 1000
    for i in range(test_points):
        x = domain_start + i * (domain_end - domain_start) / test_points
        y_true = math.sin(x)
        y_pred = predict_sin(x, base_val, initial_slope, domain_start, params)
        max_error = max(max_error, abs(y_true - y_pred))

    # Theoretical upper bound proportional to 1/N^2. For sin(x), second deriv is max 1.
    # Bound is approx (pi/N)^2 / 8
    theoretical_bound = ((math.pi / num_neurons)**2) / 8.0

    # The experimental error should be strictly less than the upper theoretical bound (with some float tolerance)
    assert max_error <= theoretical_bound * 1.5, f"Error {max_error} exceeds bound {theoretical_bound}"
    print(f"Sinus approximation validated. Max Error: {max_error:.6f}, Bound: {theoretical_bound:.6f}")

test_sin_approximation()
```
