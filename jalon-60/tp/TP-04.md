## TP 4 : Entraînement naïf par force brute (Random Search)

```python
# Tentative d'approximation de f(x) = x^2 par recherche aleatoire
import random
import math

def target_function(x):
    return x * x

def relu(x):
    return max(0.0, x)

N = 5 # number of neurons
best_mse = float('inf')
best_params = None

# Generate points on [0, 1]
x_data = [i/10.0 for i in range(11)]
y_data = [target_function(x) for x in x_data]

# Random search over 10000 configurations
for _ in range(10000):
    weights = [random.uniform(-2, 2) for _ in range(N)]
    biases = [random.uniform(-1, 1) for _ in range(N)]
    alphas = [random.uniform(-2, 2) for _ in range(N)]

    mse = 0.0
    for x, y in zip(x_data, y_data):
        pred = sum(a * relu(w * x + b) for w, b, a in zip(weights, biases, alphas))
        mse += (pred - y)**2
    mse /= len(x_data)

    if mse < best_mse:
        best_mse = mse
        best_params = (weights, biases, alphas)

print("Best MSE found:", best_mse)
assert best_mse < 0.1 # Should find at least a decent fit

```
