# TP 4 : Preuve algorithmique du Manifold Folding (Profondeur)

Nous simulons l'exercice de cours (Exo 10) : l'application successive de la fonction dent de scie génère un nombre exponentiel d'oscillations.

## Code Python Rigoureux

```python
def relu(x):
    return max(0.0, x)

def saw_tooth_operator(x):
    # G(x) = 2*relu(x) - 4*relu(x - 0.5) + 2*relu(x - 1)
    return 2.0*relu(x) - 4.0*relu(x - 0.5) + 2.0*relu(x - 1.0)

def compose_network(x, depth):
    val = x
    for _ in range(depth):
        val = saw_tooth_operator(val)
    return val

def count_peaks(depth, resolution=1000):
    points = [i/resolution for i in range(resolution + 1)]
    vals = [compose_network(p, depth) for p in points]

    peaks = 0
    # Un pic est un maximum local
    for i in range(1, resolution):
        if vals[i-1] < vals[i] and vals[i] > vals[i+1]:
            peaks += 1

    return peaks

def main():
    depths = [1, 2, 3, 4, 5]
    for L in depths:
        p = count_peaks(L)
        expected = 2**(L-1)
        print(f"Profondeur {L} : {p} pics détectés (Attendu : {expected}).")

        # Le théorème mathématique affirme que p = 2^(L-1)
        assert p == expected, f"Erreur théorique à la profondeur {L}"

    print("La profondeur du réseau multiplie exponentiellement sa capacité d'oscillation géométrique.")
    print("TP 4 Validé.")

if __name__ == "__main__":
    main()
```
