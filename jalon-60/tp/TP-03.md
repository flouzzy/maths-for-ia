# TP 03 : Évaluation de la limite discriminatoire d'une Sigmoïde

**Objectif :** Construction d'une fonction échelon (porte) de précision arbitraire via la soustraction de sigmoïdes tendues.

```python
import math

def sigmoid(x):
    # Clamping to avoid math overflow
    if x < -500: return 0.0
    if x > 500: return 1.0
    return 1.0 / (1.0 + math.exp(-x))

def universal_gate_approximation(x, center, width, sharpness):
    # Represents the combination: sigmoid(lambda*(x - left)) - sigmoid(lambda*(x - right))
    left_bound = center - (width / 2.0)
    right_bound = center + (width / 2.0)

    val1 = sigmoid(sharpness * (x - left_bound))
    val2 = sigmoid(sharpness * (x - right_bound))
    return val1 - val2

def test_gate():
    center = 0.0
    width = 2.0 # Support from -1 to 1
    sharpness = 1000.0 # High lambda for sharp corners

    # Outside the gate (should be near 0)
    assert abs(universal_gate_approximation(-2.0, center, width, sharpness) - 0.0) < 1e-5
    assert abs(universal_gate_approximation(2.0, center, width, sharpness) - 0.0) < 1e-5

    # Inside the gate (should be near 1)
    assert abs(universal_gate_approximation(0.0, center, width, sharpness) - 1.0) < 1e-5

    print("Universal gate approximation via sigmoid linear combination validated.")

test_gate()
```
