## TP 3 : Approximation d'un créneau (marche)

```python
# Demonstration experimentale de l'approximation d'une fonction porte
import math

def sigmoid(x):
    return 1.0 / (1.0 + math.exp(-x))

def step_approximation(x, k):
    # G_k(x) = sigmoid(k(x+1)) - sigmoid(k(x-1))
    return sigmoid(k * (x + 1.0)) - sigmoid(k * (x - 1.0))

# Evaluate at different points and different sharpness (k)
test_points = [-1.5, -0.5, 0.0, 0.5, 1.5]
for k in [1.0, 10.0, 100.0]:
    print("Sharpness k =", k)
    for x in test_points:
        val = step_approximation(x, k)
        print("  x=", x, " => ", round(val, 4))

# Assertions for large k
assert step_approximation(0.0, 100.0) > 0.99
assert step_approximation(1.5, 100.0) < 0.01
assert step_approximation(-1.5, 100.0) < 0.01

```
