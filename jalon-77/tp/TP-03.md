# TP 3 : Convergence des Polynômes de Bernstein

**Objectif :** Illustrer le théorème de Stone-Weierstrass (et donc la densité des polynômes) en approximant une fonction continue via les polynômes de Bernstein.

```python
import math

def f_target(x):
    return abs(x - 0.5)

def bernstein_poly(n, k, x):
    comb = math.factorial(n) // (math.factorial(k) * math.factorial(n - k))
    return comb * (x ** k) * ((1 - x) ** (n - k))

def bernstein_approx(f, n, x):
    approx = 0.0
    for k in range(n + 1):
        approx += f(k / n) * bernstein_poly(n, k, x)
    return approx

def max_error(n, num_points=1000):
    err = 0.0
    for i in range(num_points + 1):
        x = i / num_points
        e = abs(f_target(x) - bernstein_approx(f_target, n, x))
        if e > err:
            err = e
    return err

if __name__ == "__main__":
    n_values = [10, 50, 100]

    for n in n_values:
        err = max_error(n)
        print("Degre n =", n, "-> Erreur max =", err)

    # L'erreur doit etre decroissante
    err_10 = max_error(10)
    err_100 = max_error(100)

    assert err_100 < err_10, "L'erreur ne decroit pas avec le degre"
    print("Le theoreme d'approximation de Weierstrass est verifie empiriquement.")
```
