# TP 3 : Convolution avec une fonction porte

Nous allons approcher l'identité de convolution. La convolution d'une fonction avec un noyau étroit converge ponctuellement.

```python
def porte(x, n):
    if abs(x) <= 1/n:
        return n / 2.0
    return 0.0

def f_cible(x):
    return abs(x) # Continua, not differentiable at 0

def convolution_discrete(f, kernel, x, grid_radius, step, n):
    res = 0.0
    grid_points = int(grid_radius / step)
    for i in range(-grid_points, grid_points + 1):
        y = i * step
        res += f(x - y) * kernel(y, n) * step
    return res

step = 0.01
grid_radius = 2.0

for n in [2, 10]:
    val_approx = convolution_discrete(f_cible, porte, 0.0, grid_radius, step, n)
    print("f(0)=", f_cible(0.0), ", Approx(0, n=", n, ")=", val_approx)

val_10 = convolution_discrete(f_cible, porte, 0.0, grid_radius, step, 10)
# Validation
assert val_10 < 0.2, "La convolution ne se rapproche pas assez du point cible."
```
