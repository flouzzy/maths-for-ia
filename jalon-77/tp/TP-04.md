# TP 4 : Densité des fonctions continues (métrique de Hausdorff)

Ce TP simule l'approximation d'une fonction non continue (saut) par un raccordement linéaire affine (fonction trapèze).

```python
def indicatrice_positive(x):
    if 0 <= x <= 1:
        return 1.0
    return 0.0

def trapeze(x, delta):
    if 0 <= x <= 1:
        return 1.0
    elif -delta <= x < 0:
        return (x + delta) / delta
    elif 1 < x <= 1 + delta:
        return (1 + delta - x) / delta
    return 0.0

def integrer_erreur_L1(delta, steps):
    step = 4.0 / steps
    total = 0.0
    for i in range(steps):
        x = -2.0 + i * step
        total += abs(indicatrice_positive(x) - trapeze(x, delta)) * step
    return total

err1 = integrer_erreur_L1(0.5, 1000)
err2 = integrer_erreur_L1(0.1, 1000)
print("Erreur L1 pour delta=0.5 :", err1)
print("Erreur L1 pour delta=0.1 :", err2)

# Validation empirique
assert err2 < err1, "L'erreur L1 ne diminue pas avec un delta plus petit."
assert abs(err1 - 0.5) < 0.1, "L'integrale empirique est erronee."
```
