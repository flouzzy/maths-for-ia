# TP 4 : Densité par l'unité approchée (Régularisation)

**Objectif :** Implémenter un mollifier scalaire simple (approximation de l'identité) et l'appliquer par convolution sur une fonction discontinue pour observer la convergence L1.

```python
def target_func(x):
    return 1.0 if -1 <= x <= 1 else 0.0

def mollifier(z):
    # Fonction porte normalisée
    return 0.5 if -1 <= z <= 1 else 0.0

def conv_approx(x, n, num_points=1000):
    # Integrale de f(x-y) * n * mollifier(ny) dy
    # Le support de mollifier(ny) est [-1/n, 1/n]
    start = -1.0 / n
    end = 1.0 / n
    dy = (end - start) / num_points

    val = 0.0
    for i in range(num_points):
        y = start + i * dy + dy / 2
        val += target_func(x - y) * n * mollifier(n * y) * dy
    return val

def l1_error_conv(n, num_points=500):
    start = -2.0
    end = 2.0
    dx = (end - start) / num_points
    err = 0.0

    for i in range(num_points):
        x = start + i * dx + dx/2
        e = abs(target_func(x) - conv_approx(x, n))
        err += e * dx
    return err

if __name__ == "__main__":
    for n in [2, 5, 10]:
        e = l1_error_conv(n)
        print("Mollifier resolution n =", n, "-> Erreur L1 =", e)

    e_2 = l1_error_conv(2)
    e_10 = l1_error_conv(10)
    assert e_10 < e_2, "La convolution ne converge pas."
    print("Test reussi : la regularisation converge en norme L1.")
```
