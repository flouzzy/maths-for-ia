# TP 1 : Approximation d'une indicatrice en Python pur

L'objectif est d'approcher la fonction indicatrice $f = 1_{[-1, 1]}$ par une fonction polynomiale sur $[-2, 2]$ et de mesurer l'erreur empirique dans $L^2$.

```python
def f_indicatrice(x):
    if -1 <= x <= 1:
        return 1.0
    return 0.0

def generate_grid(a, b, n_points):
    step = (b - a) / (n_points - 1)
    return [a + i * step for i in range(n_points)]

def riemann_sum_L2_error(f_true, f_approx, grid, step):
    error = 0.0
    for x in grid:
        diff = f_true(x) - f_approx(x)
        error += diff * diff * step
    return error ** 0.5

# Polynomial approximation of degree 4 fitted empirically for demonstration
def f_poly(x):
    return max(0.0, 1.0 - x*x) ** 2  # Close to indicator

grid = generate_grid(-2, 2, 1000)
step = 4 / 999

err = riemann_sum_L2_error(f_indicatrice, f_poly, grid, step)
print("Erreur L2 estimee :", err)

# Validation
assert err < 1.0, "L'erreur de l'approximation polynomiale est trop elevee."
```
