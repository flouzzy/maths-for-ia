# TP 1 : Approximation d'une indicatrice par des fonctions affines par morceaux

**Objectif :** Implémenter en Python pur la construction de la fonction continue approchant $\mathbf{1}_{[a, b]}$ dans $L^1$ et vérifier numériquement que l'erreur d'approximation correspond à la théorie géométrique.

```python
def indicator(x, a, b):
    if a <= x <= b:
        return 1.0
    return 0.0

def approx_continuous(x, a, b, epsilon):
    if a <= x <= b:
        return 1.0
    elif a - epsilon <= x < a:
        return (x - (a - epsilon)) / epsilon
    elif b < x <= b + epsilon:
        return (b + epsilon - x) / epsilon
    return 0.0

def compute_l1_error(a, b, epsilon, num_points=10000):
    start = a - epsilon - 1
    end = b + epsilon + 1
    dx = (end - start) / num_points
    error = 0.0

    for i in range(num_points):
        x = start + i * dx + dx / 2
        err_val = abs(indicator(x, a, b) - approx_continuous(x, a, b, epsilon))
        error += err_val * dx

    return error

if __name__ == "__main__":
    a, b = 0.0, 1.0
    epsilon = 0.1
    error = compute_l1_error(a, b, epsilon)

    # Theorie : La difference forme deux triangles de base epsilon et hauteur 1
    # Aire = 2 * (epsilon * 1 / 2) = epsilon
    expected_error = epsilon

    print("Erreur L1 calculee :", error)
    print("Erreur L1 theorique :", expected_error)

    assert abs(error - expected_error) < 1e-2, "L'erreur numerique differe de la theorie."
    print("Test reussi.")
```
