# TP 2 : Suite de fonctions étagées convergentes

Dans ce TP, nous encodons l'approximation de Lebesgue d'une fonction positive $f(x) = x^2$ par des fonctions étagées.

```python
def f_carré(x):
    return x * x

def s_n(f, x, n):
    val = f(x)
    for k in range(n * (2 ** n)):
        lower = k / (2 ** n)
        upper = (k + 1) / (2 ** n)
        if lower <= val < upper:
            return lower
    if val >= n:
        return float(n)
    return 0.0

grid = [i * 0.1 for i in range(21)] # Points in [0, 2]

for n in [1, 2, 4]:
    errors = []
    for x in grid:
        errors.append(abs(f_carré(x) - s_n(f_carré, x, n)))
    max_err = max(errors)
    print("Erreur maximale pour n=", n, ":", max_err)

# Validation
err_4 = max([abs(f_carré(x) - s_n(f_carré, x, 4)) for x in grid])
assert err_4 <= 1/(2**4), "L'erreur de la fonction etagee devrait etre inferieure a 1/2^n."
```
