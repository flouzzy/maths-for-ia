## TP 1 : Simulation numerique du defaut de convergence dominee

```python
# TP 1 : La bosse glissante et le defaut de domination
# Cet exemple montre pourquoi la convergence simple ne suffit pas pour intervertir limite et integrale.
import math

# Suite de fonctions : bosse de hauteur n et de largeur 1/n
def fn(x, n):
    # decale legerement selon i pour varier
    shift = 0.01
    if shift < x < shift + 1.0/n:
        return n
    return 0.0

def integrale(func, x_min=0, x_max=2, num_points=2000):
    dx = (x_max - x_min) / num_points
    total = 0.0
    for j in range(num_points):
        x = x_min + j * dx + dx/2
        total += func(x) * dx
    return total

print("Approximation des integrales I_n :")
for n in [10, 50, 100, 500]:
    val = integrale(lambda x: fn(x, n))
    print(f"n = {n:3} : I_n = {val:.6f} (theorique = 1.0)")

# La limite simple est 0 presque partout, donc l'integrale de la limite est 0.
# Pourtant la limite des integrales est 1. Le TCD ne s'applique pas car la
# fonction maximale (enveloppe) est g(x) = 1/(x-shift), qui n'est pas integrable en x=shift.
assert abs(integrale(lambda x: fn(x, 50)) - 1.0) < 0.1, "L'integrale doit valoir 1"
print("Validation : la limite de l'integrale ne correspond pas a l'integrale de la limite.")
```
