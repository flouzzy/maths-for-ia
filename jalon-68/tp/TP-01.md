# TP 1 : Décomposition fonctionnelle (Parties positive et négative)

L'objectif de ce TP est d'implémenter la décomposition canonique d'une fonction arbitraire en sa partie positive et sa partie négative, qui est le socle de la construction de l'intégrale de Lebesgue pour les fonctions de signe quelconque.

```python
def f_plus(f, x):
    # Retourne la partie positive de f en x
    val = f(x)
    return val if val > 0 else 0.0

def f_moins(f, x):
    # Retourne la partie negative de f en x (rendue positive)
    val = f(x)
    return -val if val < 0 else 0.0

def test_decomposition():
    # Fonction test: f(x) = x^2 - 4
    def my_func(x):
        return x**2 - 4

    # En x = 3, f(3) = 5
    assert f_plus(my_func, 3) == 5.0
    assert f_moins(my_func, 3) == 0.0
    assert f_plus(my_func, 3) - f_moins(my_func, 3) == my_func(3)

    # En x = 1, f(1) = -3
    assert f_plus(my_func, 1) == 0.0
    assert f_moins(my_func, 1) == 3.0
    assert f_plus(my_func, 1) - f_moins(my_func, 1) == my_func(1)

    # En x = 2, f(2) = 0
    assert f_plus(my_func, 2) == 0.0
    assert f_moins(my_func, 2) == 0.0
    assert f_plus(my_func, 2) - f_moins(my_func, 2) == my_func(2)

    print("TP 1 : Tests de decomposition valides.")

if __name__ == "__main__":
    test_decomposition()
```
