# TP 4 : Non-intégrabilité de l'intégrale de Dirichlet

Nous montrons numériquement pourquoi $\int |(\sin x) / x| dx$ diverge, validant ainsi la distinction stricte entre l'intégrale de Riemann généralisée et celle de Lebesgue sur les intervalles non bornés.

```python
import math

def dirichlet_abs_integral(K):
    # Calcule l'integrale absolue sur les K premieres arches de 0 a K*pi
    # par la methode des rectangles
    total_area = 0.0
    dx = 0.01

    for k in range(K):
        start = k * math.pi
        # eviter la division par 0
        if start == 0:
            start = dx
        end = (k + 1) * math.pi

        x = start
        while x < end:
            val = abs(math.sin(x)) / x
            total_area += val * dx
            x += dx

    return total_area

def test_dirichlet_divergence():
    area_10 = dirichlet_abs_integral(10)
    area_50 = dirichlet_abs_integral(50)
    area_100 = dirichlet_abs_integral(100)

    # La serie harmonique diverge, donc l'aire grandit
    assert area_50 > area_10
    assert area_100 > area_50

    # On sait que sum_{k=1}^N (1/k) grandit comme log(N)
    assert area_100 > 5.0 # Verifie empiriquement

    print("TP 4 : Divergence de l'integrale absolue de Dirichlet constatee.")

if __name__ == "__main__":
    test_dirichlet_divergence()
```
