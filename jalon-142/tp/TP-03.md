### Implémentation en Pure Python

```python
import numpy as np

# Suppose policy_evaluation is available
def policy_evaluation(P, R, pi, gamma=0.9, epsilon=1e-6):
    num_states = P.shape[0]
    V = np.zeros(num_states)
    while True:
        V_next = np.zeros(num_states)
        for s in range(num_states):
            V_next[s] = R[s, pi[s]] + gamma * np.sum(P[s, pi[s]] * V)
        if np.max(np.abs(V_next - V)) < epsilon:
            break
        V = V_next
    return V

def policy_iteration(P, R, gamma=0.9):
    num_states, num_actions, _ = P.shape
    pi = np.zeros(num_states, dtype=int)

    while True:
        V = policy_evaluation(P, R, pi, gamma)
        policy_stable = True

        for s in range(num_states):
            old_action = pi[s]
            q_values = np.zeros(num_actions)
            for a in range(num_actions):
                q_values[a] = R[s, a] + gamma * np.sum(P[s, a] * V)
            pi[s] = np.argmax(q_values)

            if old_action != pi[s]:
                policy_stable = False

        if policy_stable:
            return pi, V

# Example
P = np.array([[[1.0, 0.0], [0.5, 0.5]], [[0.5, 0.5], [1.0, 0.0]]])
R = np.array([[10, 2], [5, 10]])
pi_star, V_star = policy_iteration(P, R)
print("Policy:", pi_star, "Value:", V_star)
```
