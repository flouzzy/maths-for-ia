# TP 2 : Génération de Fractale par IFS (Iterated Function Systems)

```python
# Une application géométrique : la fougère de Barnsley (approximation par point fixe de formes)
# Les IFS sont des unions d'applications contractantes
import random

def barnsley_fern(iterations=10000):
    # Points de la fractale (l'attracteur est l'unique compact point fixe)
    x, y = 0.0, 0.0
    points = []

    for _ in range(iterations):
        r = random.random()
        # Les 4 contractions affines
        if r < 0.01:
            x_next, y_next = 0.0, 0.16 * y
        elif r < 0.86:
            x_next = 0.85 * x + 0.04 * y
            y_next = -0.04 * x + 0.85 * y + 1.6
        elif r < 0.93:
            x_next = 0.20 * x - 0.26 * y
            y_next = 0.23 * x + 0.22 * y + 1.6
        else:
            x_next = -0.15 * x + 0.28 * y
            y_next = 0.26 * x + 0.24 * y + 0.44

        x, y = x_next, y_next
        points.append((x, y))

    return points

pts = barnsley_fern(100)
assert len(pts) == 100
assert all(isinstance(p[0], float) for p in pts)
print("Points de la fractale générés avec succès via contractions itérées.")
```
