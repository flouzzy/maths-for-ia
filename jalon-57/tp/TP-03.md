# TP 3 : Résolution de système d'équations linéaires (Méthode de Jacobi)

```python
def jacobi_method(A, b, x0, tol=1e-7, max_iter=100):
    # A = D + R, D est la diagonale, R le reste
    # On cherche x t.q Dx = -Rx + b, d'où x = D^(-1)(-Rx + b) (l'opérateur contractant)
    n = len(A)
    x = list(x0)

    # Assert stricte dominance diagonale (implique norme subordonnée < 1)
    for i in range(n):
        diag_val = abs(A[i][i])
        sum_off = sum(abs(A[i][j]) for j in range(n) if i != j)
        assert diag_val > sum_off, "Non diagonale strictement dominante, convergence non garantie."

    for it in range(max_iter):
        x_next = [0] * n
        for i in range(n):
            s = sum(A[i][j] * x[j] for j in range(n) if i != j)
            x_next[i] = (b[i] - s) / A[i][i]

        # Résidu L-infini
        err = max(abs(x_next[i] - x[i]) for i in range(n))
        if err < tol:
            return x_next, it
        x = list(x_next)

    return x, max_iter

# Système
A = [[4, 1, 1],
     [2, 7, 1],
     [1, 2, 8]]
b = [6, 10, 11]
x_init = [0, 0, 0]

sol, it = jacobi_method(A, b, x_init)
assert abs(sol[0] - 1.0) < 1e-5
assert abs(sol[1] - 1.0) < 1e-5
assert abs(sol[2] - 1.0) < 1e-5
print(f"Solution {sol} trouvée en {it} itérations")

```
