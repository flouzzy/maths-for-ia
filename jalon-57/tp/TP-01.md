# TP 1 : Implémentation de l'itération du point fixe (Racine de Héron)

```python
def heron_sqrt(a, x0, tol=1e-7, max_iter=100):
    # La fonction contractante
    def f(x):
        return 0.5 * (x + a / x)

    x = x0
    iterations = []

    for i in range(max_iter):
        x_next = f(x)
        iterations.append(x_next)

        # Critère d'arrêt a posteriori
        if abs(x_next - x) < tol:
            break
        x = x_next

    return x, iterations

# Validation mathématique
val, seq = heron_sqrt(2, 1.5)
assert abs(val**2 - 2) < 1e-6, "Le point fixe n'est pas la racine de 2"
print(f"Point fixe atteint: {val} en {len(seq)} itérations")
```
