# TP 4 : L'Opérateur de Bellman pour un MDP simplissime

```python
def bellman_value_iteration(rewards, transitions, gamma, tol=1e-7):
    # L'espace des états S = {0, 1, ..., n-1}
    n_states = len(rewards)
    V = [0.0] * n_states

    for iteration in range(1000):
        V_next = [0.0] * n_states
        for s in range(n_states):
            # Pour simplifier, MDP avec une seule action (Processus de récompense de Markov)
            expected_future = sum(transitions[s][s_prime] * V[s_prime] for s_prime in range(n_states))
            V_next[s] = rewards[s] + gamma * expected_future

        # Norme infinie (convergence de Banach)
        error = max(abs(V_next[s] - V[s]) for s in range(n_states))
        V = list(V_next)

        if error < tol:
            return V, iteration

    return V, 1000

# 2 états: 0, 1
r = [1.0, 0.0] # Récompenses
# Matrice de transition
P = [[0.8, 0.2],
     [0.1, 0.9]]

v_opt, iters = bellman_value_iteration(r, P, gamma=0.9)
# Validation par résolution analytique V = R + gamma * P * V => (I - gamma P)V = R
# (I - 0.9 P) = [[0.28, -0.18], [-0.09, 0.19]]
# Determinant = 0.28*0.19 - 0.18*0.09 = 0.0532 - 0.0162 = 0.037
# V_0 = (0.19 * 1.0) / 0.037 ~ 5.135
assert abs(v_opt[0] - 5.135) < 0.1
print(f"Value iteration converged in {iters} steps. V={v_opt}")

```
