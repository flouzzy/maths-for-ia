# TP 5 : Un modèle DEQ (Deep Equilibrium) unidimensionnel

```python
import math

def sigmoid(x):
    return 1.0 / (1.0 + math.exp(-x))

def deep_equilibrium_solve(x_input, w, u, b, tol=1e-6, max_iter=200):
    # La fonction est h = sigmoid(W*h + U*x + b)
    # L_sigma = 0.25 (dérivée max de sigmoid)
    # Si |W| < 4, l'application est contractante
    assert abs(w) < 4.0, "La condition de Lipschitz |W| < 4 n'est pas respectée, convergence non garantie."

    h = 0.0 # État caché initial, le point de départ
    iterations = []

    for i in range(max_iter):
        # Propagation (Itération du point fixe)
        h_next = sigmoid(w * h + u * x_input + b)
        iterations.append(h_next)

        if abs(h_next - h) < tol:
            break
        h = h_next

    return h, iterations

# Validation
h_star, hist = deep_equilibrium_solve(x_input=1.0, w=2.0, u=-1.0, b=0.5)
# Validation mathématique (équilibre)
residual = abs(h_star - sigmoid(2.0 * h_star + -1.0 * 1.0 + 0.5))
assert residual < 1e-5, f"L'état n'est pas à l'équilibre: {residual}"
print(f"DEQ a convergé vers l'état stationnaire {h_star} en {len(hist)} itérations.")
```
