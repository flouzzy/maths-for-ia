# TP 2 : Instabilité des Moindres Carrés Ordinaires sous Colinéarité

## Objectif
Ce TP a pour objectif de simuler et de visualiser l'instabilité de la régression linéaire classique (MCO) lorsque les caractéristiques sont quasi-colinéaires (matrice de Gram mal conditionnée), et de montrer comment l'ajout d'une régularisation Ridge résout ce problème.

---

## Code de Simulation et de Validation

```python
import numpy as np

def train_ols(X, y):
    """
    Régression linéaire classique (MCO) : w = (X^T X)^-1 X^T y
    """
    return np.linalg.solve(np.dot(X.T, X), np.dot(X.T, y))

def train_ridge(X, y, lmbda):
    """
    Régression Ridge : w = (X^T X + n * lmbda * I)^-1 X^T y
    """
    n, d = X.shape
    return np.linalg.solve(np.dot(X.T, X) + n * lmbda * np.eye(d), np.dot(X.T, y))

# Fixer la graine
np.random.seed(42)

# Générer un jeu de données hautement colinéaire (d = 2)
n = 100
d = 2

# x1 est aléatoire, x2 est x1 + un très faible bruit
x1 = np.random.uniform(-1, 1, size=(n, 1))
noise = np.random.normal(0, 1e-4, size=(n, 1))
x2 = x1 + noise

X = np.hstack((x1, x2))
y = 2 * x1[:, 0] + np.random.normal(0, 0.1, size=n)

# 1. Entraîner les modèles originaux
w_ols = train_ols(X, y)
w_ridge = train_ridge(X, y, lmbda=0.1)

# 2. Perturber la première cible y[0] d'une petite quantité epsilon
epsilon = 0.05
y_mod = y.copy()
y_mod[0] += epsilon

# 3. Entraîner les modèles perturbés
w_ols_mod = train_ols(X, y_mod)
w_ridge_mod = train_ridge(X, y_mod, lmbda=0.1)

# 4. Calculer l'impact des perturbations sur les poids
diff_ols = np.linalg.norm(w_ols - w_ols_mod)
diff_ridge = np.linalg.norm(w_ridge - w_ridge_mod)

print("--- ANALYSE DE LA COLINÉARITÉ ---")
print(f"Poids OLS originaux           : {w_ols}")
print(f"Poids OLS modifiés            : {w_ols_mod}")
print(f"Écart OLS (||w - w_mod||_2)   : {diff_ols:.6f}")

print(f"\nPoids Ridge originaux         : {w_ridge}")
print(f"Poids Ridge modifiés          : {w_ridge_mod}")
print(f"Écart Ridge (||w - w_mod||_2) : {diff_ridge:.6f}")

# Assertions de validation mathématique
# La régression Ridge doit être exponentiellement ou significativement plus stable que l'OLS
assert diff_ridge < diff_ols, "Erreur: Ridge n'a pas amélioré la stabilité !"
print("\n=> La régulation Ridge stabilise le vecteur de poids (diff_ridge < diff_ols).")

# Vérifier que la borne théorique Ridge est respectée
R = np.max(np.linalg.norm(X, axis=1))
Y_max = np.max(np.abs(y))
w_diff_bound = (2 * R * Y_max) / (0.1 * n)
assert diff_ridge <= w_diff_bound, "Erreur: La borne théorique Ridge est violée !"
print("=> La borne de stabilité Ridge est respectée.")
```
