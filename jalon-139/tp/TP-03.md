# TP 3 : Stabilité et Instabilité du Classifieur K-Plus Proches Voisins

## Objectif
Ce TP démontre de manière empirique l'instabilité du classifieur 1-PPV (1-Plus Proche Voisin) et montre comment l'augmentation du paramètre $K$ stabilise les décisions du modèle en lissant la frontière de décision stochastique.

---

## Code de Simulation et de Validation

```python
import numpy as np

class KNNClassifier:
    """
    Classifieur K-Plus Proches Voisins (K-NN) implémenté à partir de zéro.
    """
    def __init__(self, k=1):
        self.k = k
        self.X_train = None
        self.y_train = None
        
    def fit(self, X, y):
        self.X_train = X
        self.y_train = y
        
    def predict(self, X_test):
        predictions = []
        for x in X_test:
            # Calculer toutes les distances euclidiennes
            dists = np.linalg.norm(self.X_train - x, axis=1)
            # Récupérer les indices des K voisins les plus proches
            nearest_indices = np.argsort(dists)[:self.k]
            # Extraire leurs étiquettes
            nearest_labels = self.y_train[nearest_indices]
            # Vote majoritaire
            pred = 1 if np.mean(nearest_labels) >= 0 else -1
            predictions.append(pred)
        return np.array(predictions)

# Graine aléatoire
np.random.seed(42)

# Générer des données bidimensionnelles (n = 50)
n = 50
d = 2

X = np.random.uniform(-1, 1, size=(n, d))
y = np.sign(X[:, 0] + X[:, 1] + np.random.normal(0, 0.1, size=n)) # classification binaire

# Espace de test pour mesurer le changement de décision
num_test = 1000
X_test = np.random.uniform(-1, 1, size=(num_test, d))

# Lancer la simulation pour K = 1 (instable) et K = 15 (stable)
for k_val in [1, 15]:
    knn = KNNClassifier(k=k_val)
    knn.fit(X, y)
    preds_original = knn.predict(X_test)
    
    # Perturber le 10-ème point en inversant son étiquette
    X_mod = X.copy()
    y_mod = y.copy()
    y_mod[9] = -y_mod[9]
    
    knn_mod = KNNClassifier(k=k_val)
    knn_mod.fit(X_mod, y_mod)
    preds_modified = knn_mod.predict(X_test)
    
    # Calculer la proportion de points de test dont la prédiction a changé
    fraction_changed = np.mean(preds_original != preds_modified)
    
    print(f"Pour K = {k_val:2d} : fraction des prédictions modifiées = {fraction_changed * 100:.2f}%")
    
    if k_val == 1:
        frac_1 = fraction_changed
    else:
        frac_15 = fraction_changed

# Assertion de validation mathématique : K=15 doit être plus stable que K=1
assert frac_15 <= frac_1, "Erreur: K=15 est moins stable que K=1 !"
print("=> La stabilité s'améliore lorsque K augmente (l'influence d'un point est diluée par K).")
```
