# TP 3 : Calcul exact de l'Espérance et Variance (Loi Discrète)

On implémente de manière exacte le calcul de l'espérance et de la variance pour une variable aléatoire discrète arbitraire, sans approximation stochastique.
Ceci correspond à l'application directe de la formule E[X] = Somme(x * P(X=x)) et de la formule de Koenig-Huygens pour la variance.

```python
def expected_value(distribution):
    '''
    Calcule E[X] pour une distribution discrete.
    distribution est un dictionnaire de type {valeur: probabilite}
    '''
    return sum(x * prob for x, prob in distribution.items())

def variance(distribution):
    '''
    Calcule Var(X) = E[X^2] - (E[X])^2
    '''
    e_x = expected_value(distribution)

    # E[X^2] par theoreme de transfert
    e_x_squared = sum((x ** 2) * prob for x, prob in distribution.items())

    return e_x_squared - (e_x ** 2)

if __name__ == '__main__':
    # Exemple 1 : Lancer de dé équilibré
    dice_dist = {1: 1/6, 2: 1/6, 3: 1/6, 4: 1/6, 5: 1/6, 6: 1/6}

    e_dice = expected_value(dice_dist)
    v_dice = variance(dice_dist)

    print(f"De - Esperance : {e_dice:.2f}, Variance : {v_dice:.2f}")
    assert abs(e_dice - 3.5) < 1e-9
    # Variance théorique d'un dé = (6^2 - 1) / 12 = 35 / 12 = 2.9166...
    assert abs(v_dice - (35/12)) < 1e-9

    # Exemple 2 : Loi asymétrique (Bernoulli parametre p=0.8)
    bernoulli_dist = {1: 0.8, 0: 0.2}
    e_bern = expected_value(bernoulli_dist)
    v_bern = variance(bernoulli_dist)

    print(f"Bernoulli - Esperance : {e_bern:.2f}, Variance : {v_bern:.2f}")
    assert abs(e_bern - 0.8) < 1e-9
    # Variance = p(1-p) = 0.8 * 0.2 = 0.16
    assert abs(v_bern - 0.16) < 1e-9

    print("Validation algebrique reussie.")
```
