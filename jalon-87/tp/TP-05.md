# TP 5 : Espérance conditionnelle (Régression au sens de la MSE)

En apprentissage automatique (Machine Learning), la régression cherche la fonction f(x) qui minimise l'erreur quadratique moyenne E[(Y - f(X))^2].
Le théorème fondamental de la régression affirme que cette fonction optimale est précisément l'espérance conditionnelle : f(x) = E[Y | X=x].

Dans ce script, on génère un jeu de données (X, Y) où Y dépend de X avec un bruit aléatoire.
On calcule la moyenne de Y pour chaque sous-ensemble (bin) de X, ce qui approxime empiriquement E[Y | X=x].

```python
import random
import math

def generate_data(n):
    '''Genere (x, y) avec Y = sin(X) + bruit gaussien'''
    data = []
    for _ in range(n):
        x = random.uniform(0, 2 * math.pi)

        # Bruit normal approxime par Box-Muller simplifié
        u1, u2 = random.random(), random.random()
        noise = math.sqrt(-2.0 * math.log(u1)) * math.cos(2.0 * math.pi * u2)
        # variance du bruit = 0.25 (std = 0.5)
        noise *= 0.5

        y = math.sin(x) + noise
        data.append((x, y))
    return data

def compute_conditional_expectation(data, num_bins):
    '''Approxime E[Y | X=x] en discrétisant X en num_bins intervalles (bins)'''
    # Trouver min et max de X
    min_x = 0.0
    max_x = 2 * math.pi
    bin_width = (max_x - min_x) / num_bins

    # Dictionnaires pour sommer Y et compter
    sum_y = {i: 0.0 for i in range(num_bins)}
    count_y = {i: 0 for i in range(num_bins)}

    for x, y in data:
        # Trouver l'indice du bin
        bin_idx = int((x - min_x) / bin_width)
        # Gestion du cas limite
        if bin_idx == num_bins:
            bin_idx -= 1

        sum_y[bin_idx] += y
        count_y[bin_idx] += 1

    # Calculer l'espérance (moyenne) par bin
    conditional_exp = {}
    for i in range(num_bins):
        if count_y[i] > 0:
            conditional_exp[i] = sum_y[i] / count_y[i]
        else:
            conditional_exp[i] = 0.0

    return conditional_exp, bin_width, min_x

if __name__ == '__main__':
    n = 200000
    data = generate_data(n)

    num_bins = 20
    cond_exp, width, min_x = compute_conditional_expectation(data, num_bins)

    print("Approximation de l'Esperance Conditionnelle E[Y|X=x] :")

    # On verifie que l'espérance conditionnelle reconstitue bien la fonction theorique sin(x) (sans le bruit)
    for i in range(num_bins):
        center_x = min_x + (i + 0.5) * width
        theoretical = math.sin(center_x)
        empirical = cond_exp[i]

        # Test sur un point précis (ex: autour de pi/2 = 1.57)
        if abs(center_x - math.pi/2) < width:
            print(f"Autour de x=pi/2 : Empirique={empirical:.4f} | Theorique={theoretical:.4f}")
            assert abs(empirical - theoretical) < 0.1, "L'espérance conditionnelle n'a pas reconstitué le signal."

    print("Validation reussie : E[Y|X=x] approxime correctement la fonction génératrice (sin(x)).")
```
