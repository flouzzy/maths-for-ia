# TP 2 : Estimation Empirique du Vecteur Moyen et de la Matrice de Covariance

## Objectif
Implémenter *from scratch* les estimateurs sans biais du vecteur espérance $\mu$ et de la matrice de covariance $\Sigma$ d'un échantillon i.i.d. de vecteurs gaussiens.

## Code Python

```python
import numpy as np

def estimate_mean_and_covariance(X_samples):
    # X_samples shape: (N, d) where N is number of samples, d is dimension
    X = np.array(X_samples, dtype=float)
    N, d = X.shape

    if N <= 1:
        raise ValueError("Need at least 2 samples to estimate covariance.")

    # 1. Estimate the mean vector mu_hat = (1/N) * sum(X_i)
    # Initialize mean vector with zeros
    mu_hat = np.zeros(d)
    for i in range(N):
        for j in range(d):
            mu_hat[j] += X[i, j]
    mu_hat /= N

    # Or simply: mu_hat = np.sum(X, axis=0) / N

    # 2. Estimate the unbiased covariance matrix Sigma_hat
    # Sigma_hat = (1 / (N - 1)) * sum((X_i - mu_hat)(X_i - mu_hat)^T)
    Sigma_hat = np.zeros((d, d))

    for i in range(N):
        # Compute centered sample (column vector representation conceptually)
        x_centered = X[i, :] - mu_hat

        # Outer product for this sample
        for row in range(d):
            for col in range(d):
                Sigma_hat[row, col] += x_centered[row] * x_centered[col]

    Sigma_hat /= (N - 1)

    return mu_hat, Sigma_hat

# --- Validation ---
if __name__ == "__main__":
    np.random.seed(123)

    # Create synthetic data
    N = 50000
    d = 3
    true_mu = np.array([5.0, -2.0, 0.0])
    # Create a positive definite symmetric matrix
    A = np.random.rand(d, d)
    true_Sigma = A @ A.T + np.eye(d)

    # Generate samples using numpy's built-in function to focus on testing our estimators
    samples = np.random.multivariate_normal(true_mu, true_Sigma, N)

    # Estimate using our implementation
    est_mu, est_Sigma = estimate_mean_and_covariance(samples)

    # Estimate using numpy as baseline
    np_mu = np.mean(samples, axis=0)
    np_Sigma = np.cov(samples, rowvar=False, ddof=1)

    print("Estimated Mean:\n", est_mu)
    print("Numpy Mean:\n", np_mu)
    print("\nEstimated Cov:\n", est_Sigma)
    print("Numpy Cov:\n", np_Sigma)

    # Mathematical assertions
    assert np.allclose(est_mu, np_mu), "Mean computation is incorrect"
    assert np.allclose(est_Sigma, np_Sigma), "Covariance computation is incorrect"
    # Verify unbiasedness over a large sample
    assert np.allclose(true_mu, est_mu, atol=0.05), "Mean not converging to true parameter"
    assert np.allclose(true_Sigma, est_Sigma, atol=0.05), "Covariance not converging to true parameter"

    print("\nValidation successful! Algorithms are numerically correct.")
```
