# TP 4 : Conditionnement d'un Vecteur Gaussien

## Objectif
Implémenter les formules théoriques du conditionnement gaussien. Étant donné un vecteur gaussien partitionné en $Z = (X, Y)^T$, calculer le vecteur moyenne conditionnelle $\mathbb{E}[X|Y=y]$ et la matrice de covariance conditionnelle $\text{Cov}(X|Y=y)$.

## Code Python

```python
import numpy as np

def gaussian_conditioning(mu_X, mu_Y, Sigma_XX, Sigma_YY, Sigma_XY, y_observed):
    # Convert all inputs to numpy arrays
    mu_X = np.array(mu_X, dtype=float)
    mu_Y = np.array(mu_Y, dtype=float)
    Sigma_XX = np.array(Sigma_XX, dtype=float)
    Sigma_YY = np.array(Sigma_YY, dtype=float)
    Sigma_XY = np.array(Sigma_XY, dtype=float)
    y_observed = np.array(y_observed, dtype=float)

    # Dimensions
    dx = len(mu_X)
    dy = len(mu_Y)

    # Assertions on dimensions
    assert Sigma_XX.shape == (dx, dx), "Sigma_XX shape mismatch"
    assert Sigma_YY.shape == (dy, dy), "Sigma_YY shape mismatch"
    assert Sigma_XY.shape == (dx, dy), "Sigma_XY shape mismatch"
    Sigma_YX = Sigma_XY.T

    # Calculate inverse of Sigma_YY
    try:
        Sigma_YY_inv = np.linalg.inv(Sigma_YY)
    except np.linalg.LinAlgError:
        raise ValueError("Sigma_YY must be invertible for conditioning.")

    # 1. Conditional Mean: mu_{X|Y=y} = mu_X + Sigma_XY * Sigma_YY^{-1} * (y - mu_Y)
    # Note: Using matrix multiplication '@'
    innovation = y_observed - mu_Y
    mu_X_given_Y = mu_X + Sigma_XY @ Sigma_YY_inv @ innovation

    # 2. Conditional Covariance: Sigma_{X|Y} = Sigma_XX - Sigma_XY * Sigma_YY^{-1} * Sigma_YX
    # Note that it does not depend on the specific value of y_observed!
    Sigma_X_given_Y = Sigma_XX - Sigma_XY @ Sigma_YY_inv @ Sigma_YX

    # Symmetrize to fix numerical rounding errors
    Sigma_X_given_Y = (Sigma_X_given_Y + Sigma_X_given_Y.T) / 2.0

    return mu_X_given_Y, Sigma_X_given_Y

# --- Validation ---
if __name__ == "__main__":
    # Consider a 3D Gaussian Z = (X_1, X_2, Y_1)
    # We want X = (X_1, X_2) given Y = (Y_1)

    mu_X = [1.0, -1.0]
    mu_Y = [2.0]

    Sigma_XX = [[2.0, 0.5],
                [0.5, 1.0]]
    Sigma_YY = [[3.0]]
    Sigma_XY = [[0.8],
                [-0.4]] # Covariance between X components and Y

    y_val = [5.0] # We observe Y_1 = 5.0 (which is greater than its mean 2.0)

    mu_cond, Sigma_cond = gaussian_conditioning(mu_X, mu_Y, Sigma_XX, Sigma_YY, Sigma_XY, y_val)

    print("Prior Mean of X:", mu_X)
    print("Prior Covariance of X:\n", np.array(Sigma_XX))
    print("\nObserved Y =", y_val)
    print("Conditional Mean of X given Y:\n", mu_cond)
    print("Conditional Covariance of X given Y:\n", Sigma_cond)

    # Mathematical assertions
    # Since y_val > mu_Y, and Cov(X_1, Y) = 0.8 > 0, we expect E[X_1 | Y=y] > E[X_1]
    assert mu_cond[0] > mu_X[0], "Positive correlation should increase the conditional mean."
    # Since Cov(X_2, Y) = -0.4 < 0, we expect E[X_2 | Y=y] < E[X_2]
    assert mu_cond[1] < mu_X[1], "Negative correlation should decrease the conditional mean."

    # Conditional variance should be smaller than prior variance (information reduces uncertainty)
    assert Sigma_cond[0, 0] < Sigma_XX[0][0], "Conditioning must reduce variance."
    assert Sigma_cond[1, 1] < Sigma_XX[1][1], "Conditioning must reduce variance."

    print("\nValidation successful!")
```
