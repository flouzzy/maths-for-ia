# TP 1 : Génération de Vecteurs Gaussiens par Décomposition de Cholesky

## Objectif
Générer un échantillon de vecteurs gaussiens multivariés $\mathcal{N}(\mu, \Sigma)$ à partir de lois normales univariées centrées réduites $\mathcal{N}(0, 1)$, en utilisant la décomposition de Cholesky de la matrice de covariance.

## Code Python

```python
import numpy as np

def generate_gaussian_vector(mu, Sigma, num_samples):
    # Ensure inputs are numpy arrays
    mu = np.array(mu, dtype=float)
    Sigma = np.array(Sigma, dtype=float)

    # Dimension check
    n = len(mu)
    assert Sigma.shape == (n, n), "Sigma must be a square matrix of size len(mu)"

    # Check if Sigma is symmetric positive semi-definite
    assert np.allclose(Sigma, Sigma.T), "Sigma must be symmetric"
    eigenvalues = np.linalg.eigvalsh(Sigma)
    assert np.all(eigenvalues >= -1e-8), "Sigma must be positive semi-definite"

    # 1. Cholesky decomposition: Sigma = L * L^T
    try:
        L = np.linalg.cholesky(Sigma)
    except np.linalg.LinAlgError:
        # Fallback for semi-definite (singular) matrices: Eigenvalue decomposition
        # Sigma = U * D * U^T = (U * sqrt(D)) * (U * sqrt(D))^T
        D, U = np.linalg.eigh(Sigma)
        D[D < 0] = 0 # Handle numerical precision issues
        L = U @ np.diag(np.sqrt(D))

    # 2. Generate standard normal vectors Z ~ N(0, I_n)
    Z = np.random.randn(n, num_samples)

    # 3. Apply affine transformation X = mu + L * Z
    # We need to reshape mu to be a column vector for broadcasting
    mu_col = mu.reshape(n, 1)
    X = mu_col + L @ Z

    return X.T # Return as shape (num_samples, n)

# --- Validation ---
if __name__ == "__main__":
    np.random.seed(42)

    # Parameters
    true_mu = [1.0, 2.0]
    true_Sigma = [[2.0, 0.8],
                  [0.8, 1.5]]
    n_samples = 100000

    # Generation
    samples = generate_gaussian_vector(true_mu, true_Sigma, n_samples)

    # Estimation to verify
    est_mu = np.mean(samples, axis=0)
    # ddof=1 for unbiased estimator
    est_Sigma = np.cov(samples, rowvar=False, ddof=1)

    print("True Mean:", true_mu)
    print("Estimated Mean:", est_mu)
    print("\nTrue Covariance Matrix:\n", np.array(true_Sigma))
    print("Estimated Covariance Matrix:\n", est_Sigma)

    # Assertions for mathematical validation
    assert np.allclose(true_mu, est_mu, atol=0.05), "Mean estimation failed"
    assert np.allclose(true_Sigma, est_Sigma, atol=0.05), "Covariance estimation failed"
    print("\nValidation successful!")
```
