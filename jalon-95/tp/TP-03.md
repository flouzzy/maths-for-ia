# TP 3 : Calcul de la Densité de Probabilité Multivariée

## Objectif
Écrire une fonction robuste pour évaluer la fonction de densité de probabilité (PDF) d'un vecteur gaussien multivarié $\mathcal{N}(\mu, \Sigma)$. Gérer correctement l'inversion de la matrice de covariance et le calcul de son déterminant.

## Code Python

```python
import numpy as np

def multivariate_gaussian_pdf(X, mu, Sigma):
    # Evaluates the pdf at points X. X can be a single vector (d,) or multiple samples (N, d)
    X = np.atleast_2d(np.array(X, dtype=float))
    mu = np.array(mu, dtype=float)
    Sigma = np.array(Sigma, dtype=float)

    d = len(mu)
    N = X.shape[0]

    assert X.shape[1] == d, "Dimensions of X and mu do not match"
    assert Sigma.shape == (d, d), "Dimensions of Sigma and mu do not match"

    # We need the inverse and the determinant of Sigma.
    # For numerical stability, especially in high dimensions,
    # it is often better to use Cholesky or SVD, but we will use standard linear algebra here
    # while checking for singularity.

    try:
        Sigma_inv = np.linalg.inv(Sigma)
        det_Sigma = np.linalg.det(Sigma)
    except np.linalg.LinAlgError:
        raise ValueError("Covariance matrix is singular, PDF is degenerate.")

    if det_Sigma <= 0:
        raise ValueError("Covariance matrix must be positive definite (det > 0).")

    # Normalization constant: 1 / sqrt((2*pi)^d * det(Sigma))
    norm_const = 1.0 / np.sqrt((2 * np.pi) ** d * det_Sigma)

    pdf_values = np.zeros(N)

    for i in range(N):
        x_centered = X[i] - mu
        # Quadratic form: (x - mu)^T * Sigma^{-1} * (x - mu)
        # Using dot products:
        quad_form = np.dot(x_centered, np.dot(Sigma_inv, x_centered))

        # Exponential term
        pdf_values[i] = norm_const * np.exp(-0.5 * quad_form)

    # Return scalar if input was 1D
    if pdf_values.shape[0] == 1 and not hasattr(X, "shape") or X.shape[0] == 1:
        return pdf_values[0]
    return pdf_values

# --- Validation ---
if __name__ == "__main__":
    from scipy.stats import multivariate_normal

    mu = [0.5, -0.2]
    Sigma = [[1.2, 0.4],
             [0.4, 0.9]]

    # Test points
    test_points = np.array([
        [0.5, -0.2],   # At the mean
        [1.0, 1.0],    # Away from mean
        [-2.0, 0.0]    # Far away
    ])

    # Our implementation
    our_pdf = multivariate_gaussian_pdf(test_points, mu, Sigma)

    # Scipy reference
    scipy_pdf = multivariate_normal.pdf(test_points, mean=mu, cov=Sigma)

    for i in range(len(test_points)):
        print(f"Point {test_points[i]}: Our PDF = {our_pdf[i]:.6f}, Scipy PDF = {scipy_pdf[i]:.6f}")

    # Assertions
    assert np.allclose(our_pdf, scipy_pdf), "PDF computation does not match reference."
    assert our_pdf[0] > our_pdf[1], "PDF should be maximum at the mean."
    print("\nValidation successful!")
```
