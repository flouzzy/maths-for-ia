# TP 5 : Calcul de la Distance de Mahalanobis

## Objectif
Implémenter la distance de Mahalanobis, une mesure fondamentale en machine learning pour quantifier la dissimilarité entre des observations issues d'une distribution gaussienne. Vérifier sa relation avec la forme quadratique de la densité gaussienne.

## Code Python

```python
import numpy as np

def mahalanobis_distance(x, y, Sigma):
    \"\"\"
    Computes the Mahalanobis distance between two vectors x and y,
    given the covariance matrix Sigma.
    d_M(x, y) = sqrt( (x-y)^T * Sigma^{-1} * (x-y) )
    \"\"\"
    x = np.array(x, dtype=float)
    y = np.array(y, dtype=float)
    Sigma = np.array(Sigma, dtype=float)

    diff = x - y
    Sigma_inv = np.linalg.inv(Sigma)

    # Calculate quadratic form: diff^T * Sigma^{-1} * diff
    quad_form = np.dot(diff, np.dot(Sigma_inv, diff))

    # Assert it's non-negative (due to positive definiteness of Sigma)
    assert quad_form >= -1e-10, "Quadratic form must be non-negative"

    # Handle minor negative values due to floating point precision
    quad_form = max(0.0, quad_form)

    return np.sqrt(quad_form)

# --- Validation ---
if __name__ == "__main__":
    from scipy.spatial.distance import mahalanobis as scipy_mahalanobis

    mu = np.array([0.0, 0.0])
    # Highly anisotropic covariance
    Sigma = np.array([[10.0, 0.0],
                      [0.0,  0.1]])

    point1 = np.array([3.0, 0.0]) # Far along the high-variance axis (X-axis)
    point2 = np.array([0.0, 3.0]) # Far along the low-variance axis (Y-axis)

    # Euclidean distances from origin
    euc1 = np.linalg.norm(point1 - mu)
    euc2 = np.linalg.norm(point2 - mu)
    print(f"Euclidean distance origin to Point1: {euc1:.2f}")
    print(f"Euclidean distance origin to Point2: {euc2:.2f}")
    assert np.isclose(euc1, euc2), "Euclidean distances are equal"

    # Mahalanobis distances
    mah1 = mahalanobis_distance(point1, mu, Sigma)
    mah2 = mahalanobis_distance(point2, mu, Sigma)
    print(f"\nMahalanobis distance origin to Point1: {mah1:.4f}")
    print(f"Mahalanobis distance origin to Point2: {mah2:.4f}")

    # Since Y-axis has tiny variance, moving 3 units along Y is statistically a huge deviation
    # Since X-axis has huge variance, moving 3 units along X is statistically normal
    assert mah2 > mah1 * 5, "Mahalanobis distance should penalize variance."

    # Verify against scipy
    inv_Sigma = np.linalg.inv(Sigma)
    scipy_mah1 = scipy_mahalanobis(point1, mu, inv_Sigma)
    scipy_mah2 = scipy_mahalanobis(point2, mu, inv_Sigma)

    assert np.isclose(mah1, scipy_mah1), "Mismatch with scipy"
    assert np.isclose(mah2, scipy_mah2), "Mismatch with scipy"

    print("\nValidation successful! Mahalanobis distance correctly scales by covariance.")
```
