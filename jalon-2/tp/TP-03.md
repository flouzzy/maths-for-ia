```python
def check_square_root_2_rational():
    '''
    Simule la recherche d'une fraction p/q = sqrt(2).
    On borne la recherche, mais mathématiquement, on montre qu'elle échoue toujours.
    '''
    for p in range(1, 100):
        for q in range(1, 100):
            # p^2 = 2 * q^2
            if p**2 == 2 * q**2:
                raise AssertionError("Trouvé une fraction rationnelle, impossible !")
    assert True

check_square_root_2_rational()
```
