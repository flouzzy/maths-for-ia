# TP 5 : Simulation dichotomique pour TVI

## 1. Code
```python
def f(x: float) -> float:
    return x**2 - 2

def dichotomie(a: float, b: float, epsilon: float = 1e-6) -> float:
    assert f(a) * f(b) <= 0, "Les bornes doivent encadrer 0"
    while b - a > epsilon:
        c = (a + b) / 2
        if f(a) * f(c) <= 0:
            b = c
        else:
            a = c
    return (a + b) / 2

racine2 = dichotomie(1.0, 2.0)
assert abs(racine2**2 - 2) < 1e-5, f"Racine trouvée: {racine2}"

```
