# TP 1 : Contraposée de la parité

## 1. Code
```python
def est_pair(n: int) -> bool:
    return n % 2 == 0

def test_contraposition():
    '''
    Validation: si n^2 est impair => n est impair
    Contraposée: si n pair => n^2 pair
    '''
    for n in range(100):
        if est_pair(n):
            assert est_pair(n**2), f"Contradiction trouvée pour {n}"
        if not est_pair(n):
            assert not est_pair(n**2), f"Contradiction trouvée pour {n}"

test_contraposition()
```
