# TP 4 : Analyse-synthèse de parité fonctionnelle

## 1. Code
```python
def f(x: float) -> float:
    return x**3 + x**2 + 2*x + 1

def partie_paire(func, x: float) -> float:
    return (func(x) + func(-x)) / 2

def partie_impaire(func, x: float) -> float:
    return (func(x) - func(-x)) / 2

def verif_decomposition():
    for x in range(-10, 11):
        x = float(x)
        # Synthèse: P(x) + I(x) == f(x)
        val_f = f(x)
        val_sum = partie_paire(f, x) + partie_impaire(f, x)
        # Comparaison de floats
        assert abs(val_f - val_sum) < 1e-9, f"Décomposition échouée en {x}"

verif_decomposition()
```
