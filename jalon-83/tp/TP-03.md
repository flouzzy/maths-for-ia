# TP 03 - Filtrage par la Valeur Principale (vp 1/x)

**Script Python `tp_03.py` :**
```python
import math

def cauchy_principal_value(phi, epsilon, R, n_steps=10000):
    # Intègre phi(x)/x sur [-R, -epsilon] U [epsilon, R]
    def integrate_interval(a, b):
        dx = (b - a) / n_steps
        total = 0.0
        for i in range(n_steps):
            x = a + i * dx + dx / 2.0
            total += phi(x) / x * dx
        return total

    int_left = integrate_interval(-R, -epsilon)
    int_right = integrate_interval(epsilon, R)
    return int_left + int_right

if __name__ == '__main__':
    # Fonction test symétrique : phi(x) = exp(-x^2).
    # La vp(1/x) d'une fonction paire est 0.
    def phi_even(x):
        return math.exp(-x*x)

    vp_even = cauchy_principal_value(phi_even, 1e-4, 5.0)
    print(f"Action vp(1/x) sur fonction paire : {vp_even:.6f}")
    assert abs(vp_even) < 1e-3, "L'action sur une fonction paire doit être nulle"

    # Fonction test antisymétrique : phi(x) = x * exp(-x^2)
    # L'intégrale de (x exp(-x^2))/x = exp(-x^2) sur [-R, R] -> sqrt(pi)
    def phi_odd(x):
        return x * math.exp(-x*x)

    vp_odd = cauchy_principal_value(phi_odd, 1e-4, 5.0)
    print(f"Action vp(1/x) sur fonction impaire : {vp_odd:.6f} (Attendu: ~1.7724)")
    assert abs(vp_odd - math.sqrt(math.pi)) < 1e-2, "Calcul vp(1/x) incorrect pour fonction impaire"

    print("TP 03 Validé.")

```
