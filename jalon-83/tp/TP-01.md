# TP 01 - Approximation numérique de la dérivée de Heaviside

**Script Python `tp_01.py` :**
```python
import math

def heaviside_approx(x, eps=1e-3):
    # Approximation lisse : H_eps(x) = 0.5 * (1 + tanh(x / eps))
    return 0.5 * (1.0 + math.tanh(x / eps))

def dirac_approx(x, eps=1e-3):
    # Dérivée classique de H_eps(x)
    # d/dx tanh(x/eps) = (1/eps) * (1 - tanh^2(x/eps))
    # Donc H_eps'(x) = 0.5 / eps * (1 - tanh^2(x/eps))
    t = math.tanh(x / eps)
    return 0.5 / eps * (1.0 - t * t)

def integration_numerique(f, a, b, n=10000):
    # Méthode des rectangles
    dx = (b - a) / n
    total = 0.0
    for i in range(n):
        x = a + i * dx + dx / 2.0
        total += f(x) * dx
    return total

if __name__ == '__main__':
    eps_val = 0.01

    # 1. Vérifions que l'intégrale de l'approximation du Dirac vaut 1 (masse totale)
    def f_dirac(x):
        return dirac_approx(x, eps_val)

    masse = integration_numerique(f_dirac, -1.0, 1.0, 10000)
    print(f"Masse totale de l'approximation : {masse:.6f}")
    assert abs(masse - 1.0) < 1e-3, "La masse de l'approximation du Dirac doit être 1"

    # 2. Action sur une fonction test phi(x) = exp(-x^2)
    def phi(x):
        return math.exp(-x*x)

    def integrand(x):
        return dirac_approx(x, eps_val) * phi(x)

    action_approx = integration_numerique(integrand, -1.0, 1.0, 10000)
    action_theorique = phi(0.0)

    print(f"Action sur phi(x) : {action_approx:.6f} (Théorique : {action_theorique})")
    assert abs(action_approx - action_theorique) < 1e-3, "L'action du Dirac approché ne correspond pas à l'évaluation en 0"

    print("TP 01 Validé.")

```
