# TP 04 - Simulation d'un oscillateur excité par un Dirac (Fonction de Green)

**Script Python `tp_04.py` :**
```python
import math

def simulate_oscillator(dt, n_steps):
    # Simule l'équation y'' + y = F(t) par Verlet
    # avec F(t) une approximation de Dirac en t=0 (force impulsive)

    y = [0.0] * n_steps
    v = 0.0

    # Force impulsive : pour intégrer à 1 sur le pas de temps 0, on applique F = 1/dt
    # y(dt) = y(0) + v(0)dt + 1/2 (F - y(0)) dt^2 -> v(dt) sera modifié.
    # Pour simplifier, une impulsion de Dirac en t=0 confère instantanément une vitesse v=1.

    y[0] = 0.0
    v = 1.0 # Condition post-impulsion

    for i in range(1, n_steps):
        # Euler symplectique
        v = v - y[i-1] * dt
        y[i] = y[i-1] + v * dt

    return y

if __name__ == '__main__':
    dt = 0.01
    n = 1000
    y = simulate_oscillator(dt, n)

    # La théorie dit que la réponse à un Dirac est sin(t) pour t > 0
    t_test = 5.0
    idx_test = int(t_test / dt)

    y_approx = y[idx_test]
    y_theory = math.sin(t_test)

    print(f"Simulation t={t_test}: {y_approx:.4f} | Théorie: {y_theory:.4f}")
    assert abs(y_approx - y_theory) < 0.05, "La réponse impulsionnelle simulée diverge de la théorie (sin(t))"

    print("TP 04 Validé.")

```
