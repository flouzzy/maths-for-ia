# TP 05 - Opérateur de dérivation matricielle (Différences finies 2D)

**Script Python `tp_05.py` :**
```python
def laplacian_1d(signal, dx):
    # Applique le Laplacien discret : (u_{i+1} - 2u_i + u_{i-1}) / dx^2
    n = len(signal)
    lap = [0.0] * n
    for i in range(1, n - 1):
        lap[i] = (signal[i+1] - 2 * signal[i] + signal[i-1]) / (dx * dx)
    return lap

if __name__ == '__main__':
    # Signal f(x) = |x| échantillonné sur [-1, 1]
    dx = 0.1
    n = int(2.0 / dx) + 1
    x_vals = [-1.0 + i * dx for i in range(n)]

    signal = [abs(x) for x in x_vals]

    lap = laplacian_1d(signal, dx)

    # La dérivée seconde de |x| est 2 * Dirac(0)
    # Dans le domaine discret, le Dirac en 0 correspond à la valeur au point d'indice central.
    # L'intégrale (somme * dx) du laplacien doit valoir 2.

    center_idx = int(1.0 / dx)

    print(f"Laplacien au centre (x=0) : {lap[center_idx]:.2f}")

    integral = sum(lap) * dx
    print(f"Masse totale du Laplacien discret : {integral:.2f}")

    assert abs(integral - 2.0) < 1e-5, "La somme des dérivées secondes discrètes ne correspond pas à 2 * Dirac"
    print("TP 05 Validé.")

```
