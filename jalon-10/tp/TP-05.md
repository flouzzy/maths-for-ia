## Partie 1 : Implémentation des Opérations Matricielle de Base

Pour ce TP, une matrice sera représentée comme une liste de listes en Python, où chaque sous-liste représente une ligne de la matrice. Par exemple, `[[1, 2], [3, 4]]` représente la matrice $\begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$.

Implémentez les fonctions suivantes :

1.  `is_square_matrix(matrix)` : Vérifie si une matrice est carrée.
2.  `get_matrix_dimensions(matrix)` : Retourne un tuple `(rows, cols)` des dimensions de la matrice.
3.  `matrix_add(A, B)` : Ajoute deux matrices `A` et `B`.
4.  `scalar_multiply(scalar, A)` : Multiplie une matrice `A` par un scalaire.
5.  `matrix_multiply(A, B)` : Multiplie deux matrices `A` et `B`.
6.  `identity_matrix(n)` : Crée une matrice identité de taille `n x n`.
7.  `transpose(A)` : Transpose une matrice `A`.

**Contraintes :**
*   Gérez les erreurs (par exemple, dimensions incompatibles pour l'addition ou la multiplication).
*   Utilisez des boucles `for` ou des compréhensions de listes, mais pas de fonctions de bibliothèques externes.

### Code Python (Partie 1)

```python
import math

# Constante pour la comparaison de flottants
EPSILON = 1e-9

def get_matrix_dimensions(matrix):
    """Retourne les dimensions (lignes, colonnes) d'une matrice."""
    if not matrix or not isinstance(matrix, list) or not all(isinstance(row, list) for row in matrix):
        raise ValueError("La matrice doit être une liste de listes.")
    rows = len(matrix)
    if rows == 0:
        return (0, 0)
    cols = len(matrix[0])
    if not all(len(row) == cols for row in matrix):
        raise ValueError("Toutes les lignes de la matrice doivent avoir la même longueur.")
    return (rows, cols)

def is_square_matrix(matrix):
    """Vérifie si une matrice est carrée."""
    rows, cols = get_matrix_dimensions(matrix)
    return rows > 0 and rows == cols

def matrix_add(A, B):
    """Additionne deux matrices A et B."""
    rows_A, cols_A = get_matrix_dimensions(A)
    rows_B, cols_B = get_matrix_dimensions(B)

    if rows_A != rows_B or cols_A != cols_B:
        raise ValueError("Les matrices doivent avoir les mêmes dimensions pour l'addition.")

    result = [[0 for _ in range(cols_A)] for _ in range(rows_A)]
    for i in range(rows_A):
        for j in range(cols_A):
            result[i][j] = A[i][j] + B[i][j]
    return result

def scalar_multiply(scalar, A):
    """Multiplie une matrice A par un scalaire."""
    rows, cols = get_matrix_dimensions(A)
    result = [[0 for _ in range(cols)] for _ in range(rows)]
    for i in range(rows):
        for j in range(cols):
            result[i][j] = scalar * A[i][j]
    return result

def matrix_multiply(A, B):
    """Multiplie deux matrices A et B."""
    rows_A, cols_A = get_matrix_dimensions(A)
    rows_B, cols_B = get_matrix_dimensions(B)

    if cols_A != rows_B:
        raise ValueError("Le nombre de colonnes de A doit être égal au nombre de lignes de B pour la multiplication.")

    result = [[0 for _ in range(cols_B)] for _ in range(rows_A)]
    for i in range(rows_A):
        for j in range(cols_B):
            for k in range(cols_A): # ou rows_B
                result[i][j] += A[i][k] * B[k][j]
    return result

def identity_matrix(n):
    """Crée une matrice identité de taille n x n."""
    if not isinstance(n, int) or n <= 0:
        raise ValueError("La taille de la matrice identité doit être un entier positif.")
    matrix = [[0 for _ in range(n)] for _ in range(n)]
    for i in range(n):
        matrix[i][i] = 1
    return matrix

def transpose(A):
    """Transpose une matrice A."""
    rows, cols = get_matrix_dimensions(A)
    result = [[0 for _ in range(rows)] for _ in range(cols)]
    for i in range(rows):
        for j in range(cols):
            result[j][i] = A[i][j]
    return result

def are_matrices_equal(A, B, epsilon=EPSILON):
    """Vérifie si deux matrices sont égales avec une tolérance pour les flottants."""
    rows_A, cols_A = get_matrix_dimensions(A)
    rows_B, cols_B = get_matrix_dimensions(B)

    if rows_A != rows_B or cols_A != cols_B:
        return False

    for i in range(rows_A):
        for j in range(cols_A):
            if abs(A[i][j] - B[i][j]) > epsilon:
                return False
    return True

# --- Assertions pour la Partie 1 ---
print("--- Tests Partie 1 : Opérations Matricielle de Base ---")

# Test get_matrix_dimensions
A = [[1, 2, 3], [4, 5, 6]]
assert get_matrix_dimensions(A) == (2, 3)
assert get_matrix_dimensions([[1]]) == (1, 1)
try:
    get_matrix_dimensions([[1, 2], [3]])
    assert False, "Devrait lever une erreur pour des lignes de longueur différente"
except ValueError:
    assert True

# Test is_square_matrix
assert is_square_matrix([[1, 2], [3, 4]]) == True
assert is_square_matrix([[1, 2, 3], [4, 5, 6]]) == False
assert is_square_matrix([[1]]) == True

# Test matrix_add
A = [[1, 2], [3, 4]]
B = [[5, 6], [7, 8]]
expected_add = [[6, 8], [10, 12]]
assert are_matrices_equal(matrix_add(A, B), expected_add)

try:
    matrix_add(A, [[1]])
    assert False, "Devrait lever une erreur pour des dimensions incompatibles"
except ValueError:
    assert True

# Test scalar_multiply
A = [[1, 2], [3, 4]]
scalar = 2
expected_scalar_mult = [[2, 4], [6, 8]]
assert are_matrices_equal(scalar_multiply(scalar, A), expected_scalar_mult)

# Test matrix_multiply
A = [[1, 2], [3, 4]]
B = [[5, 6], [7, 8]]
expected_mult = [[19, 22], [43, 50]] # (1*5+2*7, 1*6+2*8), (3*5+4*7, 3*6+4*8)
assert are_matrices_equal(matrix_multiply(A, B), expected_mult)

C = [[1, 2, 3], [4, 5, 6]] # 2x3
D = [[7, 8], [9, 10], [11, 12]] # 3x2
expected_mult_CD = [[(1*7+2*9+3*11), (1*8+2*10+3*12)],
                    [(4*7+5*9+6*11), (4*8+5*10+6*12)]] # 2x2
# [[7+18+33, 8+20+36], [28+45+66, 32+50+72]] = [[58, 64], [139, 154]]
assert are_matrices_equal(matrix_multiply(C, D), [[58, 64], [139, 154]])

try:
    matrix_multiply(A, C) # (2x2) * (2x3) -> cols_A != rows_C
    assert False, "Devrait lever une erreur pour des dimensions incompatibles"
except ValueError:
    assert True

# Test identity_matrix
I3 = [[1, 0, 0], [0, 1, 0], [0, 0, 1]]
assert are_matrices_equal(identity_matrix(3), I3)

# Test transpose
```

## Partie 2 : Implémentation de l'Inversion de Matrice (Gauss-Jordan)

L'inversion de matrice est une opération fondamentale pour les changements de base. Vous allez implémenter l'algorithme de Gauss-Jordan pour trouver l'inverse d'une matrice carrée.

L'algorithme de Gauss-Jordan consiste à :
1.  Créer une matrice augmentée `[A | I]`, où `I` est la matrice identité de même taille que `A`.
2.  Appliquer une série d'opérations élémentaires sur les lignes (échange de lignes, multiplication d'une ligne par un scalaire non nul, addition d'un multiple d'une ligne à une autre) pour transformer la partie `A` en `I`.
3.  Lorsque la partie `A` est devenue `I`, la partie `I` de la matrice augmentée est devenue `A^-1`.
4.  Gérer les cas où la matrice n'est pas inversible (déterminant nul, pivot nul impossible à remplacer).

**Implémentez la fonction :**
*   `matrix_inverse(A)` : Calcule l'inverse de la matrice carrée `A`.

**Contraintes :**
*   Implémentez l'algorithme de Gauss-Jordan manuellement.
*   Gérez les erreurs pour les matrices non carrées ou singulières.
*   Utilisez une petite tolérance (`EPSILON`) pour les comparaisons de flottants afin de gérer les imprécisions.

### Code Python (Partie 2)

```python
# ... (Code de la Partie 1 doit être présent ou importé) ...

def matrix_inverse(A):
    """Calcule l'inverse d'une matrice carrée A en utilisant l'algorithme de Gauss-Jordan."""
    rows, cols = get_matrix_dimensions(A)

    if not is_square_matrix(A):
        raise ValueError("La matrice doit être carrée pour être inversible.")

    n = rows
    # Créer la matrice augmentée [A | I]
    augmented_matrix = [row[:] for row in A] # Copie profonde de A
    identity = identity_matrix(n)
    for i in range(n):
        augmented_matrix[i].extend(identity[i])

    for i in range(n):
        # 1. Trouver le pivot pour la colonne i
        pivot_row = i
        for r in range(i + 1, n):
            if abs(augmented_matrix[r][i]) > abs(augmented_matrix[pivot_row][i]):
                pivot_row = r

        # Échanger la ligne actuelle avec la ligne pivot
        augmented_matrix[i], augmented_matrix[pivot_row] = augmented_matrix[pivot_row], augmented_matrix[i]

        pivot = augmented_matrix[i][i]
        if abs(pivot) < EPSILON:
            raise ValueError("La matrice est singulière et non inversible.")

        # 2. Normaliser la ligne pivot (rendre le pivot égal à 1)
        for j in range(2 * n):
            augmented_matrix[i][j] /= pivot

        # 3. Éliminer les autres éléments de la colonne i (rendre les autres éléments de la colonne 0)
        for r in range(n):
            if r != i:
                factor = augmented_matrix[r][i]
                for j in range(2 * n):
                    augmented_matrix[r][j] -= factor * augmented_matrix[i][j]

    # Extraire la matrice inverse de la partie droite de la matrice augmentée
    inverse_matrix = [row[n:] for row in augmented_matrix]
    return inverse_matrix

# --- Assertions pour la Partie 2 ---
print("\n--- Tests Partie 2 : Inversion de Matrice (Gauss-Jordan) ---")

# Test 2x2
A_2x2 = [[2, 1], [1, 1]]
A_2x2_inv_expected = [[1, -1], [-1, 2]]

assert are_matrices_equal(A_2x2_inv, A_2x2_inv_expected)
assert are_matrices_equal(matrix_multiply(A_2x2, A_2x2_inv), identity_matrix(2))
assert are_matrices_equal(matrix_multiply(A_2x2_inv, A_2x2), identity_matrix(2))

# Calculé avec un outil externe pour vérification:

                      [20, -15, -4],
                      [-5, 4, 1]]

assert are_matrices_equal(A_3x3_inv, A_3x3_inv_expected)
assert are_matrices_equal(matrix_multiply(A_3x3, A_3x3_inv), identity_matrix(3))
assert are_matrices_equal(matrix_multiply(A_3x3_inv, A_3x3), identity_matrix(3))

# Test matrice singulière

    matrix_inverse(singular_matrix)
    assert False, "Devrait lever une erreur pour une matrice singulière"

    assert "singulière" in str(e)
```

## Partie 3 : Calcul des Matrices de Passage et Changement de Coordonnées

Maintenant que vous avez les outils de base, vous pouvez implémenter les fonctions pour gérer les changements de base.

Considérons deux bases $B = \{b_1, b_2, \dots, b_n\}$ et $C = \{c_1, c_2, \dots, c_n\}$ d'un espace vectoriel de dimension $n$. Les vecteurs $b_i$ et $c_i$ sont donnés par leurs coordonnées dans la base canonique (standard).

1.  **Matrice de passage de la base $B$ à la base canonique ($P_{B \to Std}$)** :
    Si les vecteurs de la base $B$ sont $b_1, \dots, b_n$ exprimés dans la base canonique, alors la matrice $P_{B \to Std}$ est simplement la matrice dont les colonnes sont les vecteurs $b_i$.

2.  **Matrice de passage de la base canonique à la base $B$ ($P_{Std \to B}$)** :
    C'est l'inverse de $P_{B \to Std}$, c'est-à-dire $P_{Std \to B} = (P_{B \to Std})^{-1}$.

3.  **Matrice de passage de la base $B$ à la base $C$ ($P_{B \to C}$)** :
    Pour passer des coordonnées dans la base $B$ aux coordonnées dans la base $C$, on utilise la formule :
    $P_{B \to C} = P_{Std \to C} \cdot P_{B \to Std} = (P_{C \to Std})^{-1} \cdot P_{B \to Std}$
    Où $P_{C \to Std}$ est la matrice dont les colonnes sont les vecteurs de la base $C$ exprimés dans la base canonique.

**Implémentez les fonctions suivantes :**

1.  `basis_to_standard_matrix(basis_vectors)` : Prend une liste de vecteurs (chaque vecteur étant une liste de flottants, représentant ses coordonnées dans la base canonique) et retourne la matrice de passage de cette base à la base canonique.
2.  `get_transition_matrix(basis_B_vectors, basis_C_vectors)` : Calcule la matrice de passage de la base $B$ à la base $C$. Les `basis_B_vectors` et `basis_C_vectors` sont des listes de vecteurs exprimés dans la base canonique.
3.  `change_coordinates(vector_in_B, transition_matrix_B_to_C)` : Prend un vecteur exprimé dans la base $B$ et la matrice de passage de $B$ à $C$, et retourne le vecteur exprimé dans la base $C$.

### Code Python (Partie 3)

```python
# ... (Code des Parties 1 et 2 doit être présent ou importé) ...

def basis_to_standard_matrix(basis_vectors):
    """
    Construit la matrice de passage d'une base donnée vers la base canonique.
    Les vecteurs de la base sont donnés par leurs coordonnées dans la base canonique.
    La matrice résultante a ces vecteurs comme colonnes.
    """
    if not basis_vectors:
        raise ValueError("La liste des vecteurs de base ne peut pas être vide.")

    n = len(basis_vectors)
    dim_vector = len(basis_vectors[0])

    if not all(len(v) == dim_vector for v in basis_vectors):
        raise ValueError("Tous les vecteurs de la base doivent avoir la même dimension.")

    if n != dim_vector:
        raise ValueError("Le nombre de vecteurs de base doit être égal à la dimension de l'espace.")

    # Transposer la liste des vecteurs pour qu'ils deviennent des colonnes
    # Si basis_vectors = [[b1x, b1y], [b2x, b2y]]
    # On veut [[b1x, b2x], [b1y, b2y]]
    matrix = transpose(basis_vectors)
    return matrix

def get_transition_matrix(basis_B_vectors, basis_C_vectors):
    """
    Calcule la matrice de passage de la base B à la base C.
    Les vecteurs des bases B et C sont donnés par leurs coordonnées dans la base canonique.
    P_B_to_C = (P_C_to_Std)^-1 * P_B_to_Std
    """
    P_B_to_Std = basis_to_standard_matrix(basis_B_vectors)
    P_C_to_Std = basis_to_standard_matrix(basis_C_vectors)

    P_Std_to_C = matrix_inverse(P_C_to_Std)
    P_B_to_C = matrix_multiply(P_Std_to_C, P_B_to_Std)

    return P_B_to_C

def change_coordinates(vector_in_B, transition_matrix_B_to_C):
    """
    Change les coordonnées d'un vecteur de la base B à la base C.
    vector_in_B est un vecteur colonne (liste de listes, ex: [[x], [y]])
    """
    rows_vec, cols_vec = get_matrix_dimensions(vector_in_B)
    rows_trans, cols_trans = get_matrix_dimensions(transition_matrix_B_to_C)

    if cols_vec != 1:
        raise ValueError("Le vecteur doit être un vecteur colonne.")
    if rows_vec != cols_trans:
        raise ValueError("La dimension du vecteur ne correspond pas aux dimensions de la matrice de transition.")

    vector_in_C = matrix_multiply(transition_matrix_B_to_C, vector_in_B)
    return vector_in_C
```
