\section{Présentation du concept clé}
L'essence du concept peut être approchée par l'analogie suivante :
\begin{itemize}
\item \textbf{Forme linéaire :} Imaginez que vous soyez un inspecteur de qualité. Un vecteur est un objet complexe (un produit avec poids, taille, prix). Une \textbf{forme linéaire}, c'est votre test : vous donnez une note unique (un nombre) à cet objet. Si vous testez deux produits ensemble, la note est la somme des notes. C'est un instrument de mesure simple et puissant.
\item \textbf{Hyperplan :} C'est la frontière parfaite. En 2D, c'est une ligne qui sépare le plan en deux. En 3D, c'est une feuille de papier infinie qui sépare l'espace. Un hyperplan, c'est l'ensemble de tous les vecteurs qui reçoivent la note "zéro" par votre test.
\item \textbf{Dualité :} C'est le monde des instruments de mesure. Si les vecteurs sont les "points", les formes linéaires sont les "regards" portés sur ces points.
Nécessité historique et mathématique : Parfois, il est plus facile de décrire un objet par la manière dont il réagit à des tests (le dual) plutôt que par sa structure interne. C'est fondamental pour définir la notion de "perpendiculaire" ou pour séparer des données.
Représentation géométrique : Imaginez une montagne. La hauteur en chaque point est une fonction (peut-être linéaire localement). L'hyperplan, c'est le "niveau de la mer" (altitude 0). L'espace dual, c'est l'ensemble de tous les plans inclinés possibles qui pourraient toucher la montagne.
\end{itemize}

\section{Formalisation}
\subsection*{A. Définitions Formelles}
Soit $E$ un $\mathbb{K}$-espace vectoriel de dimension $n$.
\textbf{1.} \textbf{Forme linéaire :} Une application $\phi : E \to \mathbb{K}$ est une forme linéaire si elle est un morphisme de $\mathbb{K}$-espaces vectoriels. Le typage chirurgical est le suivant : l'espace de départ $E$ est le domaine des objets vectoriels abstraits, tandis que l'espace d'arrivée $\mathbb{K}$ (un corps commutatif, usuellement $\mathbb{R}$ ou $\mathbb{C}$) représente la quantification scalaire. Formellement :
   $$\forall (x, y) \in E \times E, \forall (\lambda, \mu) \in \mathbb{K} \times \mathbb{K}, \quad \phi(\lambda x + \mu y) = \lambda \phi(x) + \mu \phi(y)$$
\textbf{2.} \textbf{Espace Dual ($E^*$) :} L'espace vectoriel $\mathcal{L}(E, \mathbb{K})$ de toutes les formes linéaires sur $E$.
\textbf{3.} \textbf{Hyperplan :} Un sous-espace vectoriel $H$ de $E$ est un hyperplan s'il existe une forme linéaire non nulle $\phi \in E^*$ telle que $H = \ker \phi$.
\textbf{4.} \textbf{Base Duale :} Soit $\mathcal{B} = (e_1, ..., e_n)$ une base de $E$. La base duale $\mathcal{B}^* = (e^*_1, ..., e^*_n)$ est définie par :
   $$e^*_i(e_j) = \delta_{i,j} \quad (\text{symbole de Kronecker})$$
\textbf{5.} \textbf{Orthogonalité (au sens de la dualité) :} Soit $A \subseteq E$. On définit l'orthogonal de $A$ dans $E^*$ par $A^\perp = \{ \phi \in E^* \mid \forall x \in A, \phi(x) = 0 \}$.

\subsection*{B. Théorèmes, Propositions et Lemmes}
\begin{quote}
\textbf{Théorème de la Dimension du Dual :}
\end{quote}
\begin{quote}
Si $E$ est de dimension finie, alors $\dim E^* = \dim E$.
\end{quote}
\begin{quote}
De plus, tout hyperplan $H$ de $E$ est de dimension $n-1$.
\end{quote}

\begin{quote}
\textbf{Isomorphisme de Dualité :}
\end{quote}
\begin{quote}
L'application qui à un vecteur $x$ associe son évaluation $\text{ev}_x : \phi \mapsto \phi(x)$ définit un isomorphisme canonique entre $E$ et son bidual $E^{**}$ (en dimension finie).
\end{quote}


\subsection*{C. Exemples et Cas Pathologiques}
\begin{itemize}
\item \textbf{Exemple Trivial :} La forme linéaire nulle $\phi(x) = 0$ pour tout $x$. Son noyau est $E$ entier. L'image est $\{0\}$. Elle ne définit pas un hyperplan.
\item \textbf{Cas Limite (Dimension Infinie) :} Le théorème de la dimension du dual ($\dim E^* = \dim E$) est \textbf{absolument faux} en dimension infinie. Par exemple, l'espace des polynômes $\mathbb{R}[X]$ a une base dénombrable, mais son dual abstrait a une base de cardinalité indénombrable. L'isomorphisme canonique vers le bidual s'effondre (l'application $\Psi$ est seulement injective, plus surjective).
\end{itemize}

\section{Démonstrations}
\subsection*{Démonstration du Théorème Pivot : Dimension d'un hyperplan}
Soit $H$ un hyperplan de $E$ (dimension $n$). Montrons que $\dim H = n-1$.

\textbf{1.} \textbf{Initialisation / Cadre :} Par définition d'un hyperplan, il existe une forme linéaire $\phi \in E^*$ telle que $\phi \neq 0$ et $H = \ker \phi$.

\textbf{2.} \textbf{Étape 1 : Application du théorème du rang}
   Appliquons le théorème du rang à l'application linéaire $\phi : E \to \mathbb{K}$.
   $$\dim E = \dim(\ker \phi) + \text{rg}(\phi)$$
   On sait que $\ker \phi = H$ et $\text{rg}(\phi) = \dim(\text{Im } \phi)$.

\textbf{3.} \textbf{Étape 2 : Détermination de l'Image de $\phi$}
\begin{itemize}
\item $\phi$ est une application de $E$ vers le corps de base $\mathbb{K}$.
\item L'image $\text{Im } \phi$ est donc un sous-espace vectoriel de $\mathbb{K}$.
\item Les seuls sous-espaces vectoriels de $\mathbb{K}$ (qui est un espace vectoriel de dimension 1 sur lui-même) sont $\{0\}$ et $\mathbb{K}$ lui-même.
\item Comme $\phi \neq 0$ par hypothèse, il existe au moins un vecteur $x$ tel que $\phi(x) \neq 0$.
\item Donc $\text{Im } \phi$ contient au moins un élément non nul, ce qui implique $\text{Im } \phi = \mathbb{K}$.
\end{itemize}

\textbf{4.} \textbf{Étape 3 : Calcul final}
\begin{itemize}
\item On a donc $\dim(\text{Im } \phi) = \dim \mathbb{K} = 1$.
\item L'égalité du théorème du rang devient :
     $n = \dim H + 1$.
\item D'où $\dim H = n - 1$.
\end{itemize}

\textbf{5.} \textbf{Conclusion :} Tout hyperplan d'un espace de dimension $n$ possède une dimension égale à $n-1$.



\subsection*{C. Le Bidual}
L'espace bidual $E^{**}$ est l'espace dual de l'espace dual $E^*$.
Pour tout vecteur $x \in E$, on peut définir l'application d'évaluation $\text{ev}_x : E^* \to \mathbb{K}$ par :
$$\forall \phi \in E^*, \quad \text{ev}_x(\phi) = \phi(x)$$
\begin{quote}
\textbf{Théorème de l'isomorphisme canonique :}
\end{quote}
\begin{quote}
Si $E$ est de dimension finie, l'application $\Psi : x \mapsto \text{ev}_x$ est un isomorphisme canonique (indépendant du choix d'une base) entre $E$ et $E^{**}$.
\end{quote}

\subsection*{Démonstration de l'isomorphisme canonique $E \cong E^{**}$}
\textbf{1.} \textbf{Initialisation :} Montrons que l'application $\Psi : x \mapsto \text{ev}_x$ est linéaire et injective.
\textbf{2.} \textbf{Linéarité :} Soient $x, y \in E$ et $\lambda, \mu \in \mathbb{K}$. Pour toute $\phi \in E^*$,
   $$\Psi(\lambda x + \mu y)(\phi) = \phi(\lambda x + \mu y)$$
   Par linéarité de $\phi$, on a :
   $$\phi(\lambda x + \mu y) = \lambda \phi(x) + \mu \phi(y) = \lambda \Psi(x)(\phi) + \mu \Psi(y)(\phi)$$
   Donc $\Psi(\lambda x + \mu y) = \lambda \Psi(x) + \mu \Psi(y)$.
\textbf{3.} \textbf{Injectivité :} Cherchons le noyau de $\Psi$. Soit $x \in \ker \Psi$. Alors $\Psi(x) = 0_{E^{**}}$, c'est-à-dire que pour toute $\phi \in E^*$, $\phi(x) = 0$.
   Si $x \neq 0$, on peut le compléter en une base $(x, e_2, \dots, e_n)$ de $E$. Considérons la forme coordonnée $e_1^*$ associée à cette base, qui vérifie $e_1^*(x) = 1$. Ceci contredit $\phi(x) = 0$ pour toute forme. Donc $x = 0$, et $\ker \Psi = \{0\}$. $\Psi$ est injective.
\textbf{4.} \textbf{Conclusion :} Puisque $\dim E^{**} = \dim E^* = \dim E = n$ (la dimension finie est cruciale), et que $\Psi$ est une application linéaire injective entre deux espaces de même dimension, c'est un isomorphisme.


\section{Exercices d'Application}
\subsection*{Exercice 1 : Application Directe (Base Duale)}
\textbf{Énoncé :} Dans $\mathbb{R}^2$, soit $\mathcal{B} = (e_1, e_2)$ avec $e_1 = (1, 1)$ et $e_2 = (1, 0)$. Exprimer les formes linéaires $e^*_1$ et $e^*_2$ en fonction des coordonnées canoniques $(x, y)$.
\textbf{Correction Détaillée :}
\textbf{1.} Soit $\phi(x, y) = ax + by$ une forme linéaire.
\textbf{2.} Pour $e^*_1$ :
\begin{itemize}
\item $e^*_1(e_1) = 1 \implies e^*_1(1, 1) = a+b = 1$
\item $e^*_1(e_2) = 0 \implies e^*_1(1, 0) = a = 0$
\item On en tire $a=0$ et $b=1$. Donc $e^*_1(x, y) = y$.
\textbf{3.} Pour $e^*_2$ :
\item $e^*_2(e_1) = 0 \implies e^*_2(1, 1) = a+b = 0$
\item $e^*_2(e_2) = 1 \implies e^*_2(1, 0) = a = 1$
\item On en tire $a=1$ et $b=-1$. Donc $e^*_2(x, y) = x - y$.
\textbf{Conclusion :} $\mathcal{B}^* = (y, x-y)$.
\end{itemize}

\subsection*{Exercice 2 : Niveau Avancé (Intersection d'hyperplans)}
\textbf{Énoncé :} Montrer que l'intersection de deux hyperplans distincts $H_1$ et $H_2$ est un sous-espace de dimension $n-2$.
\textbf{Correction Détaillée :}
\textbf{1.} Soient $\phi_1, \phi_2$ les formes linéaires associées telles que $H_1 = \ker \phi_1$ et $H_2 = \ker \phi_2$.
\textbf{2.} Définissons l'application $\Phi : E \to \mathbb{K}^2$ par $\Phi(x) = (\phi_1(x), \phi_2(x))$.
\textbf{3.} Le noyau de $\Phi$ est $\ker \phi_1 \cap \ker \phi_2 = H_1 \cap H_2$.
\textbf{4.} Par le théorème du rang : $\dim E = \dim(H_1 \cap H_2) + \text{rg}(\Phi)$.
\textbf{5.} Comme $H_1$ et $H_2$ sont distincts, $\phi_1$ et $\phi_2$ ne sont pas proportionnelles. La famille $(\phi_1, \phi_2)$ est donc libre dans $E^*$.
\textbf{6.} L'image de $\Phi$ est donc de dimension 2 (surjection sur $\mathbb{K}^2$).
\textbf{7.} $\text{rg}(\Phi) = 2 \implies n = \dim(H_1 \cap H_2) + 2$.
\textbf{Conclusion :} $\dim(H_1 \cap H_2) = n - 2$.

\section{Application en Intelligence Artificielle}
\begin{itemize}
\item \textbf{Le Pont Théorique :} Les hyperplans sont les \textbf{Séparateurs Linéaires} fondamentaux de l'apprentissage automatique.
\item \textbf{Exemple Concret :} Dans les \textbf{SVM (Support Vector Machines)}, l'algorithme cherche l'hyperplan optimal qui sépare deux classes de données (ex: Spam vs Non-Spam). L'équation de l'hyperplan $\phi(x) + b = 0$ (où $\phi$ est une forme linéaire) définit la frontière de décision. La \textbf{Dualité de Lagrange}, utilisée pour résoudre ce problème d'optimisation, repose entièrement sur le passage de l'espace des données (primal) à l'espace des contraintes (dual).
\end{itemize}

\chapter{Exercice 1: Espace dual en dimension 2}
\section{Énoncé}
Soit $E = \mathbb{R}^2$. On considère la base canonique $(e_1, e_2)$. Déterminer la base duale $(e_1^*, e_2^*)$ et calculer $e_1^*(3e_1 - 2e_2)$.

\section{Correction détaillée}
\textbf{1.} \textbf{Définition de la base duale :} Par définition, la base duale $(e_1^*, e_2^*)$ d'une base $(e_1, e_2)$ vérifie $e_i^*(e_j) = \delta_{ij}$.
\textbf{2.} \textbf{Explicitation des formes linéaires :}
\begin{itemize}
\item $e_1^*(x, y) = e_1^*(xe_1 + ye_2) = x e_1^*(e_1) + y e_1^*(e_2) = x \cdot 1 + y \cdot 0 = x$.
\item $e_2^*(x, y) = e_2^*(xe_1 + ye_2) = x e_2^*(e_1) + y e_2^*(e_2) = x \cdot 0 + y \cdot 1 = y$.
\textbf{3.} \textbf{Calcul de l'évaluation :} On cherche à évaluer $e_1^*$ sur le vecteur $v = 3e_1 - 2e_2$.
\textbf{4.} \textbf{Développement complet :}
   $$e_1^*(3e_1 - 2e_2) = 3 e_1^*(e_1) - 2 e_1^*(e_2)$$
   $$= 3 \times 1 - 2 \times 0$$
   $$= 3$$
\textbf{5.} \textbf{Conclusion :} La valeur de la forme linéaire $e_1^*$ sur le vecteur $3e_1 - 2e_2$ est $3$.
\end{itemize}

\chapter{Exercice 2: Noyau d'une forme linéaire}
\section{Énoncé}
Soit $\phi : \mathbb{R}^3 \to \mathbb{R}$ définie par $\phi(x, y, z) = 2x - y + 3z$. Déterminer une base de $\ker \phi$.

\section{Correction détaillée}
\textbf{1.} \textbf{Définition du noyau :} Le noyau de $\phi$ est l'ensemble des vecteurs sur lesquels la forme linéaire s'annule.
   $$\ker \phi = \{ (x,y,z) \in \mathbb{R}^3 \mid 2x - y + 3z = 0 \}$$
\textbf{2.} \textbf{Résolution de l'équation cartésienne :} L'équation $2x - y + 3z = 0$ équivaut à exprimer une variable en fonction des autres. Isolons $y$ :
   $$y = 2x + 3z$$
\textbf{3.} \textbf{Paramétrisation des vecteurs du noyau :} Un vecteur $v \in \ker \phi$ s'écrit donc :
   $$v = (x, 2x+3z, z)$$
\textbf{4.} \textbf{Décomposition en combinaison linéaire :} On sépare les paramètres libres $x$ et $z$ :
   $$v = (x, 2x, 0) + (0, 3z, z) = x(1, 2, 0) + z(0, 3, 1)$$
\textbf{5.} \textbf{Famille génératrice :} Les vecteurs $u_1 = (1, 2, 0)$ et $u_2 = (0, 3, 1)$ engendrent donc $\ker \phi$.
\textbf{6.} \textbf{Indépendance linéaire :} Supposons $\lambda u_1 + \mu u_2 = 0$.
   $$\lambda(1, 2, 0) + \mu(0, 3, 1) = (\lambda, 2\lambda+3\mu, \mu) = (0, 0, 0)$$
   On obtient immédiatement $\lambda = 0$ et $\mu = 0$. La famille $(u_1, u_2)$ est libre.
\textbf{7.} \textbf{Conclusion :} La famille $((1,2,0), (0,3,1))$ est une base de $\ker \phi$, qui est bien un hyperplan (dimension $3 - 1 = 2$).

\chapter{Exercice 3: Orthogonal d'un sous-espace}
\section{Énoncé}
Soit $F$ un sous-espace vectoriel d'un $\mathbb{K}$-espace vectoriel $E$ de dimension finie $n$. Démontrer le théorème de la dimension de l'orthogonal : $\dim F + \dim F^\perp = n$.

\section{Correction détaillée}
\textbf{1.} \textbf{Définition de l'application restriction :} On considère l'application linéaire $\rho : E^* \to F^*$ définie par $\rho(\phi) = \phi_{|F}$, c'est-à-dire que pour toute forme linéaire $\phi$ sur $E$, $\rho(\phi)$ est sa restriction au sous-espace $F$.
\textbf{2.} \textbf{Étude du noyau de $\rho$ :} Par définition,
   $$\ker \rho = \{ \phi \in E^* \mid \rho(\phi) = 0 \} = \{ \phi \in E^* \mid \forall x \in F, \phi(x) = 0 \}$$
   Or, l'ensemble des formes linéaires qui s'annulent sur $F$ est exactement la définition de l'orthogonal $F^\perp$. Donc $\ker \rho = F^\perp$.
\textbf{3.} \textbf{Étude de l'image de $\rho$ :} L'application $\rho$ est surjective. En effet, soit $\psi \in F^*$. On peut compléter une base $(e_1, \dots, e_p)$ de $F$ en une base $(e_1, \dots, e_p, e_{p+1}, \dots, e_n)$ de $E$. On définit alors une forme $\phi \in E^*$ par $\phi(e_i) = \psi(e_i)$ pour $1 \le i \le p$ et $\phi(e_i) = 0$ pour $i > p$. Ainsi $\phi_{|F} = \psi$.
   Donc $\text{Im}(\rho) = F^*$.
\textbf{4.} \textbf{Application du théorème du rang :} Le théorème du rang appliqué à $\rho : E^* \to F^*$ donne :
   $$\dim E^* = \dim \ker \rho + \dim \text{Im}(\rho)$$
\textbf{5.} \textbf{Substitutions des dimensions :}
\begin{itemize}
\item $\dim E^* = \dim E = n$
\item $\dim \ker \rho = \dim F^\perp$
\item $\dim \text{Im}(\rho) = \dim F^* = \dim F$
   On obtient : $n = \dim F^\perp + \dim F$.
\textbf{6.} \textbf{Conclusion :} L'égalité $\dim F + \dim F^\perp = n$ est rigoureusement démontrée.
\end{itemize}

\chapter{Exercice 4: Hyperplans et formes proportionnelles}
\section{Énoncé}
Montrer que deux formes linéaires non nulles $\phi$ et $\psi$ ont le même noyau si et seulement si elles sont proportionnelles.

\section{Correction détaillée}
\textbf{1.} \textbf{Sens direct (proportionnalité implique même noyau) :}
   Supposons qu'il existe un scalaire $\lambda \neq 0$ tel que $\phi = \lambda \psi$.
   Soit $x \in \ker \phi$. Alors $\phi(x) = 0$, donc $\lambda \psi(x) = 0$. Comme $\lambda \neq 0$, on a $\psi(x) = 0$, donc $x \in \ker \psi$. Ainsi $\ker \phi \subset \ker \psi$.
   Symétriquement, $\psi = \frac{1}{\lambda} \phi$, ce qui donne $\ker \psi \subset \ker \phi$. Donc $\ker \phi = \ker \psi$.
\textbf{2.} \textbf{Sens réciproque (même noyau implique proportionnalité) :}
   Supposons que $\ker \phi = \ker \psi = H$. Comme $\phi \neq 0$, il existe un vecteur $e_0 \in E$ tel que $\phi(e_0) \neq 0$. Quitte à diviser par $\phi(e_0)$, on peut choisir $e_0$ tel que $\phi(e_0) = 1$.
\textbf{3.} \textbf{Décomposition d'un vecteur :} Soit $x \in E$ quelconque. Posons $x' = x - \phi(x)e_0$.
\textbf{4.} \textbf{Évaluation de $x'$ :} Évaluons $\phi$ en $x'$ :
   $$\phi(x') = \phi(x - \phi(x)e_0) = \phi(x) - \phi(x)\phi(e_0) = \phi(x) - \phi(x)(1) = 0$$
   Donc $x' \in \ker \phi$.
\textbf{5.} \textbf{Utilisation de l'égalité des noyaux :} Puisque $\ker \phi = \ker \psi$, on a nécessairement $x' \in \ker \psi$, ce qui implique $\psi(x') = 0$.
\textbf{6.} \textbf{Relation de proportionnalité :}
   $$\psi(x - \phi(x)e_0) = 0 \implies \psi(x) - \phi(x)\psi(e_0) = 0 \implies \psi(x) = \psi(e_0)\phi(x)$$
   Ceci étant vrai pour tout $x \in E$, en posant $\lambda = \psi(e_0)$, on obtient l'égalité des formes linéaires $\psi = \lambda \phi$.
\textbf{7.} \textbf{Conclusion :} Les deux formes linéaires sont proportionnelles.

\chapter{Exercice 5: Équation d'un hyperplan en dimension n}
\section{Énoncé}
Soit $E$ un espace vectoriel de dimension $n$ et $(e_1, \dots, e_n)$ une base de $E$. Montrer qu'un hyperplan $H$ est caractérisé par une équation cartésienne de la forme $\sum_{i=1}^n a_i x_i = 0$ où les $(a_i)$ ne sont pas tous nuls.

\section{Correction détaillée}
\textbf{1.} \textbf{Caractérisation par une forme linéaire :} Par définition, un hyperplan $H$ est le noyau d'une forme linéaire non nulle, notons-la $\phi : E \to \mathbb{K}$.
   $$H = \{ x \in E \mid \phi(x) = 0 \}$$
\textbf{2.} \textbf{Décomposition dans la base :} Soit un vecteur quelconque $x \in E$. Il admet une unique décomposition dans la base $(e_1, \dots, e_n)$ :
   $$x = \sum_{i=1}^n x_i e_i$$
   où les $x_i$ sont les coordonnées de $x$.
\textbf{3.} \textbf{Application de la linéarité :} Appliquons $\phi$ au vecteur $x$ :
   $$\phi(x) = \phi\left(\sum_{i=1}^n x_i e_i\right)$$
   Par linéarité de $\phi$, la somme et les scalaires sortent :
   $$\phi(x) = \sum_{i=1}^n x_i \phi(e_i)$$
\textbf{4.} \textbf{Identification des coefficients :} Posons pour tout $i \in \{1, \dots, n\}$, $a_i = \phi(e_i)$. L'équation d'appartenance à $H$ devient alors :
   $$\sum_{i=1}^n a_i x_i = 0$$
\textbf{5.} \textbf{Non-nullité des coefficients :} Comme $\phi$ est une forme linéaire non nulle, il existe au moins un vecteur de base $e_{i_0}$ tel que $\phi(e_{i_0}) \neq 0$. Donc, il existe au moins un $a_i$ tel que $a_i \neq 0$.
\textbf{6.} \textbf{Conclusion :} Tout hyperplan est rigoureusement caractérisé par une équation linéaire homogène dont les coefficients ne sont pas tous nuls.

\chapter{Exercice 6: Bidual d'un espace vectoriel}
\section{Énoncé}
Pour tout $x \in E$, on définit l'application d'évaluation $\text{ev}_x : E^* \to \mathbb{K}$ par $\text{ev}_x(\phi) = \phi(x)$. Montrer que l'application $\Psi : x \mapsto \text{ev}_x$ est linéaire et injective.

\section{Correction détaillée}
\textbf{1.} \textbf{Étape 1:} $\text{ev}_x$ est bien une forme linéaire sur $E^*$ car $\text{ev}_x(\phi_1 + \lambda \phi_2) = (\phi_1 + \lambda \phi_2)(x) = \phi_1(x) + \lambda \phi_2(x) = \text{ev}_x(\phi_1) + \lambda \text{ev}_x(\phi_2)$.
\textbf{2.} \textbf{Étape 2:} L'application $\Psi$ est linéaire. Soit $x, y \in E$ et $\lambda, \mu \in \mathbb{K}$. Évaluons $\Psi(\lambda x + \mu y)$ sur un élément quelconque $\phi \in E^*$ :
   $$\Psi(\lambda x + \mu y)(\phi) = \phi(\lambda x + \mu y) = \lambda \phi(x) + \mu \phi(y) = \lambda \Psi(x)(\phi) + \mu \Psi(y)(\phi)$$
   Ce qui démontre la linéarité.
\textbf{3.} \textbf{Étape 3:} Montrons que $\Psi$ est injective. Soit $x \in \ker \Psi$. Alors pour toute forme linéaire $\phi \in E^*$, on a $\phi(x) = 0$.
\textbf{4.} \textbf{Étape 4:} Si $x \neq 0$, on pourrait le compléter en une base $(x, e_2, \dots, e_n)$ et définir la forme coordonnée $\phi=x^*$ telle que $x^*(x)=1$, ce qui contredit $\phi(x)=0$.
\textbf{5.} \textbf{Conclusion:} Par conséquent $x=0$, d'où $\ker \Psi = \{0\}$. $\Psi$ est injective.

\chapter{Exercice 7: Trace comme forme linéaire}
\section{Énoncé}
L'application $\text{Tr} : M_n(\mathbb{K}) \to \mathbb{K}$ est une forme linéaire. Montrer que tout hyperplan $H$ de $M_n(\mathbb{K})$ contient au moins une matrice inversible (pour $n \ge 2$).

\section{Correction détaillée}
\textbf{1.} \textbf{Étape 1:} Un hyperplan de $M_n(\mathbb{K})$ est le noyau d'une forme linéaire non nulle $\phi$. Il est connu que toute forme linéaire sur $M_n(\mathbb{K})$ s'écrit $\phi(M) = \text{Tr}(AM)$ pour une unique matrice $A \in M_n(\mathbb{K})$. Ainsi, $H = \{ M \in M_n(\mathbb{K}) \mid \text{Tr}(AM) = 0 \}$.
\textbf{2.} \textbf{Étape 2:} On cherche $M \in H$ telle que $\det(M) \neq 0$. Si $A=0$, $\phi=0$ ce qui est exclu. Supposons par l'absurde que $H$ ne contient aucune matrice inversible.
\textbf{3.} \textbf{Étape 3:} L'hyperplan $H$ est un sous-espace vectoriel de dimension $n^2-1$. Si $H$ ne contient que des matrices singulières, on a une contradiction avec les résultats de la théorie des espaces de matrices de rang borné (théorème de Dieudonné), car la dimension maximale d'un sous-espace de matrices non-inversibles est $n(n-1)$, et pour $n \ge 2$, $n^2-1 > n(n-1)$.
\textbf{4.} \textbf{Conclusion:} L'hypothèse de départ est fausse, donc on en déduit formellement qu'un hyperplan contient toujours des éléments inversibles.

\chapter{Exercice 8: Orthogonalité croisée}
\section{Énoncé}
Soient $F_1, F_2$ deux sous-espaces de $E$. Montrer que $(F_1 + F_2)^\perp = F_1^\perp \cap F_2^\perp$.

\section{Correction détaillée}
\textbf{1.} \textbf{Étape 1 :} Soit $\phi \in (F_1 + F_2)^\perp$. Alors pour tout $x \in F_1 + F_2, \phi(x)=0$.
   En particulier, pour tout $x_1 \in F_1$, $\phi(x_1)=0$ (car $F_1 \subset F_1+F_2$), donc $\phi \in F_1^\perp$.
   De même $\phi \in F_2^\perp$. Ainsi $(F_1 + F_2)^\perp \subset F_1^\perp \cap F_2^\perp$.
\textbf{2.} \textbf{Étape 2 :} Réciproquement, soit $\phi \in F_1^\perp \cap F_2^\perp$. Pour tout $x \in F_1+F_2$, on peut écrire $x = x_1 + x_2$ avec $x_1 \in F_1, x_2 \in F_2$.
\textbf{3.} \textbf{Étape 3 :} On a par linéarité $\phi(x) = \phi(x_1+x_2) = \phi(x_1) + \phi(x_2)$.
\textbf{4.} \textbf{Étape 4 :} Puisque $\phi \in F_1^\perp$, $\phi(x_1)=0$. Puisque $\phi \in F_2^\perp$, $\phi(x_2)=0$. Donc $\phi(x) = 0 + 0 = 0$. Ainsi $\phi \in (F_1+F_2)^\perp$. On a l'inclusion réciproque.
\textbf{5.} \textbf{Conclusion:} Par double inclusion, l'égalité $(F_1 + F_2)^\perp = F_1^\perp \cap F_2^\perp$ est strictement démontrée.

\chapter{Exercice 9: Indépendance linéaire de formes linéaires}
\section{Énoncé}
Soient $\phi_1, \dots, \phi_p \in E^*$. Montrer qu'elles sont linéairement indépendantes si et seulement si l'intersection de leurs noyaux $\bigcap_{i=1}^p \ker \phi_i$ est de dimension $n-p$.

\section{Correction détaillée}
\textbf{1.} \textbf{Étape 1:} On définit l'application linéaire $\Phi : E \to \mathbb{K}^p$ par $\Phi(x) = (\phi_1(x), \dots, \phi_p(x))$.
\textbf{2.} \textbf{Étape 2:} Le noyau de $\Phi$ est exactement l'intersection des noyaux : $\ker \Phi = \bigcap_{i=1}^p \ker \phi_i$.
\textbf{3.} \textbf{Étape 3:} D'après le théorème du rang, $\dim E = \dim \ker \Phi + \dim \text{Im}(\Phi)$. Donc $\dim \bigcap \ker \phi_i = n - \dim \text{Im}(\Phi)$.
\textbf{4.} \textbf{Étape 4:} L'image de la transposée $\Phi^t : (\mathbb{K}^p)^* \to E^*$ est le sous-espace engendré par les $\phi_i$. Or $\text{rg}(\Phi) = \text{rg}(\Phi^t) = \dim \text{Vect}(\phi_1, \dots, \phi_p)$.
\textbf{5.} \textbf{Étape 5:} Ainsi, $\dim \text{Vect}(\phi_1, \dots, \phi_p) = p$ (c'est-à-dire que la famille est libre) si et seulement si $\text{rg}(\Phi) = p$, ce qui équivaut à $\dim \bigcap \ker \phi_i = n - p$.
\textbf{6.} \textbf{Conclusion:} L'équivalence est rigoureusement prouvée via l'application associée et le théorème du rang.

\chapter{Exercice 10: Polynômes de Lagrange et dualité}
\section{Énoncé}
Dans $E = \mathbb{R}_{n-1}[X]$, on se donne $n$ scalaires distincts $a_1, \dots, a_n$. Montrer que les formes linéaires d'évaluation $\phi_i(P) = P(a_i)$ forment une base de $E^*$.

\section{Correction détaillée}
\textbf{1.} \textbf{Étape 1:} La dimension de $E$ est $n$, donc $\dim E^* = n$. Pour démontrer qu'une famille de $n$ vecteurs forme une base, il suffit de montrer que la famille $(\phi_1, \dots, \phi_n)$ est libre.
\textbf{2.} \textbf{Étape 2:} Supposons une combinaison linéaire nulle: $\sum_{i=1}^n \lambda_i \phi_i = 0$. Cela signifie que pour tout polynôme $P \in E$, $\sum_{i=1}^n \lambda_i P(a_i) = 0$.
\textbf{3.} \textbf{Étape 3:} On introduit les polynômes interpolateurs de Lagrange $L_j(X) = \prod_{k \neq j} \frac{X-a_k}{a_j-a_k}$. Ce sont des éléments de $E$ car leur degré est exactement $n-1$.
\textbf{4.} \textbf{Étape 4:} Par construction, $L_j(a_i) = \delta_{ij}$ (vaut 1 si $i=j$, 0 sinon).
\textbf{5.} \textbf{Étape 5:} Évaluons la combinaison linéaire nulle sur le polynôme $L_j$ :
   $$0 = \sum_{i=1}^n \lambda_i \phi_i(L_j) = \sum_{i=1}^n \lambda_i L_j(a_i) = \lambda_j$$
\textbf{6.} \textbf{Conclusion:} Pour tout indice $j$, on obtient $\lambda_j = 0$. La famille est donc libre, et c'est par conséquent une base. Les polynômes $(L_j)$ forment précisément la base antéduale associée.

def intersect_2d_lines(a1, b1, d1, a2, b2, d2):
    # Calcul du determinant du systeme
    det = a1*b2 - a2*b1
    if det == 0:
        return None # Les hyperplans (droites) sont paralleles ou confondus

    # Resolution exacte par la regle de Cramer
    x = (d1*b2 - d2*b1) / det
    y = (a1*d2 - a2*d1) / det
    return (x, y)

# Intersection de x - y = 0 et x + y = 2
assert intersect_2d_lines(1, -1, 0, 1, 1, 2) == (1.0, 1.0)

\end{lstlisting}

def riesz_representation(form_eval_func, dimension):
    # La forme lineaire est entierement determinee par ses valeurs sur la base canonique.
    # On reconstruit le vecteur representant `u` tel que phi(x) = <u, x>

    representing_vector = []
    for i in range(dimension):
        # Creation du i-eme vecteur de la base canonique e_i
        e_i = [0] * dimension
        e_i[i] = 1

        representing_vector.append(a_i)

    return representing_vector
