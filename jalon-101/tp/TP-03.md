## Introduction au Conditionnement

Un problème est dit **bien posé** si :
1.  Une solution existe.
2.  La solution est unique.
3.  La solution dépend continûment des données d'entrée (de petites modifications des entrées n'entraînent que de petites modifications des sorties).

Un problème qui ne satisfait pas la troisième condition est dit **mal posé**. Le conditionnement d'une matrice $A$ (noté `kappa(A)`) quantifie la sensibilité de la solution $x$ du système linéaire $Ax = b$ à de petites variations de $A$ ou de $b$. Un grand nombre de conditionnement indique une grande sensibilité, rendant le problème potentiellement mal posé numériquement. Le conditionnement est typiquement défini comme $kappa(A) = ||A|| \cdot ||A^{-1}||$, où $||.||$ est une norme matricielle.

import sys

# --- Partie 1 : Opérations matricielles de base (implémentation from scratch) ---

def creer_matrice_nulle(lignes, colonnes):
    """Crée une matrice remplie de zéros."""
    return [[0.0 for _ in range(colonnes)] for _ in range(lignes)]

def multiplier_matrices(A, B):
    """
    Multiplie deux matrices A et B.
    A: liste de listes (matrice de taille Lignes_A x Colonnes_A)
    B: liste de listes (matrice de taille Lignes_B x Colonnes_B)
    """
    lignes_A = len(A)
    colonnes_A = len(A[0])
    lignes_B = len(B)
    colonnes_B = len(B[0])

    assert colonnes_A == lignes_B, \
        "Le nombre de colonnes de A doit être égal au nombre de lignes de B pour la multiplication."

    C = creer_matrice_nulle(lignes_A, colonnes_B)

    for i in range(lignes_A):
        for j in range(colonnes_B):
            for k in range(colonnes_A):
                C[i][j] += A[i][k] * B[k][j]
    return C

def multiplier_matrice_vecteur(A, v):
    """
    Multiplie une matrice A par un vecteur v.
    A: liste de listes (matrice de taille Lignes_A x Colonnes_A)
    v: liste (vecteur colonne de taille Lignes_A)
    """
    lignes_A = len(A)
    colonnes_A = len(A[0])
    lignes_v = len(v)

    assert colonnes_A == lignes_v, \
        "Le nombre de colonnes de A doit être égal à la dimension du vecteur v."

    resultat = [0.0] * lignes_A
    for i in range(lignes_A):
        for j in range(colonnes_A):
            resultat[i] += A[i][j] * v[j]
    return resultat

def ajouter_vecteurs(v1, v2):
    """Ajoute deux vecteurs."""
    assert len(v1) == len(v2), "Les vecteurs doivent avoir la même dimension pour l'addition."
    return [v1[i] + v2[i] for i in range(len(v1))]

def soustraire_vecteurs(v1, v2):
    """Soustrait deux vecteurs."""
    assert len(v1) == len(v2), "Les vecteurs doivent avoir la même dimension pour la soustraction."
    return [v1[i] - v2[i] for i in range(len(v1))]

def multiplier_scalaire_vecteur(scalaire, v):
    """Multiplie un vecteur par un scalaire."""
    return [scalaire * x for x in v]

def calculer_norme_euclidienne_vecteur(v):
    """Calcule la norme euclidienne (L2) d'un vecteur."""
    return math.sqrt(sum(x**2 for x in v))

print("--- Vérification des opérations de base ---")
A_test = [[1, 2], [3, 4]]
B_test = [[5, 6], [7, 8]]
C_expected = [[19, 22], [43, 50]] # (1*5+2*7, 1*6+2*8), (3*5+4*7, 3*6+4*8)
C_actual = multiplier_matrices(A_test, B_test)
print(f"Multiplication matricielle A*B: {C_actual}")
assert C_actual == C_expected, "La multiplication matricielle est incorrecte!"

v_test = [1, 2]
Av_expected = [5, 11] # (1*1+2*2, 3*1+4*2)
Av_actual = multiplier_matrice_vecteur(A_test, v_test)
print(f"Multiplication matrice-vecteur A*v: {Av_actual}")
assert Av_actual == Av_expected, "La multiplication matrice-vecteur est incorrecte!"
print("Opérations de base vérifiées avec succès.\n")

# --- Partie 2 : Résolution de systèmes linéaires et conditionnement ---

def resoudre_systeme_lineaire(A_np, b_np):
    """
    Résout le système linéaire Ax = b.
    Utilise numpy.linalg.solve pour des raisons de robustesse et performance.
    A_np: matrice numpy
    b_np: vecteur numpy
    """
    try:
        x_np = np.linalg.solve(A_np, b_np)
        return x_np
    except np.linalg.LinAlgError as e:
        print(f"Erreur lors de la résolution du système linéaire : {e}")
        return None

def calculer_conditionnement(A_np):
    """
    Calcule le nombre de conditionnement d'une matrice.
    Utilise numpy.linalg.cond. Par défaut, utilise la norme 2.
    A_np: matrice numpy
    """
    return np.linalg.cond(A_np)

def calculer_erreur_relative(valeur_originale, valeur_perturbee):
    """
    Calcule l'erreur relative ||valeur_perturbee - valeur_originale|| / ||valeur_originale||
    Pour les vecteurs, nous utilisons la norme euclidienne.
    """
    if isinstance(valeur_originale, (list, np.ndarray)):
        # Si c'est un vecteur ou un tableau numpy, utiliser la norme euclidienne
        norme_diff = calculer_norme_euclidienne_vecteur(soustraire_vecteurs(valeur_perturbee, valeur_originale))
        norme_originale = calculer_norme_euclidienne_vecteur(valeur_originale)
    else:
        # Si c'est un scalaire
        norme_diff = abs(valeur_perturbee - valeur_originale)
        norme_originale = abs(valeur_originale)

    if norme_originale == 0:
        return float('inf') if norme_diff != 0 else 0.0
    return norme_diff / norme_originale

print("--- Illustration avec une matrice bien conditionnée ---")

# Définition d'une matrice bien conditionnée
A_bien_conditionnee = np.array([[2.0, 1.0], [1.0, 2.0]])
b_original = np.array([3.0, 3.0])

print(f"Matrice bien conditionnée A:\n{A_bien_conditionnee}")
print(f"Vecteur b original: {b_original}")

# Calcul du nombre de conditionnement
cond_A_bien = calculer_conditionnement(A_bien_conditionnee)
print(f"Nombre de conditionnement de A: {cond_A_bien:.2f}")

# Résolution du système Ax = b
x_original = resoudre_systeme_lineaire(A_bien_conditionnee, b_original)
print(f"Solution x originale: {x_original}")

# Vérification de la solution
b_reconstruit = np.dot(A_bien_conditionnee, x_original)
assert np.allclose(b_reconstruit, b_original), "La solution x_original n'est pas correcte pour la matrice bien conditionnée."
print(f"A * x_original = {b_reconstruit} (vérifié)\n")

# Petite perturbation du vecteur b
perturbation = 0.01 # Une petite perturbation
b_perturbe = b_original + np.array([perturbation, -perturbation]) # Perturbation dans des directions opposées
print(f"Vecteur b perturbé: {b_perturbe}")

# Résolution du système avec b perturbé
x_perturbe = resoudre_systeme_lineaire(A_bien_conditionnee, b_perturbe)
print(f"Solution x perturbée: {x_perturbe}")

# Calcul des erreurs relatives
erreur_relative_b = calculer_erreur_relative(b_original, b_perturbe)
erreur_relative_x = calculer_erreur_relative(x_original, x_perturbe)

print(f"Erreur relative sur b: {erreur_relative_b:.4e}")
print(f"Erreur relative sur x: {erreur_relative_x:.4e}")

# Assertion pour vérifier que l'amplification d'erreur est raisonnable
# Pour un problème bien conditionné, l'amplification (erreur_x / erreur_b) doit être proche du cond.
# Nous utilisons un facteur de tolérance pour cette vérification.
amplification_erreur = erreur_relative_x / erreur_relative_b
print(f"Amplification de l'erreur (erreur_x / erreur_b): {amplification_erreur:.2f}")
assert amplification_erreur < 2 * cond_A_bien, \
    f"L'amplification de l'erreur {amplification_erreur:.2f} est trop élevée " \
    f"pour une matrice bien conditionnée (cond={cond_A_bien:.2f})."
print("La matrice bien conditionnée montre une bonne stabilité face aux perturbations.\n")


print("--- Illustration avec une matrice mal conditionnée ---")

# Définition d'une matrice mal conditionnée (proche de la singularité)
# Les deux lignes sont très similaires
A_mal_conditionnee = np.array([[1.000, 0.999], [0.999, 0.998]]) # Matrice de Hilbert 2x2 approchée
b_original_mal = np.array([2.0, 2.0])

print(f"Matrice mal conditionnée A:\n{A_mal_conditionnee}")
print(f"Vecteur b original: {b_original_mal}")

# Calcul du nombre de conditionnement
cond_A_mal = calculer_conditionnement(A_mal_conditionnee)
print(f"Nombre de conditionnement de A: {cond_A_mal:.2f}")

# Résolution du système Ax = b
x_original_mal = resoudre_systeme_lineaire(A_mal_conditionnee, b_original_mal)
print(f"Solution x originale: {x_original_mal}")

# Vérification de la solution
b_reconstruit_mal = np.dot(A_mal_conditionnee, x_original_mal)
assert np.allclose(b_reconstruit_mal, b_original_mal), \
    "La solution x_original n'est pas correcte pour la matrice mal conditionnée."
print(f"A * x_original = {b_reconstruit_mal} (vérifié)\n")

# Petite perturbation du vecteur b
perturbation_mal = 0.001 # Une perturbation encore plus petite pour une matrice mal conditionnée
b_perturbe_mal = b_original_mal + np.array([perturbation_mal, -perturbation_mal])
print(f"Vecteur b perturbé: {b_perturbe_mal}")

# Résolution du système avec b perturbé
x_perturbe_mal = resoudre_systeme_lineaire(A_mal_conditionnee, b_perturbe_mal)
print(f"Solution x perturbée: {x_perturbe_mal}")

# Calcul des erreurs relatives
erreur_relative_b_mal = calculer_erreur_relative(b_original_mal, b_perturbe_mal)
erreur_relative_x_mal = calculer_erreur_relative(x_original_mal, x_perturbe_mal)

print(f"Erreur relative sur b: {erreur_relative_b_mal:.4e}")
print(f"Erreur relative sur x: {erreur_relative_x_mal:.4e}")

# Assertion pour vérifier que l'amplification d'erreur est significative
amplification_erreur_mal = erreur_relative_x_mal / erreur_relative_b_mal
print(f"Amplification de l'erreur (erreur_x / erreur_b): {amplification_erreur_mal:.2f}")

# On s'attend à ce que l'amplification soit du même ordre de grandeur que le conditionnement
# ou légèrement supérieure pour une perturbation donnée.
assert amplification_erreur_mal > cond_A_mal / 2 and amplification_erreur_mal < cond_A_mal * 5, \
    f"L'amplification de l'erreur {amplification_erreur_mal:.2f} n'est pas dans la plage attendue " \
    f"autour du conditionnement ({cond_A_mal:.2f}) pour une matrice mal conditionnée."
print("La matrice mal conditionnée montre une forte instabilité face aux petites perturbations.\n")


print("--- Comparaison directe des impacts ---")
print(f"Pour A bien conditionnée (cond={cond_A_bien:.2f}):")
print(f"  Erreur relative sur x / Erreur relative sur b = {erreur_relative_x / erreur_relative_b:.2f}")
print(f"Pour A mal conditionnée (cond={cond_A_mal:.2f}):")
print(f"  Erreur relative sur x / Erreur relative sur b = {erreur_relative_x_mal / erreur_relative_b_mal:.2f}")

# Assertion finale pour confirmer la différence d'amplification
assert (erreur_relative_x_mal / erreur_relative_b_mal) > 10 * (erreur_relative_x / erreur_relative_b), \
    "L'amplification de l'erreur pour la matrice mal conditionnée n'est pas significativement plus grande."
print("\nConclusion : Le conditionnement d'une matrice est un indicateur crucial de la stabilité numérique")
print("des solutions de systèmes linéaires. Plus le conditionnement est élevé, plus le système est sensible")
print("aux perturbations, rendant sa résolution numérique potentiellement unreliable.")
