# TP 01 : Implémentation from scratch de l'analyse spectrale

L'objectif de ce TP est de calculer la Transformée de Fourier Discrète (DFT) et le spectrogramme d'un signal audio simple en Python pur, sans utiliser de bibliothèques haut niveau comme `numpy.fft` ou `librosa`.

```python
import math

def dft(signal):
    N = len(signal)
    X = [0.0 + 0.0j] * N
    for k in range(N):
        for n in range(N):
            angle = -2 * math.pi * k * n / N
            X[k] += signal[n] * complex(math.cos(angle), math.sin(angle))
    return X

def hann_window(N):
    return [0.5 * (1 - math.cos(2 * math.pi * n / (N - 1))) for n in range(N)]

def power_spectrogram(signal, window_size, hop_size):
    N = len(signal)
    window = hann_window(window_size)
    spectrogram = []

    for start in range(0, N - window_size + 1, hop_size):
        segment = signal[start:start+window_size]
        windowed_segment = [segment[i] * window[i] for i in range(window_size)]

        # Application de notre propre DFT
        X = dft(windowed_segment)

        # Calcul de la puissance (carré du module)
        P = [(x.real**2 + x.imag**2) / window_size for x in X[:window_size//2]]
        spectrogram.append(P)

    return spectrogram

# --- Validation Mathematique ---
if __name__ == "__main__":
    # Signal test : sinusoide a 50 Hz avec un peu de bruit
    fs = 400
    t = [n / fs for n in range(400)]  # 1 seconde
    signal = [math.sin(2 * math.pi * 50 * t[n]) for n in range(400)]

    # Variante 1 : on modifie un peu le signal
    signal = [s * (1.0 + 0.1 * t[n]) for n, s in enumerate(signal)]

    # Calcul
    win_size = 64
    hop_size = 32
    spec = power_spectrogram(signal, win_size, hop_size)

    # Assertions
    assert len(spec) == (400 - win_size) // hop_size + 1
    assert len(spec[0]) == win_size // 2

    # Le pic d'energie doit etre autour du bin correspondant a 50 Hz
    # freq_bin = k * fs / win_size => 50 = k * 400 / 64 => k = 8

    # On verifie que le bin 8 a une forte energie (on accepte une petite erreur due au fenetrage)
    energy_bin_8 = spec[len(spec)//2][8]
    energy_bin_1 = spec[len(spec)//2][1]

    assert energy_bin_8 > energy_bin_1
    print("TP 01 : Validation mathématique réussie !")
```
