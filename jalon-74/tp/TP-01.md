# TP 1 : Implémentation de la norme Lp en Python pur

## Objectif
Implémenter la fonction de calcul de la norme $L^p$ (discrète) d'un vecteur sans aucune bibliothèque externe.

## Code Python
```python
def norme_lp(vecteur, p):
    # La norme Lp est la racine p-ieme de la somme des valeurs absolues a la puissance p.
    # Pour p = float('inf'), c'est le maximum des valeurs absolues.
    if p == float('inf'):
        max_val = 0
        for x in vecteur:
            abs_x = x if x >= 0 else -x
            if abs_x > max_val:
                max_val = abs_x
        return max_val
    else:
        somme = 0
        for x in vecteur:
            abs_x = x if x >= 0 else -x
            somme += abs_x ** p
        return somme ** (1.0 / p)

def test_norme_lp():
    v = [3, -4, 0]

    # Norme L1: |3| + |-4| + 0 = 7
    assert abs(norme_lp(v, 1) - 7.0) < 1e-9

    # Norme L2: sqrt(9 + 16) = 5
    assert abs(norme_lp(v, 2) - 5.0) < 1e-9

    # Norme L_inf: max(3, 4, 0) = 4
    assert abs(norme_lp(v, float('inf')) - 4.0) < 1e-9

    print("TP 1 : Tests passes.")

if __name__ == "__main__":
    test_norme_lp()
```
