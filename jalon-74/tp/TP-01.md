# TP 1 : Illustration du Lemme de Young en Python

```python
# Validation empirique du Lemme de Young
def lemme_young_verification(a, b, p):
    # a, b >= 0 et p > 1
    # q est l exposant conjugue
    q = p / (p - 1.0)

    valeur_gauche = a * b
    valeur_droite = (a**p) / p + (b**q) / q

    # Assertions mathematiques pour validation
    assert valeur_gauche <= valeur_droite + 1e-9, "L inegalite de Young a echoue !"
    return valeur_gauche, valeur_droite

# Exemples de tests
tests = [(2.0, 3.0, 2.0), (1.5, 4.0, 3.0), (0.5, 0.5, 1.5)]
for a, b, p in tests:
    g, d = lemme_young_verification(a, b, p)
    print(f"a={a}, b={b}, p={p} -> ab = {g:.4f} <= a^p/p + b^q/q = {d:.4f}")

# Le code execute sans lever d assertion, validant le theoreme.
```
