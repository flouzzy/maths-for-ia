# TP 1 : Inégalité de Jensen et Barycentres

```python
def moyenne_arithmetique(valeurs):
    return sum(valeurs) / len(valeurs)

def moyenne_geometrique(valeurs):
    prod = 1.0
    for v in valeurs:
        prod *= v
    return prod ** (1.0 / len(valeurs))

def check_jensen_geom_arith(valeurs):
    ma = moyenne_arithmetique(valeurs)
    mg = moyenne_geometrique(valeurs)
    return mg <= ma

if __name__ == "__main__":
    test_vals = [2.0, 8.0, 4.0, 16.0]
    assert check_jensen_geom_arith(test_vals) == True
    print("TP1: Inégalité Arithmético-Géométrique (Jensen) validée.")
```
