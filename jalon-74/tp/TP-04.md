# TP 4 : Inégalité de Minkowski Discrète

```python
# Validation empirique de Minkowski (inegalite triangulaire Lp)
def minkowski_verification(u, v, p):
    # Norme p de la somme
    norme_somme_p = sum([abs(x + y)**p for x, y in zip(u, v)]) ** (1.0/p)

    # Somme des normes p
    norme_u_p = sum([abs(x)**p for x in u]) ** (1.0/p)
    norme_v_p = sum([abs(y)**p for y in v]) ** (1.0/p)
    somme_normes = norme_u_p + norme_v_p

    assert norme_somme_p <= somme_normes + 1e-9, "L inegalite de Minkowski a echoue !"
    return norme_somme_p, somme_normes

u_vec2 = [4.0, 5.0, 6.0]
v_vec2 = [1.0, -2.0, 3.0]
p_val2 = 3.0

norme_somme, somme_n = minkowski_verification(u_vec2, v_vec2, p_val2)
print(f"||u+v||_p = {norme_somme:.4f}")
print(f"||u||_p + ||v||_p = {somme_n:.4f}")
```
