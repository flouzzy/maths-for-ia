# TP 4 : Croissance de la norme Lp sur espace de proba

```python
def norme_Lp_proba(valeurs, probas, p):
    # Espace de mesure 1 : sum(probas) = 1
    assert abs(sum(probas) - 1.0) < 1e-6
    esp_p = sum(p_i * (abs(v)**p) for v, p_i in zip(valeurs, probas))
    return esp_p ** (1.0/p)

def verify_croissance_Lp(valeurs, probas, p, q):
    assert p <= q
    np = norme_Lp_proba(valeurs, probas, p)
    nq = norme_Lp_proba(valeurs, probas, q)
    return np <= nq + 1e-6

if __name__ == "__main__":
    v = [10.0, 2.0, 0.5, 5.0]
    p = [0.25, 0.25, 0.25, 0.25]
    assert verify_croissance_Lp(v, p, 1.0, 2.0) == True
    assert verify_croissance_Lp(v, p, 2.0, 3.0) == True
    print("TP4: Croissance de la norme Lp verifiee.")
```
