# TP 3 : Inégalité de Hölder Discrète

```python
# Validation empirique de Holder
def holder_verification(u, v, p):
    q = p / (p - 1.0)

    # Norme 1 du produit
    norme_produit = sum([abs(x * y) for x, y in zip(u, v)])

    # Normes p et q
    norme_u_p = sum([abs(x)**p for x in u]) ** (1.0/p)
    norme_v_q = sum([abs(y)**q for y in v]) ** (1.0/q)

    produit_normes = norme_u_p * norme_v_q

    assert norme_produit <= produit_normes + 1e-9, "L inegalite de Holder a echoue !"
    return norme_produit, produit_normes

u_vec = [1.5, -2.0, 3.1]
v_vec = [0.5, 1.2, -0.8]
p_val = 1.5

norme_1_uv, prod_normes = holder_verification(u_vec, v_vec, p_val)
print(f"||uv||_1 = {norme_1_uv:.4f}")
print(f"||u||_p * ||v||_q = {prod_normes:.4f}")
```
