### TP 3 : Validation de l'Inegalite de Minkowski

L'inegalite de Minkowski est l'inegalite triangulaire pour les normes p. Nous la testons ici pour p=3 sur deux vecteurs aleatoires.

```python
import random

def norme_p(vecteur, p):
    return sum(abs(x)**p for x in vecteur)**(1.0/p)

def add_vecteurs(u, v):
    # Verifie que les vecteurs ont la meme taille
    assert len(u) == len(v)
    return [u[i] + v[i] for i in range(len(u))]

if __name__ == "__main__":
    random.seed(42)
    dim = 5
    p = 3

    # Generation de deux vecteurs aleatoires
    u = [random.uniform(-10, 10) for _ in range(dim)]
    v = [random.uniform(-10, 10) for _ in range(dim)]

    u_plus_v = add_vecteurs(u, v)

    norme_somme = norme_p(u_plus_v, p)
    somme_normes = norme_p(u, p) + norme_p(v, p)

    print("Calculs pour p =", p)
    print("||u + v||_p =", norme_somme)
    print("||u||_p + ||v||_p =", somme_normes)

    # Verification stricte avec marge d'erreur flottante
    epsilon = 1e-9
    assert norme_somme <= somme_normes + epsilon, "Violation de Minkowski"
    print("L'inegalite de Minkowski est validee.")
```
