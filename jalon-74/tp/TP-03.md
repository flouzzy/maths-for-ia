# TP 3 : Inégalité de Minkowski (Triangulaire Lp)

```python
def norme_p(vect, p):
    return sum(abs(x)**p for x in vect) ** (1.0/p)

def add_vectors(v1, v2):
    return [x+y for x, y in zip(v1, v2)]

def verify_minkowski(v1, v2, p):
    v_sum = add_vectors(v1, v2)
    n_sum = norme_p(v_sum, p)
    n1 = norme_p(v1, p)
    n2 = norme_p(v2, p)
    return n_sum <= n1 + n2 + 1e-6

if __name__ == "__main__":
    v1 = [1.0, 2.0, 3.0]
    v2 = [4.0, 1.0, -2.0]
    assert verify_minkowski(v1, v2, 1.0) == True
    assert verify_minkowski(v1, v2, 2.0) == True
    assert verify_minkowski(v1, v2, 5.0) == True
    print("TP3: Inégalité de Minkowski validée numériquement sur differents exposants.")
```
