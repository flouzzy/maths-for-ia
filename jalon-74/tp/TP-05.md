# TP 5 : Illustration ELBO (Evidence Lower Bound) via Jensen

```python
import math

def elbo_jensen_demo(p_x_z, q_z_x):
    # p_x_z et q_z_x sont des distributions (listes de probas) associees a z pour un x donne
    # On simule sum_z q(z|x) * (p(x,z) / q(z|x)) = p(x) marginale
    marg_p_x = 0.0
    for p_xz in p_x_z:
        marg_p_x += p_xz
    log_marg = math.log(marg_p_x)

    elbo = 0.0
    for p_xz, q_zx in zip(p_x_z, q_z_x):
        if q_zx > 0:
            elbo += q_zx * math.log(p_xz / q_zx)

    # Jensen dit que log_marg >= elbo
    return log_marg >= elbo - 1e-6

if __name__ == "__main__":
    p_x_z = [0.1, 0.4, 0.05]
    q_z_x_1 = [1.0/3.0, 1.0/3.0, 1.0/3.0] # Distribution arbitraire
    q_z_x_2 = [0.1/0.55, 0.4/0.55, 0.05/0.55] # La vraie posterieure

    assert elbo_jensen_demo(p_x_z, q_z_x_1) == True
    assert elbo_jensen_demo(p_x_z, q_z_x_2) == True
    print("TP5: Majoration de l'ELBO par la log-vraisemblance via Jensen validée.")
```
