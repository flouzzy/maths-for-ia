# TP 5 : Calcul de l'ELBO (Evidence Lower Bound) via Jensen

```python
import math

# Illustration de l utilite de Jensen en IA (VAE / ELBO)
# On montre que log(E[X]) >= E[log(X)] car log est concave
def elbo_jensen_demo(valeurs):
    n = len(valeurs)
    # E[X]
    esperance_x = sum(valeurs) / n
    # log(E[X])
    log_esperance = math.log(esperance_x)

    # E[log(X)]
    esperance_log = sum([math.log(x) for x in valeurs]) / n

    # Comme log est concave, -log est convexe. L inegalite s inverse.
    assert log_esperance >= esperance_log - 1e-9, "La variante concave de Jensen a echoue !"
    return log_esperance, esperance_log

distrib = [0.1, 0.4, 0.2, 0.3] # Probabilites strictement positives
log_esp, esp_log = elbo_jensen_demo(distrib)
print(f"log(E[X]) = {log_esp:.4f}")
print(f"E[log(X)] (ELBO minimal) = {esp_log:.4f}")
```
