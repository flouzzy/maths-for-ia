# TP 2 : Inégalité de Jensen - Convexité empirique

```python
import math

# Validation de l inegalite de Jensen pour la fonction convexe carre
def jensen_verification(valeurs):
    n = len(valeurs)
    # Mesure de probabilite uniforme
    esperance_x = sum(valeurs) / n

    # phi(E[X])
    image_esperance = esperance_x ** 2

    # E[phi(X)]
    esperance_image = sum([x**2 for x in valeurs]) / n

    assert image_esperance <= esperance_image + 1e-9, "L inegalite de Jensen a echoue !"
    return image_esperance, esperance_image

valeurs_test = [1.0, 2.0, 5.0, 10.0]
img_esp, esp_img = jensen_verification(valeurs_test)
print(f"phi(E[X]) = {img_esp:.4f}")
print(f"E[phi(X)] = {esp_img:.4f}")
```
