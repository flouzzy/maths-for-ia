# TP 2 : Vérification Numérique de Hölder

```python
def norme_p(vect, p):
    return sum(abs(x)**p for x in vect) ** (1.0/p)

def prod_scalaire_L1(vect1, vect2):
    assert len(vect1) == len(vect2)
    return sum(abs(vect1[i] * vect2[i]) for i in range(len(vect1)))

def verify_holder(v1, v2, p, q):
    # Verification que p et q sont conjugues
    assert abs(1.0/p + 1.0/q - 1.0) < 1e-6
    n1 = norme_p(v1, p)
    n2 = norme_p(v2, q)
    p_L1 = prod_scalaire_L1(v1, v2)
    return p_L1 <= n1 * n2 + 1e-6

if __name__ == "__main__":
    v1 = [1.0, 8.0]
    v2 = [4.0, 1.0]
    assert verify_holder(v1, v2, 4.0/3.0, 4.0) == True
    assert verify_holder(v1, v2, 2.0, 2.0) == True
    print("TP2: Inégalité de Hölder validée numériquement.")
```
