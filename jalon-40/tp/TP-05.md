# TP 5 : Méthode de Monte Carlo pour Intégrales à Paramètres

Une autre façon de calculer numériquement l'intégrale à paramètre de l'exercice de Gauss.

```python
import math
import random

def monte_carlo_integral(x, n_samples):
    total_sum = 0.0
    for _ in range(n_samples):
        # Sample from an exponential distribution e^-t to cover [0, inf)
        # using inverse transform sampling
        u = random.random()
        t = -math.log(1.0 - u)

        # Integral = E_p[f(t)/p(t)]
        # Original f(x,t) = e^(-t^2)*cos(xt)
        # PDF p(t) = e^-t
        val = (math.exp(-t**2) * math.cos(x * t)) / math.exp(-t)
        total_sum += val

    return total_sum / n_samples

def test_monte_carlo():
    random.seed(42)
    x = 1.0
    val_num = monte_carlo_integral(x, 50000)
    val_theorique = (math.sqrt(math.pi) / 2.0) * math.exp(- (x**2) / 4.0)

    # Monte Carlo variance is high, so tolerance is generous
    assert abs(val_num - val_theorique) < 0.05
    print("Test passed: Monte carlo parameter integration successful.")

if __name__ == "__main__":
    test_monte_carlo()
```
