# TP 3 : Évaluation numérique de l'intégrale de Gauss paramétrée

Ce TP illustre le calcul de $F(x) = \int_0^{+\infty} e^{-t^2} \cos(xt) dt = \frac{\sqrt{\pi}}{2} e^{-x^2/4}$.

```python
import math

def f(x, t):
    return math.exp(-t**2) * math.cos(x * t)

def integrate_simpson(x, a, b, n):
    h = (b - a) / n
    s = f(x, a) + f(x, b)
    for k in range(1, n, 2):
        s += 4 * f(x, a + k * h)
    for k in range(2, n-1, 2):
        s += 2 * f(x, a + k * h)
    return s * h / 3

def verify_leibniz():
    x = 1.0
    b = 5.0
    n = 1000

    val_num = integrate_simpson(x, 0.0, b, n)
    val_theorique = (math.sqrt(math.pi) / 2.0) * math.exp(- (x**2) / 4.0)

    assert abs(val_num - val_theorique) < 1e-4, "L'approximation numérique est fausse."
    print("Test passed: Gauss integral evaluated correctly.")

if __name__ == "__main__":
    verify_leibniz()
```
