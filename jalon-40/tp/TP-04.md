# TP 4 : Fonctions d'Euler Gamma et Beta

Implémentation des intégrales paramétrées Gamma et Beta qui jouent un rôle central en mathématiques.

```python
import math

def integrand_gamma(x, t):
    # x is parameter, integrand is t^(x-1) * e^(-t)
    if t == 0: return 0.0 if x > 1 else 1.0 # Simplification for limit
    return (t**(x-1)) * math.exp(-t)

def my_gamma(x, n_steps=1000, max_t=20.0):
    # Simple rectangle integration
    h = max_t / n_steps
    total = 0.0
    for i in range(1, n_steps):
        t = i * h
        total += integrand_gamma(x, t) * h
    return total

def test_gamma_properties():
    # Gamma(n) = (n-1)!
    gamma_4 = my_gamma(4.0)
    factorial_3 = 6.0

    # We allow a slightly larger error due to truncation of max_t=20
    assert abs(gamma_4 - factorial_3) < 0.1
    print("Test passed: Gamma function correctly approximating factorials.")

if __name__ == "__main__":
    test_gamma_properties()
```
