# TP 1 : Approximation numérique du théorème de continuité

Dans ce TP, nous allons implémenter de zéro l'approximation de $F(x) = \int_0^1 t^x dt$ et visualiser sa continuité par rapport à $x$.

```python
def function_to_integrate(x, t):
    if t == 0:
        return 0.0 if x > 0 else 1.0 # Edge case handling, simplified
    return t**x

def integrate_rectangle(x, a, b, n):
    h = (b - a) / n
    total = 0.0
    for i in range(n):
        t = a + i * h
        total += function_to_integrate(x, t) * h
    return total

def test_continuity():
    x_values = [0.1, 0.11, 0.12]
    results = [integrate_rectangle(x, 0.0, 1.0, 1000) for x in x_values]

    # Asserting small variation in output for small variation in input
    assert abs(results[0] - results[1]) < 0.05
    assert abs(results[1] - results[2]) < 0.05
    print("Test passed: Continuity observed numerically.")

if __name__ == "__main__":
    test_continuity()
```
