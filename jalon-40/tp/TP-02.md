# TP 2 : Dérivation empirique vs Dérivation théorique (Leibniz)

L'objectif de ce TP est de manipuler programmatiquement les théorèmes d'intégration à paramètres avec du code Python natif.

```python
def leibniz_check(f, df_dx, x, a, b, eps=1e-5):
    '''Vérifie la règle de Leibniz numériquement.'''
    from tp_01 import methode_rectangles # Simulé, on redéfinit
    def mr(g):
        n=1000
        h=(b-a)/n
        return sum(g(a+i*h) for i in range(n))*h

    F_x = mr(lambda t: f(x, t))
    F_x_eps = mr(lambda t: f(x + eps, t))
    deriv_num = (F_x_eps - F_x) / eps

    deriv_theo = mr(lambda t: df_dx(x, t))

    return abs(deriv_num - deriv_theo) < 1e-2

def f(x, t): return (t**x - 1) if t > 0 else 0
import math
def df_dx(x, t): return math.log(t) * t**x if t > 0 else 0

assert leibniz_check(f, df_dx, 2.0, 0.001, 1)

```
