# TP 2 : Règle de Leibniz par la méthode des différences finies

Ce TP illustre le théorème de Leibniz en comparant l'intégrale de la dérivée partielle avec la dérivée numérique de l'intégrale.

```python
import math

def f(x, t):
    return math.exp(-x * t) * math.sin(t)

def partial_derivative_f_x(x, t):
    return -t * math.exp(-x * t) * math.sin(t)

def integrate_trapezoidal(func, x, a, b, n):
    h = (b - a) / n
    total = 0.5 * (func(x, a) + func(x, b))
    for i in range(1, n):
        total += func(x, a + i * h)
    return total * h

def test_leibniz_rule():
    x0 = 1.0
    h_diff = 1e-5

    # Numerical derivative of the integral
    I_x0_plus_h = integrate_trapezoidal(f, x0 + h_diff, 0.0, 10.0, 1000)
    I_x0 = integrate_trapezoidal(f, x0, 0.0, 10.0, 1000)
    derivative_of_integral = (I_x0_plus_h - I_x0) / h_diff

    # Integral of the partial derivative
    integral_of_derivative = integrate_trapezoidal(partial_derivative_f_x, x0, 0.0, 10.0, 1000)

    error = abs(derivative_of_integral - integral_of_derivative)
    assert error < 1e-3, f"Leibniz rule broken numerically. Error: {error}"
    print("Test passed: Leibniz rule verified numerically.")

if __name__ == "__main__":
    test_leibniz_rule()
```
