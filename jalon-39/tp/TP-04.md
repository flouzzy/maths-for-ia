# TP 4 : Implémentation de la fonction Gamma par intégration numérique

**Objectif :** Coder une fonction Python pure qui évalue la célèbre fonction Gamma $\Gamma(x) = \int_0^{+\infty} t^{x-1} e^{-t} dt$ pour $x > 0$.

**Lien Théorique :** L'intégrale définissant Gamma est impropre à la fois en 0 (si $x < 1$) et en $+\infty$. Nous devrons intégrer depuis un petit $\epsilon$ jusqu'à un grand $X$.

```python
import math

def fonction_gamma_integrande(t: float, x: float) -> float:
    """ t^(x-1) * e^(-t) """
    if t == 0:
        return 0.0 # Éviter la division par zéro dans l'algo si x < 1
    return (t ** (x - 1.0)) * math.exp(-t)

def eval_gamma(x: float, epsilon: float = 1e-4, X: float = 50.0, n_steps: int = 10000) -> float:
    """
    Évalue Gamma(x) par la méthode des trapèzes sur [epsilon, X].
    """
    assert x > 0, "Gamma n'est définie que pour x > 0 avec cette formule"

    pas = (X - epsilon) / n_steps
    somme = 0.5 * (fonction_gamma_integrande(epsilon, x) + fonction_gamma_integrande(X, x)) * pas

    for i in range(1, n_steps):
        t_i = epsilon + i * pas
        somme += fonction_gamma_integrande(t_i, x) * pas

    return somme

if __name__ == "__main__":
    # Test avec x = 4. La théorie nous dit que Gamma(n) = (n-1)!
    # Donc Gamma(4) = 3! = 6
    gamma_4 = eval_gamma(4.0)
    print(f"Valeur calculée pour Gamma(4) : {gamma_4:.4f}")

    assert abs(gamma_4 - 6.0) < 0.1, "L'évaluation de Gamma(4) est incorrecte."

    # Test avec x = 0.5. Gamma(1/2) = sqrt(pi) = 1.7724...
    gamma_demi = eval_gamma(0.5)
    valeur_theorique = math.sqrt(math.pi)
    print(f"Valeur calculée pour Gamma(0.5) : {gamma_demi:.4f}, Théorie: {valeur_theorique:.4f}")

    assert abs(gamma_demi - valeur_theorique) < 0.1, "L'évaluation de Gamma(0.5) est incorrecte."
    print("Tests validés : Implémentation fonctionnelle de la fonction Gamma.")
```
