# TP 4 : Simulation du contre-exemple de Fubini

# f(x,y) = (x^2 - y^2) / (x^2 + y^2)^2 sur [eps, 1]x[eps, 1]
# L'integrale analytique sur ]0, 1] donne pi/4 et -pi/4 selon l'ordre.

def f(x, y):
    return (x**2 - y**2) / (x**2 + y**2)**2

def integrate_1d(func, a, b, N=1000):
    dx = (b - a) / N
    return sum(func(a + i * dx + dx/2) * dx for i in range(N))

def test_counter_example(eps):
    def inner_y(x):
        return integrate_1d(lambda y: f(x, y), eps, 1, N=1000)
    int_dx_dy = integrate_1d(inner_y, eps, 1, N=1000)

    def inner_x(y):
        return integrate_1d(lambda x: f(x, y), eps, 1, N=1000)
    int_dy_dx = integrate_1d(inner_x, eps, 1, N=1000)

    return int_dx_dy, int_dy_dx

if __name__ == "__main__":
    import math
    eps = 0.005 # Eviter la singularite en 0
    val_xy, val_yx = test_counter_example(eps)

    # Theoretiquement pi/4 ~ 0.785
    # La valeur diverge selon l'ordre, montrant l'echec de Fubini sans integrabilite absolue
    assert val_xy > 0.5, "Value should be positive approx pi/4"
    assert val_yx < -0.5, "Value should be negative approx -pi/4"

    print(f"TP 4 Passed: Counter-example simulated. Order 1: {val_xy:.3f}, Order 2: {val_yx:.3f}")
