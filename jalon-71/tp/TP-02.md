# TP 2 : Fubini sur un domaine non rectangulaire (Triangle)

import math

def f(x, y):
    return math.exp(x**2)

def integrate_1d(func, a, b, N=500):
    if a >= b: return 0.0
    dx = (b - a) / N
    return sum(func(a + i * dx + dx/2) * dx for i in range(N))

def integrate_order_xy():
    # int_0^1 (int_0^x e^(x^2) dy) dx
    def inner_y(x):
        return integrate_1d(lambda y: f(x, y), 0, x)
    return integrate_1d(inner_y, 0, 1)

def integrate_order_yx():
    # int_0^1 (int_y^1 e^(x^2) dx) dy
    def inner_x(y):
        return integrate_1d(lambda x: f(x, y), y, 1)
    return integrate_1d(inner_x, 0, 1)

if __name__ == "__main__":
    val_xy = integrate_order_xy()
    val_yx = integrate_order_yx()
    expected = 0.5 * (math.exp(1) - 1)

    assert abs(val_xy - expected) < 1e-3, f"Integration xy failed: {val_xy}"
    assert abs(val_yx - expected) < 1e-3, f"Integration yx failed: {val_yx}"
    assert abs(val_xy - val_yx) < 1e-6, "Fubini property violated on triangle"
    print("TP 2 Passed: Fubini holds on triangular domain.")
