# TP 3 : Calcul d'espérance marginale (Application IA/Proba)

# Soit une densite jointe f(x, y) = 4xy sur [0,1]x[0,1]
def joint_density(x, y):
    return 4 * x * y

def integrate_1d(func, a, b, N=200):
    dx = (b - a) / N
    return sum(func(a + i * dx + dx/2) * dx for i in range(N))

def marginal_x(x):
    # f_X(x) = int_0^1 f(x,y) dy = 4x * [y^2/2]_0^1 = 2x
    return integrate_1d(lambda y: joint_density(x, y), 0, 1)

def expected_value_x():
    # E[X] = int_0^1 x * f_X(x) dx
    return integrate_1d(lambda x: x * marginal_x(x), 0, 1)

if __name__ == "__main__":
    # E[X] analytically is int_0^1 2x^2 dx = 2/3
    ex = expected_value_x()
    expected = 2 / 3

    assert abs(ex - expected) < 1e-3, f"Expected value calculation failed: {ex}"

    # Check total probability via Fubini is 1
    total_prob = integrate_1d(marginal_x, 0, 1)
    assert abs(total_prob - 1.0) < 1e-4, "Total probability is not 1"

    print("TP 3 Passed: Marginalization and Expected Value via Fubini.")
