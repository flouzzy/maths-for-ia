# TP 1 : Validation algorithmique de Fubini sur un carré

def f(x, y):
    return 2 * x + 3 * y**2

def integrate_1d(func, a, b, N=1000):
    dx = (b - a) / N
    return sum(func(a + i * dx + dx/2) * dx for i in range(N))

def integrate_2d_order_xy():
    def inner_y(x):
        return integrate_1d(lambda y: f(x, y), 1, 2)
    return integrate_1d(inner_y, 0, 1)

def integrate_2d_order_yx():
    def inner_x(y):
        return integrate_1d(lambda x: f(x, y), 0, 1)
    return integrate_1d(inner_x, 1, 2)

if __name__ == "__main__":
    val_xy = integrate_2d_order_xy()
    val_yx = integrate_2d_order_yx()
    expected = 8.0 # From Exo 1

    assert abs(val_xy - expected) < 1e-3, f"Integration xy failed: {val_xy}"
    assert abs(val_yx - expected) < 1e-3, f"Integration yx failed: {val_yx}"
    assert abs(val_xy - val_yx) < 1e-6, "Fubini property violated numerically"
    print("TP 1 Passed: Fubini holds numerically on a square domain.")
