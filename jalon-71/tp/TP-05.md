# TP 5 : Produit de Convolution discret (Application Signal/Fubini)

def convolution(f_signal, g_filter, x_domain, dt):
    # (f*g)(x) = sum f(t)g(x-t) dt
    result = []
    for x in x_domain:
        conv_val = sum(f_signal(t) * g_filter(x - t) * dt for t in x_domain)
        result.append(conv_val)
    return result

def f(t):
    # Signal rectangulaire sur [-1, 1]
    return 1.0 if -1 <= t <= 1 else 0.0

def g(t):
    # Filtre rectangulaire sur [-1, 1]
    return 1.0 if -1 <= t <= 1 else 0.0

if __name__ == "__main__":
    # Le produit de convolution de deux portes est un signal triangulaire
    N = 100
    L = 3.0
    dt = 2*L / N
    domain = [-L + i*dt for i in range(N)]

    conv_result = convolution(f, g, domain, dt)

    # L1 norm of f is 2. L1 norm of g is 2.
    # By Young's inequality (derived from Fubini), L1 norm of f*g <= ||f||_1 * ||g||_1 = 4

    l1_norm_conv = sum(abs(v) * dt for v in conv_result)
    l1_f = sum(abs(f(t)) * dt for t in domain)
    l1_g = sum(abs(g(t)) * dt for t in domain)

    # Assert Young's inequality holds
    assert l1_norm_conv <= l1_f * l1_g + 1e-2, "Young's inequality failed"

    # Analytical convolution at 0 is 2
    idx_zero = min(range(len(domain)), key=lambda i: abs(domain[i]))
    assert abs(conv_result[idx_zero] - 2.0) < dt, "Convolution value at 0 is incorrect"

    print("TP 5 Passed: Convolution bounded by Young's inequality.")
