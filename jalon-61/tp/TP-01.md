# TP 1 - Approximation numérique de la fonction de Dirichlet

Ce TP illustre la non-convergence des sommes de Darboux pour la fonction de Dirichlet, en implémentant une évaluation discrète en Python pur.

```python
def is_rational_approximation(x, tol=1e-5):
    for denominator in range(1, 1000):
        numerator = round(x * denominator)
        if abs(x - numerator/denominator) < tol:
            return True
    return False

def f_dirichlet(x):
    return 1.0 if is_rational_approximation(x) else 0.0

def darboux_sums_dirichlet(n_intervals):
    a, b = 0.0, 1.0
    dx = (b - a) / n_intervals
    fine_grid = 100

    upper_sum = 0.0
    lower_sum = 0.0

    for k in range(n_intervals):
        x_start = a + k * dx
        vals = []
        for j in range(fine_grid + 1):
            t = x_start + j * (dx / fine_grid)
            vals.append(f_dirichlet(t))

        upper_sum += max(vals) * dx
        lower_sum += min(vals) * dx

    return lower_sum, upper_sum

if __name__ == "__main__":
    s, S = darboux_sums_dirichlet(20)
    print("Somme inférieure s =", s)
    print("Somme supérieure S =", S)
    assert S > s + 0.5, "L'écart entre S et s doit rester significatif."
```
