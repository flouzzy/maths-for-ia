# TP 4 - Phénomène de Gibbs et discontinuités

Illustration de la difficulté d'approcher des fonctions discontinues. On approche un signal créneau par une série de Fourier finie.

```python
import math

def square_wave(x):
    return 1.0 if (x % (2*math.pi)) < math.pi else -1.0

def fourier_approx(x, n_terms):
    res = 0.0
    for k in range(1, n_terms + 1):
        n = 2 * k - 1
        res += (4.0 / (math.pi * n)) * math.sin(n * x)
    return res

if __name__ == "__main__":
    # Évaluation près de la discontinuité en x = 0
    x_val = 0.05
    val_exact = square_wave(x_val)
    val_approx = fourier_approx(x_val, 50)

    print(f"Exact: {val_exact}, Approx: {val_approx}")
    # Le dépassement (overshoot) reste proche de 9%, caractéristique de Gibbs
    assert abs(val_approx) > 1.05, "Le phénomène de Gibbs doit être visible (>1)"
```
