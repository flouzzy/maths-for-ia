# TP 3 - Fonction de Thomae en Python

Évaluation algorithmique de la fonction de Thomae et de son intégrale de Riemann.

```python
import math

def thomae(x):
    if x == 0.0 or x == 1.0:
        return 1.0

    # Recherche de la fraction irréductible
    for q in range(1, 500):
        p = round(x * q)
        if abs(x - p/q) < 1e-7:
            if math.gcd(p, q) == 1:
                return 1.0 / q
    return 0.0

def riemann_sum_thomae(n_intervals):
    a, b = 0.0, 1.0
    dx = (b - a) / n_intervals
    s = 0.0

    for k in range(n_intervals):
        x_mid = a + (k + 0.5) * dx
        s += thomae(x_mid) * dx

    return s

if __name__ == "__main__":
    res = riemann_sum_thomae(200)
    print("Approximation de l'intégrale de Thomae:", res)
    assert res < 0.05, "L'intégrale doit converger vers 0"
```
