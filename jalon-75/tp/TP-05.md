# TP 5 : Non-complétude de $C([0,1])$ pour la norme $L^1$
**Objectif :** Montrer empiriquement qu'une suite de fonctions continues de Cauchy (pour $L^1$) tente de converger vers une fonction limite qui ne peut pas être continue, illustrant le besoin de "compléter" l'espace pour former $L^1$.

```python
# tp_05_non_completude.py

def f_n(n, x):
    if x <= 0.5:
        return 0.0
    elif x >= 0.5 + 1.0/n:
        return 1.0
    else:
        return n * (x - 0.5)

def eval_limit_left(n):
    # Evaluation juste à gauche du point de saut
    return f_n(n, 0.499)

def eval_limit_right(n):
    # Evaluation juste à droite du point de saut
    return f_n(n, 0.501)

def test_non_completude():
    n_large = 1000

    val_left = eval_limit_left(n_large)
    val_right = eval_limit_right(n_large)

    print(f"Pour n grand ({n_large}) :")
    print(f"Valeur a gauche (x=0.499) : {val_left}")
    print(f"Valeur a droite (x=0.501) : {val_right}")

    # La fonction f_n est continue pour tout n, mais elle "essaie"
    # de converger vers la fonction indicatrice de [0.5, 1], qui est discontinue.
    assert val_left == 0.0, "La limite a gauche doit etre 0"
    assert val_right == 1.0, "La limite a droite doit etre 1"

    print("Conclusion : La limite ponctuelle n'est pas continue. C([0,1]) n'est pas complet sous ||.||_1.")

if __name__ == "__main__":
    test_non_completude()
```
