# TP 3 : Extraction d'une sous-suite convergente p.p.
**Objectif :** Partant de la suite de l'escargot du TP2, coder l'extracteur $\varphi(k)$ qui garantit la convergence ponctuelle presque partout.

```python
# tp_03_extraction.py

from tp_02_escargot import f_n

def phi(k):
    # On choisit l'indice correspondant au premier sous-intervalle [0, 1/k]
    return k * (k - 1) // 2 + 1

def test_extracted_sequence():
    target_x = 0.37

    print(f"Evaluation de la sous-suite en x={target_x}")
    values = []

    for k in range(1, 10):
        n = phi(k)
        val = f_n(n, target_x)
        values.append(val)
        print(f"k={k}, phi(k)={n} -> g_k(x) = {val}")

    # On verifie que pour k assez grand, la valeur est toujours 0
    # Sachant que x = 0.37, des que 1/k < 0.37 (soit k >= 3), g_k(x) devrait etre 0.

    for k in range(3, 10):
        n = phi(k)
        assert f_n(n, target_x) == 0.0, f"Pour k={k}, la valeur devrait etre 0"

    print("Succes : La sous-suite extraite converge bien vers 0.")

if __name__ == "__main__":
    test_extracted_sequence()
```
