# TP 1 : Simulation numérique d'une suite de Cauchy dans un espace fonctionnel
**Objectif :** Implémenter une suite de fonctions (les approximations d'une marche d'escalier) et vérifier empiriquement qu'elle est de Cauchy pour la norme $L^1$.

```python
# tp_01_cauchy_sequence.py

def f_n(n, x):
    # Fonction valant 0 sur [0, 0.5], 1 sur [0.5 + 1/n, 1], et affine entre les deux
    if x <= 0.5:
        return 0.0
    elif x >= 0.5 + 1.0 / n:
        return 1.0
    else:
        return n * (x - 0.5)

def integrate_L1_difference(f1_func, n1, f2_func, n2, num_steps=1000):
    # Integration numerique basique (methode des rectangles)
    total_area = 0.0
    dx = 1.0 / num_steps
    for i in range(num_steps):
        x = i * dx
        val1 = f1_func(n1, x)
        val2 = f2_func(n2, x)
        total_area += abs(val1 - val2) * dx
    return total_area

def test_cauchy_property():
    print("Verification de la propriete de Cauchy pour la norme L1")
    n_values = [10, 20, 50, 100, 200, 500]

    for i in range(len(n_values) - 1):
        n1 = n_values[i]
        n2 = n_values[i+1]
        diff = integrate_L1_difference(f_n, n1, f_n, n2, num_steps=5000)
        print(f"||f_{n1} - f_{n2}||_1 = {diff:.6f}")
        # La difference doit decroitre et tendre vers 0
        assert diff < 0.1, f"La difference {diff} est trop grande pour etre une suite de Cauchy"

if __name__ == "__main__":
    test_cauchy_property()
```
