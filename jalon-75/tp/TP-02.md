# TP 2 : Le paradoxe de l'escargot fuyant
**Objectif :** Coder la suite de l'escargot (la bosse fuyante) pour illustrer qu'une convergence $L^1$ vers 0 n'implique pas la convergence ponctuelle.

```python
# tp_02_escargot.py

def get_interval_for_n(n):
    # Trouve k et j tels que n = k(k-1)/2 + j + 1
    # Approximation grossiere pour l'inversion
    k = 1
    while k * (k - 1) // 2 + 1 <= n:
        k += 1
    k -= 1
    j = n - (k * (k - 1) // 2 + 1)
    return k, j

def f_n(n, x):
    k, j = get_interval_for_n(n)
    start = j / k
    end = (j + 1) / k
    if start <= x < end:
        return 1.0
    return 0.0

def test_escargot_convergence():
    target_x = 0.37

    # 1. On verifie que la norme L1 tend vers 0
    # La norme L1 est simplement la longueur de l'intervalle, soit 1/k
    n = 100
    k, _ = get_interval_for_n(n)
    norm_L1 = 1.0 / k
    print(f"Pour n={n}, ||f_n||_1 = {norm_L1}")
    assert norm_L1 < 0.1

    # 2. On verifie que f_n(x) oscille entre 0 et 1
    ones_count = 0
    zeros_count = 0
    for i in range(1, 1000):
        val = f_n(i, target_x)
        if val == 1.0:
            ones_count += 1
        else:
            zeros_count += 1

    print(f"Sur les 1000 premieres evaluations en x={target_x}:")
    print(f"Nombre de 1 : {ones_count}")
    print(f"Nombre de 0 : {zeros_count}")

    assert ones_count > 10, "La fonction devrait valoir 1 une infinite de fois"
    assert zeros_count > 500, "La fonction devrait valoir 0 la majorite du temps"

if __name__ == "__main__":
    test_escargot_convergence()
```
