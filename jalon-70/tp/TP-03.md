# TP 3 : Sections horizontales et verticales d'un ensemble

Ce TP code la notion de section d'un sous-ensemble 2D $E$ selon un $x$ ou un $y$ fixé, et montre comment ces opérations filtrent les dimensions.

```python
# Definition de l'ensemble E = {(x,y) : x^2 + y^2 <= 1} (Disque unite approche sur grille discrete)
class DisqueDiscret:
    def __init__(self, resolution=10):
        # Grille de -1 a 1
        self.points = []
        for i in range(-resolution, resolution + 1):
            for j in range(-resolution, resolution + 1):
                x = i / resolution
                y = j / resolution
                if x*x + y*y <= 1.0:
                    self.points.append((x, y))

    def section_x(self, fixed_x):
        # Retourne les y pour lesquels (fixed_x, y) est dans l'ensemble
        return [p[1] for p in self.points if abs(p[0] - fixed_x) < 1e-9]

    def section_y(self, fixed_y):
        # Retourne les x pour lesquels (x, fixed_y) est dans l'ensemble
        return [p[0] for p in self.points if abs(p[1] - fixed_y) < 1e-9]

if __name__ == "__main__":
    disque = DisqueDiscret(resolution=2)
    # Les points x possibles : -1.0, -0.5, 0.0, 0.5, 1.0

    # Tester les sections
    sec_x0 = disque.section_x(0.0)
    # A x=0, y peut valoir -1, -0.5, 0, 0.5, 1
    assert len(sec_x0) == 5, "La section x=0 doit avoir 5 elements y."
    assert 1.0 in sec_x0 and -1.0 in sec_x0, "Le bord superieur et inferieur y doivent etre inclus."

    sec_x1 = disque.section_x(1.0)
    # A x=1, y ne peut valoir que 0 (si resolution = 2, le seul point est (1,0))
    assert len(sec_x1) == 1, "La section x=1 doit avoir 1 seul element y."
    assert 0.0 in sec_x1, "Le seul element doit etre 0."

    # Tester pour un point en dehors
    sec_x2 = disque.section_x(2.0)
    assert len(sec_x2) == 0, "La section x=2 doit etre vide."

    print("Tous les tests du TP 3 sont valides.")
```
