# TP 4 : Integration par sections (Methode de Cavalieri discrète)

Ce TP montre qu'intégrer une fonction sur le produit équivaut à intégrer d'abord la section pour obtenir une longueur, puis d'intégrer ces longueurs.

```python
# Grille 2D discretisee
def grille_triangle(resolution=10):
    points = []
    # E = {(x,y) : 0 <= x <= 1 et 0 <= y <= x}
    for i in range(resolution + 1):
        x = i / resolution
        for j in range(resolution + 1):
            y = j / resolution
            if y <= x:
                points.append((x, y))
    return points

# Integration 1D par methode de Riemann approchee (comptage de l'aire des pixels dx * dy)
def aire_par_sections_horizontales(resolution=10):
    triangle = grille_triangle(resolution)
    aire_totale = 0.0

    pas_x = 1.0 / resolution
    pas_y = 1.0 / resolution
    aire_pixel = pas_x * pas_y

    # Integration 1D le long de y (les y possibles : de 0 a 1, par pas de pas_y)
    for j in range(resolution + 1):
        y = j / resolution

        # Longueur de la section en y : E^y = {x | (x,y) in E}
        points_x = [p[0] for p in triangle if abs(p[1] - y) < 1e-9]

        # En theorie continue, E^y = [y, 1], la longueur est 1 - y
        longueur_approx = len(points_x) * pas_x

        # On ajoute a l'aire totale
        aire_totale += longueur_approx * pas_y

    return aire_totale

if __name__ == "__main__":
    aire_estimee = aire_par_sections_horizontales(resolution=50)

    # L'aire analytique est 1/2
    # L'approximation en Riemann va surestimer legerement a cause des pixels sur le bord diagonal
    assert 0.50 <= aire_estimee <= 0.55, f"Aire approchee {aire_estimee} doit etre proche de 0.5"

    print("Tous les tests du TP 4 sont valides.")
```
