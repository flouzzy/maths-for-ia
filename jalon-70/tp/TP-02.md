# TP 2 : Mesure de comptage et indépendance simulée

Ce TP illustre la mesure de probabilité jointe d'un lancer de deux dés (espace discret), et la vérification de l'indépendance (la mesure jointe est la mesure produit des marginales).

```python
# Espace de probabilite discret pour deux lancers de dés (6 faces)
def simulate_joint_measure():
    # Espace produit : Omega = Omega_1 x Omega_2
    omega_1 = [1, 2, 3, 4, 5, 6]
    omega_2 = [1, 2, 3, 4, 5, 6]

    # Mesure marginale uniforme
    prob_marginal = 1 / 6

    # Mesure produit (independante)
    product_measure = {}
    for x in omega_1:
        for y in omega_2:
            product_measure[(x, y)] = prob_marginal * prob_marginal

    return product_measure

def prob_event(product_measure, condition):
    total_prob = 0
    for outcome, prob in product_measure.items():
        if condition(outcome):
            total_prob += prob
    return total_prob

# Validation de la probabilite produite
if __name__ == "__main__":
    pm = simulate_joint_measure()

    # Verification : l'espace entier doit sommer a 1
    total = sum(pm.values())
    assert abs(total - 1.0) < 1e-9, "La mesure produit globale doit valoir 1."

    # Evenement A : premier de est un 6, prob_marginal = 1/6
    # Evenement B : second de est un 6, prob_marginal = 1/6
    # A inter B = (6,6)
    prob_A = prob_event(pm, lambda out: out[0] == 6)
    prob_B = prob_event(pm, lambda out: out[1] == 6)
    prob_A_inter_B = prob_event(pm, lambda out: out[0] == 6 and out[1] == 6)

    # Validation theorique de l'independance de P(A inter B) = P(A) * P(B)
    assert abs(prob_A - 1/6) < 1e-9, "Probabilite marginale A erronee"
    assert abs(prob_B - 1/6) < 1e-9, "Probabilite marginale B erronee"
    assert abs(prob_A_inter_B - (1/36)) < 1e-9, "La mesure de l'intersection doit etre 1/36."

    print("Tous les tests du TP 2 sont valides.")
```
