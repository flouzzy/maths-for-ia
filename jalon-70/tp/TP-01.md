# TP 1 : Rectangle cartésien et produit de l'aire cartésienne

Ce TP modélise la notion de "rectangle cartésien" comme la base de la tribu produit et simule le théorème qui définit que l'aire cartésienne d'un rectangle 2D est le produit des longueurs des intervalles 1D.

```python
# Definition des intervalles de dimension 1 (X et Y)
def length_interval(interval):
    return max(0, interval[1] - interval[0])

# Definition d'un rectangle en dimension 2 (A x B)
class RectangleCartesian:
    def __init__(self, interval_x, interval_y):
        self.interval_x = interval_x
        self.interval_y = interval_y

    def measure_product(self):
        # Implementation de la regle pi(A x B) = mu(A) * mu(B)
        measure_x = length_interval(self.interval_x)
        measure_y = length_interval(self.interval_y)
        return measure_x * measure_y

    def contains(self, point_x, point_y):
        in_x = self.interval_x[0] <= point_x <= self.interval_x[1]
        in_y = self.interval_y[0] <= point_y <= self.interval_y[1]
        return in_x and in_y

# Validation des calculs et du theoreme fondamental de l'aire produite
if __name__ == "__main__":
    rect = RectangleCartesian((0, 2), (1, 4))

    # Test 1 : verification de la mesure du rectangle
    assert rect.measure_product() == 6, "L'aire euclidienne du rectangle [0,2]x[1,4] doit valoir 6."

    # Test 2 : Appartenance
    assert rect.contains(1, 2) == True, "Le point (1,2) appartient au rectangle."
    assert rect.contains(3, 2) == False, "Le point (3,2) n'appartient pas au rectangle."

    print("Tous les tests du TP 1 sont valides.")
```
