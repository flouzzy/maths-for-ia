# TP 4 : Connexité par arcs: Génération de trajectoires convexes

```python
def generate_convex_path(x_start, x_end, steps=10):
    """
    Génère un chemin (arc) entre deux points d'un espace vectoriel
    en utilisant la combinaison convexe temporelle.
    Prouve par simulation la connexité par arcs des espaces convexes.
    """
    assert steps > 0, "Le nombre d'étapes doit être positif"

    path = []
    for i in range(steps + 1):
        t = i / steps
        # Combinaison barycentrique : chemin continu de x_start à x_end
        point = tuple((1 - t) * a + t * b for a, b in zip(x_start, x_end))
        path.append(point)

    return path

# Validation mathématique en 3D
start_pt = (0.0, 0.0, 0.0)
end_pt = (10.0, 5.0, -5.0)

traj = generate_convex_path(start_pt, end_pt, steps=5)

assert traj[0] == start_pt, "Le chemin doit commencer au point de départ"
assert traj[-1] == end_pt, "Le chemin doit finir au point d'arrivée"
# Vérification du point médian
assert traj[2][0] == pytest_approx(4.0) if 'pytest_approx' in globals() else abs(traj[2][0] - 4.0) < 1e-9
print("TP04: Génération de l'arc continu validée.")
```
