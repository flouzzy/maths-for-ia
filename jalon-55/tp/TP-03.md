# TP 3 : Simulation du théorème des valeurs intermédiaires via dichotomie

```python
def bisection_method(f, a, b, tol=1e-7):
    """
    Trouve une racine de la fonction f sur [a, b] en exploitant la connexité.
    Si f(a)*f(b) < 0, l'image du connexe [a,b] contient forcément 0.
    """
    assert f(a) * f(b) <= 0, "Les bornes doivent être de signes opposés."

    while (b - a) / 2.0 > tol:
        midpoint = (a + b) / 2.0
        if f(midpoint) == 0:
            return midpoint
        elif f(a) * f(midpoint) < 0:
            b = midpoint
        else:
            a = midpoint

    return (a + b) / 2.0

# Validation mathématique : trouver racine de x^3 - x - 2
def func(x):
    return x**3 - x - 2

# f(1) = -2, f(2) = 4. L'intervalle est connexe, donc il y a un zéro.
root = bisection_method(func, 1, 2)
assert abs(func(root)) < 1e-6, "La racine trouvée n'est pas assez précise"
print("TP03: Méthode de dichotomie validée.")
```
