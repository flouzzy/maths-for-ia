# TP 1 : Vérification topologique de la non-connexité des rationnels

```python
def check_rational_disconnection(x, alpha, y):
    """
    Démontre informatiquement l'idée topologique que les rationnels
    ne sont pas connexes en identifiant un "mur" irrationnel entre deux points.
    """
    # x et y sont nos points d'intérêt, alpha est un irrationnel entre eux.
    # On partitionne implicitement via la position par rapport à alpha.

    assert x < alpha < y, "Alpha doit séparer x et y"

    # On définit les deux "ouverts" (ensembles de rationnels de chaque côté)
    # par une fonction d'appartenance.
    def in_U(val): return val < alpha
    def in_V(val): return val > alpha

    # Vérification que x est dans U et y dans V
    assert in_U(x) == True, "x doit être dans le premier ouvert"
    assert in_V(y) == True, "y doit être dans le second ouvert"

    # Le fait qu'un irrationnel (alpha) les sépare illustre que
    # tout intervalle de Q peut être scindé.
    return True

# Validation mathématique
import math
assert check_rational_disconnection(1.0, math.sqrt(2), 2.0) == True
print("TP01: Validation des coupures irrationnelles réussie.")
```
