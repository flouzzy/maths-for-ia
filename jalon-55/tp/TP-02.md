# TP 2 : Algorithme de détection des composantes connexes d'un graphe

```python
def dfs(graph, start, visited):
    """
    Parcours en profondeur pour explorer une composante connexe.
    """
    visited.add(start)
    for neighbor in graph.get(start, []):
        if neighbor not in visited:
            dfs(graph, neighbor, visited)

def find_connected_components(graph):
    """
    Trouve toutes les composantes connexes d'un graphe non orienté.
    """
    visited = set()
    components = []

    for node in graph:
        if node not in visited:
            component = set()
            dfs(graph, node, component)
            components.append(component)
            visited.update(component)

    return components

# Validation mathématique sur un petit graphe
# Composante 1: {1, 2, 3}, Composante 2: {4, 5}
G = {
    1: [2, 3],
    2: [1, 3],
    3: [1, 2],
    4: [5],
    5: [4]
}

comps = find_connected_components(G)
assert len(comps) == 2, "Le graphe doit avoir 2 composantes connexes"
assert {1, 2, 3} in comps, "La composante {1, 2, 3} est manquante"
assert {4, 5} in comps, "La composante {4, 5} est manquante"
print("TP02: Algorithme DFS validé.")
```
