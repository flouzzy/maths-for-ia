# TP 5 : Union d'ensembles d'intersection non vide (Connexité)

```python
def merge_connected_sets(sets):
    """
    Simule le théorème: l'union de connexes ayant une intersection commune
    non vide est connexe. Ici on fusionne des ensembles (discrets).
    """
    if not sets:
        return set()

    merged = set(sets[0])
    for s in sets[1:]:
        # L'analogue de l'intersection topologique non vide
        assert len(merged.intersection(s)) > 0, "Les ensembles doivent s'intersecter pour fusionner (analogie connexité)"
        merged = merged.union(s)

    return merged

# Validation mathématique discrète
# On a un point pivot commun 'x0'
A = {'x1', 'x2', 'x0'}
B = {'y1', 'x0'}
C = {'z1', 'z2', 'x0'}

famille_connexe = [A, B, C]
grand_ensemble = merge_connected_sets(famille_connexe)

assert 'x1' in grand_ensemble and 'y1' in grand_ensemble and 'z1' in grand_ensemble
assert len(grand_ensemble) == 6
print("TP05: Fusion par intersection non vide validée.")
```
