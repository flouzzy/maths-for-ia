## Implémentation algorithmique des Normes Vectorielles

L'objectif de ce TP est d'implémenter en Python pur le calcul des normes usuelles $L^1$, $L^2$, et $L^\infty$ pour un vecteur de dimension $n$, et de vérifier empiriquement l'équivalence de ces normes sur des données générées aléatoirement.

### Code Python Pur (Zéro Framework)

```python
import math
import random

def norm_l1(vector: list[float]) -> float:
    """Calcule la norme L1 (Manhattan) du vecteur."""
    return sum(abs(x) for x in vector)

def norm_l2(vector: list[float]) -> float:
    """Calcule la norme L2 (Euclidienne) du vecteur."""
    return math.sqrt(sum(x * x for x in vector))

def norm_linf(vector: list[float]) -> float:
    """Calcule la norme L_infini (Maximum) du vecteur."""
    if not vector:
        return 0.0
    return max(abs(x) for x in vector)

def test_norm_equivalence(dim: int, num_tests: int = 1000):
    """
    Vérifie l'encadrement théorique de l'équivalence entre L2 et L_infini:
    ||x||_inf <= ||x||_2 <= sqrt(dim) * ||x||_inf
    """
    for _ in range(num_tests):
        # Génération d'un vecteur aléatoire de dimension `dim`
        vec = [random.uniform(-100.0, 100.0) for _ in range(dim)]

        n_inf = norm_linf(vec)
        n_2 = norm_l2(vec)

        # Validation stricte des bornes
        assert n_inf <= n_2 + 1e-9, f"Violation de la borne inférieure: n_inf {n_inf} > n_2 {n_2}"
        assert n_2 <= math.sqrt(dim) * n_inf + 1e-9, f"Violation de la borne supérieure: n_2 {n_2} > sqrt(n)*n_inf {math.sqrt(dim) * n_inf}"

    print(f"TP 4 - Test d'équivalence de normes validé avec succès sur {num_tests} vecteurs de dimension {dim}.")

if __name__ == '__main__':
    # Test avec la dimension variant pour chaque TP
    test_norm_equivalence(dim=40)
```
