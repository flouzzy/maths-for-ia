# TP-04 : Flot sur un potentiel de double puits

On étudie la dynamique non linéaire associée au potentiel en double puits $V(x) = \frac{1}{4}x^4 - \frac{1}{2}x^2$.
Les points critiques sont à $x = -1, 0, 1$. L'EDO du gradient est $x' = x - x^3$.

```python
def v_prime(x):
    # Derivee du potentiel V(x) = x^4/4 - x^2/2
    return x**3 - x

def rk4_step(t, x, dt):
    # EDO : x' = -V'(x)
    f = lambda _t, _x: -v_prime(_x)
    k1 = f(t, x)
    k2 = f(t + dt/2, x + dt/2 * k1)
    k3 = f(t + dt/2, x + dt/2 * k2)
    k4 = f(t + dt, x + dt * k3)
    return x + (dt/6) * (k1 + 2*k2 + 2*k3 + k4)

def simulate_double_well():
    t_end = 10.0
    dt = 0.05

    # Si on demarre juste a droite de 0 (point selle instable)
    x1 = 0.01
    t = 0.0
    while t < t_end:
        x1 = rk4_step(t, x1, dt)
        t += dt

    # Le flot doit tomber dans le puits stable à droite (x=1)
    assert abs(x1 - 1.0) < 1e-2, "Le flot n'a pas atteint l'attracteur stable +1"

    # Si on demarre juste a gauche de 0
    x2 = -0.01
    t = 0.0
    while t < t_end:
        x2 = rk4_step(t, x2, dt)
        t += dt

    # Le flot doit tomber dans le puits stable à gauche (x=-1)
    assert abs(x2 - (-1.0)) < 1e-2, "Le flot n'a pas atteint l'attracteur stable -1"

    print("TP-04 validé : La dynamique du double puits et les bassins d'attraction sont corrects.")

if __name__ == '__main__':
    simulate_double_well()
```
