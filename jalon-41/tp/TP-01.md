# TP-01 : Discrétisation et méthode d'Euler

Ce TP implémente la méthode d'Euler explicite pour résoudre numériquement une EDO du premier ordre : $y'(t) = f(t, y(t))$.
On comparera la solution numérique à la solution analytique exacte pour l'équation $y' = y$, qui donne l'exponentielle.

```python
def euler_explicit(f, y0, t0, t_end, dt):
    '''
    Résout y' = f(t, y) par la méthode d'Euler explicite.
    '''
    times = [t0]
    y_values = [y0]

    t = t0
    y = y0

    while t < t_end:
        # y_{n+1} = y_n + dt * f(t_n, y_n)
        y = y + dt * f(t, y)
        t = t + dt

        times.append(t)
        y_values.append(y)

    return times, y_values

def test_euler_exponential():
    # Equation: y' = y => analytique: y(t) = exp(t)
    def f(t, y):
        return y

    y0 = 1.0
    t0 = 0.0
    t_end = 1.0
    dt = 0.001

    times, y_num = euler_explicit(f, y0, t0, t_end, dt)

    # La vraie valeur de exp(1) est environ 2.71828
    true_value = 2.718281828459
    num_value = y_num[-1]

    error = abs(true_value - num_value)

    # Avec dt=0.001, l'erreur d'Euler explicite est en O(dt)
    assert error < 0.01, f"Erreur trop grande : {error}"
    print("TP-01 validé : La méthode d'Euler approche correctement l'exponentielle.")

if __name__ == '__main__':
    test_euler_exponential()
```
