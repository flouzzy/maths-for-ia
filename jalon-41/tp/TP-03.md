# TP-03 : Méthode de Runge-Kutta d'ordre 4

Pour une meilleure précision sur la résolution numérique des EDO, on remplace Euler par Runge-Kutta d'ordre 4 (RK4). L'erreur globale passe de $O(dt)$ à $O(dt^4)$.

```python
import math

def rk4_step(f, t, y, dt):
    k1 = f(t, y)
    k2 = f(t + dt/2, y + dt/2 * k1)
    k3 = f(t + dt/2, y + dt/2 * k2)
    k4 = f(t + dt, y + dt * k3)

    return y + (dt/6) * (k1 + 2*k2 + 2*k3 + k4)

def solve_rk4(f, y0, t0, t_end, dt):
    t = t0
    y = y0

    while t < t_end:
        y = rk4_step(f, t, y, dt)
        t += dt

    return y

def test_rk4_precision():
    # Test sur y' = y - t^2 + 1, y(0)=0.5
    # Analytique: y(t) = (t+1)^2 - 0.5 * exp(t)
    def f(t, y):
        return y - t**2 + 1.0

    def y_exact(t):
        return (t + 1.0)**2 - 0.5 * math.exp(t)

    y0 = 0.5
    t_end = 2.0
    dt = 0.1

    y_num = solve_rk4(f, y0, 0.0, t_end, dt)
    y_true = y_exact(t_end)

    error = abs(y_num - y_true)

    # RK4 est extremement precis meme avec un dt relativement grand
    assert error < 1e-4, f"Erreur de RK4 trop elevee : {error}"
    print("TP-03 validé : RK4 garantit une haute précision d'intégration.")

if __name__ == '__main__':
    test_rk4_precision()
```
