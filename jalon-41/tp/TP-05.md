# TP-05 : Introduction jouet aux Neural ODEs

Une Neural ODE modélise la transformation continue de données (un tenseur $h$) par une dynamique $dh/dt = f_\theta(h)$.
Ici on simule une dynamique paramétrée simplifiée $h' = W \cdot \text{tanh}(h)$ en dimension 1, et on cherche à intégrer le flot.

```python
import math

class ToyNeuralODE:
    def __init__(self, W):
        self.W = W

    def f(self, t, h):
        # Dynamique decrite par un pseudo-neurone
        return self.W * math.tanh(h)

    def forward(self, h0, t_end, dt=0.1):
        # Integre l'ODE avec Euler explicite
        t = 0.0
        h = h0
        while t < t_end:
            h = h + dt * self.f(t, h)
            t += dt
        return h

def test_neural_ode():
    # Pour W < 0, l'origine est un puits attractif
    model_stable = ToyNeuralODE(W=-1.0)
    h_out_stable = model_stable.forward(h0=5.0, t_end=10.0)
    assert abs(h_out_stable) < 0.1, "Avec W=-1, le flot devrait decroître vers 0."

    # Pour W > 0, l'origine est répulsive
    model_unstable = ToyNeuralODE(W=1.0)
    h_out_unstable = model_unstable.forward(h0=0.1, t_end=5.0)
    assert h_out_unstable > 1.0, "Avec W=1, le flot devrait diverger de 0."

    print("TP-05 validé : L'intégration paramétrée continue fonctionne correctement.")

if __name__ == '__main__':
    test_neural_ode()
```
