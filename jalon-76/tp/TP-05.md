# TP 5 : Projection Orthogonale et Théorème de Pythagore Généralisé

**Objectif :** Implémenter la projection d'un vecteur sur un sous-espace et vérifier le théorème de Pythagore dans L2.

```python
import math

def dot(f, g, dx): return sum(a*b for a,b in zip(f,g))*dx
def norm(f, dx): return math.sqrt(dot(f, f, dx))

def test_tp5():
    N = 1000
    dx = 1.0 / (N - 1)
    x = [i*dx for i in range(N)]

    # Base orthonormale de sous-espace : b1=1, b2=sqrt(3)*(2x-1)
    b1 = [1.0 for _ in x]
    b2 = [math.sqrt(3)*(2*xi - 1) for xi in x]

    # Fonction a projeter f(x) = x^2
    f = [xi**2 for xi in x]

    # Calcul des coefficients de projection
    c1 = dot(f, b1, dx)
    c2 = dot(f, b2, dx)

    proj_f = [c1*v1 + c2*v2 for v1, v2 in zip(b1, b2)]
    reste_f = [f_val - p_val for f_val, p_val in zip(f, proj_f)]

    # Theoreme de Pythagore: ||f||^2 = ||proj_f||^2 + ||reste_f||^2
    norm_f_sq = norm(f, dx)**2
    norm_proj_sq = norm(proj_f, dx)**2
    norm_reste_sq = norm(reste_f, dx)**2

    assert abs(norm_f_sq - (norm_proj_sq + norm_reste_sq)) < 1e-3, "Echec Pythagore"

    # Verifier que l'erreur est orthogonale au plan de projection
    dot_reste_b1 = dot(reste_f, b1, dx)
    dot_reste_b2 = dot(reste_f, b2, dx)

    assert abs(dot_reste_b1) < 1e-3
    assert abs(dot_reste_b2) < 1e-3
    print("TP 5 : Projection orthogonale et theoreme de Pythagore verifie.")

if __name__ == "__main__":
    test_tp5()
```
