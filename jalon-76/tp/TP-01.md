# TP 1 : Simulation numérique du Produit Scalaire L2

**Objectif :** Implémenter le calcul du produit scalaire par intégration numérique (méthode des rectangles) et vérifier l'orthogonalité de fonctions simples.

```python
import math

def linspace(start, stop, num):
    if num == 1: return [float(start)]
    step = (stop - start) / (num - 1)
    return [start + i * step for i in range(num)]

def dot_product_l2(f_vals, g_vals, dx):
    assert len(f_vals) == len(g_vals)
    return sum(f * g for f, g in zip(f_vals, g_vals)) * dx

def test_tp1():
    N = 10000
    a, b = -math.pi, math.pi
    dx = (b - a) / (N - 1)
    x = linspace(a, b, N)

    f = [math.cos(xi) for xi in x]
    g = [math.sin(xi) for xi in x]

    ps = dot_product_l2(f, g, dx)
    assert abs(ps) < 1e-3, f"Echec: Produit scalaire = {ps}, attendu ~0"
    print("TP 1 : Produit scalaire <cos, sin> == 0 verifie.")

if __name__ == "__main__":
    test_tp1()
```
