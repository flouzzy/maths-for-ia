# TP 2 : Norme L2 et Inégalité de Cauchy-Schwarz

**Objectif :** Calculer des normes L2 et vérifier empiriquement l'inégalité de Cauchy-Schwarz sur des signaux.

```python
import math

def linspace(start, stop, num):
    step = (stop - start) / (num - 1)
    return [start + i * step for i in range(num)]

def dot_product_l2(f_vals, g_vals, dx):
    return sum(f * g for f, g in zip(f_vals, g_vals)) * dx

def norm_l2(f_vals, dx):
    return math.sqrt(dot_product_l2(f_vals, f_vals, dx))

def test_tp2():
    N = 5000
    dx = 1.0 / (N - 1)
    x = linspace(0, 1, N)

    f = [xi for xi in x]        # f(x) = x
    g = [xi**2 for xi in x]     # g(x) = x^2

    ps = dot_product_l2(f, g, dx)
    norm_f = norm_l2(f, dx)
    norm_g = norm_l2(g, dx)

    assert abs(ps - 0.25) < 1e-2
    assert abs(norm_f - math.sqrt(1/3)) < 1e-2
    assert abs(norm_g - math.sqrt(1/5)) < 1e-2

    assert abs(ps) <= norm_f * norm_g, "Echec de l'inegalite de Cauchy-Schwarz"
    print("TP 2 : Inegalite de Cauchy-Schwarz verifiee.")

if __name__ == "__main__":
    test_tp2()
```
