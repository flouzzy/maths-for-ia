# Travail Pratique 3 : Le Singe Savant - Simulation Simple

## Énoncé et Code de Simulation

```python
def test_singe_savant():
    """
    Simule la frappe aleatoire d'un mot par un singe pour
    illustrer le second lemme avec des probabilites constantes.
    """
    import random

    alphabet = "abcdefghijklmnopqrstuvwxyz"
    target = "bc" # Un mot tres court pour des raisons de temps de calcul
    length = len(target)

    # On frappe un grand nombre de caracteres
    N = 50000
    texte = "".join(random.choice(alphabet) for _ in range(N))

    # On compte les occurrences de 'bc'
    occurrences = 0
    for i in range(N - length + 1):
        if texte[i:i+length] == target:
            occurrences += 1

    esperance = (N - length + 1) * ((1/26.0)**2)
    assert occurrences > 0, "Le singe devrait taper la sequence courte"
    print(f"Occurrences de '{target}' :", occurrences, "(Esperance:", esperance, ")")

if __name__ == '__main__':
    test_singe_savant()
```
