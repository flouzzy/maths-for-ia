# Travail Pratique 5 : Convergence des Valeurs Extremes (Loi Normale)

## Énoncé et Code de Simulation

```python
def test_extremes_normaux():
    """
    Verifie que pour une sequence i.i.d. gaussienne standard,
    l'evenement X_n > 2*sqrt(ln(n)) ne se realise qu'un nombre fini de fois.
    """
    import random
    import math

    # Implementation maison d'un generateur normal via Box-Muller
    def random_normal():
        u1 = random.random()
        u2 = random.random()
        z0 = math.sqrt(-2.0 * math.log(u1)) * math.cos(2.0 * math.pi * u2)
        return z0

    N = 10000
    realisations = 0

    for n in range(2, N + 1):
        x = random_normal()
        seuil = 2.0 * math.sqrt(math.log(n))
        if abs(x) > seuil:
            realisations += 1

    # Le premier lemme predit que cet evenement est extremement rare
    assert realisations < 10, "L'evenement doit se realiser tres peu souvent"
    print("Nombre de depassements du seuil critique:", realisations)

if __name__ == '__main__':
    test_extremes_normaux()
```
