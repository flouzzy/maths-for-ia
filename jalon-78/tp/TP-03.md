# Évaluation de l'erreur quadratique moyenne

Le théorème de Parseval nous assure que l'énergie du signal est conservée. Nous allons mesurer l'erreur quadratique entre le signal original et sa reconstruction tronquée (ordre N).

```python
import math

def f_triangle(t):
    # Signal triangulaire (valeur absolue sur [-pi, pi])
    t = (t + math.pi) % (2 * math.pi) - math.pi
    return abs(t)

def theory_coeffs_triangle(n_max):
    a = [math.pi] # a0
    b = [0.0]
    for n in range(1, n_max + 1):
        b.append(0.0)
        if n % 2 != 0:
            a.append(-4.0 / (math.pi * n * n))
        else:
            a.append(0.0)
    return a, b

def mse_error(f, a, b, n_max, num_points=500):
    error = 0.0
    h = 2 * math.pi / num_points
    for i in range(num_points):
        t = -math.pi + i * h
        true_val = f(t)
        approx = a[0]/2.0
        for n in range(1, n_max + 1):
            approx += a[n] * math.cos(n * t)
        diff = true_val - approx
        error += diff * diff * h
    return error / (2 * math.pi)

a_tri, b_tri = theory_coeffs_triangle(5)
error_N5 = mse_error(f_triangle, a_tri, b_tri, 5)
a_tri_20, b_tri_20 = theory_coeffs_triangle(20)
error_N20 = mse_error(f_triangle, a_tri_20, b_tri_20, 20)

print(f"MSE N=5  : {error_N5}")
print(f"MSE N=20 : {error_N20}")
assert error_N20 < error_N5, "L'erreur doit decroitre avec N"
```
