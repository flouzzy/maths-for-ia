# TP 03 : Calcul de la Transformée de Fourier Discrète (DFT 1D)


Nous programmons ici l'algorithme direct (naïf) de la Transformée de Fourier Discrète en complexes purs, en $O(N^2)$.

```python
import cmath
import math

def dft(x):
    N = len(x)
    X = []
    for k in range(N):
        somme = 0j
        for n in range(N):
            angle = -2j * math.pi * k * n / N
            somme += x[n] * cmath.exp(angle)
        X.append(somme)
    return X

def inverse_dft(X):
    N = len(X)
    x = []
    for n in range(N):
        somme = 0j
        for k in range(N):
            angle = 2j * math.pi * k * n / N
            somme += X[k] * cmath.exp(angle)
        x.append(somme / N)
    return x

# Test sur un signal de 4 points
signal_test = [1.0, 2.0, 3.0, 4.0]
X_k = dft(signal_test)

# Verification de la composante continue k=0
assert abs(X_k[0] - (1+2+3+4)) < 1e-10

# Inversion pour retrouver le signal de depart
signal_reconstruit = inverse_dft(X_k)

for i in range(len(signal_test)):
    assert abs(signal_reconstruit[i].real - signal_test[i]) < 1e-10
    assert abs(signal_reconstruit[i].imag) < 1e-10

print("TP 03 valide : DFT et iDFT codées de zero avec succes.")
```
