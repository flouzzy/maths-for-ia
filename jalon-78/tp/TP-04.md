# TP 04 : Algorithme Fast Fourier Transform (FFT) Radix-2


Implémentation algorithmique de la FFT récursive (Cooley-Tukey Radix-2), réduisant la complexité de $O(N^2)$ à $O(N \log N)$.

```python
import cmath
import math

def fft(x):
    N = len(x)
    if N <= 1:
        return x

    # Verification que N est une puissance de 2
    if (N & (N - 1)) != 0:
        raise ValueError("La taille du tableau doit etre une puissance de 2.")

    # Separation des indices pairs et impairs
    x_pair = x[0::2]
    x_impair = x[1::2]

    # Appels recursifs
    X_pair = fft(x_pair)
    X_impair = fft(x_impair)

    X = [0j] * N

    # Recombinaison (le papillon)
    for k in range(N // 2):
        W_k = cmath.exp(-2j * math.pi * k / N) * X_impair[k]
        X[k] = X_pair[k] + W_k
        X[k + N // 2] = X_pair[k] - W_k

    return X

# Test de la FFT sur un signal simple de taille 8
signal_test = [1, 1, 1, 1, 0, 0, 0, 0]
X_fft = fft(signal_test)

# Comparaison avec la DFT naive du TP 03
def dft_naive(x):
    N = len(x)
    return [sum(x[n] * cmath.exp(-2j * math.pi * k * n / N) for n in range(N)) for k in range(N)]

X_dft = dft_naive(signal_test)

for k in range(len(signal_test)):
    assert abs(X_fft[k] - X_dft[k]) < 1e-10

print("TP 04 valide : L'algorithme FFT recursif Radix-2 est exact par rapport a la DFT.")
```
