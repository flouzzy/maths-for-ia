# Observation du Phénomène de Gibbs

Le phénomène de Gibbs décrit une sur-oscillation de 9% environ aux points de discontinuité. Nous allons traquer ce maximum numériquement sur le signal carré.

```python
import math

def reconstruct_square(t, n_max):
    val = 0.0
    for n in range(1, n_max + 1):
        if n % 2 != 0:
            b_n = 4.0 / (n * math.pi)
            val += b_n * math.sin(n * t)
    return val

def find_maximum_overshoot(n_max, search_points=1000):
    # On cherche le max sur [0, pi]
    max_val = 0.0
    t_max = 0.0
    for i in range(search_points):
        # On concentre la recherche pres de 0
        t = (i / search_points) * math.pi
        val = reconstruct_square(t, n_max)
        if val > max_val:
            max_val = val
            t_max = t
    return max_val, t_max

n_test = 50
max_gibbs, t_pos = find_maximum_overshoot(n_test)
print(f"Pour N={n_test}, depassement max = {max_gibbs} en t={t_pos}")

# Le pic theorique se trouve en t = pi / (2 * (N_impaire_max + 1)) environ
# L'amplitude doit etre proche de 1.1789 (constante de Wilbraham-Gibbs)
assert 1.15 < max_gibbs < 1.25, "Le phenomene de Gibbs doit osciller autour de 1.18"
```
