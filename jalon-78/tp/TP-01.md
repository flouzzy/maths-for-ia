# TP 01 : Calcul numérique naïf des coefficients de Fourier


L'objectif de ce TP est de programmer le calcul des coefficients de Fourier réels d'un signal périodique par intégration numérique simple (méthode des rectangles).

```python
import math

def integrale_rectangles(func, a, b, n_steps=1000):
    dx = (b - a) / n_steps
    somme = 0.0
    for i in range(n_steps):
        x = a + i * dx
        somme += func(x) * dx
    return somme

def fourier_coeffs_reels(f, T, N, n_steps=1000):
    omega = 2.0 * math.pi / T

    # Calcul de a0
    a0 = (2.0 / T) * integrale_rectangles(f, 0, T, n_steps)

    coeffs = [(a0, 0.0)] # Stockage sous forme de tuple (an, bn)

    for n in range(1, N + 1):
        # Fonctions integrandes pour an et bn
        def integrande_an(t, n=n):
            return f(t) * math.cos(n * omega * t)

        def integrande_bn(t, n=n):
            return f(t) * math.sin(n * omega * t)

        an = (2.0 / T) * integrale_rectangles(integrande_an, 0, T, n_steps)
        bn = (2.0 / T) * integrale_rectangles(integrande_bn, 0, T, n_steps)

        coeffs.append((an, bn))

    return coeffs

# Test sur le signal carre
def signal_carre(t):
    # Periode 2*pi, vaut 1 sur [0, pi[, -1 sur [pi, 2pi[
    t_mod = t % (2 * math.pi)
    if t_mod < math.pi:
        return 1.0
    else:
        return -1.0

# Calcul des 3 premiers harmoniques
coeffs = fourier_coeffs_reels(signal_carre, 2*math.pi, 3, 5000)

# Verifications (valeurs theoriques : a0=0, an=0, bn = 4/(n*pi) pour n impair, 0 pour n pair)
a0, b0 = coeffs[0]
a1, b1 = coeffs[1]
a2, b2 = coeffs[2]
a3, b3 = coeffs[3]

assert abs(a0) < 1e-2
assert abs(a1) < 1e-2
assert abs(b1 - 4/math.pi) < 1e-2
assert abs(a2) < 1e-2
assert abs(b2) < 1e-2
assert abs(a3) < 1e-2
assert abs(b3 - 4/(3*math.pi)) < 1e-2
print("TP 01 valide : Coefficients du signal carre estimes correctement.")
```
