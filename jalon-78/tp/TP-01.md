# Calcul numérique des coefficients de Fourier

L'objectif de ce TP est de calculer numériquement les coefficients de Fourier $a_n$ et $b_n$ d'un signal périodique en utilisant la méthode d'intégration des trapèzes.

```python
import math

def f_square(t):
    # Signal carre de periode 2*pi, amplitude 1 et -1
    t = t % (2 * math.pi)
    if t < math.pi:
        return 1.0
    else:
        return -1.0

def trapz_integral(f, a, b, N=1000):
    h = (b - a) / N
    s = 0.5 * (f(a) + f(b))
    for i in range(1, N):
        s += f(a + i * h)
    return s * h

def fourier_coefficients(f, n_max, num_points=2000):
    a = []
    b = []
    a0 = (1 / math.pi) * trapz_integral(f, -math.pi, math.pi, num_points)
    a.append(a0)
    b.append(0.0) # b0 n'existe pas, on met 0 par convention

    for n in range(1, n_max + 1):
        def f_cos(t): return f(t) * math.cos(n * t)
        def f_sin(t): return f(t) * math.sin(n * t)

        an = (1 / math.pi) * trapz_integral(f_cos, -math.pi, math.pi, num_points)
        bn = (1 / math.pi) * trapz_integral(f_sin, -math.pi, math.pi, num_points)
        a.append(an)
        b.append(bn)

    return a, b

# Test et validation
a, b = fourier_coefficients(f_square, 5)
print("Coefficient a0 :", a[0])
for i in range(1, 6):
    print(f"n={i} -> an={a[i]:.4f}, bn={b[i]:.4f}")

# Assertions : an doivent etre nuls (fonction impaire)
for i in range(6):
    assert abs(a[i]) < 1e-2, "Les coefficients an d'un signal impair doivent etre nuls"
```
