## TP 1 : Calcul numerique des coefficients de Fourier


Dans ce TP, nous allons ecrire une fonction qui calcule de maniere numerique les coefficients de Fourier (reels) d'une fonction periodique arbitraire.

\begin{lstlisting}[language=Python]
import math

def calculate_fourier_coefficients(func, T, N, num_points=1000):
    a = []
    b = []
    omega = 2 * math.pi / T
    dt = T / num_points

    # Calcul de a0
    a0 = 0.0
    for i in range(num_points):
        t = i * dt
        a0 += func(t) * dt
    a0 = a0 * (2.0 / T)
    a.append(a0)
    b.append(0.0) # b0 n'est pas utilise mais on garde l'index

    # Calcul des an et bn
    for n in range(1, N + 1):
        an = 0.0
        bn = 0.0
        for i in range(num_points):
            t = i * dt
            an += func(t) * math.cos(n * omega * t) * dt
            bn += func(t) * math.sin(n * omega * t) * dt
        an = an * (2.0 / T)
        bn = bn * (2.0 / T)
        a.append(an)
        b.append(bn)

    return a, b

# Test avec le signal carre
def square_wave(t):
    # Periode 2*pi, vaut 1 sur [0, pi[ et -1 sur [pi, 2*pi[
    t_mod = t % (2 * math.pi)
    if t_mod < math.pi:
        return 1.0
    else:
        return -1.0

a_coeffs, b_coeffs = calculate_fourier_coefficients(square_wave, 2*math.pi, 5)
print("a:", a_coeffs)
print("b:", b_coeffs)
