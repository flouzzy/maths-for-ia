# TP 02 : Synthèse de signal et phénomène de Gibbs


Dans ce TP, nous allons reconstruire un signal à partir de ses coefficients de Fourier et observer l'erreur d'approximation aux points de discontinuité.

```python
import math

def synthese_fourier(coeffs, T, t):
    omega = 2.0 * math.pi / T
    valeur = coeffs[0][0] / 2.0 # a0 / 2

    for n in range(1, len(coeffs)):
        an, bn = coeffs[n]
        valeur += an * math.cos(n * omega * t) + bn * math.sin(n * omega * t)

    return valeur

# Generation des coefficients theoriques exacts du signal carre pour N grand
N_max = 50
coeffs_theoriques = [(0.0, 0.0)] # a0, b0

for n in range(1, N_max + 1):
    an = 0.0
    if n % 2 == 1:
        bn = 4.0 / (n * math.pi)
    else:
        bn = 0.0
    coeffs_theoriques.append((an, bn))

T = 2 * math.pi

# Evaluation au point de continuite t = pi/2
val_pi_sur_2 = synthese_fourier(coeffs_theoriques, T, math.pi / 2)
# Le signal theorique vaut 1
assert abs(val_pi_sur_2 - 1.0) < 0.05

# Evaluation au point de discontinuite t = 0
val_zero = synthese_fourier(coeffs_theoriques, T, 0)
# Par Dirichlet, la somme est la moyenne 0
assert abs(val_zero) < 1e-10

# Observation du phenomene de Gibbs pres de la discontinuite (t = pi / (2*(N_max+1)) approx)
t_gibbs = math.pi / (2 * (N_max + 1))
val_gibbs = synthese_fourier(coeffs_theoriques, T, t_gibbs)

# Le depassement theorique maximal est d'environ 1.1789 (au-dessus de 1)
assert val_gibbs > 1.15

print("TP 02 valide : Synthese de signal et depassement de Gibbs verifies.")
```
