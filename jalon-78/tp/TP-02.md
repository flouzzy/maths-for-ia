## TP 2 : Reconstruction du signal a partir des harmoniques


Ce TP consiste a reconstruire un signal a partir de ses coefficients de Fourier (la somme partielle) pour observer le phenomene de Gibbs.

\begin{lstlisting}[language=Python]
import math

def reconstruct_signal(a, b, T, t_values):
    omega = 2 * math.pi / T
    N = len(a) - 1
    reconstructed = []

    for t in t_values:
        val = a[0] / 2.0
        for n in range(1, N + 1):
            val += a[n] * math.cos(n * omega * t)
            val += b[n] * math.sin(n * omega * t)
        reconstructed.append(val)

    return reconstructed

# Coefficients theoriques pour un signal dent de scie
def sawtooth_coeffs(N):
    a = [0.0] * (N + 1)
    b = [0.0] * (N + 1)
    for n in range(1, N + 1):
        b[n] = 2.0 * (-1.0)**(n+1) / n
    return a, b

# Reconstruction sur une grille de temps
N_harmonics = 10
a_theo, b_theo = sawtooth_coeffs(N_harmonics)

t_grid = [i * 0.1 for i in range(100)]
signal_recon = reconstruct_signal(a_theo, b_theo, 2*math.pi, t_grid)

for i in range(5):
    print("t = {:.1f}, f_approx(t) = {:.4f}".format(t_grid[i], signal_recon[i]))
\end{lstlisting}
