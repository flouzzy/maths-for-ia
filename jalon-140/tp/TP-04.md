# TP 4 - Perte Charnière (Hinge Loss) et SVM

## Introduction Mathématique
Consistance du Hinge Loss : lien avec la marge maximale.

## Implémentation (Python Pur)
```python
import numpy as np

def generate_separable_data(n=100):
    X = np.random.randn(n, 2)
    # On écarte les points pour garantir la séparabilité
    y = np.sign(X[:, 0] + X[:, 1])
    X += y[:, None] * 1.5
    return X, y

def hinge_loss_grad(w, X, y, C=1.0):
    margins = y * (X @ w)
    loss = np.maximum(0, 1 - margins)

    # Gradient: -y*x if margin < 1, else 0
    grad_mask = (margins < 1).astype(float)
    grad_w = -C * (grad_mask * y)[:, None] * X
    return np.sum(grad_w, axis=0) + w  # + w for L2 regularization

X_train, y_train = generate_separable_data(200)

w = np.zeros(2)
lr = 0.01

for epoch in range(500):
    grad = hinge_loss_grad(w, X_train, y_train, C=1.0)
    w -= lr * grad

margins = y_train * (X_train @ w)
support_vectors = np.sum(margins < 1.05) # Approximativement sur la marge

preds = np.sign(X_train @ w)
acc = np.mean(preds == y_train)

# Assertion mathématique
assert acc == 1.0, "Le problème est linéairement séparable, Hinge Loss doit trouver une séparation parfaite."
print(f"Modèle entraîné. Précision : {acc*100}%. Nombre estimé de vecteurs de support : {support_vectors}.")

```
