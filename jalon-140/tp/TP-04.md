# TP 4 : Classifieur de Bayes Optimal et Fonctions de Perte de Substitution

## Objectif
Ce TP a pour objectif de comprendre et d'implémenter le classifieur de Bayes optimal, ainsi que d'explorer le concept des fonctions de perte de substitution (surrogate loss functions) et leur lien avec la minimisation du risque empirique (ERM). Nous allons coder ces concepts en Python pur, sans utiliser de bibliothèques externes pour l'algèbre linéaire, les statistiques ou le machine learning.

Nous allons :
1.  Générer des données synthétiques pour un problème de classification binaire.
2.  Implémenter un classifieur de Bayes optimal capable de prendre en compte une matrice de coûts arbitraire.
3.  Implémenter différentes fonctions de perte : la perte 0-1, la perte quadratique et la perte d'entropie croisée binaire.
4.  Démontrer comment les fonctions de perte de substitution peuvent être utilisées pour évaluer les performances et comment elles se rapportent à la perte 0-1.
5.  Valider nos implémentations avec des assertions mathématiques.

## Code Python
```python
import math

# --- 1. Data Generation ---

def gaussian_pdf(x, mu, sigma):
    """
    Calcule la densité de probabilité (PDF) d'une distribution gaussienne.
    """
    if sigma <= 0:
        raise ValueError("L'écart-type (sigma) doit être positif.")
    coefficient = 1 / (sigma * math.sqrt(2 * math.pi))
    exponent = -((x - mu)**2) / (2 * sigma**2)
    return coefficient * math.exp(exponent)

def generate_synthetic_data(n_samples_per_class, mu1, sigma1, mu2, sigma2):
    """
    Génère des données synthétiques 1D pour deux classes gaussiennes.
    Pour respecter la contrainte "pas de Matplotlib, sklearn, tensorflow, etc.",
    nous générons des points de manière déterministe et représentative
    autour des moyennes, plutôt que de manière aléatoire.
    """
    X = []
    y = []

    # Classe 0
    for i in range(n_samples_per_class):
        # Création de points espacés autour de mu1
        x_val = mu1 + (i - n_samples_per_class / 2) * (sigma1 / (n_samples_per_class / 4))
        X.append(x_val)
        y.append(0)

    # Classe 1
    for i in range(n_samples_per_class):
        # Création de points espacés autour de mu2
        x_val = mu2 + (i - n_samples_per_class / 2) * (sigma2 / (n_samples_per_class / 4))
        X.append(x_val)
        y.append(1)

    # Note: Les données ne sont pas mélangées pour la simplicité et la reproductibilité
    # sans utiliser de module 'random'.

    return X, y

# --- 2. Bayes Classifier (Optimal pour une matrice de perte donnée) ---

def estimate_parameters(X, y):
    """
    Estime les paramètres (moyenne, écart-type, prior) pour chaque classe
    à partir des données d'entraînement.
    """
    class_0_data = [X[i] for i in range(len(X)) if y[i] == 0]
    class_1_data = [X[i] for i in range(len(X)) if y[i] == 1]

    n_0 = len(class_0_data)
    n_1 = len(class_1_data)
    n_total = n_0 + n_1

    # Priors (probabilités a priori)
    prior_0 = n_0 / n_total if n_total > 0 else 0
    prior_1 = n_1 / n_total if n_total > 0 else 0

    # Moyennes
    mu_0 = sum(class_0_data) / n_0 if n_0 > 0 else 0
    mu_1 = sum(class_1_data) / n_1 if n_1 > 0 else 0

    # Écart-types (variance non biaisée)
    # Gérer le cas de n_x <= 1 pour éviter la division par zéro
    sigma_0 = math.sqrt(sum([(x - mu_0)**2 for x in class_0_data]) / (n_0 - 1)) if n_0 > 1 else 1e-6
    sigma_1 = math.sqrt(sum([(x - mu_1)**2 for x in class_1_data]) / (n_1 - 1)) if n_1 > 1 else 1e-6

    return {
        'prior_0': prior_0, 'mu_0': mu_0, 'sigma_0': sigma_0,
        'prior_1': prior_1, 'mu_1': mu_1, 'sigma_1': sigma_1
    }

def calculate_posteriors(x, params):
    """
    Calcule les probabilités a posteriori P(Y=y|X=x) pour un point x.
    Utilise le théorème de Bayes : P(Y|X) = P(X|Y)P(Y) / P(X)
    """
    prior_0 = params['prior_0']
    mu_0 = params['mu_0']
    sigma_0 = params['sigma_0']

    prior_1 = params['prior_1']
    mu_1 = params['mu_1']
    sigma_1 = params['sigma_1']

    # P(X=x|Y=0) : vraisemblance de x étant donné la classe 0
    likelihood_x_given_y0 = gaussian_pdf(x, mu_0, sigma_0)
    # P(X=x|Y=1) : vraisemblance de x étant donné la classe 1
    likelihood_x_given_y1 = gaussian_pdf(x, mu_1, sigma_1)

    # P(X=x) : évidence (probabilité marginale de X)
    # P(X=x) = P(X=x|Y=0)P(Y=0) + P(X=x|Y=1)P(Y=1)
    evidence = likelihood_x_given_y0 * prior_0 + likelihood_x_given_y1 * prior_1

    if evidence == 0:
        # Si le point est extrêmement improbable pour les deux classes,
        # on retourne des probabilités égales pour éviter la division par zéro.
        return 0.5, 0.5

    # P(Y=0|X=x) : probabilité a posteriori de la classe 0
    posterior_0 = (likelihood_x_given_y0 * prior_0) / evidence
    # P(Y=1|X=x) : probabilité a posteriori de la classe 1
    posterior_1 = (likelihood_x_given_y1 * prior_1) / evidence

    return posterior_0, posterior_1

def optimal_bayes_classifier(X_test, params, loss_matrix):
    """
    Implémente le classifieur de Bayes optimal pour une matrice de perte donnée.
    La matrice de perte `loss_matrix[i][j]` représente le coût de prédire la classe `j`
    quand la vraie classe est `i`.
    Pour 2 classes (0, 1), la matrice est de la forme :
    [[L(0,0), L(0,1)],
     [L(1,0), L(1,1)]]
    """
    predictions = []
    for x in X_test:
        posterior_0, posterior_1 = calculate_posteriors(x, params)

        # Calcul du risque attendu pour chaque décision possible (prédire 0 ou 1)
        # Risque si on décide de prédire la classe 0 : R(a=0|x) = L(0,0)P(Y=0|X=x) + L(1,0)P(Y=1|X=x)
        risk_predict_0 = loss_matrix[0][0] * posterior_0 + loss_matrix[1][0] * posterior_1
        # Risque si on décide de prédire la classe 1 : R(a=1|x) = L(0,1)P(Y=0|X=x) + L(1,1)P(Y=1|X=x)
        risk_predict_1 = loss_matrix[0][1] * posterior_0 + loss_matrix[1][1] * posterior_1

        # La décision optimale est celle qui minimise le risque attendu
        if risk_predict_0 < risk_predict_1:
            predictions.append(0)
        else:
            # Si les risques sont égaux, on peut choisir arbitrairement, ici la classe 1
            predictions.append(1)

    return predictions

# --- 3. Fonctions de Perte (Loss Functions) ---

def zero_one_loss(y_true, y_pred):
    """
    Calcule la perte 0-1 (erreur de classification).
    Retourne 1 si y_true est différent de y_pred, 0 sinon.
    """
    return 1 if y_true != y_pred else 0

def total_zero_one_loss(y_true_list, y_pred_list):
    """
    Calcule la perte 0-1 totale (nombre d'erreurs) pour une liste de prédictions.
    """
    assert len(y_true_list) == len(y_pred_list), "Les listes de vérité terrain et de prédictions doivent avoir la même taille."
    errors = 0
    for i in range(len(y_true_list)):
        errors += zero_one_loss(y_true_list[i], y_pred_list[i])
    return errors

def squared_loss(y_true, y_pred_score):
    """
    Calcule la perte quadratique.
    Pour la classification binaire, y_true est 0 ou 1.
    y_pred_score est une valeur continue, souvent une probabilité (P(Y=1|X)) ou un score brut.
    """
    # Ici, nous supposons que y_pred_score est la probabilité prédite pour la classe 1.
    # La cible pour la classe 0 est 0, et pour la classe 1 est 1.
    return (y_true - y_pred_score)**2

def binary_cross_entropy_loss(y_true, y_pred_proba):
    """
    Calcule la perte d'entropie croisée binaire (logistique).
    y_true est la vraie étiquette (0 ou 1).
    y_pred_proba est la probabilité prédite pour la classe 1 (P(Y=1|X)).
    """
    # Assurer que les probabilités sont dans [epsilon, 1-epsilon] pour éviter log(0) ou log(1)
    # qui peuvent être infinis et causer des erreurs numériques.
    epsilon = 1e-10
    y_pred_proba = max(epsilon, min(1 - epsilon, y_pred_proba))

    if y_true == 1:
        return -math.log(y_pred_proba)
    else: # y_true == 0
        return -math.log(1 - y_pred_proba)

# --- 4. Démonstration et Assertions ---

if __name__ == "__main__":
    print("--- TP 4 : Classifieur de Bayes Optimal et Fonctions de Perte de Substitution ---")

    # Paramètres pour la génération de données synthétiques
    n_samples_per_class = 50
    mu1, sigma1 = 2, 0.8  # Paramètres pour la classe 0
    mu2, sigma2 = 5, 1.2  # Paramètres pour la classe 1

    # Génération des données d'entraînement et de test
    X_train, y_train = generate_synthetic_data(n_samples_per_class, mu1, sigma1, mu2, sigma2)
    X_test, y_test = generate_synthetic_data(n_samples_per_class // 2, mu1, sigma1, mu2, sigma2) # Moins de points pour le test

    print(f"\nDonnées générées: {len(X_train)} points d'entraînement, {len(X_test)} points de test.")
    print(f"Exemple X_train[:5]: {[f'{x:.2f}' for x in X_train[:5]]}, y_train[:5]: {y_train[:5]}")

    # --- Assertions de validation pour la génération de données et PDF ---
    assert len(X_train) == len(y_train), "Les listes de caractéristiques et d'étiquettes doivent avoir la même taille."
    assert all(isinstance(x, (int, float)) for x in X_train), "Toutes les caractéristiques doivent être numériques."
    assert all(y_val in [0, 1] for y_val in y_train), "Toutes les étiquettes doivent être 0 ou 1."
    assert gaussian_pdf(mu1, mu1, sigma1) > 0, "La PDF doit être positive au mode (moyenne)."
    assert gaussian_pdf(mu1 + 10*sigma1, mu1, sigma1) < gaussian_pdf(mu1, mu1, sigma1), "La PDF doit décroître loin de la moyenne."
    try:
        gaussian_pdf(0, 0, -1)
        assert False, "Sigma négatif devrait lever une erreur ValueError."
    except ValueError:
        assert True, "Sigma négatif lève bien une erreur ValueError."


    # Estimation des paramètres à partir des données d'entraînement
    params = estimate_parameters(X_train, y_train)
    print("\nParamètres estimés à partir des données d'entraînement:")
    for k, v in params.items():
        print(f"  {k}: {v:.4f}")

    # --- Assertions de validation pour les paramètres estimés ---
    assert 0 <= params['prior_0'] <= 1 and 0 <= params['prior_1'] <= 1, "Les probabilités a priori doivent être entre 0 et 1."
    assert abs(params['prior_0'] + params['prior_1'] - 1.0) < 1e-9, "La somme des probabilités a priori doit être 1."
    assert params['sigma_0'] > 0 and params['sigma_1'] > 0, "Les écart-types estimés doivent être positifs."

    # --- Classifieur de Bayes Optimal avec Perte 0-1 ---
    print("\n--- Classifieur de Bayes Optimal (Minimisation de la Perte 0-1) ---")
    # Matrice de perte 0-1: L(i,j) = 0 si i=j (pas de coût pour une bonne prédiction), 1 si i!=j (coût de 1 pour une erreur)
    loss_matrix_01 = [[0, 1], [1, 0]]
    predictions_01 = optimal_bayes_classifier(X_test, params, loss_matrix_01)

    errors_01 = total_zero_one_loss(y_test, predictions_01)
    accuracy_01 = (len(y_test) - errors_01) / len(y_test)
    print(f"Prédictions (0-1 loss) pour les 5 premiers points de test: {predictions_01[:5]}")
    print(f"Vraies étiquettes pour les 5 premiers points de test: {y_test[:5]}")
    print(f"Nombre d'erreurs (0-1 loss): {errors_01} sur {len(y_test)} points.")
    print(f"Précision (Accuracy) (0-1 loss): {accuracy_01:.4f}")

    # --- Assertions de validation pour le classifieur 0-1 ---
    assert all(p in [0, 1] for p in predictions_01), "Les prédictions du classifieur doivent être 0 ou 1."
    assert errors_01 >= 0, "Le nombre d'erreurs ne peut pas être négatif."
    assert 0 <= accuracy_01 <= 1, "La précision doit être entre 0 et 1."

    # --- Classifieur de Bayes Optimal avec Coûts Asymétriques ---
    print("\n--- Classifieur de Bayes Optimal (Minimisation d'un Coût Asymétrique) ---")
    # Supposons qu'une erreur de type I (prédire 1 quand la vraie classe est 0) est 5 fois plus coûteuse
    # qu'une erreur de type II (prédire 0 quand la vraie classe est 1).
    # L(0,1) = 5 (coût de prédire 1 quand c'est 0)
    # L(1,0) = 1 (coût de prédire 0 quand c'est 1)
    loss_matrix_asymmetric = [[0, 5], [1, 0]]
    predictions_asymmetric = optimal_bayes_classifier(X_test, params, loss_matrix_asymmetric)

    errors_asymmetric = total_zero_one_loss(y_test, predictions_asymmetric)
    # Calcul du coût total selon la matrice de perte asymétrique
    total_cost_asymmetric = sum(loss_matrix_asymmetric[y_test[i]][predictions_asymmetric[i]] for i in range(len(y_test)))
    print(f"Prédictions (asymmetric loss) pour les 5 premiers points de test: {predictions_asymmetric[:5]}")
    print(f"Nombre d'erreurs (0-1 loss) avec classifieur asymétrique: {errors_asymmetric} sur {len(y_test)} points.")
    print(f"Coût total (asymmetric loss): {total_cost_asymmetric:.2f}")

    # --- Assertions de validation pour le classifieur asymétrique ---
    assert all(p in [0, 1] for p in predictions_asymmetric), "Les prédictions du classifieur asymétrique doivent être 0 ou 1."
    assert total_cost_asymmetric >= 0, "Le coût total ne peut pas être négatif."

    # --- Fonctions de Perte de Substitution (Surrogate Loss Functions) ---
    print("\n--- Fonctions de Perte de Substitution ---")

    # Pour illustrer les pertes de substitution, nous utilisons les probabilités a posteriori P(Y=1|X)
    # comme score de prédiction pour chaque point de test.
    test_posteriors_1 = []
    for x_val in X_test:
        _, p_1 = calculate_posteriors(x_val, params)
        test_posteriors_1.append(p_1)

    print(f"Exemple de P(Y=1|X) pour les 5 premiers points de test: {[f'{p:.4f}' for p in test_posteriors_1[:5]]}")

    # Calcul de la perte quadratique moyenne (Mean Squared Error - MSE)
    total_squared_loss = 0
    for i in range(len(y_test)):
        total_squared_loss += squared_loss(y_test[i], test_posteriors_1[i])
    mean_squared_loss = total_squared_loss / len(y_test)
    print(f"Perte Quadratique Moyenne (MSE) sur les probabilités: {mean_squared_loss:.4f}")

    # Calcul de la perte d'entropie croisée binaire moyenne (Binary Cross-Entropy - BCE)
    total_bce_loss = 0
    for i in range(len(y_test)):
        total_bce_loss += binary_cross_entropy_loss(y_test[i], test_posteriors_1[i])
    mean_bce_loss = total_bce_loss / len(y_test)
    print(f"Perte d'Entropie Croisée Binaire Moyenne (BCE): {mean_bce_loss:.4f}")

    # --- Assertions de validation pour les pertes de substitution ---
    assert mean_squared_loss >= 0, "La perte quadratique doit être non-négative."
    assert mean_bce_loss >= 0, "La perte d'entropie croisée doit être non-négative."
    assert all(0 <= p <= 1 for p in test_posteriors_1), "Les probabilités a posteriori doivent être entre 0 et 1."

    # --- Minimisation du Risque Empirique (ERM) (Conceptuel) ---
    print("\n--- Minimisation du Risque Empirique (ERM) (Conceptuel) ---")
    print("Le classifieur de Bayes optimal minimise le risque attendu (Expected Risk).")
    print("En pratique, avec des données d'entraînement finies, on ne peut pas calculer le risque attendu directement.")
    print("On minimise alors le risque empirique (Empirical Risk), qui est la moyenne des pertes sur les données d'entraînement.")
    print("Les fonctions de perte de substitution (comme la perte quadratique ou l'entropie croisée) sont cruciales pour l'ERM.")
    print("Elles sont préférées à la perte 0-1 car elles sont différentiables et plus faciles à optimiser avec des algorithmes comme la descente de gradient.")
    print("Minimiser une perte de substitution 'convexe' sur les données d'entraînement conduit souvent à un modèle qui se rapproche du classifieur de Bayes optimal (qui minimise la perte 0-1).")
    print(f"Le risque empirique 0-1 pour le classifieur de Bayes optimal est: {errors_01 / len(y_test):.4f}")
    print(f"Le risque empirique quadratique pour les probabilités du classifieur de Bayes est: {mean_squared_loss:.4f}")
    print(f"Le risque empirique d'entropie croisée pour les probabilités du classifieur de Bayes est: {mean_bce_loss:.4f}")

    # Vérification des probabilités a posteriori
    p0_sample, p1_sample = calculate_posteriors(X_test[0], params)
    assert abs(p0_sample + p1_sample - 1.0) < 1e-9, "La somme des probabilités a posteriori doit être 1."
    assert 0 <= p0_sample <= 1 and 0 <= p1_sample <= 1, "Les probabilités a posteriori doivent être entre 0 et 1."

    print("\nTP 4 terminé avec succès et toutes les assertions validées.")
```
