# TP 3 : Classifieur de Bayes Optimal et Fonctions de Perte de Substitution

## Objectif
Ce TP a pour objectif d'implémenter un classifieur de Bayes optimal à partir de zéro, en se concentrant sur des données discrètes pour simplifier les calculs de probabilités sans l'aide de bibliothèques externes. Nous explorerons la règle de décision du classifieur de Bayes qui minimise la perte 0-1. Ensuite, nous introduirons le concept de fonctions de perte de substitution (surrogate loss functions) comme alternatives continues et différentiables à la perte 0-1, et discuterons de leur rôle dans l'apprentissage par minimisation du risque empirique (ERM), même si l'optimisation complète n'est pas l'objet direct de ce TP.

Nous allons :
1.  Générer un petit jeu de données avec des caractéristiques discrètes.
2.  Implémenter les fonctions pour calculer les probabilités a priori et les vraisemblances conditionnelles.
3.  Construire un classifieur de Bayes pour prédire la classe d'un échantillon.
4.  Calculer la perte 0-1 pour évaluer le classifieur.
5.  Définir et illustrer des fonctions de perte de substitution simples.

## Code Python
```python
# --- Partie 1 : Génération de données discrètes ---

# Un petit jeu de données avec des caractéristiques discrètes et une étiquette de classe binaire (0 ou 1).
# Chaque entrée est un dictionnaire représentant un échantillon.
data = [
    {'couleur': 'rouge', 'taille': 'petit', 'classe': 0},
    {'couleur': 'bleu', 'taille': 'moyen', 'classe': 1},
    {'couleur': 'rouge', 'taille': 'grand', 'classe': 0},
    {'couleur': 'vert', 'taille': 'petit', 'classe': 1},
    {'couleur': 'bleu', 'taille': 'petit', 'classe': 1},
    {'couleur': 'rouge', 'taille': 'moyen', 'classe': 0},
    {'couleur': 'vert', 'taille': 'grand', 'classe': 1},
    {'couleur': 'bleu', 'taille': 'grand', 'classe': 1},
    {'couleur': 'rouge', 'taille': 'petit', 'classe': 0},
    {'couleur': 'vert', 'taille': 'moyen', 'classe': 1},
    {'couleur': 'rouge', 'taille': 'grand', 'classe': 0},
    {'couleur': 'bleu', 'taille': 'moyen', 'classe': 1},
]

# Séparer les caractéristiques et les étiquettes pour faciliter le traitement
features = [d.copy() for d in data]
for f in features:
    del f['classe']
labels = [d['classe'] for d in data]

print("--- Données générées ---")
for i, sample in enumerate(data):
    print(f"Échantillon {i+1}: {sample}")
print("-" * 30)

# --- Partie 2 : Calcul des probabilités a priori P(Y) ---

def calculate_priors(labels_list):
    """
    Calcule les probabilités a priori P(Y=y) pour chaque classe.
    """
    total_samples = len(labels_list)
    class_counts = {}
    for label in labels_list:
        class_counts[label] = class_counts.get(label, 0) + 1

    priors = {
        class_label: count / total_samples
        for class_label, count in class_counts.items()
    }
    return priors

priors = calculate_priors(labels)
print("\n--- Probabilités a priori P(Y) ---")
for class_label, prob in priors.items():
    print(f"P(Y={class_label}) = {prob:.2f}")

# Assertions pour les probabilités a priori
assert abs(sum(priors.values()) - 1.0) < 1e-9, "La somme des probabilités a priori doit être égale à 1."
assert priors[0] == 6/12, "P(Y=0) incorrect."
assert priors[1] == 6/12, "P(Y=1) incorrect."
print("Assertions P(Y) passées avec succès.")
print("-" * 30)

# --- Partie 3 : Calcul des vraisemblances conditionnelles P(X|Y) ---

def calculate_likelihoods(data_list):
    """
    Calcule les vraisemblances conditionnelles P(X_i=x_i|Y=y) pour chaque caractéristique et valeur.
    Retourne un dictionnaire imbriqué: likelihoods[class_label][feature_name][feature_value]
    """
    likelihoods = {}
    class_counts = {}

    # Initialiser les compteurs
    for sample in data_list:
        class_label = sample['classe']
        class_counts[class_label] = class_counts.get(class_label, 0) + 1
        if class_label not in likelihoods:
            likelihoods[class_label] = {}

        for feature_name, feature_value in sample.items():
            if feature_name == 'classe':
                continue
            if feature_name not in likelihoods[class_label]:
                likelihoods[class_label][feature_name] = {}
            likelihoods[class_label][feature_name][feature_value] = \
                likelihoods[class_label][feature_name].get(feature_value, 0) + 1

    # Convertir les comptes en probabilités
    for class_label, feature_data in likelihoods.items():
        for feature_name, value_counts in feature_data.items():
            for feature_value, count in value_counts.items():
                likelihoods[class_label][feature_name][feature_value] = \
                    count / class_counts[class_label]
    return likelihoods

likelihoods = calculate_likelihoods(data)
print("\n--- Vraisemblances conditionnelles P(X|Y) ---")
for class_label, feature_data in likelihoods.items():
    print(f"Pour la classe Y={class_label}:")
    for feature_name, value_probs in feature_data.items():
        print(f"  {feature_name}: {value_probs}")

# Assertions pour les vraisemblances conditionnelles
# Exemple: P(couleur='rouge'|Y=0)
# Classe 0: 6 échantillons. 'rouge': 4 fois. P(couleur='rouge'|Y=0) = 4/6 = 0.666...
assert abs(likelihoods[0]['couleur']['rouge'] - (4/6)) < 1e-9, "P(couleur='rouge'|Y=0) incorrect."
# Classe 1: 6 échantillons. 'bleu': 3 fois. P(couleur='bleu'|Y=1) = 3/6 = 0.5
assert abs(likelihoods[1]['couleur']['bleu'] - (3/6)) < 1e-9, "P(couleur='bleu'|Y=1) incorrect."
# Classe 0: 6 échantillons. 'petit': 3 fois. P(taille='petit'|Y=0) = 3/6 = 0.5
assert abs(likelihoods[0]['taille']['petit'] - (3/6)) < 1e-9, "P(taille='petit'|Y=0) incorrect."
print("Assertions P(X|Y) passées avec succès.")
print("-" * 30)

# --- Partie 4 : Classifieur de Bayes Optimal ---

def bayes_predict(sample_features, priors, likelihoods):
    """
    Prédit la classe d'un échantillon en utilisant le classifieur de Bayes.
    Assumons l'indépendance conditionnelle des caractéristiques (Naïve Bayes).
    """
    scores = {}
    for class_label, prior in priors.items():
        # P(Y=y)
        score = prior
        # Multiplier par P(X_i=x_i|Y=y) pour chaque caractéristique
        for feature_name, feature_value in sample_features.items():
            if feature_name in likelihoods[class_label] and \
               feature_value in likelihoods[class_label][feature_name]:
                score *= likelihoods[class_label][feature_name][feature_value]
            else:
                # Gérer les cas où une caractéristique/valeur n'a pas été vue pour cette classe
                # Une approche simple est d'assigner une probabilité de 0.
                # Pour la démonstration, nous nous assurerons que les échantillons de test sont dans les données d'entraînement.
                score *= 0 # Si une caractéristique n'est pas vue, la probabilité est 0.
        scores[class_label] = score

    # Trouver la classe avec le score le plus élevé (argmax P(Y=y|X=x))
    # P(X=x) est un facteur de normalisation constant pour toutes les classes, donc on peut l'ignorer.
    predicted_class = None
    max_score = -1
    for class_label, score in scores.items():
        if score > max_score:
            max_score = score
            predicted_class = class_label
        # Gérer les égalités (arbitrairement choisir la classe avec la plus petite étiquette)
        elif score == max_score and predicted_class is not None:
            if class_label < predicted_class:
                predicted_class = class_label

    return predicted_class, scores

# Tester le classifieur de Bayes
test_sample_1 = {'couleur': 'rouge', 'taille': 'petit'}
predicted_class_1, scores_1 = bayes_predict(test_sample_1, priors, likelihoods)
print(f"\n--- Test du Classifieur de Bayes ---")
print(f"Échantillon de test 1: {test_sample_1}")
print(f"Scores par classe: {scores_1}")
print(f"Classe prédite: {predicted_class_1}") # Devrait être 0

test_sample_2 = {'couleur': 'vert', 'taille': 'grand'}
predicted_class_2, scores_2 = bayes_predict(test_sample_2, priors, likelihoods)
print(f"Échantillon de test 2: {test_sample_2}")
print(f"Scores par classe: {scores_2}")
print(f"Classe prédite: {predicted_class_2}") # Devrait être 1

# Assertions pour les prédictions
# Pour test_sample_1: {'couleur': 'rouge', 'taille': 'petit'}
# P(Y=0|X) ~ P(Y=0) * P(rouge|Y=0) * P(petit|Y=0) = (6/12) * (4/6) * (3/6) = 0.5 * 0.666... * 0.5 = 0.166...
# P(Y=1|X) ~ P(Y=1) * P(rouge|Y=1) * P(petit|Y=1) = (6/12) * (0/6) * (2/6) = 0.5 * 0 * 0.333... = 0
assert predicted_class_1 == 0, "Prédiction incorrecte pour l'échantillon 1."
assert abs(scores_1[0] - (0.5 * (4/6) * (3/6))) < 1e-9, "Score pour classe 0 de l'échantillon 1 incorrect."
assert abs(scores_1[1] - (0.5 * (0/6) * (2/6))) < 1e-9, "Score pour classe 1 de l'échantillon 1 incorrect."

# Pour test_sample_2: {'couleur': 'vert', 'taille': 'grand'}
# P(Y=0|X) ~ P(Y=0) * P(vert|Y=0) * P(grand|Y=0) = (6/12) * (0/6) * (2/6) = 0
# P(Y=1|X) ~ P(Y=1) * P(vert|Y=1) * P(grand|Y=1) = (6/12) * (2/6) * (2/6) = 0.5 * 0.333... * 0.333... = 0.055...
assert predicted_class_2 == 1, "Prédiction incorrecte pour l'échantillon 2."
assert abs(scores_2[0] - (0.5 * (0/6) * (2/6))) < 1e-9, "Score pour classe 0 de l'échantillon 2 incorrect."
assert abs(scores_2[1] - (0.5 * (2/6) * (2/6))) < 1e-9, "Score pour classe 1 de l'échantillon 2 incorrect."
print("Assertions de prédiction passées avec succès.")
print("-" * 30)

# --- Partie 5 : Perte 0-1 (Zero-One Loss) ---

def zero_one_loss(true_labels, predicted_labels):
    """
    Calcule la perte 0-1.
    Retourne le nombre d'erreurs de classification.
    """
    errors = 0
    for i in range(len(true_labels)):
        if true_labels[i] != predicted_labels[i]:
            errors += 1
    return errors

# Évaluer le classifieur sur les données d'entraînement (pour démonstration)
predicted_labels_train = []
for sample_data in data:
    sample_features = sample_data.copy()
    del sample_features['classe']
    pred, _ = bayes_predict(sample_features, priors, likelihoods)
    predicted_labels_train.append(pred)

total_errors = zero_one_loss(labels, predicted_labels_train)
accuracy = (len(labels) - total_errors) / len(labels)

print(f"\n--- Évaluation avec Perte 0-1 ---")
print(f"Vraies étiquettes: {labels}")
print(f"Étiquettes prédites: {predicted_labels_train}")
print(f"Nombre total d'erreurs (perte 0-1): {total_errors}")
print(f"Précision (Accuracy): {accuracy:.2f}")

# Assertions pour la perte 0-1
# Pour ce petit dataset, toutes les prédictions du classifieur de Bayes sont correctes.
assert total_errors == 0, "Le nombre d'erreurs 0-1 est incorrect."
assert accuracy == 1.0, "La précision est incorrecte."
print("Assertions de perte 0-1 passées avec succès.")
print("-" * 30)

# --- Partie 6 : Fonctions de Perte de Substitution (Surrogate Loss Functions) ---

# La perte 0-1 est non-différentiable et difficile à optimiser directement.
# Les fonctions de perte de substitution sont des approximations continues et souvent différentiables.
# Elles sont utilisées dans l'apprentissage par Minimisation du Risque Empirique (ERM).

# Pour illustrer, considérons un classifieur qui produit un "score de confiance" s(x) pour la classe 1.
# Si s(x) > 0.5, on prédit 1. Si s(x) <= 0.5, on prédit 0.
# Les vraies étiquettes sont y (0 ou 1).

def zero_one_loss_score_based(y_true, s_score, threshold=0.5):
    """
    Perte 0-1 basée sur un score de confiance.
    """
    y_pred = 1 if s_score > threshold else 0
    return 1 if y_true != y_pred else 0

def squared_loss_surrogate(y_true, s_score):
    """
    Fonction de perte de substitution : Perte quadratique (Squared Loss).
    Souvent utilisée pour la régression, mais peut être adaptée pour la classification
    en traitant les étiquettes comme des valeurs numériques (0 ou 1) et le score comme une probabilité.
    """
    return (y_true - s_score) ** 2

def hinge_like_loss_surrogate(y_true, s_score):
    """
    Fonction de perte de substitution : Perte de type Hinge (simplifiée).
    La perte Hinge classique est souvent utilisée avec des étiquettes y_true de -1 ou 1
    et un score f(x) non borné, sous la forme max(0, 1 - y_true * f(x)).
    Ici, nous adaptons pour y_true de 0 ou 1, et s_score est un score entre 0 et 1 (comme une probabilité).
    Nous voulons que s_score soit proche de 1 pour y_true=1 et proche de 0 pour y_true=0.
    """
    if y_true == 1:
        # Si la vraie classe est 1, on veut s_score proche de 1. Pénalité si s_score est faible.
        return max(0, 1 - s_score)
    else: # y_true == 0
        # Si la vraie classe est 0, on veut s_score proche de 0. Pénalité si s_score est élevé.
        return max(0, s_score)

print(f"\n--- Fonctions de Perte de Substitution ---")

# Exemples de scores et de vraies étiquettes
true_label_ex = 1
score_ex_correct = 0.9
score_ex_incorrect = 0.2
score_ex_borderline = 0.5

print(f"\nExemple pour y_true = {true_label_ex}:")
print(f"  Score correct (0.9):")
print(f"    Perte 0-1: {zero_one_loss_score_based(true_label_ex, score_ex_correct)}")
print(f"    Perte Quadratique: {squared_loss_surrogate(true_label_ex, score_ex_correct):.2f}")
print(f"    Perte Hinge-like: {hinge_like_loss_surrogate(true_label_ex, score_ex_correct):.2f}")

print(f"  Score incorrect (0.2):")
print(f"    Perte 0-1: {zero_one_loss_score_based(true_label_ex, score_ex_incorrect)}")
print(f"    Perte Quadratique: {squared_loss_surrogate(true_label_ex, score_ex_incorrect):.2f}")
print(f"    Perte Hinge-like: {hinge_like_loss_surrogate(true_label_ex, score_ex_incorrect):.2f}")

print(f"  Score limite (0.5):")
print(f"    Perte 0-1: {zero_one_loss_score_based(true_label_ex, score_ex_borderline)}")
print(f"    Perte Quadratique: {squared_loss_surrogate(true_label_ex, score_ex_borderline):.2f}")
print(f"    Perte Hinge-like: {hinge_like_loss_surrogate(true_label_ex, score_ex_borderline):.2f}")

true_label_ex_0 = 0
score_ex_correct_0 = 0.1
score_ex_incorrect_0 = 0.8

print(f"\nExemple pour y_true = {true_label_ex_0}:")
print(f"  Score correct (0.1):")
print(f"    Perte 0-1: {zero_one_loss_score_based(true_label_ex_0, score_ex_correct_0)}")
print(f"    Perte Quadratique: {squared_loss_surrogate(true_label_ex_0, score_ex_correct_0):.2f}")
print(f"    Perte Hinge-like: {hinge_like_loss_surrogate(true_label_ex_0, score_ex_correct_0):.2f}")

print(f"  Score incorrect (0.8):")
print(f"    Perte 0-1: {zero_one_loss_score_based(true_label_ex_0, score_ex_incorrect_0)}")
print(f"    Perte Quadratique: {squared_loss_surrogate(true_label_ex_0, score_ex_incorrect_0):.2f}")
print(f"    Perte Hinge-like: {hinge_like_loss_surrogate(true_label_ex_0, score_ex_incorrect_0):.2f}")


# Assertions pour les fonctions de perte de substitution
# Perte 0-1
assert zero_one_loss_score_based(1, 0.9) == 0, "0-1 loss incorrect for correct score."
assert zero_one_loss_score_based(1, 0.2) == 1, "0-1 loss incorrect for incorrect score."
assert zero_one_loss_score_based(0, 0.1) == 0, "0-1 loss incorrect for correct score (class 0)."
assert zero_one_loss_score_based(0, 0.8) == 1, "0-1 loss incorrect for incorrect score (class 0)."

# Perte Quadratique
assert abs(squared_loss_surrogate(1, 0.9) - 0.01) < 1e-9, "Squared loss incorrect."
assert abs(squared_loss_surrogate(1, 0.2) - 0.64) < 1e-9, "Squared loss incorrect."
assert abs(squared_loss_surrogate(0, 0.1) - 0.01) < 1e-9, "Squared loss incorrect (class 0)."
assert abs(squared_loss_surrogate(0, 0.8) - 0.64) < 1e-9, "Squared loss incorrect (class 0)."

# Perte Hinge-like
assert abs(hinge_like_loss_surrogate(1, 0.9) - 0.1) < 1e-9, "Hinge-like loss incorrect."
assert abs(hinge_like_loss_surrogate(1, 0.2) - 0.8) < 1e-9, "Hinge-like loss incorrect."
assert abs(hinge_like_loss_surrogate(0, 0.1) - 0.1) < 1e-9, "Hinge-like loss incorrect (class 0)."
assert abs(hinge_like_loss_surrogate(0, 0.8) - 0.8) < 1e-9, "Hinge-like loss incorrect (class 0)."
print("Assertions des fonctions de perte de substitution passées avec succès.")
```
