# TP 1 : Classifieur de Bayes optimal

## Objectif
Ce TP a pour objectif d'implémenter de zéro le concept fondamental du classifieur de Bayes optimal pour un problème de classification binaire. Nous allons définir un modèle génératif simple avec des distributions gaussiennes pour les classes, calculer les probabilités a posteriori en utilisant le théorème de Bayes, appliquer la règle de décision de Bayes (minimisant le risque pour la fonction de perte 0-1), et estimer l'erreur de Bayes par simulation.

## Code Python
```python
# Importations nécessaires (standard library uniquement)
import math
import random

# --- 1. Définition du Modèle Génératif ---
# Nous considérons un problème de classification binaire avec deux classes, C0 et C1,
# et une seule caractéristique X.
# Les distributions conditionnelles P(X|C) sont gaussiennes.

# Paramètres des priors (probabilités a priori des classes)
PRIOR_C0 = 0.5
PRIOR_C1 = 0.5
assert abs(PRIOR_C0 + PRIOR_C1 - 1.0) < 1e-9, "Les probabilités a priori doivent sommer à 1."
assert 0 <= PRIOR_C0 <= 1 and 0 <= PRIOR_C1 <= 1, "Les probabilités a priori doivent être entre 0 et 1."

# Paramètres des distributions gaussiennes conditionnelles
# Classe C0: N(mu_C0, sigma_C0^2)
MU_C0 = 0.0
SIGMA_C0 = 1.0
assert SIGMA_C0 > 0, "L'écart-type de C0 doit être positif."

# Classe C1: N(mu_C1, sigma_C1^2)
MU_C1 = 2.0
SIGMA_C1 = 1.0
assert SIGMA_C1 > 0, "L'écart-type de C1 doit être positif."

# --- 2. Implémentation de la Fonction de Densité de Probabilité (PDF) Gaussienne ---
def gaussian_pdf(x: float, mu: float, sigma: float) -> float:
    """
    Calcule la valeur de la fonction de densité de probabilité (PDF)
    pour une distribution gaussienne (normale).
    """
    assert sigma > 0, "L'écart-type (sigma) doit être positif."
    coefficient = 1 / (sigma * math.sqrt(2 * math.pi))
    exponent = -0.5 * ((x - mu) / sigma)**2
    return coefficient * math.exp(exponent)

# Assertions de validation mathématique pour la PDF Gaussienne
# Test à la moyenne (point le plus haut)
assert abs(gaussian_pdf(0, 0, 1) - (1 / math.sqrt(2 * math.pi))) < 1e-9, "PDF à la moyenne incorrecte."
# Test avec des valeurs connues
assert gaussian_pdf(0, 0, 1) > 0, "PDF doit être positive."
assert gaussian_pdf(1, 0, 1) < gaussian_pdf(0, 0, 1), "PDF doit décroître en s'éloignant de la moyenne."
try:
    gaussian_pdf(0, 0, 0)
    assert False, "Devrait lever une AssertionError pour sigma=0."
except AssertionError:
    pass # Attendu

# --- 3. Calcul des Probabilités a Posteriori P(C|X) ---
def calculate_posterior_probabilities(x: float) -> tuple[float, float]:
    """
    Calcule les probabilités a posteriori P(C0|X) et P(C1|X)
    pour une observation X donnée, en utilisant le théorème de Bayes.
    """
    # P(X|C0) et P(X|C1)
    p_x_given_c0 = gaussian_pdf(x, MU_C0, SIGMA_C0)
    p_x_given_c1 = gaussian_pdf(x, MU_C1, SIGMA_C1)

    # P(X) = P(X|C0)P(C0) + P(X|C1)P(C1) (probabilité marginale de X)
    p_x = p_x_given_c0 * PRIOR_C0 + p_x_given_c1 * PRIOR_C1
    assert p_x >= 0, "La probabilité marginale P(X) ne peut pas être négative."

    # Gérer le cas où P(X) est quasi-nulle pour éviter une division par zéro.
    # Si X est un événement extrêmement rare, les probabilités a posteriori sont indéfinies.
    # Pour ce TP, nous retournons 0.0 pour les deux, ce qui est une simplification.
    if p_x < 1e-12:
        return 0.0, 0.0

    # P(C0|X) = P(X|C0)P(C0) / P(X)
    p_c0_given_x = (p_x_given_c0 * PRIOR_C0) / p_x
    # P(C1|X) = P(X|C1)P(C1) / P(X)
    p_c1_given_x = (p_x_given_c1 * PRIOR_C1) / p_x

    # Assertions de validation mathématique pour les probabilités a posteriori
    assert 0 <= p_c0_given_x <= 1, f"P(C0|X) doit être entre 0 et 1, mais est {p_c0_given_x}"
    assert 0 <= p_c1_given_x <= 1, f"P(C1|X) doit être entre 0 et 1, mais est {p_c1_given_x}"
    assert abs(p_c0_given_x + p_c1_given_x - 1.0) < 1e-9, "La somme des probabilités a posteriori doit être 1."

    return p_c0_given_x, p_c1_given_x

# Assertions de validation mathématique pour les probabilités a posteriori
# Test avec X = MU_C0 (devrait favoriser C0 si priors et sigmas sont égaux)
p_c0_at_mu0, p_c1_at_mu0 = calculate_posterior_probabilities(MU_C0)
assert p_c0_at_mu0 > p_c1_at_mu0, f"P(C0|X={MU_C0}) devrait être > P(C1|X={MU_C0})"
assert abs(p_c0_at_mu0 + p_c1_at_mu0 - 1.0) < 1e-9, "Somme des postérioris incorrecte."

# Test avec X = MU_C1 (devrait favoriser C1)
p_c0_at_mu1, p_c1_at_mu1 = calculate_posterior_probabilities(MU_C1)
assert p_c1_at_mu1 > p_c0_at_mu1, f"P(C1|X={MU_C1}) devrait être > P(C0|X={MU_C1})"

# Test avec X au point d'intersection (si priors et sigmas égaux, c'est au milieu)
# Pour MU_C0=0, MU_C1=2, SIGMA_C0=SIGMA_C1=1, PRIOR_C0=PRIOR_C1=0.5, l'intersection est à X=1
p_c0_at_mid, p_c1_at_mid = calculate_posterior_probabilities(1.0)
assert abs(p_c0_at_mid - p_c1_at_mid) < 1e-9, "Au point d'intersection, les postérioris devraient être égales."
assert abs(p_c0_at_mid - 0.5) < 1e-9, "Au point d'intersection avec priors égaux, les postérioris devraient être 0.5."

# --- 4. Règle de Décision de Bayes (pour la fonction de perte 0-1) ---
def bayes_classifier(x: float) -> int:
    """
    Applique la règle de décision de Bayes pour la fonction de perte 0-1.
    Classe X dans la classe ayant la probabilité a posteriori la plus élevée.
    Retourne 0 pour C0, 1 pour C1.
    """
    p_c0_given_x, p_c1_given_x = calculate_posterior_probabilities(x)

    if p_c0_given_x > p_c1_given_x:
        return 0  # Classer dans C0
    else:
        return 1  # Classer dans C1 (inclut le cas d'égalité, arbitrairement à C1)

# Assertions de validation mathématique pour le classifieur de Bayes
assert bayes_classifier(MU_C0) == 0, f"X={MU_C0} devrait être classé en C0."
assert bayes_classifier(MU_C1) == 1, f"X={MU_C1} devrait être classé en C1."
assert bayes_classifier(1.0) == 1, f"X=1.0 devrait être classé en C1 (égalité arbitrairement à C1)."
assert bayes_classifier(-1.0) == 0, f"X=-1.0 devrait être classé en C0."
assert bayes_classifier(3.0) == 1, f"X=3.0 devrait être classé en C1."

# --- 5. Estimation de l'Erreur de Bayes par Simulation ---
# L'erreur de Bayes est l'erreur minimale théoriquement atteignable pour la fonction de perte 0-1.
# Nous l'estimons en générant un grand nombre d'échantillons selon le modèle génératif
# et en comptant les erreurs de classification du classifieur de Bayes.

def generate_samples(num_samples: int) -> list[tuple[float, int]]:
    """
    Génère des échantillons (X, label_vrai) selon le modèle génératif défini.
    """
    samples = []
    for _ in range(num_samples):
        # Choisir la classe selon les priors
        if random.random() < PRIOR_C0:
            true_label = 0 # C0
            x = random.gauss(MU_C0, SIGMA_C0)
        else:
            true_label = 1 # C1
            x = random.gauss(MU_C1, SIGMA_C1)
        samples.append((x, true_label))
    return samples

def estimate_bayes_error(samples: list[tuple[float, int]]) -> float:
    """
    Estime l'erreur de Bayes en classifiant les échantillons générés
    et en comparant avec leurs vraies étiquettes.
    """
    misclassifications = 0
    for x, true_label in samples:
        predicted_label = bayes_classifier(x)
        if predicted_label != true_label:
            misclassifications += 1

    error_rate = misclassifications / len(samples)
    return error_rate

# Exécution de la simulation
NUM_SIMULATION_SAMPLES = 100000
print(f"\n--- Simulation pour estimer l'erreur de Bayes avec {NUM_SIMULATION_SAMPLES} échantillons ---")
simulation_samples = generate_samples(NUM_SIMULATION_SAMPLES)
estimated_error = estimate_bayes_error(simulation_samples)

print(f"Erreur de Bayes estimée : {estimated_error:.4f}")

# Assertions de validation mathématique pour l'erreur de Bayes estimée
assert 0 <= estimated_error <= 1, "L'erreur de Bayes doit être entre 0 et 1."
# Pour ce problème spécifique (Gaussiennes qui se chevauchent), l'erreur ne devrait pas être 0 ni 1.
# Elle devrait être inférieure à 0.5 (sinon, un classifieur aléatoire ferait mieux).
assert estimated_error > 0, "L'erreur de Bayes ne devrait pas être nulle si les distributions se chevauchent."
assert estimated_error < 0.5, "L'erreur de Bayes devrait être inférieure à 0.5 pour un problème non trivial."

# --- 6. Démonstration et Visualisation (simplifiée) ---
print("\n--- Démonstration du classifieur pour quelques points X ---")
test_points = [-1.5, 0.0, 0.5, 1.0, 1.5, 2.0, 3.0]

for x_val in test_points:
    p_c0, p_c1 = calculate_posterior_probabilities(x_val)
    prediction = bayes_classifier(x_val)
    print(f"X = {x_val:.2f}: P(C0|X) = {p_c0:.4f}, P(C1|X) = {p_c1:.4f} -> Prédiction: C{prediction}")

print("\nLe classifieur de Bayes optimal minimise le risque pour la fonction de perte 0-1.")
print("L'erreur de Bayes estimée représente la limite inférieure d'erreur pour ce problème.")
```
