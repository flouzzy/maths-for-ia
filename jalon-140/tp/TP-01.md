# TP 1 - Estimation non-paramétrique et Classifieur de Bayes

## Introduction Mathématique
Implémentation d'un estimateur à noyau pour approcher la règle de Bayes.

## Implémentation (Python Pur)
```python
import numpy as np

def generate_data(n_samples=200):
    X = np.random.uniform(-3, 3, n_samples)
    # P(Y=1 | x) = sigmoid(sin(x) * x)
    eta = 1 / (1 + np.exp(-np.sin(X) * X * 2))
    Y = np.where(np.random.rand(n_samples) < eta, 1, -1)
    return X, Y, eta

def nadaraya_watson(x_query, X, Y, h=0.5):
    # Kernel Gaussien
    K = np.exp(-0.5 * ((x_query[:, None] - X[None, :]) / h)**2)
    # Regression locale pour estimer P(Y=1|x) - P(Y=-1|x)
    weights = K / (np.sum(K, axis=1, keepdims=True) + 1e-8)
    return np.sum(weights * Y, axis=1)

X_train, Y_train, _ = generate_data(500)
X_test = np.linspace(-3, 3, 200)

# Estimate f(x)
f_hat = nadaraya_watson(X_test, X_train, Y_train, h=0.3)
predictions = np.sign(f_hat)

# Vrai Bayes
eta_test = 1 / (1 + np.exp(-np.sin(X_test) * X_test * 2))
bayes_predictions = np.where(eta_test > 0.5, 1, -1)

# Assertion mathématique: La consistance veut que l'erreur converge vers l'erreur de Bayes.
# On vérifie si notre estimateur converge vers le classifieur de Bayes
accuracy_vs_bayes = np.mean(predictions == bayes_predictions)
assert accuracy_vs_bayes > 0.85, "L'estimateur non-paramétrique ne s'approche pas suffisamment de Bayes."
print(f"Précision par rapport au classifieur de Bayes optimal : {accuracy_vs_bayes*100:.2f}%")

```
