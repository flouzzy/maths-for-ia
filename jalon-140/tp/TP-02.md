# TP 2 : Classifieur de Bayes Optimal, Fonctions de Perte et Minimisation du Risque Empirique (ERM)

## Objectif
Ce TP vise à implémenter de zéro le classifieur de Bayes optimal pour un problème de classification binaire, en se basant sur les probabilités a priori et les vraisemblances. Nous explorerons également différentes fonctions de perte (0-1 loss et squared loss comme fonction de perte de substitution) et démontrerons le principe de la Minimisation du Risque Empirique (ERM) en comparant les performances du classifieur de Bayes avec un classifieur simple sur un jeu de données synthétique.

## Code Python
```python
import random

# --- 1. Définition des distributions sous-jacentes (pour un X discret) ---
# Nous considérons un problème de classification binaire où Y peut être 0 ou 1.
# La caractéristique X peut prendre des valeurs discrètes, par exemple 0, 1, 2, 3, 4.

# Probabilités a priori P(Y)
prior_y = {
    0: 0.5,  # P(Y=0)
    1: 0.5   # P(Y=1)
}
assert sum(prior_y.values()) == 1.0, "Les probabilités a priori P(Y) doivent sommer à 1."

# Vraisemblances P(X|Y)
# Ces probabilités sont définies arbitrairement pour l'exemple.
likelihood_x_given_y = {
    0: {0: 0.1, 1: 0.2, 2: 0.4, 3: 0.2, 4: 0.1}, # P(X|Y=0) pour chaque valeur de X
    1: {0: 0.1, 1: 0.1, 2: 0.2, 3: 0.3, 4: 0.3}  # P(X|Y=1) pour chaque valeur de X
}

# Assertions pour vérifier que les vraisemblances somment à 1 pour chaque classe
for y_val in likelihood_x_given_y:
    assert abs(sum(likelihood_x_given_y[y_val].values()) - 1.0) < 1e-9, \
        f"Les vraisemblances pour Y={y_val} doivent sommer à 1."

# --- 2. Implémentation du Théorème de Bayes ---

def calculate_evidence_px(x_val, prior_y, likelihood_x_given_y):
    """
    Calcule la probabilité marginale P(X=x_val) (l'évidence).
    P(X) = sum_y P(X|Y) * P(Y)
    """
    px = 0.0
    for y_val in prior_y:
        # Utilise .get(x_val, 0.0) pour gérer les cas où x_val n'est pas dans la distribution
        px += likelihood_x_given_y[y_val].get(x_val, 0.0) * prior_y[y_val]
    return px

def calculate_posterior_py_given_x(y_val, x_val, prior_y, likelihood_x_given_y):
    """
    Calcule la probabilité postérieure P(Y=y_val|X=x_val) en utilisant le Théorème de Bayes.
    P(Y|X) = P(X|Y) * P(Y) / P(X)
    """
    px = calculate_evidence_px(x_val, prior_y, likelihood_x_given_y)
    if px == 0:
        # Si P(X) est zéro, cela signifie que x_val est impossible,
        # donc la probabilité postérieure est zéro.
        return 0.0

    # P(Y|X) = P(X|Y) * P(Y) / P(X)
    return (likelihood_x_given_y[y_val].get(x_val, 0.0) * prior_y[y_val]) / px

# --- 3. Implémentation du Classifieur de Bayes Optimal (pour la perte 0-1) ---

def bayes_classifier(x_val, prior_y, likelihood_x_given_y):
    """
    Implémente le classifieur de Bayes optimal pour la fonction de perte 0-1.
    Il prédit la classe y qui maximise P(Y=y|X=x_val).
    """
    posteriors = {}
    for y_val in prior_y:
        posteriors[y_val] = calculate_posterior_py_given_x(y_val, x_val, prior_y, likelihood_x_given_y)

    # Trouver la classe avec la probabilité postérieure maximale.
    # En cas d'égalité, nous choisissons la classe avec l'étiquette la plus petite (e.g., 0).
    best_y = None
    max_posterior = -1.0
    for y_val, posterior in posteriors.items():
        if posterior > max_posterior:
            max_posterior = posterior
            best_y = y_val
        elif posterior == max_posterior and best_y is not None and y_val < best_y:
            best_y = y_val # Règle de départage
    return best_y, posteriors

# --- Assertions de validation mathématique pour le classifieur de Bayes ---

# Test de P(X) pour X=2
# P(X=2) = P(X=2|Y=0)P(Y=0) + P(X=2|Y=1)P(Y=1)
#        = (0.4 * 0.5) + (0.2 * 0.5) = 0.2 + 0.1 = 0.3
px_for_x2 = calculate_evidence_px(2, prior_y, likelihood_x_given_y)
assert abs(px_for_x2 - 0.3) < 1e-9, f"Erreur de calcul de P(X=2): {px_for_x2}"

# Test de P(Y|X) pour X=2
# P(Y=0|X=2) = P(X=2|Y=0)P(Y=0) / P(X=2) = (0.4 * 0.5) / 0.3 = 0.2 / 0.3 = 2/3
posterior_y0_given_x2 = calculate_posterior_py_given_x(0, 2, prior_y, likelihood_x_given_y)
assert abs(posterior_y0_given_x2 - (2/3)) < 1e-9, f"Erreur de calcul de P(Y=0|X=2): {posterior_y0_given_x2}"
# P(Y=1|X=2) = P(X=2|Y=1)P(Y=1) / P(X=2) = (0.2 * 0.5) / 0.3 = 0.1 / 0.3 = 1/3
posterior_y1_given_x2 = calculate_posterior_py_given_x(1, 2, prior_y, likelihood_x_given_y)
assert abs(posterior_y1_given_x2 - (1/3)) < 1e-9, f"Erreur de calcul de P(Y=1|X=2): {posterior_y1_given_x2}"
assert abs(posterior_y0_given_x2 + posterior_y1_given_x2 - 1.0) < 1e-9, "Les probabilités postérieures pour X=2 doivent sommer à 1."

# Test du classifieur de Bayes pour X=2
# P(Y=0|X=2) = 2/3 > P(Y=1|X=2) = 1/3, donc la prédiction doit être 0.
predicted_class_x2, _ = bayes_classifier(2, prior_y, likelihood_x_given_y)
assert predicted_class_x2 == 0, f"Le classifieur de Bayes pour X=2 devrait prédire 0, a obtenu {predicted_class_x2}."

# Test du classifieur de Bayes pour X=4
# P(Y=0|X=4) = (0.1 * 0.5) / P(X=4) = 0.05 / P(X=4)
# P(Y=1|X=4) = (0.3 * 0.5) / P(X=4) = 0.15 / P(X=4)
# P(X=4) = 0.05 + 0.15 = 0.2
# P(Y=0|X=4) = 0.05 / 0.2 = 0.25
# P(Y=1|X=4) = 0.15 / 0.2 = 0.75
# P(Y=1|X=4) > P(Y=0|X=4), donc la prédiction doit être 1.
predicted_class_x4, _ = bayes_classifier(4, prior_y, likelihood_x_given_y)
assert predicted_class_x4 == 1, f"Le classifieur de Bayes pour X=4 devrait prédire 1, a obtenu {predicted_class_x4}."


# --- 4. Fonctions de Perte (Loss Functions) ---

def zero_one_loss(y_true, y_pred):
    """
    Fonction de perte 0-1 (misclassification error).
    Retourne 0 si la prédiction est correcte, 1 sinon.
    """
    return 0 if y_true == y_pred else 1

def squared_loss(y_true, y_pred):
    """
    Fonction de perte quadratique (squared loss), une fonction de perte de substitution courante.
    Elle mesure la distance euclidienne entre la vraie valeur et la prédiction.
    """
    return (y_true - y_pred)**2

# --- Assertions pour les fonctions de perte ---
assert zero_one_loss(0, 0) == 0, "0-1 loss (0,0) devrait être 0"
assert zero_one_loss(1, 1) == 0, "0-1 loss (1,1) devrait être 0"
assert zero_one_loss(0, 1) == 1, "0-1 loss (0,1) devrait être 1"
assert zero_one_loss(1, 0) == 1, "0-1 loss (1,0) devrait être 1"

assert squared_loss(0, 0) == 0, "Squared loss (0,0) devrait être 0"
assert squared_loss(1, 1) == 0, "Squared loss (1,1) devrait être 0"
assert squared_loss(0, 1) == 1, "Squared loss (0,1) devrait être 1"
assert squared_loss(1, 0) == 1, "Squared loss (1,0) devrait être 1"
# Test avec une prédiction continue pour la perte quadratique
assert abs(squared_loss(0.5, 0) - 0.25) < 1e-9, "Squared loss (0.5,0) devrait être 0.25"


# --- 5. Démonstration de la Minimisation du Risque Empirique (ERM) ---

# Fonction pour générer un échantillon (X, Y) selon les distributions définies
def generate_sample(prior_y, likelihood_x_given_y):
    """
    Génère un échantillon (x, y) en respectant les distributions P(Y) et P(X|Y).
    """
    # Échantillonner Y
    r_y = random.random()
    current_prob_sum = 0.0
    y_sample = None
    for y_val, prob_y in prior_y.items():
        current_prob_sum += prob_y
        if r_y <= current_prob_sum:
            y_sample = y_val
            break

    # Échantillonner X étant donné Y
    r_x = random.random()
    current_prob_sum = 0.0
    x_sample = None
    likelihood_for_y = likelihood_x_given_y[y_sample]
    # Trier les clés pour assurer un ordre d'échantillonnage cohérent
    sorted_x_keys = sorted(likelihood_for_y.keys())
    for x_val in sorted_x_keys:
        prob_x_given_y = likelihood_for_y[x_val]
        current_prob_sum += prob_x_given_y
        if r_x <= current_prob_sum:
            x_sample = x_val
            break

    return x_sample, y_sample

# Générer un jeu de données synthétique
num_samples = 1000
dataset = [generate_sample(prior_y, likelihood_x_given_y) for _ in range(num_samples)]

# Séparer les caractéristiques (X) et les étiquettes (Y)
X_data = [sample[0] for sample in dataset]
Y_data = [sample[1] for sample in dataset]

# Définir un classifieur "sous-optimal" simple pour comparaison
def simple_threshold_classifier(x_val, threshold_x=2):
    """
    Un classifieur simple qui prédit 0 si x_val < threshold_x, sinon 1.
    """
    return 0 if x_val < threshold_x else 1

# Fonction pour calculer le risque empirique
def calculate_empirical_risk(classifier_func, X_data, Y_data, loss_func, *args_for_classifier):
    """
    Calcule le risque empirique pour un classifieur et une fonction de perte donnés.
    """
    total_loss = 0.0
    for i in range(len(X_data)):
        x_val = X_data[i]
        y_true = Y_data[i]
        # Appelle le classifieur avec x_val et les arguments supplémentaires
        y_pred = classifier_func(x_val, *args_for_classifier)
        total_loss += loss_func(y_true, y_pred)
    return total_loss / len(X_data)

# Calculer le risque empirique pour le classifieur de Bayes (avec perte 0-1)
empirical_risk_bayes_01 = calculate_empirical_risk(
    lambda x, p_y, l_x_gy: bayes_classifier(x, p_y, l_x_gy)[0], # Le lambda extrait la prédiction (y_pred)
    X_data, Y_data, zero_one_loss, prior_y, likelihood_x_given_y
)

# Calculer le risque empirique pour le classifieur à seuil simple (avec perte 0-1)
empirical_risk_threshold_01 = calculate_empirical_risk(
    simple_threshold_classifier, X_data, Y_data, zero_one_loss, 2 # threshold_x=2
)

# Calculer le risque empirique pour le classifieur de Bayes (avec perte quadratique)
empirical_risk_bayes_squared = calculate_empirical_risk(
    lambda x, p_y, l_x_gy: bayes_classifier(x, p_y, l_x_gy)[0],
    X_data, Y_data, squared_loss, prior_y, likelihood_x_given_y
)

# Calculer le risque empirique pour le classifieur à seuil simple (avec perte quadratique)
empirical_risk_threshold_squared = calculate_empirical_risk(
    simple_threshold_classifier, X_data, Y_data, squared_loss, 2
)

# --- Assertions pour le Risque Empirique ---
assert 0 <= empirical_risk_bayes_01 <= 1, "Le risque empirique 0-1 doit être entre 0 et 1."
assert 0 <= empirical_risk_threshold_01 <= 1, "Le risque empirique 0-1 doit être entre 0 et 1."
assert empirical_risk_bayes_squared >= 0, "Le risque empirique quadratique doit être non-négatif."
assert empirical_risk_threshold_squared >= 0, "Le risque empirique quadratique doit être non-négatif."


# --- Affichage des résultats pour la démonstration ---
print("--- Classifieur de Bayes Optimal et Fonctions de Perte ---")
print(f"Probabilités a priori P(Y): {prior_y}")
print(f"Vraisemblances P(X|Y): {likelihood_x_given_y}")

print("\nDécisions du Classifieur de Bayes pour les valeurs de X possibles:")
# Itérer sur toutes les valeurs de X possibles dans nos définitions
all_x_values = sorted(list(set(list(likelihood_x_given_y[0].keys()) + list(likelihood_x_given_y[1].keys()))))
for x_val in all_x_values:
    predicted_y, posteriors = bayes_classifier(x_val, prior_y, likelihood_x_given_y)
    print(f"  Pour X={x_val}: P(Y=0|X={x_val})={posteriors.get(0,0):.3f}, P(Y=1|X={x_val})={posteriors.get(1,0):.3f} -> Prédiction Y={predicted_y}")

print(f"\nJeu de données synthétique généré avec {num_samples} échantillons pour la démonstration ERM.")
print(f"Risque Empirique (perte 0-1) - Classifieur de Bayes: {empirical_risk_bayes_01:.4f}")
print(f"Risque Empirique (perte 0-1) - Classifieur à Seuil Simple (seuil=2): {empirical_risk_threshold_01:.4f}")
print(f"Risque Empirique (perte quadratique) - Classifieur de Bayes: {empirical_risk_bayes_squared:.4f}")
print(f"Risque Empirique (perte quadratique) - Classifieur à Seuil Simple (seuil=2): {empirical_risk_threshold_squared:.4f}")

if empirical_risk_bayes_01 <= empirical_risk_threshold_01:
    print("\nObservation: Le Classifieur de Bayes (optimal pour la perte 0-1) atteint généralement un risque empirique 0-1 inférieur ou égal.")
else:
    print("\nObservation: Pour cet échantillon spécifique, le risque empirique 0-1 du Classifieur de Bayes était légèrement supérieur à celui du classifieur simple. Cela peut arriver en raison de la variance d'échantillonnage, mais théoriquement, le classifieur de Bayes minimise le risque réel.")
```
