# TP 2 - Minimisation du Risque Empirique (ERM) avec Surrogate Loss

## Introduction Mathématique
Minimisation de la perte logistique via descente de gradient pur NumPy.

## Implémentation (Python Pur)
```python
import numpy as np

def generate_linear_data(n=200):
    X = np.random.randn(n, 2)
    true_w = np.array([1.5, -2.0])
    logits = X @ true_w
    probs = 1 / (1 + np.exp(-logits))
    Y = np.where(np.random.rand(n) < probs, 1, -1)
    return X, Y

def logistic_loss(w, X, y):
    margins = y * (X @ w)
    return np.mean(np.log(1 + np.exp(-margins)))

def logistic_loss_grad(w, X, y):
    margins = y * (X @ w)
    # The gradient is -y * x / (1 + exp(y * w^T x))
    probs = 1 / (1 + np.exp(margins))
    grad = -np.mean((probs * y)[:, None] * X, axis=0)
    return grad

X_train, Y_train = generate_linear_data(1000)
X_test, Y_test = generate_linear_data(500)

w = np.zeros(2)
lr = 0.5
epochs = 500

for epoch in range(epochs):
    grad = logistic_loss_grad(w, X_train, Y_train)
    w -= lr * grad

# Evaluation
train_loss = logistic_loss(w, X_train, Y_train)
test_loss = logistic_loss(w, X_test, Y_test)

preds = np.sign(X_test @ w)
acc = np.mean(preds == Y_test)

# Validation mathématique : L'ERM avec fonction convexe doit donner de bons résultats
assert train_loss < 0.5, "La loss d'entrainement n'a pas convergé."
assert acc > 0.75, f"La précision de test est anormalement basse: {acc}"
print(f"Test Accuracy: {acc*100:.2f}%, Test Surrogate Loss: {test_loss:.4f}")
```
