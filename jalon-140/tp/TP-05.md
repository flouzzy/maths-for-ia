## Objectif
Ce TP a pour objectif d'implémenter de zéro les concepts fondamentaux du Jalon 140 : le classifieur de Bayes optimal, les fonctions de perte de substitution (surrogate loss functions) et le principe de minimisation du risque empirique (Empirical Risk Minimization - ERM). Nous allons générer un jeu de données synthétique où les distributions sous-jacentes sont connues, ce qui nous permettra de calculer et de comparer le classifieur de Bayes optimal (qui minimise le risque 0-1 théorique) avec un classifieur appris via ERM en utilisant une fonction de perte substitut.

Nous allons :
1.  Générer des données synthétiques pour un problème de classification binaire 1D à partir de distributions Gaussiennes connues.
2.  Implémenter le calcul des probabilités a posteriori et le classifieur de Bayes optimal.
3.  Implémenter la perte 0-1 (misclassification error) ainsi que des fonctions de perte substituts comme la perte logistique et la perte charnière.
4.  Mettre en œuvre le principe d'ERM en entraînant un modèle de régression logistique linéaire 1D par descente de gradient, minimisant ainsi le risque empirique logistique.
5.  Comparer les performances (risque 0-1) du classifieur de Bayes optimal et du modèle ERM appris.

Toutes les implémentations seront réalisées en Python pur, sans l'aide de bibliothèques d'apprentissage automatique ou de visualisation externes (comme `numpy`, `sklearn`, `matplotlib`, `tensorflow`, etc.), et incluront des assertions pour valider les étapes mathématiques.

```python
import math
import random

# --- Fonctions utilitaires mathématiques de base ---

def gaussian_pdf(x, mu, sigma):
    """
    Calcule la densité de probabilité d'une distribution Gaussienne (normale)
    pour une valeur x donnée, avec moyenne mu et écart-type sigma.
    """
    if sigma <= 0:
        raise ValueError("L'écart-type (sigma) doit être positif.")
    coefficient = 1 / (math.sqrt(2 * math.pi) * sigma)
    exponent = -((x - mu)**2) / (2 * sigma**2)
    return coefficient * math.exp(exponent)

def sigmoid(z):
    """
    Calcule la fonction sigmoïde, utilisée pour mapper une valeur réelle à une probabilité [0, 1].
    """
    return 1 / (1 + math.exp(-z))

# --- 1. Génération de données synthétiques ---

def generate_synthetic_data(n_samples, mu1, sigma1, mu2, sigma2, p_y1):
    """
    Génère un dataset synthétique 1D pour la classification binaire.
    Les données sont générées à partir de deux distributions Gaussiennes.

    Args:
        n_samples (int): Nombre total d'échantillons à générer.
        mu1 (float): Moyenne de la distribution Gaussienne pour la classe 1.
        sigma1 (float): Écart-type de la distribution Gaussienne pour la classe 1.
        mu2 (float): Moyenne de la distribution Gaussienne pour la classe 0.
        sigma2 (float): Écart-type de la distribution Gaussienne pour la classe 0.
        p_y1 (float): Probabilité a priori de la classe 1 (P(Y=1)).

    Returns:
        tuple: (list de floats X, list de ints Y) où X sont les caractéristiques et Y les labels.
    """
    data = []
    labels = []

    n_samples_y1 = int(n_samples * p_y1)
    n_samples_y0 = n_samples - n_samples_y1

    # Utilisation d'un générateur de nombres aléatoires pour la reproductibilité
    # (Box-Muller transform pour générer des nombres gaussiens à partir d'uniformes)
    random.seed(42)

    for _ in range(n_samples_y1):
        u1 = random.random()
        u2 = random.random()
        z0 = math.sqrt(-2 * math.log(u1)) * math.cos(2 * math.pi * u2)
        data.append(mu1 + z0 * sigma1)
        labels.append(1)

    for _ in range(n_samples_y0):
        u1 = random.random()
        u2 = random.random()
        z0 = math.sqrt(-2 * math.log(u1)) * math.cos(2 * math.pi * u2)
        data.append(mu2 + z0 * sigma2)
        labels.append(0)

    # Mélanger les données pour éviter un ordre séquentiel des classes
    combined = list(zip(data, labels))
    random.shuffle(combined)
    shuffled_data, shuffled_labels = zip(*combined)

    return list(shuffled_data), list(shuffled_labels)

# --- 2. Classifieur de Bayes Optimal ---

def calculate_posterior_probabilities(x, mu1, sigma1, mu2, sigma2, p_y1):
    """
    Calcule les probabilités a posteriori P(Y=1|X=x) et P(Y=0|X=x)
    en utilisant le théorème de Bayes, étant donné les paramètres des distributions.

    Args:
        x (float): La valeur de la caractéristique.
        mu1, sigma1 (float): Paramètres de la Gaussienne pour la classe 1.
        mu2, sigma2 (float): Paramètres de la Gaussienne pour la classe 0.
        p_y1 (float): Probabilité a priori de la classe 1.

    Returns:
        tuple: (P(Y=1|X=x), P(Y=0|X=x))
    """
    p_y0 = 1 - p_y1

    # P(X|Y=1) : vraisemblance de x étant donné la classe 1
    likelihood_x_given_y1 = gaussian_pdf(x, mu1, sigma1)
    # P(X|Y=0) : vraisemblance de x étant donné la classe 0
    likelihood_x_given_y0 = gaussian_pdf(x, mu2, sigma2)

    # P(X) = P(X|Y=1)P(Y=1) + P(X|Y=0)P(Y=0) (loi des probabilités totales)
    marginal_x = likelihood_x_given_y1 * p_y1 + likelihood_x_given_y0 * p_y0

    if marginal_x == 0: # Éviter la division par zéro si x est très improbable
        # Dans ce cas, les probabilités a posteriori sont indéterminées,
        # on peut retourner une probabilité égale ou gérer comme une erreur.
        # Pour la classification, une décision aléatoire est souvent prise.
        return 0.5, 0.5

    # P(Y=1|X) = P(X|Y=1)P(Y=1) / P(X)
    posterior_y1 = (likelihood_x_given_y1 * p_y1) / marginal_x
    # P(Y=0|X) = P(X|Y=0)P(Y=0) / P(X)
    posterior_y0 = (likelihood_x_given_y0 * p_y0) / marginal_x

    return posterior_y1, posterior_y0

def bayes_optimal_classifier(x, mu1, sigma1, mu2, sigma2, p_y1):
    """
    Implémente le classifieur de Bayes optimal pour minimiser la perte 0-1.
    Prédit la classe avec la probabilité a posteriori la plus élevée.

    Args:
        x (float): La valeur de la caractéristique.
        mu1, sigma1 (float): Paramètres de la Gaussienne pour la classe 1.
        mu2, sigma2 (float): Paramètres de la Gaussienne pour la classe 0.
        p_y1 (float): Probabilité a priori de la classe 1.

    Returns:
        int: La classe prédite (0 ou 1).
    """
    posterior_y1, posterior_y0 = calculate_posterior_probabilities(x, mu1, sigma1, mu2, sigma2, p_y1)

    if posterior_y1 >= posterior_y0:
        return 1
    else:
        return 0

# --- 3. Fonctions de Perte Substituts et Risque Empirique ---

def zero_one_loss(y_true, y_pred):
    """
    Calcule la perte 0-1 (misclassification error) pour une seule prédiction.
    Retourne 1 si la prédiction est incorrecte, 0 sinon.
    """
    return 1 if y_true != y_pred else 0

def empirical_zero_one_risk(y_true_list, y_pred_list):
    """
    Calcule le risque empirique 0-1 (taux d'erreur) sur un ensemble de prédictions.
    """
    if not y_true_list:
        return 0.0
    errors = sum(zero_one_loss(yt, yp) for yt, yp in zip(y_true_list, y_pred_list))
    return errors / len(y_true_list)

def logistic_loss(y_true, y_pred_proba_y1):
    """
    Calcule la perte logistique (Binary Cross-Entropy) pour une seule prédiction.
    C'est une fonction de perte substitut convexe pour la perte 0-1.

    Args:
        y_true (int): Le vrai label (0 ou 1).
        y_pred_proba_y1 (float): La probabilité prédite que Y=1.

    Returns:
        float: La valeur de la perte logistique.
    """
    # Assurer que les probabilités sont dans un intervalle sûr pour le log
    epsilon = 1e-10
    y_pred_proba_y1 = max(epsilon, min(1 - epsilon, y_pred_proba_y1))

    if y_true == 1:
        return -math.log(y_pred_proba_y1)
    else: # y_true == 0
        return -math.log(1 - y_pred_proba_y1)

def empirical_logistic_risk(y_true_list, y_pred_proba_y1_list):
    """
    Calcule le risque empirique logistique sur un ensemble de prédictions.
    """
    if not y_true_list:
        return 0.0
    total_loss = sum(logistic_loss(yt, yp) for yt, yp in zip(y_true_list, y_pred_proba_y1_list))
    return total_loss / len(y_true_list)

def hinge_loss(y_true, y_score):
    """
    Calcule la perte charnière (Hinge Loss) pour une seule prédiction.
    Utilisée notamment par les SVM, c'est une autre fonction de perte substitut convexe.

    Args:
        y_true (int): Le vrai label (0 ou 1). Sera transformé en -1 ou 1.
        y_score (float): La sortie du classifieur (e.g., w*x + b).

    Returns:
        float: La valeur de la perte charnière.
    """
    # Convertir y_true (0 ou 1) en -1 ou 1 pour la perte charnière standard
    y_true_transformed = 1 if y_true == 1 else -1
    return max(0, 1 - y_true_transformed * y_score)

def empirical_hinge_risk(y_true_list, y_score_list):
    """
    Calcule le risque empirique charnière sur un ensemble de prédictions.
    """
    if not y_true_list:
        return 0.0
    total_loss = sum(hinge_loss(yt, ys) for yt, ys in zip(y_true_list, y_score_list))
    return total_loss / len(y_true_list)

# --- 4. Minimisation du Risque Empirique (ERM) avec Régression Logistique ---

# Pour des données 1D, un modèle linéaire est simplement: score = w*x + b
# La probabilité P(Y=1|X=x) est donnée par sigmoid(w*x + b)

def predict_proba_logistic(x, w, b):
    """
    Prédit la probabilité P(Y=1|X=x) pour un modèle logistique linéaire 1D.
    """
    score = w * x + b
    return sigmoid(score)

def predict_class_logistic(x, w, b):
    """
    Prédit la classe (0 ou 1) pour un modèle logistique linéaire 1D.
    Le seuil est 0.5 pour la probabilité.
    """
    proba_y1 = predict_proba_logistic(x, w, b)
    return 1 if proba_y1 >= 0.5 else 0

def calculate_gradient_logistic_loss(x, y_true, w, b):
    """
    Calcule les gradients de la perte logistique par rapport à w et b
    pour un seul échantillon (x, y_true).
    """
    proba_y1 = predict_proba_logistic(x, w, b)

    # La dérivée de la perte logistique par rapport au score (w*x + b) est (proba_y1 - y_true)
    error = proba_y1 - y_true

    grad_w = error * x
    grad_b = error

    return grad_w, grad_b

def gradient_descent_logistic_regression(X, Y, learning_rate, n_iterations):
    """
    Implémente la descente de gradient par lots (Batch Gradient Descent)
    pour minimiser le risque logistique empirique pour un modèle logistique linéaire 1D.

    Args:
        X (list): Liste des caractéristiques (valeurs 1D).
        Y (list): Liste des labels (0 ou 1).
        learning_rate (float): Taux d'apprentissage pour la descente de gradient.
        n_iterations (int): Nombre d'itérations de la descente de gradient.

    Returns:
        tuple: (w, b) les poids et le biais appris du modèle.
    """
    # Initialisation aléatoire des poids pour la reproductibilité
    random.seed(42)
    w = random.uniform(-1, 1)
    b = random.uniform(-1, 1)

    print(f"Initialisation des paramètres du modèle: w={w:.4f}, b={b:.4f}")

    previous_risk = float('inf') # Pour l'assertion de convergence

    for i in range(n_iterations):
        grad_w_sum = 0
        grad_b_sum = 0

        # Calcul des gradients moyens sur tout le dataset (Batch Gradient Descent)
        for x_val, y_val in zip(X, Y):
            grad_w, grad_b = calculate_gradient_logistic_loss(x_val, y_val, w, b)
            grad_w_sum += grad_w
            grad_b_sum += grad_b

        # Mise à jour des poids
        w -= learning_rate * (grad_w_sum / len(X))
        b -= learning_rate * (grad_b_sum / len(X))

        # Affichage et validation de la convergence toutes les 10% des itérations
        if (i + 1) % (n_iterations // 10) == 0 or i == 0:
            y_pred_proba_list = [predict_proba_logistic(x_val, w, b) for x_val in X]
            current_risk = empirical_logistic_risk(Y, y_pred_proba_list)
            print(f"Itération {i+1}/{n_iterations}: w={w:.4f}, b={b:.4f}, Risque Logistique Empirique={current_risk:.4f}")

            # Assertion pour vérifier que le risque diminue (ou reste stable)
            assert current_risk <= previous_risk + 1e-6, "Le risque logistique devrait diminuer ou rester stable au fil des itérations."
            previous_risk = current_risk

    return w, b

# --- Bloc d'exécution principal et Assertions ---
if __name__ == "__main__":
    print("--- TP 5 : Classifieur de Bayes Optimal et Fonctions de Perte Substituts ---")

    print("\n## 1. Génération de données synthétiques et validation")

    # Paramètres des distributions Gaussiennes pour les deux classes
    mu1_true, sigma1_true = 2.0, 1.0  # Classe 1 (ex: des valeurs positives)
    mu2_true, sigma2_true = -2.0, 1.0 # Classe 0 (ex: des valeurs négatives)
    p_y1_true = 0.5 # Probabilité a priori de la classe 1 (classes équilibrées)

    n_samples = 1000
    X_data, Y_labels = generate_synthetic_data(n_samples, mu1_true, sigma1_true, mu2_true, sigma2_true, p_y1_true)

    print(f"Données générées : {len(X_data)} échantillons.")
    print(f"Exemple de données (X, Y) : {list(zip(X_data[:5], Y_labels[:5]))}")
    assert len(X_data) == n_samples, "Assertion échouée: Le nombre d'échantillons générés est incorrect."
    assert all(y in [0, 1] for y in Y_labels), "Assertion échouée: Les labels doivent être 0 ou 1."
    assert abs(sum(Y_labels) / len(Y_labels) - p_y1_true) < 0.05, "Assertion échouée: La proportion de la classe 1 est trop éloignée de la probabilité a priori."

    # Assertion sur la PDF Gaussienne
    pdf_val_at_mu = gaussian_pdf(mu1_true, mu1_true, sigma1_true)
    assert pdf_val_at_mu > 0, "Assertion échouée: La PDF Gaussienne doit être positive."
    assert gaussian_pdf(mu1_true + 3 * sigma1_true, mu1_true, sigma1_true) < pdf_val_at_mu, "Assertion échouée: La PDF doit être plus faible loin de la moyenne."

    print("\n## 2. Classifieur de Bayes Optimal")

    # Calcul des probabilités a posteriori pour un point d'exemple
    # Pour des distributions symétriques avec mêmes variances et priors, le point de décision est au milieu des moyennes.
    test_x = (mu1_true + mu2_true) / 2.0
    posterior_y1, posterior_y0 = calculate_posterior_probabilities(test_x, mu1_true, sigma1_true, mu2_true, sigma2_true, p_y1_true)
    print(f"Pour x={test_x:.2f}: P(Y=1|X={test_x:.2f})={posterior_y1:.4f}, P(Y=0|X={test_x:.2f})={posterior_y0:.4f}")
    assert abs(posterior_y1 + posterior_y0 - 1.0) < 1e-9, "Assertion échouée: La somme des probabilités a posteriori doit être 1."
    assert posterior_y1 >= 0 and posterior_y1 <= 1, "Assertion échouée: La probabilité a posteriori doit être entre 0 et 1."
    # Pour ce test_x, avec des distributions symétriques et priors égaux, les probabilités devraient être égales.
    assert abs(posterior_y1 - 0.5) < 1e-9, "Assertion échouée: P(Y=1|X) devrait être 0.5 au point de décision optimal."

    # Prédiction du classifieur de Bayes optimal
    bayes_pred = bayes_optimal_classifier(test_x, mu1_true, sigma1_true, mu2_true, sigma2_true, p_y1_true)
    print(f"Prédiction Bayes optimale pour x={test_x:.2f}: {bayes_pred}")

    # Évaluation du risque 0-1 du classifieur de Bayes optimal sur les données générées
    bayes_predictions = [bayes_optimal_classifier(x, mu1_true, sigma1_true, mu2_true, sigma2_true, p_y1_true) for x in X_data]
    bayes_optimal_risk = empirical_zero_one_risk(Y_labels, bayes_predictions)
    print(f"Risque 0-1 empirique du classifieur de Bayes optimal sur les données générées: {bayes_optimal_risk:.4f}")
    # Le risque de Bayes est la borne inférieure du risque 0-1.
    # Pour des données générées à partir des vraies distributions, ce risque devrait être proche du risque théorique.
    assert bayes_optimal_risk >= 0 and bayes_optimal_risk <= 0.5, "Assertion échouée: Le risque 0-1 doit être entre 0 et 0.5 (pour un classifieur non-aléatoire)."

    print("\n## 3. Fonctions de Perte Substituts et Risque Empirique")

    # Assertion sur la perte 0-1
    assert zero_one_loss(1, 1) == 0, "Assertion échouée: Perte 0-1 incorrecte pour une prédiction correcte."
    assert zero_one_loss(1, 0) == 1, "Assertion échouée: Perte 0-1 incorrecte pour une prédiction incorrecte."

    # Assertion sur la perte logistique
    # Si y_true=1 et proba_y1=1, perte = 0
    assert abs(logistic_loss(1, 1.0 - 1e-10)) < 1e-5, "Assertion échouée: Perte logistique incorrecte pour proba_y1=1."
    # Si y_true=0 et proba_y1=0, perte = 0
    assert abs(logistic_loss(0, 1e-10)) < 1e-5, "Assertion échouée: Perte logistique incorrecte pour proba_y1=0."
    # Si y_true=1 et proba_y1=0.5, perte = -log(0.5) approx 0.693
    assert abs(logistic_loss(1, 0.5) - 0.693) < 0.001, "Assertion échouée: Perte logistique incorrecte pour proba_y1=0.5."
    assert logistic_loss(1, 0.1) > logistic_loss(1, 0.9), "Assertion échouée: La perte logistique devrait être plus élevée pour une mauvaise prédiction."

    # Assertion sur la perte charnière
    # y_true=1 (transformé en 1), y_score=2 -> max(0, 1 - 1*2) = 0
    assert hinge_loss(1, 2) == 0, "Assertion échouée: Perte charnière incorrecte pour score positif et correct."
    # y_true=1 (transformé en 1), y_score=0.5 -> max(0, 1 - 1*0.5) = 0.5
    assert hinge_loss(1, 0.5) == 0.5, "Assertion échouée: Perte charnière incorrecte pour score positif mais faible."
    # y_true=0 (transformé en -1), y_score=-2 -> max(0, 1 - (-1)*(-2)) = max(0, 1-2) = 0
    assert hinge_loss(0, -2) == 0, "Assertion échouée: Perte charnière incorrecte pour score négatif et correct."
    # y_true=0 (transformé en -1), y_score=0.5 -> max(0, 1 - (-1)*0.5) = max(0, 1+0.5) = 1.5
    assert hinge_loss(0, 0.5) == 1.5, "Assertion échouée: Perte charnière incorrecte pour score positif et incorrect."

    print("\n## 4. Minimisation du Risque Empirique (ERM) avec Régression Logistique")

    learning_rate = 0.1
    n_iterations = 100

    # Entraînement du modèle de régression logistique par descente de gradient
    print(f"\nDébut de l'entraînement de la régression logistique (learning_rate={learning_rate}, iterations={n_iterations})...")
    w_learned, b_learned = gradient_descent_logistic_regression(X_data, Y_labels, learning_rate, n_iterations)
    print(f"\nModèle logistique appris: w={w_learned:.4f}, b={b_learned:.4f}")

    # Évaluation du modèle appris
    y_pred_proba_erm = [predict_proba_logistic(x, w_learned, b_learned) for x in X_data]
    y_pred_class_erm = [predict_class_logistic(x, w_learned, b_learned) for x in X_data]

    erm_logistic_risk = empirical_logistic_risk(Y_labels, y_pred_proba_erm)
    erm_zero_one_risk = empirical_zero_one_risk(Y_labels, y_pred_class_erm)

    print(f"Risque logistique empirique du modèle ERM: {erm_logistic_risk:.4f}")
    print(f"Risque 0-1 empirique du modèle ERM: {erm_zero_one_risk:.4f}")

    print("\n## 5. Comparaison des résultats")
    print(f"Risque 0-1 du Classifieur de Bayes Optimal: {bayes_optimal_risk:.4f}")
    print(f"Risque 0-1 du modèle ERM (Régression Logistique): {erm_zero_one_risk:.4f}")

    # Assertion: Le risque ERM devrait être proche du risque de Bayes optimal
    # ou légèrement supérieur, car ERM minimise une perte substitut et non directement la perte 0-1.
    # Dans ce cas 1D avec variances égales, le classifieur de Bayes optimal est linéaire,
    # donc la régression logistique (qui est un modèle linéaire) peut s'en approcher très bien.
    assert erm_zero_one_risk >= bayes_optimal_risk - 0.02, "Assertion échouée: Le risque ERM ne devrait pas être significativement inférieur au risque de Bayes optimal."
    assert erm_zero_one_risk < 0.5, "Assertion échouée: Le classifieur ERM devrait faire mieux qu'un classifieur aléatoire."
    assert erm_zero_one_risk <= bayes_optimal_risk + 0.05, "Assertion échouée: Le risque ERM ne devrait pas être beaucoup plus élevé que le risque de Bayes optimal."

    print("\n--- Fin du TP 5 ---")
```
