## Contexte Théorique

La **quantification universelle** est un concept fondamental en logique mathématique. Elle est exprimée par le symbole "∀" (pour tout, pour chaque). Une proposition de la forme "∀x P(x)" signifie "Pour tout élément x dans un domaine donné, la propriété P(x) est vraie".

Pour vérifier la vérité d'une telle proposition sur un **domaine fini**, il suffit de tester la propriété P(x) pour *chaque* élément x du domaine.
*   Si P(x) est vraie pour *tous* les x du domaine, alors "∀x P(x)" est vraie.
*   Si P(x) est fausse pour *au moins un* x du domaine, alors "∀x P(x)" est fausse. Cet élément x pour lequel P(x) est fausse est appelé un **contre-exemple**.

La **négation** d'une proposition universelle est directement liée à l'existence d'un contre-exemple :
¬(∀x P(x)) ≡ ∃x ¬P(x)
C'est-à-dire, "Il n'est pas vrai que pour tout x, P(x) est vraie" est équivalent à "Il existe un x pour lequel P(x) est fausse".

```python
import math

def check_universal_quantifier(domain, predicate):
    """
    Vérifie si une proposition universelle (∀x P(x)) est vraie sur un domaine fini.
    Si la proposition est fausse, elle retourne le premier contre-exemple trouvé.

    Args:
        domain (iterable): L'ensemble fini sur lequel la quantification est effectuée.
        predicate (function): Une fonction qui prend un élément du domaine en argument
                              et retourne un booléen (True si la propriété est vraie, False sinon).

    Returns:
        tuple: (is_true, counter_example)
               - is_true (bool): True si la proposition universelle est vraie sur le domaine.
                                 False sinon.
               - counter_example (any): L'élément du domaine qui a rendu la proposition fausse
                                        (le premier trouvé), ou None si la proposition est vraie.
    """
    for item in domain:
        if not predicate(item):
            # Nous avons trouvé un contre-exemple, la proposition universelle est fausse.
            return False, item

    # Si la boucle se termine, cela signifie que le prédicat est vrai pour tous les éléments.
    return True, None

# --- Définition des prédicats pour les tests ---

# Prédicat 1: "x est pair"
def is_even(x):
    return x % 2 == 0

# Prédicat 2: "x * x >= x" (pour les entiers)
def square_ge_self(x):
    return x * x >= x

# Prédicat 3: "x > 5"
def is_greater_than_5(x):
    return x > 5

# Prédicat 4: "(x - 2)^2 >= 0" (toujours vrai pour tout x réel, donc aussi pour les entiers)
def always_non_negative_square(x):
    return (x - 2)**2 >= 0

# --- Définition des domaines pour les tests ---
domain_even_only = [2, 4, 6, 8]
domain_mixed_parity = [1, 2, 3, 4]
domain_integers_sq = [-2, -1, 0, 1, 2]
domain_gt5_true = [6, 7, 8]
domain_gt5_false = [3, 4, 5, 6]
domain_empty = [] # Un domaine vide rend toujours une proposition universelle vraie (vacuously true)

# --- Exécution des tests et assertions ---
print("--- Démarrage des tests de quantification universelle ---")

# Test 1.1: P(x) = "x est pair" sur [2, 4, 6, 8]
is_true, ce = check_universal_quantifier(domain_even_only, is_even)
print(f"Test 1.1: ∀x ∈ {domain_even_only}, x est pair -> Résultat: {is_true}, Contre-exemple: {ce}")
assert is_true == True, "Test 1.1 échoué: Devrait être vrai."
assert ce is None, "Test 1.1 échoué: Ne devrait pas avoir de contre-exemple."

# Test 1.2: P(x) = "x est pair" sur [1, 2, 3, 4]
is_true, ce = check_universal_quantifier(domain_mixed_parity, is_even)
print(f"Test 1.2: ∀x ∈ {domain_mixed_parity}, x est pair -> Résultat: {is_true}, Contre-exemple: {ce}")
assert is_true == False, "Test 1.2 échoué: Devrait être faux."
assert ce in [1, 3], "Test 1.2 échoué: Le contre-exemple devrait être 1 ou 3."
# Assertion de la négation: Si ∀x P(x) est faux, alors ∃x ¬P(x) est vrai.
# Le contre-exemple 'ce' est précisément cet x pour lequel ¬P(x) est vrai.
assert not is_even(ce), f"Test 1.2 échoué: Le contre-exemple {ce} devrait rendre le prédicat faux."


# Test 2.1: P(x) = "x * x >= x" sur [-2, -1, 0, 1, 2]
is_true, ce = check_universal_quantifier(domain_integers_sq, square_ge_self)
print(f"Test 2.1: ∀x ∈ {domain_integers_sq}, x*x >= x -> Résultat: {is_true}, Contre-exemple: {ce}")
assert is_true == True, "Test 2.1 échoué: Devrait être vrai."
assert ce is None, "Test 2.1 échoué: Ne devrait pas avoir de contre-exemple."


# Test 3.1: P(x) = "x > 5" sur [6, 7, 8]
is_true, ce = check_universal_quantifier(domain_gt5_true, is_greater_than_5)
print(f"Test 3.1: ∀x ∈ {domain_gt5_true}, x > 5 -> Résultat: {is_true}, Contre-exemple: {ce}")
assert is_true == True, "Test 3.1 échoué: Devrait être vrai."
assert ce is None, "Test 3.1 échoué: Ne devrait pas avoir de contre-exemple."

# Test 3.2: P(x) = "x > 5" sur [3, 4, 5, 6]
is_true, ce = check_universal_quantifier(domain_gt5_false, is_greater_than_5)
print(f"Test 3.2: ∀x ∈ {domain_gt5_false}, x > 5 -> Résultat: {is_true}, Contre-exemple: {ce}")
assert is_true == False, "Test 3.2 échoué: Devrait être faux."
assert ce in [3, 4, 5], "Test 3.2 échoué: Le contre-exemple devrait être 3, 4 ou 5."
assert not is_greater_than_5(ce), f"Test 3.2 échoué: Le contre-exemple {ce} devrait rendre le prédicat faux."


# Test 4.1: P(x) = "(x - 2)^2 >= 0" sur [1, 2, 3, 4]
is_true, ce = check_universal_quantifier(domain_mixed_parity, always_non_negative_square)
print(f"Test 4.1: ∀x ∈ {domain_mixed_parity}, (x-2)^2 >= 0 -> Résultat: {is_true}, Contre-exemple: {ce}")
assert is_true == True, "Test 4.1 échoué: Devrait être vrai."
assert ce is None, "Test 4.1 échoué: Ne devrait pas avoir de contre-exemple."

is_true, ce = check_universal_quantifier(domain_empty, is_even)
print(f"Test 5.1: ∀x ∈ {domain_empty}, x est pair -> Résultat: {is_true}, Contre-exemple: {ce}")
assert is_true == True, "Test 5.1 échoué: Une proposition universelle sur un domaine vide est toujours vraie."
assert ce is None, "Test 5.1 échoué: Ne devrait pas avoir de contre-exemple sur un domaine vide."
