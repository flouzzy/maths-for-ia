# TP 02 : Tirages avec et sans remise

**Objectif :** Modéliser une urne et observer que le tirage avec remise préserve l'indépendance, tandis que le tirage sans remise détruit l'indépendance.

```python
import random

def simulate_urn(n_simulations, with_replacement):
    urn = ["red"] * 4 + ["blue"] * 6
    count_R1 = 0
    count_R2 = 0
    count_R1_and_R2 = 0

    for _ in range(n_simulations):
        current_urn = list(urn)

        # Premier tirage
        choice1 = random.choice(current_urn)
        if not with_replacement:
            current_urn.remove(choice1)

        # Deuxieme tirage
        choice2 = random.choice(current_urn)

        is_R1 = (choice1 == "red")
        is_R2 = (choice2 == "red")

        if is_R1: count_R1 += 1
        if is_R2: count_R2 += 1
        if is_R1 and is_R2: count_R1_and_R2 += 1

    return count_R1 / n_simulations, count_R2 / n_simulations, count_R1_and_R2 / n_simulations

def validate_simulations():
    N = 100000
    p_R1_rep, p_R2_rep, p_R1R2_rep = simulate_urn(N, True)

    # Avec remise : Independance (p_R1 * p_R2 = p_R1R2 = 0.16)
    assert abs(p_R1_rep - 0.4) < 0.01
    assert abs(p_R2_rep - 0.4) < 0.01
    assert abs(p_R1R2_rep - (p_R1_rep * p_R2_rep)) < 0.01, "Doit etre independant"

    p_R1_norep, p_R2_norep, p_R1R2_norep = simulate_urn(N, False)

    # Sans remise : Dependance (p_R1R2 = 0.133, produit = 0.16)
    assert abs(p_R1_norep - 0.4) < 0.01
    assert abs(p_R2_norep - 0.4) < 0.01
    assert abs(p_R1R2_norep - 2/15) < 0.01
    assert abs(p_R1R2_norep - (p_R1_norep * p_R2_norep)) > 0.02, "Ne doit pas etre independant"

    print("Test reussi : Tirage sans remise invalide l'independance.")

if __name__ == "__main__":
    validate_simulations()
```
