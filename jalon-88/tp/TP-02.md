\subsection*{TP 2 : Covariance nulle de variables dependantes}

Ce TP illustre le fait qu'une covariance nulle n'implique pas l'independance.
Nous allons generer une variable X uniforme sur -1, 0, 1 et Y = X**2.

\begin{lstlisting}
import random

def estimer_covariance_dependante(n_echantillons):
    somme_X = 0
    somme_Y = 0
    somme_XY = 0

    valeurs_possibles = [-1, 0, 1]

    for _ in range(n_echantillons):
        X = random.choice(valeurs_possibles)
        Y = X ** 2

        somme_X += X
        somme_Y += Y
        somme_XY += X * Y

    esp_X = somme_X / n_echantillons
    esp_Y = somme_Y / n_echantillons
    esp_XY = somme_XY / n_echantillons

    covariance = esp_XY - esp_X * esp_Y

    print(f"Esperance X : {esp_X:.4f}")
    print(f"Esperance Y : {esp_Y:.4f}")
    print(f"Esperance XY : {esp_XY:.4f}")
    print(f"Covariance estimee : {covariance:.4f}")

    assert abs(covariance) < 0.05, "La covariance doit etre proche de 0"

estimer_covariance_dependante(10000)
\end{lstlisting}
