# TP 2 : Le contre-exemple de Bernstein (Indépendance 2 à 2 vs Mutuelle)

## Objectif
Nous allons simuler l'expérience classique démontrant que l'indépendance deux à deux n'implique pas l'indépendance mutuelle.
On jette deux pièces équilibrées.
$A$ = "La 1ère pièce est Pile"
$B$ = "La 2ème pièce est Pile"
$C$ = "Les deux pièces donnent le même résultat"
Nous vérifierons que $\mathbb{P}(A \cap B \cap C) \neq \mathbb{P}(A)\mathbb{P}(B)\mathbb{P}(C)$.

## Implémentation Python pur

```python
import random

def lancer_pieces():
    # Simule le jet de deux pieces. 1 pour Pile, 0 pour Face.
    p1 = random.choice([0, 1])
    p2 = random.choice([0, 1])
    return p1, p2

def simulation_bernstein(nb_tirages):
    cpt_A = 0
    cpt_B = 0
    cpt_C = 0
    cpt_AB = 0
    cpt_AC = 0
    cpt_BC = 0
    cpt_ABC = 0

    for _ in range(nb_tirages):
        p1, p2 = lancer_pieces()

        est_A = (p1 == 1)
        est_B = (p2 == 1)
        est_C = (p1 == p2)

        if est_A: cpt_A += 1
        if est_B: cpt_B += 1
        if est_C: cpt_C += 1
        if est_A and est_B: cpt_AB += 1
        if est_A and est_C: cpt_AC += 1
        if est_B and est_C: cpt_BC += 1
        if est_A and est_B and est_C: cpt_ABC += 1

    freqs = {
        'A': cpt_A / nb_tirages,
        'B': cpt_B / nb_tirages,
        'C': cpt_C / nb_tirages,
        'AB': cpt_AB / nb_tirages,
        'AC': cpt_AC / nb_tirages,
        'BC': cpt_BC / nb_tirages,
        'ABC': cpt_ABC / nb_tirages
    }
    return freqs

nb_sim = 500000
freq = simulation_bernstein(nb_sim)

print("Independance deux a deux :")
print(f"P(A)*P(B) = {freq['A']*freq['B']:.4f}, P(AB) = {freq['AB']:.4f}")
print(f"P(A)*P(C) = {freq['A']*freq['C']:.4f}, P(AC) = {freq['AC']:.4f}")
print(f"P(B)*P(C) = {freq['B']*freq['C']:.4f}, P(BC) = {freq['BC']:.4f}")

print("\nIndependance mutuelle :")
prod_abc = freq['A'] * freq['B'] * freq['C']
print(f"P(A)*P(B)*P(C) = {prod_abc:.4f}")
print(f"P(A inter B inter C) = {freq['ABC']:.4f}")

# Assertions
assert abs(freq['A']*freq['B'] - freq['AB']) < 0.01, "A et B devraient etre independants"
assert abs(freq['A']*freq['C'] - freq['AC']) < 0.01, "A et C devraient etre independants"
assert abs(freq['B']*freq['C'] - freq['BC']) < 0.01, "B et C devraient etre independants"

# Assertion cruciale : P(ABC) est different du produit !
assert abs(prod_abc - freq['ABC']) > 0.05, "L'independance mutuelle devrait echouer !"
print("\nValidation reussie : Les evenements sont independants deux a deux, mais PAS mutuellement.")
```
