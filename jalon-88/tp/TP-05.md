\subsection*{TP 5 : Produit empirique des esperances}

Nous verifions par la simulation numerique que l'esperance du produit de deux variables aleatoires independantes est le produit de leurs esperances, en utilisant des variables continues uniformes.

\begin{lstlisting}
import random

def verifier_theoreme_esperance(n_samples):
    somme_X = 0
    somme_Y = 0
    somme_XY = 0

    for _ in range(n_samples):
        # X uniforme sur [0, 2] -> Esperance theorique = 1
        X = random.uniform(0, 2)
        # Y uniforme sur [2, 6] -> Esperance theorique = 4
        Y = random.uniform(2, 6)

        produit = X * Y

        somme_X += X
        somme_Y += Y
        somme_XY += produit

    esp_X = somme_X / n_samples
    esp_Y = somme_Y / n_samples
    esp_XY = somme_XY / n_samples

    produit_des_esp = esp_X * esp_Y

    print(f"E[X] estimee : {esp_X:.4f}")
    print(f"E[Y] estimee : {esp_Y:.4f}")
    print(f"E[XY] estimee : {esp_XY:.4f}")
    print(f"E[X]*E[Y] calcule : {produit_des_esp:.4f}")

    assert abs(esp_XY - produit_des_esp) < 0.05, "Le theoreme n'est pas verifie empiriquement"

verifier_theoreme_esperance(20000)
\end{lstlisting}
