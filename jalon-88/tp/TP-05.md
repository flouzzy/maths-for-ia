# Le classifieur Naive Bayes (Factorisation conditionnelle par l'indépendance)

Construction algorithmique pure d'un classifieur Naive Bayes. Ce code démontre comment l'hypothèse d'indépendance conditionnelle permet de simplifier radicalement l'apprentissage probabiliste multivarié.

```python
def entrainer_naive_bayes_binaire(dataset):
    """
    dataset : liste de tuples (features, label)
    features : tuple de booleens (x1, x2, x3)
    label : booleen
    """
    total_docs = len(dataset)
    count_class_1 = sum(1 for _, y in dataset if y)
    count_class_0 = total_docs - count_class_1

    prior_1 = count_class_1 / total_docs
    prior_0 = count_class_0 / total_docs

    # Probabilites conditionnelles P(x_i=1 | y)
    counts_features_given_1 = [0, 0, 0]
    counts_features_given_0 = [0, 0, 0]

    for x, y in dataset:
        for i in range(3):
            if y:
                counts_features_given_1[i] += x[i]
            else:
                counts_features_given_0[i] += x[i]

    # Ajout d'un lissage de Laplace (add-1) pour eviter les probas a 0
    p_x_given_1 = [(c + 1) / (count_class_1 + 2) for c in counts_features_given_1]
    p_x_given_0 = [(c + 1) / (count_class_0 + 2) for c in counts_features_given_0]

    return prior_0, prior_1, p_x_given_0, p_x_given_1

def predire_naive_bayes(x, priors):
    prior_0, prior_1, p_x_given_0, p_x_given_1 = priors

    # Calcul de la vraisemblance par LE PRODUIT DES PROBABILITES (L'hypothese d'independance naive)
    likelihood_1 = prior_1
    likelihood_0 = prior_0

    for i in range(3):
        likelihood_1 *= p_x_given_1[i] if x[i] else (1 - p_x_given_1[i])
        likelihood_0 *= p_x_given_0[i] if x[i] else (1 - p_x_given_0[i])

    return likelihood_1 > likelihood_0

# Donnees d'entrainement factices (x1, x2, x3), label
data = [
    ((1, 1, 0), True),
    ((1, 0, 0), True),
    ((0, 1, 0), False),
    ((0, 0, 1), False),
    ((1, 1, 1), True),
    ((0, 0, 0), False)
]

modele = entrainer_naive_bayes_binaire(data)
# Le theoreme de Bayes et l'hypothese d'independance predisent que (1,0,0) appartient a True
prediction = predire_naive_bayes((1, 0, 0), modele)
print(f"Prediction pour (1,0,0) : Classe {prediction}")

assert prediction == True, "La factorisation probabiliste devrait orienter vers la classe True"
```
