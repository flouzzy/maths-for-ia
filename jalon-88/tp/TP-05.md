# TP 05 : Minimum de variables uniformes indépendantes

**Objectif :** Valider expérimentalement la fonction de répartition du minimum de $n$ variables uniformes indépendantes.

```python
import random

def simulate_min_uniform(n_vars, n_simulations):
    empirical_probs = {}
    x_test_points = [0.1, 0.3, 0.5, 0.7, 0.9]

    for x in x_test_points:
        empirical_probs[x] = 0

    for _ in range(n_simulations):
        # Tirage de n variables U(0,1) independantes
        samples = [random.random() for _ in range(n_vars)]
        min_val = min(samples)

        for x in x_test_points:
            if min_val <= x:
                empirical_probs[x] += 1

    for x in x_test_points:
        empirical_probs[x] /= n_simulations

    return empirical_probs

def test_min_uniform():
    n_vars = 5
    n_sim = 50000
    empirical_probs = simulate_min_uniform(n_vars, n_sim)

    for x, emp_p in empirical_probs.items():
        # F(x) = 1 - (1-x)^n
        theoretical_p = 1 - (1 - x)**n_vars
        assert abs(emp_p - theoretical_p) < 0.01, f"Erreur pour x={x}"

    print("Test reussi : La fonction de repartition empirique du minimum correspond au calcul analytique.")

if __name__ == "__main__":
    test_min_uniform()
```
