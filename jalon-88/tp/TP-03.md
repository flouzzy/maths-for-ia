\subsection*{TP 3 : Somme de variables de Poisson}

Nous allons generer empiriquement la somme de deux variables de Poisson independantes et verifier que la moyenne empirique correspond a la somme des parametres.
Pour generer une variable de Poisson sans bibliotheque avancee, on utilise la methode de l'inverse de la fonction de repartition.

\begin{lstlisting}
import math
import random

def poisson_random(lambd):
    L = math.exp(-lambd)
    k = 0
    p = 1.0
    while p > L:
        k += 1
        p *= random.random()
    return k - 1

def verifier_somme_poisson(lambda1, lambda2, n_tirages):
    somme_empirique = 0

    for _ in range(n_tirages):
        X = poisson_random(lambda1)
        Y = poisson_random(lambda2)
        Z = X + Y
        somme_empirique += Z

    moyenne_estimee = somme_empirique / n_tirages
    moyenne_theorique = lambda1 + lambda2

    print(f"Moyenne theorique Z : {moyenne_theorique}")
    print(f"Moyenne estimee Z : {moyenne_estimee:.4f}")

    assert abs(moyenne_estimee - moyenne_theorique) < 0.1, "L'estimation diverge"

verifier_somme_poisson(2.5, 3.5, 10000)
\end{lstlisting}
