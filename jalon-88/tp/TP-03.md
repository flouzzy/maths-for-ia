# Stabilité de la loi de Poisson par somme de variables indépendantes

Ce script vérifie numériquement le théorème fondamental démontrant que la somme de deux variables de Poisson indépendantes de paramètres respectifs a et b suit une loi de Poisson de paramètre a+b.

```python
import random
import math

def simuler_poisson(lam):
    """Simulateur naif d'une loi de Poisson par le temps d'attente exponentiel."""
    L = math.exp(-lam)
    k = 0
    p = 1.0
    while p > L:
        k += 1
        p *= random.random()
    return k - 1

def valider_somme_poisson(lam1, lam2, n_samples):
    """
    Génère X ~ P(lam1) et Y ~ P(lam2) indépendantes.
    Vérifie si S = X + Y a les propriétés empiriques de P(lam1 + lam2).
    """
    somme_S = 0
    somme_S_carre = 0

    for _ in range(n_samples):
        X = simuler_poisson(lam1)
        Y = simuler_poisson(lam2)
        S = X + Y
        somme_S += S
        somme_S_carre += S * S

    esp_S = somme_S / n_samples
    var_S = (somme_S_carre / n_samples) - (esp_S * esp_S)

    return esp_S, var_S

L1 = 3.0
L2 = 4.5
N = 100000
esp_empirique, var_empirique = valider_somme_poisson(L1, L2, N)

theorie = L1 + L2
print(f"Esperance empirique de S : {esp_empirique:.4f} (Theorie = {theorie})")
print(f"Variance empirique de S : {var_empirique:.4f} (Theorie = {theorie})")

# Assertions confirmant que E[X] = Var(X) = lambda pour une Poisson
epsilon = 0.1
assert abs(esp_empirique - theorie) < epsilon, "L'esperance n'est pas stable."
assert abs(var_empirique - theorie) < epsilon, "La variance n'est pas stable."
```
