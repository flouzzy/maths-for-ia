# TP 03 : Covariance nulle de variables dépendantes

**Objectif :** Implémenter numériquement l'exemple où une covariance vaut zéro bien que les deux variables soient strictement dépendantes.

```python
import random

def simulate_dependent_zero_cov(n_simulations):
    sum_x = 0
    sum_y = 0
    sum_xy = 0
    count_x0_y1 = 0

    for _ in range(n_simulations):
        # Loi uniforme sur -1, 0, 1
        x = random.choice([-1, 0, 1])
        y = abs(x)

        sum_x += x
        sum_y += y
        sum_xy += (x * y)

        if x == 0 and y == 1:
            count_x0_y1 += 1

    mean_x = sum_x / n_simulations
    mean_y = sum_y / n_simulations
    mean_xy = sum_xy / n_simulations
    cov_xy = mean_xy - (mean_x * mean_y)
    prob_x0_y1 = count_x0_y1 / n_simulations

    return mean_x, mean_y, cov_xy, prob_x0_y1

def test_covariance():
    mean_x, mean_y, cov_xy, p_x0_y1 = simulate_dependent_zero_cov(50000)

    # Validations mathématiques
    assert abs(mean_x) < 0.05, "L'esperance de X doit etre 0"
    assert abs(mean_y - 2/3) < 0.05, "L'esperance de Y doit etre 2/3"
    assert abs(cov_xy) < 0.05, "La covariance doit etre nulle"

    p_x0 = 1/3
    p_y1 = 2/3
    assert abs(p_x0_y1 - 0) < 0.01, "P(X=0 et Y=1) doit valoir 0"
    assert abs(p_x0_y1 - (p_x0 * p_y1)) > 0.1, "Les variables sont dependantes !"

    print("Test reussi : Covariance nulle malgre la dependance absolue (Y = |X|).")

if __name__ == "__main__":
    test_covariance()
```
