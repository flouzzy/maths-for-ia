# TP 3 : Covariance nulle sans indépendance

## Objectif
Nous allons implémenter la construction de l'exercice 6 pour montrer empiriquement que deux variables aléatoires peuvent avoir une covariance strictement nulle (être non-corrélées) tout en étant totalement dépendantes.
Soit $X$ uniforme sur $\{-1, 0, 1\}$.
Soit $Y = X^2$.

## Implémentation Python pur

```python
import random

def generer_X():
    # Genere X selon la loi decrite : -1 avec proba 1/4, 0 avec proba 1/2, 1 avec proba 1/4.
    u = random.random()
    if u < 0.25:
        return -1
    elif u < 0.75:
        return 0
    else:
        return 1

def simuler_cov_independance(nb_tirages):
    somme_X = 0
    somme_Y = 0
    somme_XY = 0

    cpt_X0 = 0
    cpt_Y1 = 0
    cpt_X0_et_Y1 = 0

    for _ in range(nb_tirages):
        x = generer_X()
        y = x**2

        somme_X += x
        somme_Y += y
        somme_XY += (x * y)

        if x == 0:
            cpt_X0 += 1
        if y == 1:
            cpt_Y1 += 1
        if x == 0 and y == 1:
            cpt_X0_et_Y1 += 1

    esp_X = somme_X / nb_tirages
    esp_Y = somme_Y / nb_tirages
    esp_XY = somme_XY / nb_tirages

    cov = esp_XY - (esp_X * esp_Y)

    prob_X0 = cpt_X0 / nb_tirages
    prob_Y1 = cpt_Y1 / nb_tirages
    prob_X0_et_Y1 = cpt_X0_et_Y1 / nb_tirages

    return cov, prob_X0, prob_Y1, prob_X0_et_Y1

nb_sim = 500000
cov_xy, p_X0, p_Y1, p_joint = simuler_cov_independance(nb_sim)

print(f"Covariance(X,Y) estimee : {cov_xy:.6f}")
print(f"P(X=0) * P(Y=1) : {(p_X0 * p_Y1):.4f}")
print(f"P(X=0 et Y=1)   : {p_joint:.4f}")

# Assertions
assert abs(cov_xy) < 0.01, "La covariance devrait etre nulle."
assert abs(p_joint - (p_X0 * p_Y1)) > 0.1, "L'independance devrait etre fausse (non-egalite)."
print("\nAssertion reussie : Covariance nulle, mais forte dependance probabiliste validee.")
```
