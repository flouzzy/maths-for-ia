# Simulation Monte-Carlo de l'indépendance de deux événements

Ce TP vise à simuler informatiquement la notion abstraite d'indépendance de deux événements par une approche fréquentiste de Monte-Carlo, en simulant le lancer de deux dés.

```python
import random

def simuler_lancers_des(n_simulations):
    """
    Simule n lancers de deux des.
    Evenement A : Le de 1 fait un resultat pair.
    Evenement B : Le de 2 fait 5 ou 6.
    """
    count_A = 0
    count_B = 0
    count_A_and_B = 0

    for _ in range(n_simulations):
        de1 = random.randint(1, 6)
        de2 = random.randint(1, 6)

        A_realise = (de1 % 2 == 0)
        B_realise = (de2 in [5, 6])

        if A_realise:
            count_A += 1
        if B_realise:
            count_B += 1
        if A_realise and B_realise:
            count_A_and_B += 1

    prob_A = count_A / n_simulations
    prob_B = count_B / n_simulations
    prob_A_and_B = count_A_and_B / n_simulations
    prob_produit = prob_A * prob_B

    return prob_A, prob_B, prob_A_and_B, prob_produit

# Execution avec un grand nombre de simulations
N = 1000000
p_a, p_b, p_inter, p_prod = simuler_lancers_des(N)

print(f"Probabilite empirique P(A) = {p_a:.4f} (Theorie: 0.5000)")
print(f"Probabilite empirique P(B) = {p_b:.4f} (Theorie: 0.3333)")
print(f"Probabilite empirique P(A inter B) = {p_inter:.4f}")
print(f"Produit empirique P(A) * P(B) = {p_prod:.4f}")

# Assertion pour valider la convergence de Monte-Carlo vers l'independance mathematique absolue
epsilon = 0.01
assert abs(p_inter - p_prod) < epsilon, "Les evenements devraient etre empiriquement independants."
```
