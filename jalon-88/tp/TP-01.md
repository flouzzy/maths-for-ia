\subsection*{TP 1 : Simulation de l'indépendance de dés}

Dans ce TP, nous allons simuler le lancer de deux des et verifier empiriquement leur independance.
Nous calculerons la frequence d'apparition conjointe et la comparerons au produit des frequences marginales.

\begin{lstlisting}
import random

def simuler_des(nb_lancers):
    nb_A = 0 # Evt A : de 1 est pair
    nb_B = 0 # Evt B : de 2 est impair
    nb_A_inter_B = 0

    for _ in range(nb_lancers):
        de1 = random.randint(1, 6)
        de2 = random.randint(1, 6)

        est_A = (de1 % 2 == 0)
        est_B = (de2 % 2 != 0)

        if est_A:
            nb_A += 1
        if est_B:
            nb_B += 1
        if est_A and est_B:
            nb_A_inter_B += 1

    freq_A = nb_A / nb_lancers
    freq_B = nb_B / nb_lancers
    freq_inter = nb_A_inter_B / nb_lancers

    print(f"Frequence A : {freq_A:.4f} (Theorie : 0.5)")
    print(f"Frequence B : {freq_B:.4f} (Theorie : 0.5)")
    print(f"Frequence A inter B : {freq_inter:.4f} (Theorie : 0.25)")
    print(f"Produit freq_A * freq_B : {freq_A * freq_B:.4f}")

    assert abs(freq_inter - (freq_A * freq_B)) < 0.05, "Ecart trop grand"

simuler_des(10000)
\end{lstlisting}
