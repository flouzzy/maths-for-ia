# TP 01 : Simulation du jeu de dés et vérification d'indépendance

**Objectif :** Simuler des lancers de dés pour vérifier expérimentalement l'indépendance entre la parité du premier dé et la somme égale à 7.

```python
import random

def simulate_dice(n_simulations):
    count_A = 0
    count_B = 0
    count_A_and_B = 0

    for _ in range(n_simulations):
        x = random.randint(1, 6)
        y = random.randint(1, 6)

        is_A = (x % 2 == 0)
        is_B = (x + y == 7)

        if is_A: count_A += 1
        if is_B: count_B += 1
        if is_A and is_B: count_A_and_B += 1

    prob_A = count_A / n_simulations
    prob_B = count_B / n_simulations
    prob_A_and_B = count_A_and_B / n_simulations

    return prob_A, prob_B, prob_A_and_B

def test_independence():
    n_simulations = 100000
    p_A, p_B, p_AB = simulate_dice(n_simulations)

    # Validation mathématique
    expected_p_A = 1/2
    expected_p_B = 1/6
    expected_p_AB = 1/12

    assert abs(p_A - expected_p_A) < 0.01, "Probabilite de A incorrecte"
    assert abs(p_B - expected_p_B) < 0.01, "Probabilite de B incorrecte"
    assert abs(p_AB - expected_p_AB) < 0.01, "Probabilite de A inter B incorrecte"

    product_pA_pB = p_A * p_B
    assert abs(p_AB - product_pA_pB) < 0.01, "A et B ne semblent pas independants"

    print("Test reussi : Les probabilites simulees correspondent aux probabilites theoriques de l'independance.")

if __name__ == "__main__":
    test_independence()
```
