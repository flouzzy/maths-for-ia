# TP 1 : Vérification empirique de l'indépendance de deux événements

## Objectif
Le but de ce TP est de simuler le jet de deux dés équilibrés à 6 faces et de vérifier empiriquement l'indépendance de deux événements spécifiques définis sur ce lancer.
Soient $A$ l'événement "Le résultat du premier dé est pair" et $B$ l'événement "La somme des deux dés vaut 7". Nous allons estimer les probabilités par la méthode de Monte-Carlo et vérifier que $\mathbb{P}(A \cap B) \approx \mathbb{P}(A)\mathbb{P}(B)$.

## Implémentation Python pur

```python
import random

def lancer_deux_des():
    # Simule le jet de deux des a 6 faces.
    de1 = random.randint(1, 6)
    de2 = random.randint(1, 6)
    return de1, de2

def verifier_evenement_A(de1, de2):
    # Evenement A : le premier de est pair.
    return de1 % 2 == 0

def verifier_evenement_B(de1, de2):
    # Evenement B : la somme des deux des vaut 7.
    return de1 + de2 == 7

def simulation_independance(nb_tirages):
    compteur_A = 0
    compteur_B = 0
    compteur_A_et_B = 0

    for _ in range(nb_tirages):
        d1, d2 = lancer_deux_des()
        est_A = verifier_evenement_A(d1, d2)
        est_B = verifier_evenement_B(d1, d2)

        if est_A:
            compteur_A += 1
        if est_B:
            compteur_B += 1
        if est_A and est_B:
            compteur_A_et_B += 1

    # Calcul des probabilites empiriques
    prob_A = compteur_A / nb_tirages
    prob_B = compteur_B / nb_tirages
    prob_A_et_B = compteur_A_et_B / nb_tirages

    return prob_A, prob_B, prob_A_et_B

# Validation
nb_simulations = 500000
p_A, p_B, p_A_et_B = simulation_independance(nb_simulations)

produit = p_A * p_B

print(f"Probabilite empirique de A : {p_A:.4f} (Theorique: 0.5000)")
print(f"Probabilite empirique de B : {p_B:.4f} (Theorique: 0.1667)")
print(f"Probabilite empirique de A inter B : {p_A_et_B:.4f} (Theorique: 0.0833)")
print(f"Produit P(A)*P(B) : {produit:.4f}")

# Assertion mathematique a 1% de tolerance
assert abs(p_A_et_B - produit) < 0.01, "L'independance n'est pas verifiee par la simulation."
print("Assertion reussie : P(A inter B) est approximativement egal a P(A)P(B).")
```
