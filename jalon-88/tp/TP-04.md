# TP 04 : Somme de deux variables de Poisson indépendantes

**Objectif :** Vérifier expérimentalement que la somme de deux variables de Poisson indépendantes suit une loi de Poisson dont le paramètre est la somme des paramètres.

```python
import math
import random

def sample_poisson(lambda_param):
    L = math.exp(-lambda_param)
    k = 0
    p = 1.0
    while True:
        p = p * random.random()
        if p <= L:
            return k
        k += 1

def simulate_poisson_sum():
    lambda_1 = 2.0
    lambda_2 = 3.0
    lambda_sum = lambda_1 + lambda_2

    n_simulations = 50000
    counts = {}

    for _ in range(n_simulations):
        x = sample_poisson(lambda_1)
        y = sample_poisson(lambda_2)
        z = x + y
        counts[z] = counts.get(z, 0) + 1

    def theoretical_prob(k, lam):
        return (math.exp(-lam) * (lam ** k)) / math.factorial(k)

    for k in range(10):
        empirical_p = counts.get(k, 0) / n_simulations
        theoretical_p = theoretical_prob(k, lambda_sum)
        assert abs(empirical_p - theoretical_p) < 0.01, f"Ecart detecte pour Z={k}"

    print("Test reussi : Z = X+Y suit bien la loi de Poisson de parametre lambda_1 + lambda_2.")

if __name__ == "__main__":
    simulate_poisson_sum()
```
