# TP 4 : Stabilité de la loi de Poisson par somme de variables indépendantes

## Objectif
Si $X_1 \sim \mathcal{P}(\lambda_1)$ et $X_2 \sim \mathcal{P}(\lambda_2)$ sont indépendantes, alors leur somme $Y = X_1 + X_2$ suit une loi de Poisson $\mathcal{P}(\lambda_1 + \lambda_2)$.
Nous allons vérifier cela en simulant ces variables aléatoires à partir de tirages uniformes (méthode de l'inversion de la fonction de répartition).

## Implémentation Python pur

```python
import random
import math

def tirage_poisson(lam):
    # Genere une variable aleatoire suivant une loi de Poisson de parametre lambda.
    L = math.exp(-lam)
    p = 1.0
    k = 0
    while p > L:
        k += 1
        p *= random.random()
    return k - 1

def simulation_somme_poisson(lam1, lam2, nb_tirages):
    frequences_Y = {}

    somme_empirique_Y = 0

    for _ in range(nb_tirages):
        x1 = tirage_poisson(lam1)
        x2 = tirage_poisson(lam2)
        y = x1 + x2

        somme_empirique_Y += y
        if y in frequences_Y:
            frequences_Y[y] += 1
        else:
            frequences_Y[y] = 1

    esp_empirique = somme_empirique_Y / nb_tirages
    return esp_empirique, frequences_Y

lam1 = 2.0
lam2 = 3.5
nb_sim = 200000

print(f"Simulation de X1~P({lam1}) et X2~P({lam2})")
esp_y, freq_y = simulation_somme_poisson(lam1, lam2, nb_sim)

print(f"Esperance empirique de la somme : {esp_y:.4f}")
print(f"Parametre theorique attendu : {lam1 + lam2}")

# Verifions pour k=5
k_test = 5
prob_empirique = freq_y.get(k_test, 0) / nb_sim
lam_tot = lam1 + lam2
prob_theorique = math.exp(-lam_tot) * (lam_tot**k_test) / math.factorial(k_test)

print(f"P(Y={k_test}) empirique : {prob_empirique:.4f}")
print(f"P(Y={k_test}) theorique : {prob_theorique:.4f}")

# Assertions
assert abs(esp_y - lam_tot) < 0.05, "L'esperance ne correspond pas au parametre somme."
assert abs(prob_empirique - prob_theorique) < 0.01, "La distribution empirique differe de la theorie."
print("Validation reussie : La somme de deux Poisson independantes est une Poisson.")
```
