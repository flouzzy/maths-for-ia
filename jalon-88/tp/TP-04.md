\subsection*{TP 4 : Simulation du Dropout (Independance forcee)}

Ce TP simule le mecanisme de Dropout d'un reseau de neurones simplifie.
Nous appliquons un masque binaire aleatoire et independant a chaque neurone d'une couche cachee lors de la transmission du signal.

\begin{lstlisting}
import random

def relu(x):
    return x if x > 0 else 0

def forward_avec_dropout(entrees, poids, p_dropout):
    assert len(entrees) == len(poids)
    n = len(entrees)

    masque = []
    for _ in range(n):
        # Tirage de Bernoulli avec proba (1 - p_dropout) d'etre garde
        if random.random() > p_dropout:
            masque.append(1)
        else:
            masque.append(0)

    # Le facteur d'echelle corrige l'esperance pendant l'entrainement
    facteur = 1.0 / (1.0 - p_dropout)

    signal = 0
    for i in range(n):
        activation_neurone = relu(entrees[i] * poids[i])
        signal += activation_neurone * masque[i] * facteur

    return signal, masque

entrees = [1.5, -0.5, 2.0, 3.1, -1.0]
poids = [0.8, 1.2, -0.9, 0.5, 1.1]

signal_out, masque_applique = forward_avec_dropout(entrees, poids, 0.4)
print(f"Masque binaire aleatoire independant : {masque_applique}")
print(f"Signal aggrege resultant : {signal_out:.4f}")
\end{lstlisting}
