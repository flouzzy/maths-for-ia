# TP 01 : Entropie de Shannon et incertitude

**Objectif :** Implémenter le calcul de l'entropie de Shannon et vérifier ses propriétés fondamentales.

```python
import math

def shannon_entropy(probabilities):
    """
    Calcule l'entropie de Shannon d'une distribution de probabilités discrète.
    H(P) = - sum(p * log(p))
    """
    # Validation des probabilités
    assert abs(sum(probabilities) - 1.0) < 1e-6, "Les probabilités doivent sommer à 1."
    assert all(p >= 0 for p in probabilities), "Toutes les probabilités doivent être positives."

    entropy = 0.0
    for p in probabilities:
        if p > 0: # Limite p log p = 0 quand p -> 0
            entropy -= p * math.log(p) # Utilisation du log naturel (nats)

    return entropy

# --- Validation Mathématique ---

# 1. Distribution déterministe (Entropie nulle)
p_det = [1.0, 0.0, 0.0]
h_det = shannon_entropy(p_det)
assert h_det == 0.0, "L'entropie d'un événement certain doit être nulle."

# 2. Distribution uniforme (Entropie maximale)
n_states = 4
p_unif = [1.0/n_states] * n_states
h_unif = shannon_entropy(p_unif)
expected_max_entropy = math.log(n_states)
assert abs(h_unif - expected_max_entropy) < 1e-9, "L'entropie d'une loi uniforme n'est pas maximale (log(N))."

# 3. Comparaison
p_biased = [0.8, 0.1, 0.05, 0.05]
h_biased = shannon_entropy(p_biased)
assert h_biased < h_unif, "L'entropie d'une distribution biaisée doit être inférieure à celle de la loi uniforme."

print("Validation mathématique réussie : Les axiomes de Shannon sont vérifiés.")
```
