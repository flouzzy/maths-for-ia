# TP 03 : Cross-Entropy et lien fondamental $H(P, Q) = H(P) + D_{KL}(P || Q)$

**Objectif :** Vérifier expérimentalement la relation fondamentale de la théorie de l'information.

```python
import math

def shannon_entropy(p):
    return -sum(px * math.log(px) for px in p if px > 0)

def kl_divergence(p, q):
    return sum(p[i] * math.log(p[i] / q[i]) for i in range(len(p)) if p[i] > 0)

def cross_entropy(p, q):
    """
    Calcule l'entropie croisée H(P, Q) = -E_P[log Q].
    """
    assert len(p) == len(q), "Tailles différentes."

    ce = 0.0
    for i in range(len(p)):
        if p[i] > 0:
            assert q[i] > 0, "Log de zéro impossible."
            ce -= p[i] * math.log(q[i])

    return ce

# --- Validation Mathématique ---

P = [0.5, 0.25, 0.25]
Q = [0.1, 0.7, 0.2]

# Calculs séparés
h_p = shannon_entropy(P)
d_kl = kl_divergence(P, Q)
h_pq = cross_entropy(P, Q)

# Vérification du théorème
theorem_val = h_p + d_kl

print(f"H(P) = {h_p:.4f}")
print(f"D_KL(P || Q) = {d_kl:.4f}")
print(f"H(P) + D_KL(P || Q) = {theorem_val:.4f}")
print(f"H(P, Q) = {h_pq:.4f}")

assert abs(h_pq - theorem_val) < 1e-9, "Erreur dans la relation fondamentale."
print("Validation mathématique réussie : H(P, Q) = H(P) + D_KL(P || Q).")
```
