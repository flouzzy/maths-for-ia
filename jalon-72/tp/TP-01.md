# TP 1 : Implémentation Divergence KL basique

```python
import math

def kl_divergence(p, q):
    # p et q sont des listes de probabilités
    assert len(p) == len(q)
    kl = 0.0
    for pi, qi in zip(p, q):
        if pi > 0 and qi > 0:
            kl += pi * math.log(pi / qi)
        elif pi > 0 and qi == 0:
            return float('inf')
    return kl

# Test
p = [0.5, 0.5]
q = [0.1, 0.9]
res = kl_divergence(p, q)
assert res > 0
print(f"KL(P||Q) = {res}")
```
