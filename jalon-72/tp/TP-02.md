# TP 2 : Asymétrie de la Divergence KL

```python
import math

def kl_divergence(p, q):
    kl = 0.0
    for pi, qi in zip(p, q):
        if pi > 0 and qi > 0:
            kl += pi * math.log(pi / qi)
        elif pi > 0 and qi == 0:
            return float('inf')
    return kl

p = [0.2, 0.8]
q = [0.5, 0.5]

kl_pq = kl_divergence(p, q)
kl_qp = kl_divergence(q, p)

print(f"KL(P||Q) = {kl_pq}")
print(f"KL(Q||P) = {kl_qp}")

assert kl_pq != kl_qp, "La divergence KL doit etre asymetrique"
```
