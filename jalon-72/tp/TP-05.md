# TP 5 : Divergence de Jensen-Shannon

```python
import math

def kl_divergence(p, q):
    kl = 0.0
    for pi, qi in zip(p, q):
        if pi > 0 and qi > 0:
            kl += pi * math.log(pi / qi)
    return kl

def jensen_shannon(p, q):
    m = [(pi + qi) / 2.0 for pi, qi in zip(p, q)]
    return 0.5 * kl_divergence(p, m) + 0.5 * kl_divergence(q, m)

p = [1.0, 0.0]
q = [0.0, 1.0]

jsd = jensen_shannon(p, q)
print(f"JSD(P||Q) pour distributions disjointes : {jsd}")

# La borne max est ln(2)
ln2 = math.log(2)
assert abs(jsd - ln2) < 1e-9
```
