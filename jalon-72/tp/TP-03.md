# TP 3 : KL entre deux lois normales (formule fermée)

```python
def kl_normales(mu1, var1, mu2, var2):
    import math
    # Formule deduite de l'Exercice 6
    term1 = math.log(var2 / var1)
    term2 = (var1 + (mu1 - mu2)**2) / var2
    term3 = -1
    return 0.5 * (term1 + term2 + term3)

# Test pour P=N(0,1) et Q=N(0,1)
assert kl_normales(0, 1, 0, 1) == 0.0

# Test pour P=N(0,1) et Q=N(1,1)
res = kl_normales(0, 1, 1, 1)
assert res == 0.5
print(f"KL(N(0,1) || N(1,1)) = {res}")
```
