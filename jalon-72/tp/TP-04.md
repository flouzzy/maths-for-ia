# TP 4 : Entropie croisée et KL

```python
import math

def cross_entropy(p, q):
    ce = 0.0
    for pi, qi in zip(p, q):
        if pi > 0:
            assert qi > 0, "q doit avoir un support couvrant p"
            ce -= pi * math.log(qi)
    return ce

def entropy(p):
    ent = 0.0
    for pi in p:
        if pi > 0:
            ent -= pi * math.log(pi)
    return ent

def kl_divergence(p, q):
    kl = 0.0
    for pi, qi in zip(p, q):
        if pi > 0:
            kl += pi * math.log(pi / qi)
    return kl

p = [0.3, 0.7]
q = [0.5, 0.5]

ce = cross_entropy(p, q)
h_p = entropy(p)
kl = kl_divergence(p, q)

# Verifions H(P, Q) = H(P) + KL(P||Q)
diff = abs(ce - (h_p + kl))
assert diff < 1e-9
print("Egalite verifiee H(P, Q) == H(P) + KL(P||Q)")
```
