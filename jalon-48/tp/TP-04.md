# TP 04 : Softmax et entropie croisée

## Objectif
Ce TP implémente la stabilité numérique de Softmax et le calcul élégant de son gradient combiné avec l'entropie croisée.

## Code Python

```python
import math

def softmax(z):
    # Stabilisation numérique en soustrayant le maximum
    max_z = max(z)
    exp_z = [math.exp(i - max_z) for i in z]
    sum_exp = sum(exp_z)
    return [i / sum_exp for i in exp_z]

def cross_entropy_loss(y_pred, y_true):
    # On suppose y_true one-hot
    loss = 0.0
    for p, t in zip(y_pred, y_true):
        if t == 1.0:
            loss -= math.log(p + 1e-9) # Éviter log(0)
    return loss

def softmax_cross_entropy_backward(y_pred, y_true):
    # Règle démontrée : dL/dz = y_pred - y_true
    return [p - t for p, t in zip(y_pred, y_true)]

if __name__ == '__main__':
    z = [2.0, 1.0, 0.1]
    y_true = [1.0, 0.0, 0.0]

    y_pred = softmax(z)
    print(f"Prédictions Softmax : {y_pred}")

    loss = cross_entropy_loss(y_pred, y_true)
    print(f"Perte Cross-Entropy : {loss}")

    dz = softmax_cross_entropy_backward(y_pred, y_true)
    print(f"Gradient (dL/dz) : {dz}")

    # Assertions
    assert abs(sum(y_pred) - 1.0) < 1e-6
    # Le gradient de la bonne classe doit être négatif pour forcer z à augmenter
    assert dz[0] < 0
```
