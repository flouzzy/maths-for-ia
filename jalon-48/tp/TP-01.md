# TP 1 : Rétropropagation Numpy - Architecture 1 $\bigstar$$\bigstar$$\star$$\star$$\star$

**Objectif :**
Implémenter la propagation avant et la rétropropagation des erreurs pour un réseau de neurones multicouche en Python pur.
Dans ce TP, l'architecture testée a 3 couches cachées.

```python
import numpy as np

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def sigmoid_prime(x):
    s = sigmoid(x)
    return s * (1.0 - s)

class NeuralNetwork_1:
    def __init__(self, layers):
        self.num_layers = len(layers)
        self.biases = [np.random.randn(y, 1) for y in layers[1:]]
        self.weights = [np.random.randn(y, x) for x, y in zip(layers[:-1], layers[1:])]

    def backprop(self, x, y):
        nabla_b = [np.zeros(b.shape) for b in self.biases]
        nabla_w = [np.zeros(w.shape) for w in self.weights]

        # Propagation avant
        activation = x
        activations = [x]
        zs = []
        for b, w in zip(self.biases, self.weights):
            z = np.dot(w, activation) + b
            zs.append(z)
            activation = sigmoid(z)
            activations.append(activation)

        # Rétropropagation - Couche de sortie
        delta = (activations[-1] - y) * sigmoid_prime(zs[-1])
        nabla_b[-1] = delta
        nabla_w[-1] = np.dot(delta, activations[-2].transpose())

        # Rétropropagation - Couches cachées
        for l in range(2, self.num_layers):
            z = zs[-l]
            sp = sigmoid_prime(z)
            delta = np.dot(self.weights[-l+1].transpose(), delta) * sp
            nabla_b[-l] = delta
            nabla_w[-l] = np.dot(delta, activations[-l-1].transpose())

        return nabla_b, nabla_w

# Test d'intégrité avec l'architecture spécifique au TP 1
# On utilise 3 couches cachées pour vérifier la propagation sur une plus grande profondeur
layers_config = [2] + [3] * 3 + [1]
nn = NeuralNetwork_1(layers_config)
x = np.array([[0.5], [0.1]])
y = np.array([[1.0]])
nabla_b, nabla_w = nn.backprop(x, y)
assert len(nabla_w) == len(layers_config) - 1
```
