# TP 01 : Implémentation from scratch du neurone linéaire (Forward et Backward)

## Objectif
Ce TP implémente la propagation avant et arrière pour un neurone linéaire avec la perte d'erreur quadratique moyenne.

## Code Python

```python
def linear_forward(x, w, b):
    """
    Calcule la sortie d'un neurone linéaire.
    x : float, l'entrée
    w : float, le poids
    b : float, le biais
    """
    return w * x + b

def mse_loss(y_pred, y_true):
    """
    Calcule l'erreur quadratique moyenne.
    """
    return 0.5 * (y_pred - y_true)**2

def linear_backward(x, y_pred, y_true):
    """
    Calcule les gradients de la perte par rapport à w et b.
    """
    dy_pred = y_pred - y_true # dL/dy_pred
    dw = dy_pred * x
    db = dy_pred * 1.0
    return dw, db

if __name__ == '__main__':
    # Initialisation
    x = 2.0
    y_true = 5.0
    w = 1.0
    b = 0.5

    # Forward pass
    y_pred = linear_forward(x, w, b)
    loss = mse_loss(y_pred, y_true)
    print(f"Prédiction initiale : {y_pred}, Perte : {loss}")

    # Backward pass
    dw, db = linear_backward(x, y_pred, y_true)
    print(f"Gradients -> dw: {dw}, db: {db}")

    # Mise à jour (Descente de gradient)
    lr = 0.1
    w -= lr * dw
    b -= lr * db

    # Vérification
    y_pred_new = linear_forward(x, w, b)
    loss_new = mse_loss(y_pred_new, y_true)
    print(f"Nouvelle prédiction : {y_pred_new}, Nouvelle perte : {loss_new}")

    # Assertions de validation
    assert loss_new < loss, "La perte n'a pas diminué après la mise à jour."
    assert dw == -2.5 * 2.0
```
