# TP 05 : Intégration globale - Un MLP complet à 2 couches

## Objectif
Ce TP assemble tous les composants précédents pour entraîner un Perceptron Multicouche sur un problème jouet.

## Code Python

```python
import math

# Réutilisation des composants précédents
def relu(x): return [max(0.0, i) for i in x]
def relu_deriv(x): return [1.0 if i > 0 else 0.0 for i in x]

def mat_vec_mult(W, x): return [sum(w*i for w, i in zip(row, x)) for row in W]
def vec_add(v1, v2): return [x+y for x, y in zip(v1, v2)]
def transpose(mat): return [[mat[j][i] for j in range(len(mat))] for i in range(len(mat[0]))]
def outer_product(v1, v2): return [[i*j for j in v2] for i in v1]

def softmax(z):
    max_z = max(z)
    exp_z = [math.exp(i - max_z) for i in z]
    sum_exp = sum(exp_z)
    return [i / sum_exp for i in exp_z]

if __name__ == '__main__':
    # Architecture : Entrée (2) -> Cachée (3, ReLU) -> Sortie (2, Softmax)
    x = [0.5, 0.8]
    y_true = [0.0, 1.0] # Classe 2

    W1 = [[0.1, -0.2], [0.3, 0.4], [-0.1, 0.1]]
    b1 = [0.1, 0.1, -0.1]

    W2 = [[0.2, 0.5, -0.3], [0.1, -0.1, 0.2]]
    b2 = [0.0, 0.0]

    lr = 0.1

    for epoch in range(5):
        # FORWARD
        z1 = vec_add(mat_vec_mult(W1, x), b1)
        a1 = relu(z1)

        z2 = vec_add(mat_vec_mult(W2, a1), b2)
        a2 = softmax(z2)

        loss = -sum(t*math.log(p + 1e-9) for p, t in zip(a2, y_true))
        print(f"Epoch {epoch} - Perte : {loss:.4f}")

        # BACKWARD
        # Couche 2
        dz2 = [p - t for p, t in zip(a2, y_true)]
        dW2 = outer_product(dz2, a1)
        db2 = dz2

        # Couche 1
        da1 = mat_vec_mult(transpose(W2), dz2)
        dz1 = [da * dr for da, dr in zip(da1, relu_deriv(z1))]
        dW1 = outer_product(dz1, x)
        db1 = dz1

        # UPDATE
        for i in range(len(W2)):
            for j in range(len(W2[0])): W2[i][j] -= lr * dW2[i][j]
        for i in range(len(b2)): b2[i] -= lr * db2[i]

        for i in range(len(W1)):
            for j in range(len(W1[0])): W1[i][j] -= lr * dW1[i][j]
        for i in range(len(b1)): b1[i] -= lr * db1[i]

    # Vérification
    assert loss < 1.0, "Le réseau ne semble pas converger."
```
