# TP 02 : Fonctions d'activation et leurs dérivées

## Objectif
Ce TP implémente la Sigmoïde et ReLU ainsi que leurs dérivées pour le calcul du gradient local.

## Code Python

```python
import math

def sigmoid(x):
    return 1.0 / (1.0 + math.exp(-x))

def sigmoid_derivative(x):
    s = sigmoid(x)
    return s * (1 - s)

def relu(x):
    return max(0.0, x)

def relu_derivative(x):
    return 1.0 if x > 0 else 0.0

if __name__ == '__main__':
    x_val = 1.5

    s_val = sigmoid(x_val)
    ds_val = sigmoid_derivative(x_val)
    print(f"Sigmoïde({x_val}) = {s_val}, Dérivée = {ds_val}")

    r_val = relu(x_val)
    dr_val = relu_derivative(x_val)
    print(f"ReLU({x_val}) = {r_val}, Dérivée = {dr_val}")

    x_neg = -2.0
    print(f"ReLU({x_neg}) = {relu(x_neg)}, Dérivée = {relu_derivative(x_neg)}")

    # Assertions
    assert relu_derivative(2.0) == 1.0
    assert relu_derivative(-1.0) == 0.0
    assert abs(sigmoid_derivative(0.0) - 0.25) < 1e-7
```
