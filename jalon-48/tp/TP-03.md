# TP 03 : Calcul Jacobien sur une couche dense vectorielle

## Objectif
Ce TP illustre comment les gradients sont propagés à travers la multiplication matricielle d'une couche dense.

## Code Python

```python
# Pure Python, pas de Numpy (représentation par listes de listes)

def dot_product(v1, v2):
    return sum(x*y for x, y in zip(v1, v2))

def mat_vec_mult(W, x):
    return [dot_product(row, x) for row in W]

def vec_add(v1, v2):
    return [x+y for x, y in zip(v1, v2)]

def transpose(mat):
    return [[mat[j][i] for j in range(len(mat))] for i in range(len(mat[0]))]

def outer_product(v1, v2):
    return [[v1[i] * v2[j] for j in range(len(v2))] for i in range(len(v1))]

def dense_forward(x, W, b):
    return vec_add(mat_vec_mult(W, x), b)

def dense_backward(x, W, dZ):
    # dL/dW = dZ * x^T (produit extérieur)
    dW = outer_product(dZ, x)
    # dL/db = dZ
    db = dZ
    # dL/dx = W^T * dZ
    dx = mat_vec_mult(transpose(W), dZ)
    return dx, dW, db

if __name__ == '__main__':
    # x dimension 2
    x = [1.0, -1.0]
    # Couche de dimension 2 vers 3
    W = [
        [0.5, -0.2],
        [0.0, 1.0],
        [-0.5, 0.5]
    ]
    b = [0.1, 0.1, -0.1]

    # Forward
    z = dense_forward(x, W, b)
    print(f"Sortie z : {z}")

    # Supposons un gradient entrant (dL/dz)
    dZ = [1.0, -1.0, 0.5]

    # Backward
    dx, dW, db = dense_backward(x, W, dZ)

    print(f"Gradient entrant propagé (dx) : {dx}")
    print(f"Gradient des poids (dW) : {dW}")
    print(f"Gradient du biais (db) : {db}")

    # Assertions
    assert db == dZ
    assert dW[0][0] == dZ[0] * x[0]
```
