# TP 4 : Espace des fonctions continues C[0,1]

**Objectif :** Approcher numériquement la norme uniforme (infinie) d'une fonction et montrer que la dérivation est un opérateur non continu pour cette norme.

```python
def uniform_norm(f, n_points=1000):
    # On discretise [0, 1]
    step = 1.0 / n_points
    max_val = 0.0
    for i in range(n_points + 1):
        x = i * step
        val = abs(f(x))
        if val > max_val:
            max_val = val
    return max_val

# Suite de fonctions f_n(x) = x^n
def make_f(n):
    return lambda x: x ** n

def make_df(n):
    return lambda x: n * (x ** (n - 1))

norms_f = []
norms_df = []
ratios = []

for n in range(1, 10):
    f_n = make_f(n)
    df_n = make_df(n)

    norm_f = uniform_norm(f_n)
    norm_df = uniform_norm(df_n)
    ratio = norm_df / norm_f if norm_f != 0 else 0

    norms_f.append(norm_f)
    norms_df.append(norm_df)
    ratios.append(ratio)

print("Ratios ||Df_n|| / ||f_n||:", ratios)
# Le ratio grandit proportionnellement a n, prouvant la non-continuite
assert ratios[-1] > 8.0
```
