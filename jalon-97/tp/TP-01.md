# TP 1 : Norme d'opérateur d'une matrice

**Objectif :** Implémenter le calcul approché de la norme d'opérateur d'une matrice agissant sur $\mathbb{R}^n$ par la méthode de la puissance itérée.

```python
import math

def norm_2(v):
    return math.sqrt(sum(x * x for x in v))

def mat_vec_mul(A, v):
    n = len(A)
    m = len(v)
    res = [0.0] * n
    for i in range(n):
        for j in range(m):
            res[i] += A[i][j] * v[j]
    return res

def transpose(A):
    n = len(A)
    m = len(A[0])
    res = [[0.0] * n for _ in range(m)]
    for i in range(n):
        for j in range(m):
            res[j][i] = A[i][j]
    return res

def mat_mat_mul(A, B):
    n = len(A)
    m = len(B[0])
    p = len(B)
    res = [[0.0] * m for _ in range(n)]
    for i in range(n):
        for j in range(m):
            for k in range(p):
                res[i][j] += A[i][k] * B[k][j]
    return res

def power_iteration_operator_norm(A, num_iterations=100):
    # La norme d'operateur d'une matrice A est la racine carree de la plus grande valeur propre de A^T A
    At = transpose(A)
    AtA = mat_mat_mul(At, A)

    n = len(AtA)
    # Vecteur initial aleatoire (ici on prend [1, 1, ...])
    v = [1.0] * n

    for _ in range(num_iterations):
        v_next = mat_vec_mul(AtA, v)
        v_norm = norm_2(v_next)
        if v_norm == 0:
            break
        v = [x / v_norm for x in v_next]

    # Rayon spectral de A^T A (valeur propre maximale)
    Av = mat_vec_mul(AtA, v)
    lambda_max = sum(v[i] * Av[i] for i in range(n)) / sum(v[i] * v[i] for i in range(n))

    return math.sqrt(lambda_max)

# Validation
A = [[3.0, 0.0], [0.0, -2.0]]
norm_A = power_iteration_operator_norm(A)
print(f"Norme operateur: {norm_A}")
assert abs(norm_A - 3.0) < 1e-5
```
