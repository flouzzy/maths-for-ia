# TP 5 : Espace dual et forme linéaire continue

**Objectif :** Évaluer numériquement la norme d'une forme linéaire continue définie par une intégrale sur $\mathcal{C}([0,1])$.

```python
def integrate(f, a, b, n_steps=500):
    step = (b - a) / n_steps
    area = 0.0
    for i in range(n_steps):
        x = a + i * step + step / 2.0
        area += f(x) * step
    return area

# Forme lineaire L(f) = int_0^{1/2} f - int_{1/2}^1 f
def L(f):
    return integrate(f, 0, 0.5) - integrate(f, 0.5, 1.0)

# Testons avec une fonction qui approche le supremum
def make_f_n(n):
    def f(x):
        if x <= 0.5 - 1.0/n:
            return 1.0
        elif x >= 0.5 + 1.0/n:
            return -1.0
        else:
            # Pente = -n
            return 1.0 - n * (x - (0.5 - 1.0/n))
    return f

f_10 = make_f_n(10)
val_L = L(f_10)

print(f"Valeur de L(f_10) : {val_L}")
# Le supremum theorique est 1.
assert val_L > 0.8
assert val_L < 1.0

f_100 = make_f_n(100)
val_L_100 = L(f_100)
print(f"Valeur de L(f_100) : {val_L_100}")
assert val_L_100 > val_L
```
