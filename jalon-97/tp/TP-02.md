# TP 2 : Complétude et convergence dans un espace de suites

**Objectif :** Simuler une suite de Cauchy dans l'espace $\ell^1$ et vérifier numériquement sa convergence vers une limite bien définie.

```python
def l1_norm(u):
    return sum(abs(x) for x in u)

def add_seq(u, v):
    return [x + y for x, y in zip(u, v)]

def sub_seq(u, v):
    return [x - y for x, y in zip(u, v)]

def generate_cauchy_sequence(max_n, dim=100):
    # On construit une suite dans l1
    # u^{(n)} = sum_{k=1}^n (1/k^2) e_k
    sequence = []
    for n in range(1, max_n + 1):
        u = [0.0] * dim
        for k in range(1, min(n + 1, dim)):
            u[k-1] = 1.0 / (k * k)
        sequence.append(u)
    return sequence

seq = generate_cauchy_sequence(50)

# Verification du critere de Cauchy
n1 = 40
n2 = 45
diff = sub_seq(seq[n2-1], seq[n1-1])
dist = l1_norm(diff)
print(f"Distance entre termes {n1} et {n2} : {dist}")
assert dist < 0.05

# Limite theorique
limit_seq = [0.0] * 100
for k in range(1, 100):
    limit_seq[k-1] = 1.0 / (k * k)

dist_to_limit = l1_norm(sub_seq(seq[49], limit_seq))
print(f"Distance a la limite pour n=50 : {dist_to_limit}")
assert dist_to_limit < 0.05
```
