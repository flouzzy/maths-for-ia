# TP 3 : Normalisation spectrale pour IA

**Objectif :** Implémenter manuellement la normalisation spectrale d'une couche de réseau de neurones pour forcer la constante de Lipschitz (norme d'opérateur) à 1.

```python
import math

def norm_2(v):
    return math.sqrt(sum(x * x for x in v))

def mat_vec_mul(A, v):
    res = [0.0] * len(A)
    for i in range(len(A)):
        for j in range(len(v)):
            res[i] += A[i][j] * v[j]
    return res

def transpose(A):
    res = [[0.0] * len(A) for _ in range(len(A[0]))]
    for i in range(len(A)):
        for j in range(len(A[0])):
            res[j][i] = A[i][j]
    return res

def estimate_spectral_norm(W, u, num_iters=5):
    # W: matrice de taille (out, in)
    # u: vecteur de taille (out)
    Wt = transpose(W)
    v = [0.0] * len(Wt)

    for _ in range(num_iters):
        # v = W^T * u
        v_unnorm = mat_vec_mul(Wt, u)
        norm_v = norm_2(v_unnorm)
        v = [x / norm_v for x in v_unnorm]

        # u = W * v
        u_unnorm = mat_vec_mul(W, v)
        norm_u = norm_2(u_unnorm)
        u = [x / norm_u for x in u_unnorm]

    # sigma = u^T * W * v
    Wv = mat_vec_mul(W, v)
    sigma = sum(u[i] * Wv[i] for i in range(len(u)))
    return sigma, u

def spectral_normalize(W):
    out_dim = len(W)
    u_init = [1.0 / math.sqrt(out_dim)] * out_dim
    sigma, _ = estimate_spectral_norm(W, u_init, num_iters=10)

    # Division de la matrice par sa norme spectrale
    W_sn = [[W[i][j] / sigma for j in range(len(W[0]))] for i in range(out_dim)]
    return W_sn, sigma

W_initial = [[3.0, 1.0], [0.0, 4.0]]
W_normalized, original_sigma = spectral_normalize(W_initial)

# Verifions que la nouvelle norme spectrale est 1
new_sigma, _ = spectral_normalize(W_normalized)

print(f"Norme originale: {original_sigma}")
print(f"Nouvelle norme: {new_sigma}")
assert abs(new_sigma - 1.0) < 1e-2
```
