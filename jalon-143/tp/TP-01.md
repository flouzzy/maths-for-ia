# TP 1 : Implémentation complète du Laplacien Combinatoire en Python Pur

## Contexte
Ce TP vise à implémenter, à partir de zéro, les matrices d'adjacence, de degrés et laplaciennes sans utiliser aucune bibliothèque externe spécialisée comme `networkx` ou `numpy`.

## Code Python
```python
def create_adjacency_matrix(n, edges, undirected=True):
    # Initialisation d'une matrice n x n remplie de 0
    A = [[0 for _ in range(n)] for _ in range(n)]
    for u, v in edges:
        A[u][v] = 1
        if undirected:
            A[v][u] = 1
    return A

def create_degree_matrix(A):
    n = len(A)
    D = [[0 for _ in range(n)] for _ in range(n)]
    for i in range(n):
        degree = sum(A[i])
        D[i][i] = degree
    return D

def create_laplacian_matrix(D, A):
    n = len(A)
    L = [[0 for _ in range(n)] for _ in range(n)]
    for i in range(n):
        for j in range(n):
            L[i][j] = D[i][j] - A[i][j]
    return L

# --- VALIDATION MATHÉMATIQUE ---
def test_laplacian():
    # Graphe triangle (3 sommets)
    n = 3
    edges = [(0, 1), (1, 2), (2, 0)]
    A = create_adjacency_matrix(n, edges)

    assert A == [[0, 1, 1], [1, 0, 1], [1, 1, 0]], "Erreur d'adjacence"

    D = create_degree_matrix(A)
    assert D == [[2, 0, 0], [0, 2, 0], [0, 0, 2]], "Erreur de degré"

    L = create_laplacian_matrix(D, A)
    expected_L = [
        [2, -1, -1],
        [-1, 2, -1],
        [-1, -1, 2]
    ]
    assert L == expected_L, "Erreur du laplacien combinatoire"

    # Vérification que L * 1 = 0 (la somme de chaque ligne est 0)
    for row in L:
        assert sum(row) == 0, "Propriété fondamentale L * 1 = 0 non respectée"

    print("Tous les tests mathématiques du TP 1 sont passés avec succès.")

test_laplacian()
```
