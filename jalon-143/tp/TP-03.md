# TP 3 : Construction du Laplacien Normalisé $\mathcal{L}$

## Contexte
Le laplacien normalisé est défini par $\mathcal{L} = D^{-1/2} L D^{-1/2}$, ce qui revient composante par composante à $\mathcal{L}_{ij} = 1$ si $i=j$, $-1 / \sqrt{d_i d_j}$ si $\{i,j\} \in E$, et $0$ sinon. Implémentons cela en Python pur.

## Code Python
```python
import math

def create_normalized_laplacian(n, edges):
    # Calcul des degrés
    degrees = [0] * n
    for u, v in edges:
        degrees[u] += 1
        degrees[v] += 1

    L_norm = [[0.0 for _ in range(n)] for _ in range(n)]

    for i in range(n):
        if degrees[i] > 0:
            L_norm[i][i] = 1.0

    for u, v in edges:
        if degrees[u] > 0 and degrees[v] > 0:
            val = -1.0 / math.sqrt(degrees[u] * degrees[v])
            L_norm[u][v] = val
            L_norm[v][u] = val

    return L_norm

# --- VALIDATION MATHÉMATIQUE ---
def test_normalized_laplacian():
    # Graphe linéaire à 3 sommets: 0 -- 1 -- 2
    n = 3
    edges = [(0, 1), (1, 2)]

    L_norm = create_normalized_laplacian(n, edges)

    # Degrés: d(0)=1, d(1)=2, d(2)=1
    # Attentes:
    # L_norm[0][0] = 1, L_norm[1][1] = 1, L_norm[2][2] = 1
    # L_norm[0][1] = -1 / sqrt(1*2) = -1/sqrt(2)
    # L_norm[1][2] = -1 / sqrt(2*1) = -1/sqrt(2)

    expected_01 = -1.0 / math.sqrt(2)

    assert abs(L_norm[0][0] - 1.0) < 1e-9
    assert abs(L_norm[1][1] - 1.0) < 1e-9
    assert abs(L_norm[0][1] - expected_01) < 1e-9
    assert abs(L_norm[1][2] - expected_01) < 1e-9
    assert abs(L_norm[0][2] - 0.0) < 1e-9

    # Propriété spectrale: la trace de L_norm vaut le nombre de sommets isolés (ici 0, tous sont liés, mais 3 sommets ont deg>0)
    # Trace = somme des deg>0, ici trace = 3
    trace = sum(L_norm[i][i] for i in range(n))
    assert abs(trace - 3.0) < 1e-9, "Trace invalide"

    print("Tous les tests mathématiques du TP 3 sont passés avec succès.")

test_normalized_laplacian()
```
