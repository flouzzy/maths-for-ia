## Partie 1 : Représentation du Graphe

Nous allons commencer par définir une classe simple pour représenter un graphe non orienté. Nous utiliserons une liste d'adjacence pour stocker les connexions entre les nœuds.

```python
class Graph:
    """
    Représentation simple d'un graphe non orienté utilisant une liste d'adjacence.
    """
    def __init__(self, num_nodes):
        """
        Initialise un graphe avec un nombre donné de nœuds.
        Les nœuds sont indexés de 0 à num_nodes - 1.
        """
        self.num_nodes = num_nodes
        self.adj = {i: [] for i in range(num_nodes)} # Liste d'adjacence

    def add_edge(self, u, v):
        """
        Ajoute une arête entre les nœuds u et v.
        Le graphe est non orienté, donc l'arête est ajoutée dans les deux sens.
        """
        if u < 0 or u >= self.num_nodes or v < 0 or v >= self.num_nodes:
            raise ValueError(f"Nœuds {u}, {v} hors limites pour un graphe de {self.num_nodes} nœuds.")
        if v not in self.adj[u]: # Évite les arêtes multiples
            self.adj[u].append(v)
        if u not in self.adj[v]:
            self.adj[v].append(u)

    def get_degree(self, node):
        """
        Retourne le degré d'un nœud.
        """
        if node < 0 or node >= self.num_nodes:
            raise ValueError(f"Nœud {node} hors limites.")
        return len(self.adj[node])

    def get_adjacency_list(self):
        """
        Retourne la liste d'adjacence du graphe.
        """
        return self.adj

    def get_num_nodes(self):
        """
        Retourne le nombre de nœuds dans le graphe.
        """
        return self.num_nodes

    def __str__(self):
        """
        Représentation textuelle du graphe.
        """
        s = f"Graphe avec {self.num_nodes} nœuds:\n"
        for node, neighbors in self.adj.items():
            s += f"  {node}: {neighbors}\n"
        return s

# --- Fonctions utilitaires pour les matrices (implémentées from scratch) ---

def create_matrix(rows, cols, default_value=0):
    """Crée une matrice (liste de listes) de taille rows x cols."""
    return [[default_value for _ in range(cols)] for _ in range(rows)]

def matrix_add(A, B):
    """Additionne deux matrices A et B."""
    if len(A) != len(B) or len(A[0]) != len(B[0]):
        raise ValueError("Les dimensions des matrices doivent correspondre pour l'addition.")
    rows, cols = len(A), len(A[0])
    C = create_matrix(rows, cols)
    for i in range(rows):
        for j in range(cols):
            C[i][j] = A[i][j] + B[i][j]
    return C

def matrix_subtract(A, B):
    """Soustrait la matrice B de la matrice A."""
    if len(A) != len(B) or len(A[0]) != len(B[0]):
        raise ValueError("Les dimensions des matrices doivent correspondre pour la soustraction.")
    rows, cols = len(A), len(A[0])
    C = create_matrix(rows, cols)
    for i in range(rows):
        for j in range(cols):
            C[i][j] = A[i][j] - B[i][j]
    return C

def matrix_multiply(A, B):
    """Multiplie deux matrices A et B."""
    rows_A, cols_A = len(A), len(A[0])
    rows_B, cols_B = len(B), len(B[0])
    if cols_A != rows_B:
        raise ValueError("Le nombre de colonnes de A doit être égal au nombre de lignes de B pour la multiplication.")

    C = create_matrix(rows_A, cols_B)
    for i in range(rows_A):
        for j in range(cols_B):
            for k in range(cols_A):
                C[i][j] += A[i][k] * B[k][j]
    return C

def identity_matrix(size):
    """Crée une matrice identité de taille size x size."""
    I = create_matrix(size, size)
    for i in range(size):
        I[i][i] = 1
    return I

def is_symmetric(matrix):
    """Vérifie si une matrice est symétrique."""
    rows = len(matrix)
    cols = len(matrix[0])
    if rows != cols:
        return False
    for i in range(rows):
        for j in range(i + 1, cols):
            if matrix[i][j] != matrix[j][i]:
                return False
    return True

def sum_rows_zero(matrix, tolerance=1e-9):
    """Vérifie si la somme de chaque ligne d'une matrice est proche de zéro."""
    for row in matrix:
        if abs(sum(row)) > tolerance:
            return False
    return True

# --- Exemple de graphe ---
print("--- Partie 1 : Représentation du Graphe ---")

g.add_edge(0, 1)
g.add_edge(1, 2)
g.add_edge(2, 0) # Triangle 0-1-2
g.add_edge(3, 4)
g.add_edge(4, 5)
g.add_edge(5, 3) # Triangle 3-4-5
g.add_edge(2, 3) # Pont entre les deux triangles

print(g)
print(f"Degré du nœud 0: {g.get_degree(0)}")
print(f"Degré du nœud 2: {g.get_degree(2)}")
print(f"Degré du nœud 3: {g.get_degree(3)}")

# Assertions de base
assert g.get_num_nodes() == 6
assert g.get_degree(0) == 2
assert g.get_degree(2) == 3 # Connecté à 0, 1, 3
assert g.get_degree(3) == 3 # Connecté à 2, 4, 5
print("Assertions de la Partie 1 passées avec succès.")
```

## Partie 2 : Le Laplacien Combinatoire

Le Laplacien combinatoire $L$ d'un graphe est une matrice fondamentale en théorie spectrale. Il est défini comme $L = D - A$, où $D$ est la matrice des degrés (une matrice diagonale où $D_{ii}$ est le degré du nœud $i$) et $A$ est la matrice d'adjacence.

### Définition Mathématique
Pour un graphe $G=(V, E)$ avec $n$ nœuds :
*   La **matrice d'adjacence** $A$ est une matrice $n \times n$ où $A_{ij} = 1$ si une arête existe entre $i$ et $j$, et $0$ sinon.
*   La **matrice des degrés** $D$ est une matrice diagonale $n \times n$ où $D_{ii} = \text{deg}(i)$ (le degré du nœud $i$), et $D_{ij} = 0$ pour $i \neq j$.
*   Le **Laplacien combinatoire** $L = D - A$.

### Implémentation

```python
def get_adjacency_matrix(graph):
    """
    Calcule la matrice d'adjacence du graphe.
    """
    n = graph.get_num_nodes()
    adj_matrix = create_matrix(n, n)
    for u in range(n):
        for v in graph.get_adjacency_list()[u]:
            adj_matrix[u][v] = 1
    return adj_matrix

def get_degree_matrix(graph):
    """
    Calcule la matrice des degrés du graphe.
    """
    n = graph.get_num_nodes()
    degree_matrix = create_matrix(n, n)
    for i in range(n):
        degree_matrix[i][i] = graph.get_degree(i)
    return degree_matrix

def get_combinatorial_laplacian(graph):
    """
    Calcule le Laplacien combinatoire L = D - A.
    """
    adj_matrix = get_adjacency_matrix(graph)
    degree_matrix = get_degree_matrix(graph)
    laplacian = matrix_subtract(degree_matrix, adj_matrix)
    return laplacian

# --- Application et Assertions ---
print("\n--- Partie 2 : Le Laplacien Combinatoire ---")
adj_matrix = get_adjacency_matrix(g)
print("Matrice d'adjacence A:")
for row in adj_matrix:
    print(row)

degree_matrix = get_degree_matrix(g)
print("\nMatrice des degrés D:")
for row in degree_matrix:
    print(row)

laplacian_combinatorial = get_combinatorial_laplacian(g)
print("\nLaplacien Combinatoire L = D - A:")
for row in laplacian_combinatorial:
    print(row)

# Assertions de validation mathématique
# 1. Le Laplacien combinatoire est symétrique.
assert is_symmetric(laplacian_combinatorial), "Le Laplacien combinatoire doit être symétrique."

# 2. La somme de chaque ligne (et colonne) du Laplacien combinatoire est 0.
# Ceci est une propriété fondamentale car L * 1 = 0, où 1 est le vecteur de tous les uns.
assert sum_rows_zero(laplacian_combinatorial), "La somme des lignes du Laplacien combinatoire doit être 0."

print("Assertions de la Partie 2 passées avec succès.")
```

## Partie 3 : Le Laplacien Normalisé (Symétrique)

Le Laplacien normalisé symétrique, souvent noté $L_{sym}$, est une alternative au Laplacien combinatoire qui est particulièrement utile pour les graphes avec des degrés très variés. Il est défini comme $L_{sym} = I - D^{-1/2} A D^{-1/2}$, où $I$ est la matrice identité et $D^{-1/2}$ est une matrice diagonale dont les éléments diagonaux sont $1/\sqrt{\text{deg}(i)}$ (ou 0 si $\text{deg}(i)=0$).

### Définition Mathématique
Pour un graphe $G=(V, E)$ avec $n$ nœuds :
*   $D^{-1/2}$ est une matrice diagonale $n \times n$ où $(D^{-1/2})_{ii} = 1/\sqrt{\text{deg}(i)}$ si $\text{deg}(i) > 0$, et $0$ si $\text{deg}(i) = 0$.
*   Le **Laplacien normalisé symétrique** $L_{sym} = I - D^{-1/2} A D^{-1/2}$.

### Implémentation

```python
import math

def get_inverse_sqrt_degree_matrix(graph):
    """
    Calcule la matrice D^(-1/2).
    Gère les nœuds isolés (degré 0) en mettant 0 sur la diagonale.
    """
    n = graph.get_num_nodes()
    inv_sqrt_degree_matrix = create_matrix(n, n)
    for i in range(n):
        degree = graph.get_degree(i)
        if degree > 0:
            inv_sqrt_degree_matrix[i][i] = 1.0 / math.sqrt(degree)
        else:
            inv_sqrt_degree_matrix[i][i] = 0.0 # Pour les nœuds isolés
    return inv_sqrt_degree_matrix

def get_normalized_laplacian(graph):
    """
    Calcule le Laplacien normalisé symétrique L_sym = I - D^(-1/2) A D^(-1/2).
    """
    n = graph.get_num_nodes()
    adj_matrix = get_adjacency_matrix(graph)
    inv_sqrt_degree_matrix = get_inverse_sqrt_degree_matrix(graph)

    # Calcul de D^(-1/2) A
    D_inv_sqrt_A = matrix_multiply(inv_sqrt_degree_matrix, adj_matrix)

    # Calcul de D^(-1/2) A D^(-1/2)
    D_inv_sqrt_A_D_inv_sqrt = matrix_multiply(D_inv_sqrt_A, inv_sqrt_degree_matrix)

    # Calcul de I - D^(-1/2) A D^(-1/2)
    identity = identity_matrix(n)
    normalized_laplacian = matrix_subtract(identity, D_inv_sqrt_A_D_inv_sqrt)

    return normalized_laplacian

# --- Application et Assertions ---
print("\n--- Partie 3 : Le Laplacien Normalisé (Symétrique) ---")
inv_sqrt_degree_matrix = get_inverse_sqrt_degree_matrix(g)
print("Matrice D^(-1/2):")
for row in inv_sqrt_degree_matrix:
    print([f"{x:.4f}" for x in row])

laplacian_normalized = get_normalized_laplacian(g)
print("\nLaplacien Normalisé L_sym = I - D^(-1/2) A D^(-1/2):")
for row in laplacian_normalized:
    print([f"{x:.4f}" for x in row])

# Assertions de validation mathématique
# 1. Le Laplacien normalisé est symétrique.
assert is_symmetric(laplacian_normalized), "Le Laplacien normalisé doit être symétrique."

print("Assertions de la Partie 3 passées avec succès.")
```

## Partie 4 : Vecteur de Fiedler et Coupures Optimales (Min-Cut)

Le problème de la coupure minimale (Min-Cut) consiste à trouver une partition des nœuds d'un graphe en deux ensembles, telle que le nombre d'arêtes entre ces deux ensembles soit minimal. Ce problème est NP-difficile en général, mais la théorie spectrale offre une approximation efficace.

Le **vecteur de Fiedler** est le vecteur propre associé à la deuxième plus petite valeur propre (appelée connectivité algébrique $\lambda_2$) du Laplacien (combinatoire ou normalisé). Ce vecteur a la propriété remarquable de pouvoir être utilisé pour partitionner un graphe. Les nœuds dont les composantes du vecteur de Fiedler ont un signe différent (ou sont de part et d'autre d'un seuil) tendent à appartenir à des partitions différentes, souvent de manière à minimiser la coupure.

Comme mentionné dans les contraintes, le calcul des valeurs propres et vecteurs propres "from scratch" est une tâche complexe d'algèbre linéaire numérique. Pour ce TP, nous utiliserons `numpy.linalg.eig` pour cette étape spécifique, tout en gardant le reste de l'implémentation "from scratch".

### Implémentation

```python
import numpy as np

def get_fiedler_vector(laplacian_matrix):
    """
    Calcule le vecteur de Fiedler d'une matrice Laplacienne.
    Utilise numpy pour la décomposition en valeurs propres.
    """
    # Convertir la matrice Python (liste de listes) en tableau NumPy
    np_laplacian = np.array(laplacian_matrix)

    # Calculer les valeurs propres et vecteurs propres
    # np.linalg.eig retourne des valeurs propres complexes et des vecteurs propres complexes
    # même si la matrice est réelle et symétrique. Nous prenons la partie réelle.
    eigenvalues, eigenvectors = np.linalg.eig(np_laplacian)

    # Assurez-vous que les valeurs propres sont réelles (pour les matrices symétriques)
    eigenvalues = np.real(eigenvalues)
    eigenvectors = np.real(eigenvectors)

    # Trier les valeurs propres et les vecteurs propres correspondants
    # np.argsort retourne les indices qui trieraient un tableau
    sorted_indices = np.argsort(eigenvalues)
    sorted_eigenvalues = eigenvalues[sorted_indices]
    sorted_eigenvectors = eigenvectors[:, sorted_indices]

    # La première valeur propre (la plus petite) doit être proche de zéro.
    # Le vecteur de Fiedler est le vecteur propre correspondant à la deuxième plus petite valeur propre.
    # Si le graphe est connexe, la première valeur propre est 0 et la deuxième est > 0.
    # Si le graphe a plusieurs composantes, il y aura plusieurs valeurs propres 0.
    # Nous cherchons la première valeur propre non nulle, ou la deuxième si la première est 0.

    # Trouver l'indice de la deuxième plus petite valeur propre non nulle
    # (ou la deuxième valeur propre si la première est 0)
    fiedler_index = -1
    for i, val in enumerate(sorted_eigenvalues):
        if abs(val) > 1e-9: # Si la valeur propre n'est pas (presque) zéro
            fiedler_index = i
            break

    if fiedler_index == -1: # Tous les eigenvalues sont 0 (graphe sans arêtes)
        return np.zeros(len(laplacian_matrix)) # Pas de Fiedler vector significatif

    # Si la première valeur propre est 0, le Fiedler vector est le deuxième.
    # Si le graphe est déconnecté, il peut y avoir plusieurs 0.
    # Pour un graphe connexe, le Fiedler vector est le 2ème.
    # Pour un graphe déconnecté, le nombre de 0 est le nombre de composantes.
    # Nous prenons le vecteur propre correspondant à la première valeur propre strictement positive.

    # Si sorted_eigenvalues[0] est 0, alors fiedler_index sera 1 (si le graphe est connexe).
    # Si sorted_eigenvalues[0] et sorted_eigenvalues[1] sont 0 (2 composantes), fiedler_index sera 2.
    # On veut le premier vecteur propre qui n'est pas le vecteur de tous les uns (celui de la première 0).
    # Pour un graphe connexe, c'est le 2ème vecteur propre.
    # Pour un graphe déconnecté, c'est le vecteur propre associé à la première valeur propre non nulle.

    # Pour être précis, le Fiedler vector est le vecteur propre associé à la *deuxième* plus petite valeur propre.
    # Si la plus petite est 0, c'est le vecteur propre associé à la première valeur propre > 0.
    # Si le graphe est déconnecté, il y a plusieurs valeurs propres 0. Le Fiedler vector est alors le vecteur propre
    # associé à la plus petite valeur propre *non nulle*.

    # Simplifions pour ce TP: nous prenons le vecteur propre associé à la 2ème plus petite valeur propre.
    # Si la 1ère est 0, c'est la 2ème. Si la 1ère n'est pas 0, c'est la 2ème.
    # C'est la définition standard pour le Fiedler vector.

    if len(sorted_eigenvalues) < 2:
        return np.zeros(len(laplacian_matrix)) # Pas de 2ème eigenvalue

    # Le vecteur de Fiedler est le vecteur propre correspondant à la 2ème plus petite valeur propre.
    fiedler_vector = sorted_eigenvectors[:, 1] # Index 1 pour la 2ème plus petite

    return fiedler_vector

def spectral_partitioning(graph, fiedler_vector):
    """
    Partitionne le graphe en deux ensembles en utilisant le signe des composantes
    du vecteur de Fiedler.
    """
    partition_A = []
    partition_B = []

    for i in range(graph.get_num_nodes()):
        if fiedler_vector[i] >= 0: # Utilise 0 comme seuil
            partition_A.append(i)
        else:
            partition_B.append(i)

    return partition_A, partition_B

def calculate_cut_size(graph, partition_A, partition_B):
    """
    Calcule la taille de la coupure (nombre d'arêtes entre partition_A et partition_B).
    """
    cut_size = 0
    adj_list = graph.get_adjacency_list()

    # Pour chaque nœud dans la partition A
    for u in partition_A:
        # Vérifier ses voisins
        for v in adj_list[u]:
            # Si le voisin est dans la partition B, c'est une arête de coupure
            if v in partition_B:
                cut_size += 1

    # Chaque arête est comptée une seule fois (u-v est la même que v-u)
    return cut_size

# --- Application et Assertions ---
print("\n--- Partie 4 : Vecteur de Fiedler et Coupures Optimales (Min-Cut) ---")

# Utilisation du Laplacien Combinatoire pour le Fiedler vector

print(f"\nVecteur de Fiedler (Laplacien Combinatoire): {fiedler_combinatorial}")

print(f"Partition spectrale (Combinatoire): A={partition_A_comb}, B={partition_B_comb}")

print(f"Partition spectrale (Normalisé): A={partition_A_norm}, B={partition_B_norm}")

cut_size_norm = calculate_cut_size(g, partition_A_norm, partition_B_norm)
print(f"Taille de la coupure (Normalisé): {cut_size_norm}")

# Assertions de validation mathématique
# Pour notre graphe d'exemple (deux triangles connectés par un pont 2-3),
# la coupure minimale est de 1 (l'arête (2,3)).
# Le vecteur de Fiedler devrait idéalement séparer {0,1,2} de {3,4,5}.
# Les nœuds 0,1,2 devraient avoir des valeurs de même signe, et 3,4,5 de l'autre signe.

# Vérifions que les partitions sont non vides et disjointes
assert len(partition_A_comb) + len(partition_B_comb) == g.get_num_nodes()
assert not any(node in partition_B_comb for node in partition_A_comb)

# Pour le graphe d'exemple, la coupure optimale est 1 (l'arête (2,3)).
# Le Laplacien spectral devrait trouver cette coupure.
# Les nœuds {0,1,2} devraient être d'un côté et {3,4,5} de l'autre.
# Le signe du vecteur de Fiedler devrait refléter cela.
# Par exemple, si fiedler_vector[0], fiedler_vector[1], fiedler_vector[2] sont positifs
# et fiedler_vector[3], fiedler_vector[4], fiedler_vector[5] sont négatifs (ou vice-versa).

# Vérifions la validité de la partition pour notre graphe spécifique
# La partition devrait être {0,1,2} et {3,4,5} (ou l'inverse)
expected_partition_1 = {0, 1, 2}
expected_partition_2 = {3, 4, 5}

# Convertir les listes en ensembles pour une comparaison facile
set_A_comb = set(partition_A_comb)
set_B_comb = set(partition_B_comb)

# Vérifier si la partition correspond à l'une des deux configurations attendues
is_correct_partition_comb = (set_A_comb == expected_partition_1 and set_B_comb == expected_partition_2) or \
                            (set_A_comb == expected_partition_2 and set_B_comb == expected_partition_1)

assert is_correct_partition_comb, f"La partition spectrale combinatoire n'est pas celle attendue: {set_A_comb}, {set_B_comb}"
assert cut_size_comb == 1, f"La taille de la coupure combinatoire n'est pas 1, mais {cut_size_comb}"

set_A_norm = set(partition_A_norm)
set_B_norm = set(partition_B_norm)

is_correct_partition_norm = (set_A_norm == expected_partition_1 and set_B_norm == expected_partition_2) or \
                            (set_A_norm == expected_partition_2 and set_B_norm == expected_partition_1)

assert is_correct_partition_norm, f"La partition spectrale normalisée n'est pas celle attendue: {set_A_norm}, {set_B_norm}"
assert cut_size_norm == 1, f"La taille de la coupure normalisée n'est pas 1, mais {cut_size_norm}"
```

Poursuivez votre exploration de la théorie spectrale des graphes, il y a encore beaucoup à découvrir !
