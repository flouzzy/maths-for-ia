# TP 5 : Approximation du vecteur de Fiedler (Power Iteration)

## Contexte
Pour trouver la plus petite valeur propre non nulle (trou spectral) et son vecteur propre (vecteur de Fiedler), nous avons besoin d'outils d'algèbre linéaire. Puisque la méthode exacte demande de la factorisation, nous allons créer un mini solveur itératif en Python pur qui approche les vecteurs propres.

## Code Python
```python
import random
import math

def dot_product(u, v):
    return sum(a * b for a, b in zip(u, v))

def norm(u):
    return math.sqrt(dot_product(u, u))

def normalize(u):
    n = norm(u)
    if n == 0: return u
    return [x / n for x in u]

def mat_vec(M, x):
    return [dot_product(row, x) for row in M]

def mat_sub_scalar(M, scalar):
    # M - scalar * I
    n = len(M)
    res = [[M[i][j] for j in range(n)] for i in range(n)]
    for i in range(n):
        res[i][i] -= scalar
    return res

def power_iteration(M, iters=1000, tol=1e-9):
    """
    Calcule la valeur propre principale et le vecteur propre associé
    par la méthode de la puissance itérée.
    """
    n = len(M)
    # Initialisation aléatoire
    v = [random.random() for _ in range(n)]
    v = normalize(v)

    for _ in range(iters):
        v_next = mat_vec(M, v)
        v_next_norm = normalize(v_next)

        # Critère d'arrêt
        diff = sum(abs(a - b) for a, b in zip(v, v_next_norm))
        if diff < tol:
            v = v_next_norm
            break
        v = v_next_norm

    lambda_max = dot_product(v, mat_vec(M, v))
    return lambda_max, v

def fiedler_vector_approximation(L, iters=1000):
    """
    Approximation de la deuxième plus petite valeur propre (Fiedler)
    et de son vecteur propre par méthode de déflation.
    Pour une matrice Laplacienne L, la plus petite valeur propre est 0
    avec pour vecteur propre 1 = (1,...,1)^T.

    Soit L' = lambda_max * I - L.
    La plus grande valeur propre de L' sera lambda_max - 0 = lambda_max.
    Le vecteur propre associé est 1.

    Pour trouver la 2e plus petite vp de L (trou spectral), on doit trouver
    la 2e plus grande vp de L'.
    Pour ce faire, on retire la projection sur le vecteur propre connu (1)
    à chaque itération de la puissance itérée sur L'.
    """
    n = len(L)
    # 1. Trouver lambda_max de L pour la translation
    lambda_max, _ = power_iteration(L, iters)

    # 2. Construire L' = lambda_max * I - L
    L_prime = mat_sub_scalar([[-x for x in row] for row in L], -lambda_max)

    # 3. Vecteur propre trivial de L associé à vp=0 (vecteur de 1)
    u1 = normalize([1.0] * n)

    # 4. Puissance itérée avec déflation (orthogonalisation par rapport à u1)
    v = [random.random() for _ in range(n)]
    v = normalize(v)

    for _ in range(iters):
        # Multiplication matricielle
        w = mat_vec(L_prime, v)

        # Déflation : on soustrait la composante colinéaire à u1
        # w = w - (w.u1)u1
        proj = dot_product(w, u1)
        for i in range(n):
            w[i] -= proj * u1[i]

        v_next = normalize(w)

        # On pourrait mettre un critère de tolérance ici
        v = v_next

    # v est maintenant une approximation du vecteur de Fiedler.
    # L'associer à sa valeur propre de L
    lambda_2 = dot_product(v, mat_vec(L, v))

    return lambda_2, v

# --- VALIDATION MATHÉMATIQUE ---
def test_fiedler():
    # Graphe biparti complet K_2,2
    # L a pour valeurs propres : 0, 2, 2, 4
    L = [
        [ 2,  0, -1, -1],
        [ 0,  2, -1, -1],
        [-1, -1,  2,  0],
        [-1, -1,  0,  2]
    ]

    lambda_2, fiedler_vec = fiedler_vector_approximation(L, iters=1000)

    # On sait mathématiquement que la valeur de Fiedler (lambda_2) doit être 2
    assert abs(lambda_2 - 2.0) < 1e-2, f"Échec: lambda_2 estimée à {lambda_2} au lieu de 2.0"

    # Vérification que le vecteur de Fiedler est bien orthogonal au vecteur 1
    orthogonality = sum(fiedler_vec)
    assert abs(orthogonality) < 1e-5, f"Échec: le vecteur n'est pas orthogonal à 1 (somme={orthogonality})"

    print("Le calcul du vecteur de Fiedler par déflation converge correctement. Tests TP 5 validés.")

test_fiedler()
```
