L'objectif de ce TP est d'implémenter de manière purement algorithmique (from scratch, en Python sans bibliothèques de haut niveau) des briques fondamentales liées à la SVD, pour asseoir la compréhension théorique.

```python
import math

# --- UTILS Briques d'Algèbre Linéaire ---
def transpose(mat):
    return [[mat[j][i] for j in range(len(mat))] for i in range(len(mat[0]))]

def matmul(A, B):
    m = len(A)
    n = len(B[0])
    p = len(B)
    C = [[0] * n for _ in range(m)]
    for i in range(m):
        for j in range(n):
            for k in range(p):
                C[i][j] += A[i][k] * B[k][j]
    return C

# --- IMPLEMENTATION SPECIFIQUE TP 2 ---

# La déflation consiste à retirer la composante principale pour trouver la suivante
def deflation(A, v1, lambda1):
    n = len(A)
    # A_def = A - lambda1 * (v1 v1^T)
    A_def = [[0.0]*n for _ in range(n)]
    for i in range(n):
        for j in range(n):
            A_def[i][j] = A[i][j] - lambda1 * v1[i] * v1[j]
    return A_def
```
