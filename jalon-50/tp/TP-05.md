# TP 5 : Frontière de décision (Approche Machine Learning)

Ce TP implémente "from scratch" en Python pur le concept ciblé.

```python
import math

# Simuler un classifieur très simple : f(x, y) = 1 si x^2 + y^2 < 1 else 0
def classifier(x, y):
    return 1 if (x**2 + y**2) < 1 else 0

def find_decision_boundary(resolution):
    boundary_points = []
    eps = 1.0 / resolution

    # Parcourir la grille
    for i in range(int(-2*resolution), int(2*resolution)):
        for j in range(int(-2*resolution), int(2*resolution)):
            x = i * eps
            y = j * eps

            label_center = classifier(x, y)

            # Vérifier si un voisin immédiat a un label différent (changement de classe)
            neighbors_labels = [
                classifier(x+eps, y),
                classifier(x-eps, y),
                classifier(x, y+eps),
                classifier(x, y-eps)
            ]

            if any(label != label_center for label in neighbors_labels):
                boundary_points.append((x, y))

    return boundary_points

boundary = find_decision_boundary(20)

# Vérifier que les points de la frontière sont proches du cercle unité x^2 + y^2 = 1
distances_to_circle = [abs(math.sqrt(x**2 + y**2) - 1.0) for x, y in boundary]
max_error = max(distances_to_circle)

assert max_error < 0.1, f"L'erreur maximale est {max_error}, devrait être faible"
print(f"TP5: Frontière de décision calculée empiriquement avec succès. Max erreur radiale : {max_error:.4f}")

```
