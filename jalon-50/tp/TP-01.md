# TP 1 : Simulation sur un maillage discret 1D

Ce TP implémente "from scratch" en Python pur le concept ciblé.

```python
def get_neighbors_1d(x, length):
    neighbors = []
    if x > 0: neighbors.append(x - 1)
    if x < length - 1: neighbors.append(x + 1)
    return neighbors

def topologie_1d(length, A_points):
    interior = set()
    closure = set(A_points)

    for x in A_points:
        neighbors = get_neighbors_1d(x, length)
        if all(n in A_points for n in neighbors):
            interior.add(x)

    for x in range(length):
        if x not in A_points:
            neighbors = get_neighbors_1d(x, length)
            if any(n in A_points for n in neighbors):
                closure.add(x)

    boundary = closure - interior
    return interior, closure, boundary

# Test
longueur = 10
A = {2, 3, 4, 7} # Segment et point isolé
i, c, b = topologie_1d(longueur, A)

assert i == {3}, "Seul 3 a ses deux voisins dans A"
assert c == {1, 2, 3, 4, 5, 6, 7, 8}, "L'adhérence s'étend aux voisins immédiats"
assert b == c - i, "La frontière est la différence"
print("TP1: Topologie 1D validée.")

```
