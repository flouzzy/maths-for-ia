# TP 2 : Graphe quelconque

Ce TP implémente "from scratch" en Python pur le concept ciblé.

```python
def topologie_graphe(adj_list, A_nodes):
    interior = set()
    closure = set(A_nodes)

    for node in A_nodes:
        if all(neighbor in A_nodes for neighbor in adj_list.get(node, [])):
            interior.add(node)

    for node in adj_list.keys():
        if node not in A_nodes:
            if any(neighbor in A_nodes for neighbor in adj_list.get(node, [])):
                closure.add(node)

    boundary = closure - interior
    return interior, closure, boundary

# Graphe étoilé : 0 est le centre, 1, 2, 3 sont les feuilles
adj = {
    0: [1, 2, 3],
    1: [0],
    2: [0],
    3: [0]
}
A = {0, 1}
i, c, b = topologie_graphe(adj, A)

assert i == {1}, "1 est lié uniquement à 0 (qui est dans A). 0 est lié à 2 et 3 (qui ne le sont pas)."
assert c == {0, 1, 2, 3}, "Tous les noeuds touchent A"
assert b == c - i, "Validation frontière"
print("TP2: Topologie sur graphe validée.")
```
