# TP 3 : Espace euclidien 2D (Approximation discrète avec eps-voisinage)

Ce TP implémente "from scratch" en Python pur le concept ciblé.

```python
import math

def distance(p1, p2):
    return math.sqrt((p1[0]-p2[0])**2 + (p1[1]-p2[1])**2)

def echantillonnage(grid_size):
    return [(x/10.0, y/10.0) for x in range(grid_size) for y in range(grid_size)]

def get_eps_neighbors(p, space, eps):
    return [q for q in space if q != p and distance(p, q) < eps]

def topologie_eps(space, A_points, eps):
    interior = set()
    closure = set(A_points)

    for p in A_points:
        neighbors = get_eps_neighbors(p, space, eps)
        # S'il y a des voisins, ils doivent tous être dans A
        if len(neighbors) > 0 and all(n in A_points for n in neighbors):
            interior.add(p)

    for p in space:
        if p not in A_points:
            neighbors = get_eps_neighbors(p, space, eps)
            if any(n in A_points for n in neighbors):
                closure.add(p)

    boundary = closure - interior
    return interior, closure, boundary

space = echantillonnage(20) # 0 à 1.9
# Rectangle 0.5 <= x <= 1.5, 0.5 <= y <= 1.5
A = {p for p in space if 0.5 <= p[0] <= 1.5 and 0.5 <= p[1] <= 1.5}
eps = 0.15 # Voisinage immédiat sur la grille de pas 0.1

i, c, b = topologie_eps(space, A, eps)

assert len(i) < len(A)
assert len(c) > len(A)
assert b == c - i
print(f"TP3: Espace continu discrétisé validé (Int: {len(i)}, Frontière: {len(b)})")

```
