# TP 4 : Transformation affine et conservation de loi

Lorsqu'on applique une transformation affine $Y = aX + b$ à une variable aléatoire Normale, la loi image reste Normale. Nous allons le vérifier sur l'espérance et la variance.

```python
import random

def generate_normal(mu, sigma, num_samples):
    # Approximation par la somme d'uniformes (TCL basique)
    samples = []
    for _ in range(num_samples):
        s = sum(random.uniform(0, 1) for _ in range(12)) - 6
        samples.append(mu + sigma * s)
    return samples

def test_affine_transformation():
    # X ~ N(mu=0, sigma=1)
    samples_X = generate_normal(0, 1, 10000)

    # Transformation Y = 3X + 5
    a = 3.0
    b = 5.0
    samples_Y = [a * x + b for x in samples_X]

    mean_Y = sum(samples_Y) / len(samples_Y)
    var_Y = sum((y - mean_Y)**2 for y in samples_Y) / len(samples_Y)

    # Theorie : E[aX+b] = aE[X]+b = 5. Var[aX+b] = a^2 Var[X] = 9.
    assert abs(mean_Y - 5.0) < 0.1, "L'esperance n'a pas suivi la transformation affine."
    assert abs(var_Y - 9.0) < 0.5, "La variance n'a pas suivi la loi a^2 Var(X)."
    print("TP 4 : Tests passes.")

if __name__ == "__main__":
    test_affine_transformation()
```
