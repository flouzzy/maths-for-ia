# TP 1 : Simulation d'une mesure image empirique

Dans ce TP, nous allons simuler le transfert d'une mesure de probabilité d'un espace de départ abstrait vers l'espace d'arrivée via une variable aléatoire (mesure image ou *push-forward*).

```python
import random

def simulate_push_forward():
    # Espace abstrait (lancers de deux des)
    omega = [(i, j) for i in range(1, 7) for j in range(1, 7)]

    # Probabilite uniforme sur l'espace de depart
    p_omega = 1.0 / len(omega)

    # Application mesurable (Variable Aleatoire) : Somme des des
    def X(event):
        return event[0] + event[1]

    # Calcul de la loi image empirique
    loi_image = {}
    for event in omega:
        val = X(event)
        loi_image[val] = loi_image.get(val, 0) + p_omega

    return loi_image

def test_push_forward():
    loi = simulate_push_forward()
    # La somme des probabilites doit etre 1 (Conservation de la masse)
    total_prob = sum(loi.values())
    assert abs(total_prob - 1.0) < 1e-9, "La mesure image n'est pas une proba."
    # La somme 7 est la plus probable (1/6 = 6/36)
    assert abs(loi[7] - (6.0 / 36.0)) < 1e-9, "Erreur sur P(X=7)."
    print("TP 1 : Tests passes.")

if __name__ == "__main__":
    test_push_forward()
```
