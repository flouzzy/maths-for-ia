# TP 05 : Conception Théorique d'un Moteur de Recherche Sémantique : Géométrie des Espaces de Plongement et Algèbre Linéaire Fondamentale

## Objectif Mathématique
Ce TP vise à explorer les fondements mathématiques de la recherche sémantique et de l'algèbre linéaire, en se concentrant sur la géométrie des espaces vectoriels et la dualité. Nous implémenterons "from scratch" les opérations vectorielles de base, la similarité cosinus pour mesurer la proximité sémantique dans des espaces de plongement simplifiés (Bag-of-Words), et le processus d'orthogonalisation de Gram-Schmidt. Ce dernier illustrera la construction de bases orthogonales, concept clé pour comprendre la structure et les projections dans les espaces vectoriels, et résoudra un problème d'algèbre linéaire typique des concours. L'objectif est de comprendre comment les concepts de distance angulaire (similarité cosinus) et de transformation de base (Gram-Schmidt) sont essentiels à la manipulation et à l'interprétation des données vectorielles, notamment dans le contexte des plongements sémantiques. La dualité est abordée implicitement par l'interprétation du produit scalaire comme l'application d'une forme linéaire, permettant de projeter un vecteur sur un autre ou de mesurer leur alignement.


### Normalisation L2 et Conditionnement
L'étape systématique de normalisation L2 $v \mapsto \frac{v}{\|v\|_2}$ agit comme un conditionneur garantissant que tous les points résident sur la variété riemannienne $\mathbb{S}^{|V|-1}$. La recherche du plus proche voisin devient équivalente au problème du MIPS (Maximum Inner Product Search), simplifiant drastiquement l'inférence.

## Implémentation Python Pure
```python
import math

# --- Partie 1 : Opérations Vectorielles Fondamentales (From Scratch) ---

def dot_product(v1, v2):
    """
    Calcule le produit scalaire de deux vecteurs.
    Précondition: Les vecteurs doivent avoir la même dimension.
    """
    if len(v1) != len(v2):
        raise ValueError("Les vecteurs doivent avoir la même dimension pour le produit scalaire.")
    result = 0
    for i in range(len(v1)):
        result += v1[i] * v2[i]
    return result

def vector_magnitude(v):
    """
    Calcule la magnitude (norme euclidienne) d'un vecteur.
    """
    sum_sq = 0
    for x in v:
        sum_sq += x * x
    return math.sqrt(sum_sq)

def normalize_vector(v):
    """
    Normalise un vecteur pour obtenir un vecteur unitaire.
    """
    mag = vector_magnitude(v)
    if mag == 0:
        # Un vecteur nul ne peut pas être normalisé. Retourne une copie du vecteur nul.
        return [0.0] * len(v)
    return [x / mag for x in v]

def scalar_multiply(scalar, v):
    """
    Multiplie un vecteur par un scalaire.
    """
    return [scalar * x for x in v]

def vector_subtract(v1, v2):
    """
    Soustrait le vecteur v2 du vecteur v1 (v1 - v2).
    Précondition: Les vecteurs doivent avoir la même dimension.
    """
    if len(v1) != len(v2):
        raise ValueError("Les vecteurs doivent avoir la même dimension pour la soustraction.")
    return [v1[i] - v2[i] for i in range(len(v1))]

# --- Assertions pour les opérations vectorielles ---
print("--- Tests des opérations vectorielles ---")
v_a = [1, 2, 3]
v_b = [4, 5, 6]
v_zero = [0, 0, 0]
v_neg = [-1, -2, -3]

assert dot_product(v_a, v_b) == 32, f"Erreur produit scalaire: {dot_product(v_a, v_b)}"
assert abs(vector_magnitude(v_a) - math.sqrt(14)) < 1e-9, f"Erreur magnitude: {vector_magnitude(v_a)}"
assert abs(normalize_vector(v_a)[0] - (1 / math.sqrt(14))) < 1e-9, f"Erreur normalisation: {normalize_vector(v_a)[0]}"
assert scalar_multiply(2, v_a) == [2, 4, 6], f"Erreur multiplication scalaire: {scalar_multiply(2, v_a)}"
assert vector_subtract(v_b, v_a) == [3, 3, 3], f"Erreur soustraction vectorielle: {vector_subtract(v_b, v_a)}"
assert dot_product(v_a, v_zero) == 0, "Produit scalaire avec vecteur nul doit être 0"
assert vector_magnitude(v_zero) == 0, "Magnitude d'un vecteur nul doit être 0"
assert normalize_vector(v_zero) == [0.0, 0.0, 0.0], "Normalisation d'un vecteur nul"
assert dot_product(v_a, v_neg) == -14, "Produit scalaire avec vecteur négatif"
print("Tests des opérations vectorielles passés avec succès.")

# --- Partie 2 : Similarité Cosinus pour la Recherche Sémantique ---

def cosine_similarity(v1, v2):
    """
    Calcule la similarité cosinus entre deux vecteurs.
    Retourne 0 si l'un des vecteurs est nul.
    """
    mag1 = vector_magnitude(v1)
    mag2 = vector_magnitude(v2)

    if mag1 == 0 or mag2 == 0:
        return 0.0  # Convention pour les vecteurs nuls

    return dot_product(v1, v2) / (mag1 * mag2)

# --- Assertions pour la similarité cosinus ---
print("\n--- Tests de la similarité cosinus ---")
v_parallel = [2, 4, 6] # Parallèle à v_a
v_orthogonal_true = [1, 1, -1] # 1*1 + 2*1 + 3*(-1) = 0

assert abs(cosine_similarity(v_a, v_a) - 1.0) < 1e-9, f"Erreur similarité cosinus (même vecteur): {cosine_similarity(v_a, v_a)}"
assert abs(cosine_similarity(v_a, v_parallel) - 1.0) < 1e-9, f"Erreur similarité cosinus (vecteurs parallèles): {cosine_similarity(v_a, v_parallel)}"
assert abs(cosine_similarity(v_a, v_orthogonal_true) - 0.0) < 1e-9, f"Erreur similarité cosinus (vecteurs orthogonaux): {cosine_similarity(v_a, v_orthogonal_true)}"
assert cosine_similarity(v_a, v_zero) == 0.0, "Similarité cosinus avec vecteur nul doit être 0"
assert abs(cosine_similarity(v_a, v_neg) - (-1.0)) < 1e-9, f"Erreur similarité cosinus (vecteurs opposés): {cosine_similarity(v_a, v_neg)}"
print("Tests de la similarité cosinus passés avec succès.")

# --- Partie 3 : Implémentation Simplifiée d'Espaces de Plongement (Bag-of-Words) ---

def tokenize(text):
    """
    Tokenise un texte en mots, convertit en minuscules et filtre les caractères non alphabétiques.
    """
    # Simple tokenisation: supprime la ponctuation et divise par espace
    clean_text = ''.join(c.lower() if c.isalpha() or c.isspace() else ' ' for c in text)
    return [word for word in clean_text.split() if word]

def build_vocabulary(documents):
    """
    Construit un vocabulaire unique à partir d'une liste de documents.
    Retourne un dictionnaire {mot: index}.
    """
    unique_words = {}
    index = 0
    for doc in documents:
        for word in tokenize(doc):
            if word not in unique_words:
                unique_words[word] = index
                index += 1
    return unique_words

def document_to_vector(document, vocabulary):
    """
    Convertit un document en vecteur Bag-of-Words (comptes de fréquence).
    """
    vector = [0] * len(vocabulary)
    for word in tokenize(document):
        if word in vocabulary:
            vector[vocabulary[word]] += 1
    return vector

def semantic_search_engine(query, documents):
    """
    Moteur de recherche sémantique simplifié utilisant la similarité cosinus.
    """
    # Inclure la requête et tous les documents pour construire un vocabulaire global
    all_texts = documents + [query]
    vocabulary = build_vocabulary(all_texts)

    query_vector = document_to_vector(query, vocabulary)

    results = []
    for i, doc_text in enumerate(documents):
        doc_vector = document_to_vector(doc_text, vocabulary)
        similarity = cosine_similarity(query_vector, doc_vector)
        results.append((doc_text, similarity))

    # Trie les résultats par similarité décroissante
    results.sort(key=lambda x: x[1], reverse=True)
    return results

# --- Assertions pour le moteur de recherche sémantique ---
print("\n--- Tests du moteur de recherche sémantique ---")
documents = [
    "Le chat dort sur le tapis.",
    "Un chien court dans le parc.",
    "Le chat noir est mignon.",
    "Le parc est grand et vert."
]
query = "chat mignon"

# Test de tokenisation
assert tokenize("Le Chat, dort!") == ['le', 'chat', 'dort'], f"Erreur tokenisation: {tokenize('Le Chat, dort!')}"

# Test de vocabulaire
vocab = build_vocabulary(documents)
expected_vocab_size = len(set(word for doc in documents for word in tokenize(doc)))
assert len(vocab) == expected_vocab_size, f"Erreur taille vocabulaire: {len(vocab)} vs {expected_vocab_size}"

# Test de vectorisation (vérification de la somme des comptes)
doc_vec_chat_dort = document_to_vector("Le chat dort", vocab)
assert sum(doc_vec_chat_dort) == 3, f"Erreur somme vectorisation: {sum(doc_vec_chat_dort)}"

# Test de recherche
search_results = semantic_search_engine(query, documents)
# Le document "Le chat noir est mignon." devrait être le plus pertinent.
assert search_results[0][0] == "Le chat noir est mignon.", f"Erreur ordre de recherche: {search_results[0][0]}"
assert search_results[0][1] > search_results[1][1], "Le document le plus pertinent n'a pas la plus haute similarité."
print("Tests du moteur de recherche sémantique passés avec succès.")

# --- Partie 4 : Problème d'Algèbre de l'X : Orthogonalisation de Gram-Schmidt ---

def project(v, u):
    """
    Calcule la projection orthogonale du vecteur v sur le vecteur u.
    """
    if vector_magnitude(u) == 0:
        return [0.0] * len(u) # Projection sur un vecteur nul est un vecteur nul

    # proj_u(v) = (v . u / ||u||^2) * u
    scalar_factor = dot_product(v, u) / dot_product(u, u)
    return scalar_multiply(scalar_factor, u)

def gram_schmidt(vectors):
    """
    Applique le processus d'orthogonalisation de Gram-Schmidt à un ensemble de vecteurs.
    Retourne une base orthonormée.
    Précondition: Les vecteurs d'entrée sont traités séquentiellement. Les vecteurs
    linéairement dépendants des précédents seront ignorés (résultant en un vecteur nul).
    """
    if not vectors:
        return []

    orthonormal_basis = []
    for v in vectors:
        # Initialise le vecteur orthogonal potentiel avec le vecteur original
        u = list(v) # Crée une copie mutable

        # Soustrait les projections sur les vecteurs déjà orthogonalisés
        for basis_vec in orthonormal_basis:
            proj_v_basis_vec = project(v, basis_vec)
            u = vector_subtract(u, proj_v_basis_vec)

        # Normalise le vecteur résultant s'il n'est pas nul
        mag_u = vector_magnitude(u)
        if mag_u > 1e-9: # Utilise une petite tolérance pour éviter les divisions par zéro dues aux erreurs de flottants
            orthonormal_basis.append(normalize_vector(u))
        # Si mag_u est proche de zéro, cela signifie que le vecteur est linéairement dépendant
        # des vecteurs précédents, et il est ignoré.

    return orthonormal_basis

# --- Assertions pour Gram-Schmidt ---
print("\n--- Tests de Gram-Schmidt ---")

# Cas 1: Vecteurs orthogonaux initiaux (devraient juste être normalisés)
v1_gs = [1, 0]
v2_gs = [0, 1]
basis1 = gram_schmidt([v1_gs, v2_gs])
assert len(basis1) == 2, "Gram-Schmidt devrait produire 2 vecteurs pour 2 entrées indépendantes."
assert abs(vector_magnitude(basis1[0]) - 1.0) < 1e-9, "Le premier vecteur de base n'est pas unitaire."
assert abs(vector_magnitude(basis1[1]) - 1.0) < 1e-9, "Le second vecteur de base n'est pas unitaire."
assert abs(dot_product(basis1[0], basis1[1])) < 1e-9, "Les vecteurs de base ne sont pas orthogonaux."
assert abs(basis1[0][0] - 1.0) < 1e-9 and abs(basis1[0][1] - 0.0) < 1e-9, "Le premier vecteur n'est pas correct."
assert abs(basis1[1][0] - 0.0) < 1e-9 and abs(basis1[1][1] - 1.0) < 1e-9, "Le second vecteur n'est pas correct."

# Cas 2: Vecteurs non orthogonaux
v3_gs = [1, 1]
v4_gs = [2, 0]
basis2 = gram_schmidt([v3_gs, v4_gs])
assert len(basis2) == 2, "Gram-Schmidt devrait produire 2 vecteurs pour 2 entrées indépendantes."
assert abs(vector_magnitude(basis2[0]) - 1.0) < 1e-9, "Le premier vecteur de base n'est pas unitaire."
assert abs(vector_magnitude(basis2[1]) - 1.0) < 1e-9, "Le second vecteur de base n'est pas unitaire."
assert abs(dot_product(basis2[0], basis2[1])) < 1e-9, f"Les vecteurs de base ne sont pas orthogonaux: {dot_product(basis2[0], basis2[1])}"

# Vérification des valeurs spécifiques pour basis2
expected_basis2_0 = normalize_vector([1, 1])
expected_basis2_1 = normalize_vector([1, -1])
for i in range(len(expected_basis2_0)):
    assert abs(basis2[0][i] - expected_basis2_0[i]) < 1e-9, f"Erreur Gram-Schmidt basis2[0]: {basis2[0]} vs {expected_basis2_0}"
    assert abs(basis2[1][i] - expected_basis2_1[i]) < 1e-9, f"Erreur Gram-Schmidt basis2[1]: {basis2[1]} vs {expected_basis2_1}"

# Cas 3: Vecteurs en 3D
v5_gs = [1, 1, 0]
v6_gs = [1, 0, 1]
v7_gs = [0, 1, 1]
basis3 = gram_schmidt([v5_gs, v6_gs, v7_gs])
assert len(basis3) == 3, "Gram-Schmidt devrait produire 3 vecteurs pour 3 entrées indépendantes."
for i in range(3):
    assert abs(vector_magnitude(basis3[i]) - 1.0) < 1e-9, f"Vecteur {i} de basis3 n'est pas unitaire."
    for j in range(i + 1, 3):
        assert abs(dot_product(basis3[i], basis3[j])) < 1e-9, f"Vecteurs {i} et {j} de basis3 ne sont pas orthogonaux."

# Cas 4: Vecteurs linéairement dépendants (un vecteur nul ou multiple d'un précédent)
v8_gs = [1, 0]
v9_gs = [2, 0] # Linéairement dépendant de v8_gs
basis4 = gram_schmidt([v8_gs, v9_gs])
assert len(basis4) == 1, "Gram-Schmidt devrait ignorer les vecteurs linéairement dépendants."
assert abs(basis4[0][0] - 1.0) < 1e-9 and abs(basis4[0][1] - 0.0) < 1e-9, "Le vecteur de base n'est pas correct."

print("Tests de Gram-Schmidt passés avec succès.")

print("\n--- Démonstration du Moteur de Recherche Sémantique ---")
documents_demo = [

    "L'apprentissage automatique révolutionne l'intelligence artificielle.",
    "Les mathématiques sont fondamentales pour la compréhension de l'IA.",
    "Un chat noir dort paisiblement sur le canapé.",
    "Le deep learning est une branche de l'apprentissage automatique."

print("Résultats de recherche (document, similarité cosinus):")

print("\n--- Démonstration de Gram-Schmidt ---")
```
