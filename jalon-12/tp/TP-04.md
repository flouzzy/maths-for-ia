# TP 04 : Moteur de Recherche Sémantique par Similarité Cosinus et Algèbre Linéaire Fondamentale

## Objectif Mathématique
Ce TP vise à explorer la conception théorique d'un moteur de recherche sémantique en implémentant les concepts fondamentaux de la similarité cosinus. Nous aborderons la vectorisation de texte via TF-IDF, la géométrie des espaces de plongement (implicitement par la représentation vectorielle des documents), et la dualité des vecteurs. En complément, un problème d'algèbre linéaire de niveau "X" sera résolu, illustrant la projection orthogonale de vecteurs, renforçant ainsi la compréhension des opérations vectorielles essentielles.


### Structure de Données Creuses (Sparse)
Une implémentation optimale de la représentation vectorielle des documents s'appuie sur la nature creuse de la matrice document-terme. L'utilisation de dictionnaires Python confère un avantage structurel limitant les opérations du produit scalaire aux seules intersections non nulles, réduisant la complexité pratique bien en deçà de la borne théorique $\mathcal{O}(|V|)$.

```python
import math

# --- Part 1: Text Processing and Vectorization (TF-IDF from scratch) ---

def tokenize(text):
    """
    Tokenizes a given text into a list of lowercase words, removing punctuation.
    """
    cleaned_text = ""
    for char in text:
        # Keep letters, numbers, and spaces. Replace other characters with space.
        if 'a' <= char <= 'z' or 'A' <= char <= 'Z' or '0' <= char <= '9' or char == ' ':
            cleaned_text += char
        else:
            cleaned_text += ' ' # Replace punctuation with space to separate words

    tokens = []
    current_token = ""
    for char in cleaned_text.lower():
        if char == ' ':
            if current_token: # Add token if not empty
                tokens.append(current_token)
            current_token = ""
        else:
            current_token += char
    if current_token: # Add the last token if any
        tokens.append(current_token)
    return tokens

def count_term_frequency(tokens):
    """
    Counts the frequency of each term in a list of tokens.
    Returns a dictionary {term: count}.
    """
    term_freq = {}
    for token in tokens:
        term_freq[token] = term_freq.get(token, 0) + 1
    return term_freq

def compute_tf(term_freqs):
    """
    Calculates Term Frequency (TF) for a document.
    TF(t, d) = (Number of times term t appears in document d) / (Total number of terms in document d).
    Returns a dictionary {term: tf_score}.
    """
    total_terms = sum(term_freqs.values())
    if total_terms == 0:
        return {}
    tf_scores = {}
    for term, count in term_freqs.items():
        tf_scores[term] = count / total_terms
    return tf_scores

def compute_idf(documents_term_freqs):
    """
    Calculates Inverse Document Frequency (IDF) for a collection of documents.
    IDF(t) = log_e(Total number of documents / Number of documents with term t).
    A smoothing factor (+1) is applied to avoid division by zero and log(0).
    Returns a dictionary {term: idf_score}.
    """
    total_documents = len(documents_term_freqs)
    if total_documents == 0:
        return {}

    doc_count_per_term = {}
    for doc_freqs in documents_term_freqs:
        for term in doc_freqs:
            doc_count_per_term[term] = doc_count_per_term.get(term, 0) + 1

    idf_scores = {}
    for term, doc_count in doc_count_per_term.items():
        # Add 1 to numerator and denominator to avoid division by zero and log(0)
        # and to prevent common terms from having an IDF of 0.
        idf_scores[term] = math.log((total_documents + 1) / (doc_count + 1)) + 1

    return idf_scores

def compute_tfidf(tf_scores, idf_scores):
    """
    Calculates TF-IDF scores for a document.
    TF-IDF(t, d) = TF(t, d) * IDF(t).
    Returns a dictionary {term: tfidf_score}.
    """
    tfidf_scores = {}
    for term, tf in tf_scores.items():
        # If a term is not in IDF scores (e.g., it appeared in a document but not in the corpus for IDF calculation),
        # its IDF is effectively 0, making TF-IDF 0.
        tfidf_scores[term] = tf * idf_scores.get(term, 0.0)
    return tfidf_scores

def build_vocabulary(documents_term_freqs):
    """
    Builds a global vocabulary from a collection of documents.
    Returns a sorted list of unique terms.
    """
    vocabulary_set = set()
    for doc_freqs in documents_term_freqs:
        for term in doc_freqs:
            vocabulary_set.add(term)
    return sorted(list(vocabulary_set))

def vectorize(tfidf_scores, vocabulary):
    """
    Converts TF-IDF scores into a dense vector based on a global vocabulary.
    """
    vector = [0.0] * len(vocabulary)
    for i, term in enumerate(vocabulary):
        vector[i] = tfidf_scores.get(term, 0.0)
    return vector

# --- Part 2: Cosine Similarity (from scratch) ---

def dot_product(vec1, vec2):
    """
    Calculates the dot product of two vectors.
    Assumes vectors are of the same length.
    """
    if len(vec1) != len(vec2):
        raise ValueError("Vectors must have the same length for dot product.")

    result = 0.0
    for i in range(len(vec1)):
        result += vec1[i] * vec2[i]
    return result

def vector_norm(vec):
    """
    Calculates the Euclidean norm (L2 norm) of a vector.
    ||v|| = sqrt(v1^2 + v2^2 + ... + vn^2)
    """
    sum_sq = 0.0
    for x in vec:
        sum_sq += x * x
    return math.sqrt(sum_sq)

def cosine_similarity(vec1, vec2):
    """
    Calculates the cosine similarity between two vectors.
    cos(theta) = (vec1 . vec2) / (||vec1|| * ||vec2||)
    Returns 0.0 if either vector has a zero norm to avoid division by zero.
    """
    norm1 = vector_norm(vec1)
    norm2 = vector_norm(vec2)

    if norm1 == 0.0 or norm2 == 0.0:
        return 0.0 # Return 0.0 for zero vectors, indicating no similarity.

    return dot_product(vec1, vec2) / (norm1 * norm2)

# --- Part 3: Semantic Search Engine (basic) ---

class SemanticSearchEngine:
    def __init__(self, documents):
        self.documents = documents
        self.document_tokens = [tokenize(doc) for doc in documents]
        self.document_term_freqs = [count_term_frequency(tokens) for tokens in self.document_tokens]
        self.vocabulary = build_vocabulary(self.document_term_freqs)
        self.idf_scores = compute_idf(self.document_term_freqs)
        self.document_vectors = self._index_documents()

    def _index_documents(self):
        """
        Indexes all documents by converting them into TF-IDF vectors.
        """
        vectors = []
        for doc_freqs in self.document_term_freqs:
            tf_scores = compute_tf(doc_freqs)
            tfidf_scores = compute_tfidf(tf_scores, self.idf_scores)
            vectors.append(vectorize(tfidf_scores, self.vocabulary))
        return vectors

    def search(self, query, top_k=3):
        """
        Searches the indexed documents for the given query and returns
        the top_k most similar documents.
        """
        query_tokens = tokenize(query)
        query_term_freqs = count_term_frequency(query_tokens)
        query_tf_scores = compute_tf(query_term_freqs)
        query_tfidf_scores = compute_tfidf(query_tf_scores, self.idf_scores)
        query_vector = vectorize(query_tfidf_scores, self.vocabulary)

        if vector_norm(query_vector) == 0: # Query is empty or contains only unknown words
            return []

        similarities = []
        for i, doc_vector in enumerate(self.document_vectors):
            sim = cosine_similarity(query_vector, doc_vector)
            similarities.append((sim, self.documents[i]))

        # Sort by similarity in descending order.
        # Python's built-in sort is used for convenience; a custom sort (e.g., bubble sort)
        # could be implemented from scratch if strictly required for sorting logic.
        similarities.sort(key=lambda x: x[0], reverse=True)

        return similarities[:top_k]

# --- Part 4: Problème d'Algèbre de l'X (Projection Orthogonale) ---

def orthogonal_projection(u, v):
    """
    Calcule la projection orthogonale du vecteur u sur le vecteur v.
    Formule: proj_v(u) = ((u . v) / ||v||^2) * v

    Args:
        u (list): Le vecteur à projeter.
        v (list): Le vecteur sur lequel projeter.

    Returns:
        list: Le vecteur résultant de la projection.

    Raises:
        ValueError: Si les vecteurs n'ont pas la même longueur.
    """
    if len(u) != len(v):
        raise ValueError("Les vecteurs doivent avoir la même longueur.")

    norm_v_squared = dot_product(v, v) # ||v||^2

    if norm_v_squared == 0.0:
        # If v is the zero vector, the projection of any u onto v is the zero vector.
        return [0.0] * len(u)

    scalar_factor = dot_product(u, v) / norm_v_squared

    projection = []
    for x in v:
        projection.append(scalar_factor * x)
    return projection
