## Simulation : Série de Fourier divergente de la fonction d'Osgood

egin{lstlisting}[language=Python]
import numpy as np

def dirichlet_kernel(n, x):
    '''Noyau de Dirichlet D_n(x).'''
    if np.isclose(x, 0) or np.isclose(x % (2*np.pi), 0):
        return 2*n + 1
    return np.sin((n + 0.5) * x) / np.sin(x / 2)

# La norme L1 du noyau de Dirichlet (Constante de Lebesgue) diverge
# C'est la clef pour utiliser Baire / Banach-Steinhaus.
def lebesgue_constant(n):
    # Integration numerique approchee
    x = np.linspace(-np.pi, np.pi, 1000)
    dx = 2 * np.pi / 1000
    return np.sum(np.abs(dirichlet_kernel(n, x))) * dx / (2 * np.pi)

# Validation : la constante augmente logarythmiquement avec n
c1 = lebesgue_constant(1)
c10 = lebesgue_constant(10)
c50 = lebesgue_constant(50)
assert c1 < c10 < c50
