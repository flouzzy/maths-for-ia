## Simulation : Construction de fonctions de Weierstrass

egin{lstlisting}[language=Python]
import numpy as np

def weierstrass(x, a=0.5, b=3, terms=50):
    '''
    Fonction continue nulle part derivable (Weierstrass).
    a in (0, 1), b entier impair tel que ab > 1 + 3pi/2.
    '''
    result = 0.0
    for n in range(terms):
        result += (a ** n) * np.cos((b ** n) * np.pi * x)
    return result

# Calcul de taux d'accroissement (qui doit exploser)
x0 = 0.5
h = 1e-6
taux = (weierstrass(x0 + h) - weierstrass(x0)) / h
print(f"Taux d'accroissement en x=0.5 avec h={h}: {taux:.2f}")
# On observe numeriquement que la derivee n'existe pas.
