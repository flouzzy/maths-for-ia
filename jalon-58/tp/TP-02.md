## Simulation : Vérification de la densité de l'ensemble de Cantor

egin{lstlisting}[language=Python]
import random

def is_in_cantor(x, iterations=10):
    '''Verifie si un point x de [0,1] appartient a l'ensemble de Cantor.'''
    for _ in range(iterations):
        x *= 3
        if 1 < x < 2:
            return False
        if x >= 2:
            x -= 2
    return True

# Validation
assert is_in_cantor(1/3) == True
assert is_in_cantor(0.5) == False
# L'ensemble est nulle part dense : entre deux points du Cantor,
# on peut toujours trouver un point qui n'y est pas (un point du milieu retire).
\end{lstlisting}
