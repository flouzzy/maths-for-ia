# TP 1 : Illustration Numérique de la Convergence Simple et Uniforme

## Objectif
Visualiser et vérifier par le calcul la différence entre convergence simple et uniforme.

## Code Python
```python
import math

def f_n(x, n):
    # Suite f_n(x) = (nx) / (1 + n^2 x^2)
    return (n * x) / (1 + n ** 2 * x ** 2)

def f_lim(x):
    # Limite simple
    return 0.0

def test_convergence():
    n = 1000
    # Test de la convergence simple
    # Pour un x fixé (ex: x=0.5), fn(x) tend vers 0
    x_fixed = 0.5
    val = f_n(x_fixed, n)
    assert abs(val - f_lim(x_fixed)) < 1e-2, f"Convergence simple echouee: {val}"

    # Test de la convergence non-uniforme
    # Le maximum est atteint en x_n = 1/n
    x_max = 1 / n
    val_max = f_n(x_max, n)
    # fn(1/n) = 1/2 pour tout n
    assert abs(val_max - 0.5) < 1e-5, "Le max devrait etre 1/2"

if __name__ == "__main__":
    test_convergence()
    print("TP1: Tests de convergence valides.")
```
