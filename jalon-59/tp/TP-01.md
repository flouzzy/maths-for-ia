# TP 1 : Simulation de la non-convergence uniforme

Ce TP illustre la différence entre convergence simple et uniforme en codant la suite de fonctions $f_n(x) = x^n$ sur $[0, 1]$.

```python
import math

def f_n(x, n):
    # Calcule x^n. x doit etre dans [0, 1].
    return math.pow(x, n)

def f_limite(x):
    # Limite simple de f_n sur [0, 1].
    if x < 1.0:
        return 0.0
    elif x == 1.0:
        return 1.0
    else:
        raise ValueError("x doit etre dans [0, 1]")

def supremum_norm(n, num_points=1000):
    # Approxime la norme uniforme ||f_n - f_limite||_inf sur [0, 1].
    max_diff = 0.0
    for i in range(num_points + 1):
        x = i / num_points
        diff = abs(f_n(x, n) - f_limite(x))
        if diff > max_diff:
            max_diff = diff
    return max_diff

def run_tests():
    # Test de la limite simple
    assert f_limite(0.5) == 0.0
    assert f_limite(1.0) == 1.0

    # Test de l'absence de convergence uniforme
    # Pour n'importe quel n, il y a un point x (proche de 1) ou x^n est proche de 1,
    # alors que la limite simple y est 0. Donc la norme inf est toujours ~1.
    for n in [10, 100, 1000]:
        norme = supremum_norm(n)
        assert norme > 0.9, f"Pour n={n}, la norme inf vaut {norme} (devrait etre ~1.0)"

    print("TP 1 : Tests passes. La norme infinie ne tend pas vers 0, la convergence n'est pas uniforme.")

if __name__ == "__main__":
    run_tests()
```
