# TP 5 : Équation de Volterra et Méthode d'Euler

Application du théorème de Peano via une méthode d'Euler polygonale pour l'équation y' = cos(y) avec y(0) = 0.
La fonction cos est bornée par 1, donc toutes les approximations seront 1-Lipschitziennes, formant une famille équicontinue.

```python
import math

def euler_approximation(steps, T=1.0):
    # Resout y'(t) = cos(y(t)), y(0) = 0 avec la methode d'Euler explicite.
    # Renvoie une liste de couples (t, y).

    dt = T / steps
    trajectory = [(0.0, 0.0)]
    y_current = 0.0

    for i in range(1, steps + 1):
        t_current = i * dt
        # La pente est bornee par 1 car |cos| <= 1
        slope = math.cos(y_current)
        y_current += slope * dt
        trajectory.append((t_current, y_current))

    return trajectory

def run_tests():
    T_max = 1.0
    traj_10 = euler_approximation(10, T_max)
    traj_100 = euler_approximation(100, T_max)

    # Verification de la condition de Lipschitz
    for traj in [traj_10, traj_100]:
        for i in range(len(traj) - 1):
            t1, y1 = traj[i]
            t2, y2 = traj[i+1]
            slope = abs((y2 - y1) / (t2 - t1))
            # La pente doit etre bornee par 1 (max de cos) + tolerance float
            assert slope <= 1.0001, f"Propriete Lipschitz violee: pente de {slope}"

    # Point final (y_final approximatif de la solution de y' = cos(y))
    # La vraie solution est arcsin(tanh(t)) ou un equivalent.
    # Analysons juste la stabilite.
    y_final_10 = traj_10[-1][1]
    y_final_100 = traj_100[-1][1]

    # Verifions que la suite approche (Cauchy/compacite)
    assert abs(y_final_10 - y_final_100) < 0.05, "La convergence semble instable, violation du principe de compacite."

    print("TP 5 : Tests passes. Les trajectoires d'Euler sont uniformement Lipschitziennes (equicontinues).")

if __name__ == "__main__":
    run_tests()
```
