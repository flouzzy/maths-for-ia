# TP 2 : Simulation de l'Équicontinuité

## Objectif
Vérifier numériquement la propriété d'équicontinuité sur une famille de fonctions lipschitziennes.

## Code Python
```python
def generate_functions():
    # Famille de fonctions fk(x) = sin(k*x) / k pour k in [1, 100]
    # La derivee est cos(k*x), bornee par 1.
    return [lambda x, k=k: math.sin(k * x) / k for k in range(1, 101)]

import math

def test_equicontinuity():
    funcs = generate_functions()
    epsilon = 0.01
    delta = epsilon # Car les fonctions sont 1-lipschitziennes

    x = 0.5
    y = x + delta * 0.5 # |x-y| < delta

    for f in funcs:
        diff = abs(f(x) - f(y))
        assert diff < epsilon, f"Fail: {diff} >= {epsilon}"

if __name__ == "__main__":
    test_equicontinuity()
    print("TP2: Equicontinuite verifiee avec succes.")
```
