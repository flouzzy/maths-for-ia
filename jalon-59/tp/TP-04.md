# TP 4 : Compacité des fonctions Lipschitziennes

Dans cet exercice, nous allons vérifier qu'une famille générée aléatoirement, mais forcée à être 1-Lipschitzienne et bornée en 0, conserve ces propriétés.

```python
import random

class RandomLipschitzFunction:
    # Genere une approximation lineaire par morceaux d'une fonction 1-Lipschitzienne.
    def __init__(self, num_segments=10):
        self.points = [(0.0, 0.0)] # Commence en 0
        current_y = 0.0

        for i in range(1, num_segments + 1):
            x = i / num_segments
            # Pente aleatoire entre -1 et 1
            slope = random.uniform(-1.0, 1.0)
            dx = 1.0 / num_segments
            current_y += slope * dx
            self.points.append((x, current_y))

    def evaluate(self, x):
        # Evalue par interpolation lineaire.
        if x <= 0: return self.points[0][1]
        if x >= 1: return self.points[-1][1]

        for i in range(len(self.points) - 1):
            x0, y0 = self.points[i]
            x1, y1 = self.points[i+1]
            if x0 <= x <= x1:
                t = (x - x0) / (x1 - x0)
                return y0 + t * (y1 - y0)
        return 0.0

def run_tests():
    random.seed(42)
    f = RandomLipschitzFunction(20)

    # Verification condition initiale
    assert f.evaluate(0.0) == 0.0

    # Verification de la propriete 1-Lipschitz (avec marge numerique)
    num_tests = 100
    for _ in range(num_tests):
        x1 = random.uniform(0.0, 1.0)
        x2 = random.uniform(0.0, 1.0)
        if x1 == x2: continue

        diff_y = abs(f.evaluate(x1) - f.evaluate(x2))
        diff_x = abs(x1 - x2)

        # Pente max est 1, ajout d'une tolérance float
        assert diff_y / diff_x <= 1.0001, "La fonction generatrice a depasse la constante Lipschitz de 1."

    print("TP 4 : Tests passes. La famille construite respecte l'equicontinuite Lipschitz.")

if __name__ == "__main__":
    run_tests()
```
