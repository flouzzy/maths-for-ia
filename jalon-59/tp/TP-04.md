# TP 4 : Norme infinie et espaces de fonctions

## Objectif
Implémenter le calcul de la norme infinie (sup) sur une grille discrétisée.

## Code Python
```python
def norm_inf(f, a, b, num_points=1000):
    dx = (b - a) / num_points
    max_val = 0
    for i in range(num_points + 1):
        x = a + i * dx
        val = abs(f(x))
        if val > max_val:
            max_val = val
    return max_val

def test_norm_inf():
    # f(x) = x^2 sur [-2, 2] -> norm inf = 4
    f = lambda x: x ** 2
    n_inf = norm_inf(f, -2, 2)
    assert abs(n_inf - 4.0) < 1e-2, f"Norme inf incorrecte: {n_inf}"

if __name__ == "__main__":
    test_norm_inf()
    print("TP4: Calcul de la norme infinie verifie.")
```
