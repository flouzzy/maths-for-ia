# TP 2 - Algorithmique de la Nilpotence

## Objectif
Implémenter from scratch, sans aucune librairie externe (pure Python), la détection de la nilpotence et le calcul de l'indice de nilpotence d'une matrice, illustrant le concept clé de la réduction de Jordan.

## Implémentation Pure Python

```python
def matrix_multiply(A, B):
    """Multiplication de deux matrices carrées."""
    n = len(A)
    C = [[0] * n for _ in range(n)]
    for i in range(n):
        for j in range(n):
            for k in range(n):
                C[i][j] += A[i][k] * B[k][j]
    return C

def is_zero_matrix(A):
    """Vérifie si une matrice est strictement nulle."""
    for row in A:
        for val in row:
            if val != 0:
                return False
    return True

def compute_nilpotence_index(A):
    """
    Calcule l'indice de nilpotence de A.
    Retourne l'indice k si A^k = 0, sinon None si A n'est pas nilpotente (testé jusqu'à n).
    """
    n = len(A)
    current_power = A
    if is_zero_matrix(current_power):
        return 1

    for k in range(2, n + 1):
        current_power = matrix_multiply(current_power, A)
        if is_zero_matrix(current_power):
            return k

    return None

# Assertions de Validation Mathématique 2
# Test sur un bloc de Jordan J_3(0)
J3 = [
    [0, 1, 0],
    [0, 0, 1],
    [0, 0, 0]
]

index = compute_nilpotence_index(J3)
assert index == 3, f"Erreur: l'indice de nilpotence attendu est 3, obtenu {index}"
print(f"Validation réussie : Indice de nilpotence de J3 est bien {index}.")
```
