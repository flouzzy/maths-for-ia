## Implémentation Python (From Scratch)

Nous allons implémenter les fonctions nécessaires pour manipuler des matrices et des vecteurs, puis les utiliser pour calculer le noyau, l'image et le rang d'une application linéaire. La représentation des matrices sera une liste de listes (où chaque sous-liste est une ligne de la matrice) et les vecteurs seront de simples listes.

```python
import math

# --- Fonctions utilitaires pour les opérations vectorielles et matricielles ---

def _is_close(a, b, epsilon=1e-9):
    """
    Vérifie si deux nombres flottants sont proches l'un de l'autre,
    compte tenu des imprécisions de calcul en virgule flottante.
    """
    return abs(a - b) < epsilon

def _is_zero(val, epsilon=1e-9):
    """
    Vérifie si un nombre flottant est proche de zéro.
    """
    return _is_close(val, 0.0, epsilon)

def vector_add(v1, v2):
    """
    Additionne deux vecteurs.
    Args:
        v1 (list): Le premier vecteur.
        v2 (list): Le second vecteur.
    Returns:
        list: Le vecteur résultant de l'addition.
    Raises:
        ValueError: Si les vecteurs n'ont pas la même dimension.
    """
    if len(v1) != len(v2):
        raise ValueError("Les vecteurs doivent avoir la même dimension pour être additionnés.")
    return [v1[i] + v2[i] for i in range(len(v1))]

def scalar_vector_mult(scalar, v):
    """
    Multiplie un vecteur par un scalaire.
    Args:
        scalar (float or int): Le scalaire.
        v (list): Le vecteur.
    Returns:
        list: Le vecteur résultant de la multiplication.
    """
    return [scalar * x for x in v]

def dot_product(v1, v2):
    """
    Calcule le produit scalaire de deux vecteurs.
    Args:
        v1 (list): Le premier vecteur.
        v2 (list): Le second vecteur.
    Returns:
        float: Le produit scalaire.
    Raises:
        ValueError: Si les vecteurs n'ont pas la même dimension.
    """
    if len(v1) != len(v2):
        raise ValueError("Les vecteurs doivent avoir la même dimension pour le produit scalaire.")
    return sum(v1[i] * v2[i] for i in range(len(v1)))

def matrix_vector_mult(matrix, vector):
    """
    Multiplie une matrice par un vecteur.
    Args:
        matrix (list of lists): La matrice.
        vector (list): Le vecteur.
    Returns:
        list: Le vecteur résultant.
    Raises:
        ValueError: Si les dimensions ne sont pas compatibles.
    """
    num_rows = len(matrix)
    if num_rows == 0:
        return []
    num_cols = len(matrix[0])

    if num_cols != len(vector):
        raise ValueError("Le nombre de colonnes de la matrice doit correspondre à la dimension du vecteur.")

    result = [0.0] * num_rows
    for i in range(num_rows):
        result[i] = dot_product(matrix[i], vector)
    return result

def matrix_mult(A, B):
    """
    Multiplie deux matrices A et B.
    Args:
        A (list of lists): La première matrice.
        B (list of lists): La seconde matrice.
    Returns:
        list of lists: La matrice résultante A * B.
    Raises:
        ValueError: Si les dimensions ne sont pas compatibles.
    """
    rows_A = len(A)
    if rows_A == 0:
        return []
    cols_A = len(A[0])

    rows_B = len(B)
    if rows_B == 0:
        return []
    cols_B = len(B[0])

    if cols_A != rows_B:
        raise ValueError("Le nombre de colonnes de la première matrice doit être égal au nombre de lignes de la seconde.")

    C = [[0.0 for _ in range(cols_B)] for _ in range(rows_A)]

    for i in range(rows_A):
        for j in range(cols_B):
            for k in range(cols_A):
                C[i][j] += A[i][k] * B[k][j]
    return C

def transpose(matrix):
    """
    Transpose une matrice.
    Args:
        matrix (list of lists): La matrice à transposer.
    Returns:
        list of lists: La matrice transposée.
    """
    if not matrix:
        return []
    num_rows = len(matrix)
    num_cols = len(matrix[0])
    transposed = [[0.0 for _ in range(num_rows)] for _ in range(num_cols)]
    for i in range(num_rows):
        for j in range(num_cols):
            transposed[j][i] = matrix[i][j]
    return transposed

# --- Fonctions pour l'élimination de Gauss-Jordan (RREF) ---

def _swap_rows(matrix, r1, r2):
    """
    Échange deux lignes dans une matrice.
    Args:
        matrix (list of lists): La matrice.
        r1 (int): Indice de la première ligne.
        r2 (int): Indice de la seconde ligne.
    """
    matrix[r1], matrix[r2] = matrix[r2], matrix[r1]

def _scale_row(matrix, r, scalar):
    """
    Multiplie une ligne par un scalaire.
    Args:
        matrix (list of lists): La matrice.
        r (int): Indice de la ligne à modifier.
        scalar (float): Le scalaire par lequel multiplier.
    """
    matrix[r] = [x * scalar for x in matrix[r]]

def _add_rows(matrix, r_target, r_source, scalar):
    """
    Ajoute un multiple d'une ligne à une autre (r_target = r_target + scalar * r_source).
    Args:
        matrix (list of lists): La matrice.
        r_target (int): Indice de la ligne cible.
        r_source (int): Indice de la ligne source.
        scalar (float): Le scalaire par lequel multiplier la ligne source.
    """
    matrix[r_target] = vector_add(matrix[r_target], scalar_vector_mult(scalar, matrix[r_source]))

def rref(matrix_input, epsilon=1e-9):
    """
    Calcule la forme échelonnée réduite (RREF) d'une matrice en utilisant l'élimination de Gauss-Jordan.
    Args:
        matrix_input (list of lists): La matrice d'entrée.
        epsilon (float): Tolérance pour les comparaisons de flottants.
    Returns:
        tuple: (matrix_rref, pivot_cols) où matrix_rref est la matrice en RREF
               et pivot_cols est une liste des indices des colonnes pivots.
    """
    # Crée une copie profonde de la matrice pour ne pas modifier l'originale
    matrix = [list(row) for row in matrix_input]
    num_rows = len(matrix)
    if num_rows == 0:
        return [], []
    num_cols = len(matrix[0])

    lead = 0 # Indice de la colonne pivot actuelle
    pivot_cols = [] # Stocke les indices des colonnes contenant un pivot

    for r in range(num_rows): # Itère sur chaque ligne
        if lead >= num_cols: # Si on a dépassé le nombre de colonnes, on arrête
            break

        i = r
        # Cherche une ligne non nulle pour le pivot dans la colonne 'lead'
        while i < num_rows and _is_zero(matrix[i][lead], epsilon):
            i += 1

        if i < num_rows: # Un pivot a été trouvé dans la colonne 'lead'
            _swap_rows(matrix, i, r) # Place la ligne avec le pivot en position 'r'
            pivot_cols.append(lead) # Enregistre l'indice de la colonne pivot

            lv = matrix[r][lead] # Valeur du pivot
            _scale_row(matrix, r, 1.0 / lv) # Normalise la ligne pour que le pivot soit 1

            # Élimine les autres éléments de la colonne 'lead' pour qu'ils soient zéro
            for i_row in range(num_rows):
                if i_row != r: # Pour toutes les autres lignes
                    lv_other = matrix[i_row][lead]
                    _add_rows(matrix, i_row, r, -lv_other) # L_i_row = L_i_row - (lv_other) * L_r
            lead += 1 # Passe à la colonne suivante pour chercher le prochain pivot
        else: # Pas de pivot dans cette colonne, passer à la colonne suivante sans avancer de ligne
            lead += 1
            # On reste sur la même ligne r pour chercher un pivot dans la colonne suivante

    return matrix, pivot_cols

# --- Fonctions pour Noyau, Image, Rang ---

def get_rank(matrix):
    """
    Calcule le rang d'une matrice (dimension de l'image).
    Le rang est le nombre de lignes non nulles dans la RREF.
    Args:
        matrix (list of lists): La matrice.
    Returns:
        int: Le rang de la matrice.
    """
    rref_matrix, _ = rref(matrix)
    rank = 0
    for row in rref_matrix:
        # Une ligne est non nulle si au moins un de ses éléments n'est pas proche de zéro
        if not all(_is_zero(x) for x in row):
            rank += 1
    return rank

def get_kernel_basis(matrix, epsilon=1e-9):
    """
    Calcule une base du noyau (espace nul) d'une matrice.
    Args:
        matrix (list of lists): La matrice.
        epsilon (float): Tolérance pour les comparaisons de flottants.
    Returns:
        list of lists: Une liste de vecteurs formant une base du noyau.
    """
    rref_matrix, pivot_cols = rref(matrix, epsilon)
    num_rows = len(rref_matrix)
    if num_rows == 0: # Matrice vide
        return []
    num_cols = len(rref_matrix[0])

    # Identifier les variables libres (colonnes qui ne sont pas des pivots)
    free_vars_indices = [j for j in range(num_cols) if j not in pivot_cols]

    kernel_basis = []

    # Pour chaque variable libre, on construit un vecteur de base du noyau
    for free_var_idx in free_vars_indices:
        basis_vector = [0.0] * num_cols
        basis_vector[free_var_idx] = 1.0 # On attribue 1 à la variable libre actuelle

        # Exprimer les variables pivots en fonction des variables libres
        # Chaque ligne de la RREF correspond à une équation pour une variable pivot
        for i, pivot_col_idx in enumerate(pivot_cols):
            # L'équation est de la forme: x_pivot + sum(coeff * x_free) = 0
            # Donc x_pivot = - sum(coeff * x_free)
            # Ici, on ne considère que la variable libre actuelle (basis_vector[free_var_idx] = 1)
            # Les autres variables libres sont implicitement à 0 pour ce vecteur de base
            basis_vector[pivot_col_idx] = -rref_matrix[i][free_var_idx]

        kernel_basis.append(basis_vector)

    # Filtrer les vecteurs nuls si toutes les variables sont pivots (noyau trivial)
    # Cela peut arriver si le noyau est {0} et que le calcul produit un vecteur [0,0,...]
    return [v for v in kernel_basis if not all(_is_zero(x, epsilon) for x in v)]


def get_image_basis(matrix, epsilon=1e-9):
    """
    Calcule une base de l'image (espace colonne) d'une matrice.
    La base est formée par les colonnes de la matrice originale
    qui correspondent aux colonnes pivots de sa RREF.
    Args:
        matrix (list of lists): La matrice.
        epsilon (float): Tolérance pour les comparaisons de flottants.
    Returns:
        list of lists: Une liste de vecteurs (colonnes de la matrice originale)
                       formant une base de l'image.
    """
    _, pivot_cols = rref(matrix, epsilon)

    if not matrix:
        return []

    # Transposer la matrice pour accéder facilement aux colonnes originales
    original_cols = transpose(matrix)

    image_basis = []
    for col_idx in pivot_cols:
        image_basis.append(original_cols[col_idx])

    return image_basis

def check_rank_theorem(matrix):
    """
    Vérifie le théorème du rang pour une matrice donnée.
    Le théorème du rang stipule que dim(E) = dim(ker f) + rg(f).
    Args:
        matrix (list of lists): La matrice représentant l'application linéaire f.
    Returns:
        bool: True si le théorème du rang est vérifié, False sinon.
    """
    if not matrix:
        return True # Cas trivial pour une matrice vide

    # dim(E) est le nombre de colonnes de la matrice (dimension de l'espace de départ)
    dim_E = len(matrix[0])

    dim_ker_f = len(get_kernel_basis(matrix))
    rank_f = get_rank(matrix)

    return dim_E == dim_ker_f + rank_f

def is_projector(matrix, epsilon=1e-9):
    """
    Vérifie si une matrice représente un projecteur (P*P = P).
    Args:
        matrix (list of lists): La matrice à vérifier.
        epsilon (float): Tolérance pour les comparaisons de flottants.
    Returns:
        bool: True si la matrice est un projecteur, False sinon.
    """
    if not matrix:
        return False

    num_rows = len(matrix)
    num_cols = len(matrix[0])

    if num_rows != num_cols:
        # Un projecteur est un endomorphisme, donc la matrice doit être carrée
        return False

    product = matrix_mult(matrix, matrix)

    for i in range(num_rows):
        for j in range(num_cols):
            if not _is_close(product[i][j], matrix[i][j], epsilon):
                return False
    return True

def is_zero_vector(v, epsilon=1e-9):
    """
    Vérifie si un vecteur est le vecteur nul.
    Args:
        v (list): Le vecteur à vérifier.
        epsilon (float): Tolérance pour les comparaisons de flottants.
    Returns:
        bool: True si le vecteur est le vecteur nul, False sinon.
    """
    return all(_is_close(x, 0.0, epsilon) for x in v)

```

### Exercice 1 : Application Directe (Noyau et Image)

**Énoncé du cours :** Soit $f : \mathbb{R}^3 \to \mathbb{R}^2$ définie par $f(x,y,z) = (x+y, y+z)$. Déterminer $\ker f$ et $\text{rg } f$.

**Matrice associée à $f$ :**
Pour $f(x,y,z) = (x+y, y+z)$, la matrice $A$ telle que $f(x) = Ax$ est :
$A = \begin{pmatrix} 1 & 1 & 0 \\ 0 & 1 & 1 \end{pmatrix}$

**Correction Détaillée du cours :**
1.  **Noyau :** Résolvons $f(x,y,z) = (0,0)$.
    -   $x+y = 0 \implies x = -y$
    -   $y+z = 0 \implies z = -y$
    -   Donc $\ker f = \{ (-y, y, -y) \mid y \in \mathbb{R} \} = \text{Vect}((-1, 1, -1))$.
    -   $\dim(\ker f) = 1$.
2.  **Théorème du rang :** $\dim \mathbb{R}^3 = \dim(\ker f) + \text{rg } f \implies 3 = 1 + \text{rg } f$.
    -   $\text{rg } f = 2$.

**Validation avec le code Python :**

```python
# Matrice de l'application linéaire f de l'Exercice 1
matrix_ex1 = [[1, 1, 0],
              [0, 1, 1]]

print(f"Matrice A de l'Exercice 1:\n{matrix_ex1}")

# Calcul de la base du noyau
kernel_basis_ex1 = get_kernel_basis(matrix_ex1)
print(f"\nBase du noyau de f: {kernel_basis_ex1}")
dim_ker_f_ex1 = len(kernel_basis_ex1)
print(f"Dimension du noyau de f: {dim_ker_f_ex1}")

# Assertions pour le noyau
# Le cours donne Vect((-1, 1, -1)). Notre algorithme peut donner un multiple, par exemple (1, -1, 1).
# Nous vérifions la dimension et que le vecteur généré est bien dans le noyau.
assert dim_ker_f_ex1 == 1, f"Erreur: La dimension du noyau devrait être 1, mais est {dim_ker_f_ex1}"
# Vérifions que le vecteur de base trouvé est bien dans le noyau (f(v) = 0)
if kernel_basis_ex1:
    test_vec_ker = kernel_basis_ex1[0]
    result_mult = matrix_vector_mult(matrix_ex1, test_vec_ker)
    assert is_zero_vector(result_mult), f"Erreur: Le vecteur {test_vec_ker} n'est pas dans le noyau."
    print(f"Vecteur de base du noyau ({test_vec_ker}) vérifié: f(v) = {result_mult} (proche de zéro)")


# Calcul du rang
rank_f_ex1 = get_rank(matrix_ex1)
print(f"\nRang de f: {rank_f_ex1}")

# Assertions pour le rang
assert rank_f_ex1 == 2, f"Erreur: Le rang de f devrait être 2, mais est {rank_f_ex1}"

# Calcul de la base de l'image
image_basis_ex1 = get_image_basis(matrix_ex1)
print(f"\nBase de l'image de f: {image_basis_ex1}")
dim_im_f_ex1 = len(image_basis_ex1)
print(f"Dimension de l'image de f: {dim_im_f_ex1}")

# Assertions pour l'image
assert dim_im_f_ex1 == rank_f_ex1, f"Erreur: La dimension de l'image ({dim_im_f_ex1}) ne correspond pas au rang ({rank_f_ex1})."
# Pour cet exemple, l'image est R^2, donc une base canonique ou toute base de R^2 est valide.
# Les colonnes pivots de la matrice originale sont (1,0) et (1,1).
assert image_basis_ex1 == [[1, 0], [1, 1]], f"Erreur: Base de l'image inattendue: {image_basis_ex1}"


# Vérification du théorème du rang
dim_E_ex1 = len(matrix_ex1[0]) # Dimension de l'espace de départ (nombre de colonnes)
print(f"\nDimension de l'espace de départ E (R^3): {dim_E_ex1}")
is_rank_theorem_valid_ex1 = check_rank_theorem(matrix_ex1)
print(f"Théorème du rang vérifié: {is_rank_theorem_valid_ex1} (dim(E) = {dim_E_ex1}, dim(ker f) = {dim_ker_f_ex1}, rg(f) = {rank_f_ex1})")

# Assertion pour le théorème du rang
assert is_rank_theorem_valid_ex1 == True, "Erreur: Le théorème du rang n'est pas vérifié pour l'Exercice 1."

print("\n--- Validation de l'Exercice 1 terminée avec succès ---")
```

**Explication de la validation :**
Le code calcule correctement la dimension du noyau (1) et le rang (2) de l'application linéaire $f$. La base du noyau trouvée par `get_kernel_basis` est `[[1.0, -1.0, 1.0]]`, qui est un multiple scalaire du vecteur `(-1, 1, -1)` donné dans le cours, et donc génère le même sous-espace vectoriel. La base de l'image est `[[1, 0], [1, 1]]`, qui sont les colonnes pivots de la matrice originale, et forment bien une base de $\mathbb{R}^2$. Enfin, le théorème du rang est vérifié : $\dim(\mathbb{R}^3) = 3$, $\dim(\ker f) = 1$, $\text{rg}(f) = 2$, et $3 = 1 + 2$.

### Exercice 2 : Niveau Avancé (Projecteurs)

**Énoncé du cours :** Soit $p \in \mathcal{L}(E)$ tel que $p \circ p = p$. Démontrer que $E = \ker p \oplus \text{Im } p$.

**Validation avec le code Python :**
Pour valider cette propriété, nous allons utiliser une matrice de projecteur simple, par exemple la projection sur l'axe des abscisses dans $\mathbb{R}^2$.
La matrice associée est $P = \begin{pmatrix} 1 & 0 \\ 0 & 0 \end{pmatrix}$.

1.  **Vérification que $P$ est un projecteur :** $P \circ P = P$.
    $P \cdot P = \begin{pmatrix} 1 & 0 \\ 0 & 0 \end{pmatrix} \cdot \begin{pmatrix} 1 & 0 \\ 0 & 0 \end{pmatrix} = \begin{pmatrix} 1 \cdot 1 + 0 \cdot 0 & 1 \cdot 0 + 0 \cdot 0 \\ 0 \cdot 1 + 0 \cdot 0 & 0 \cdot 0 + 0 \cdot 0 \end{pmatrix} = \begin{pmatrix} 1 & 0 \\ 0 & 0 \end{pmatrix} = P$.
    La condition est remplie.

2.  **Calcul du noyau de $P$ :** $P(x,y) = (0,0) \implies (x,0) = (0,0) \implies x=0$.
    $\ker P = \{ (0,y) \mid y \in \mathbb{R} \} = \text{Vect}((0,1))$.
    $\dim(\ker P) = 1$.

3.  **Calcul de l'image de $P$ :** L'image est l'ensemble des vecteurs $(x,0)$.
    $\text{Im } P = \{ (x,0) \mid x \in \mathbb{R} \} = \text{Vect}((1,0))$.
    $\dim(\text{Im } P) = 1$.

4.  **Vérification de la somme directe :**
    -   L'intersection $\ker P \cap \text{Im } P = \{ (0,y) \} \cap \{ (x,0) \} = \{ (0,0) \}$. L'intersection est triviale.
    -   La somme des dimensions : $\dim(\ker P) + \dim(\text{Im } P) = 1 + 1 = 2$.
    -   La dimension de l'espace de départ $E = \mathbb{R}^2$ est 2.
    -   Puisque $\dim(\ker P) + \dim(\text{Im } P) = \dim E$ et que l'intersection est triviale, on a bien $E = \ker P \oplus \text{Im } P$.

**Validation avec le code Python :**

```python
# Matrice d'un projecteur (projection sur l'axe des x dans R^2)
projector_matrix = [[1, 0],
                    [0, 0]]

print(f"\nMatrice P du projecteur:\n{projector_matrix}")

# 1. Vérifier que c'est bien un projecteur (P*P = P)
is_proj = is_projector(projector_matrix)
print(f"\nLa matrice P est-elle un projecteur? {is_proj}")
assert is_proj == True, "Erreur: La matrice donnée n'est pas un projecteur."

# 2. Calculer la base du noyau
kernel_basis_proj = get_kernel_basis(projector_matrix)
print(f"\nBase du noyau de P: {kernel_basis_proj}")
dim_ker_p = len(kernel_basis_proj)
print(f"Dimension du noyau de P: {dim_ker_p}")
assert dim_ker_p == 1, f"Erreur: La dimension du noyau devrait être 1, mais est {dim_ker_p}"
# Vérifions que le vecteur de base trouvé est bien dans le noyau (P*v = 0)
if kernel_basis_proj:
    test_vec_ker_proj = kernel_basis_proj[0]
    result_mult_proj = matrix_vector_mult(projector_matrix, test_vec_ker_proj)
    assert is_zero_vector(result_mult_proj), f"Erreur: Le vecteur {test_vec_ker_proj} n'est pas dans le noyau."
    print(f"Vecteur de base du noyau ({test_vec_ker_proj}) vérifié: P*v = {result_mult_proj} (proche de zéro)")


# 3. Calculer la base de l'image
image_basis_proj = get_image_basis(projector_matrix)
print(f"\nBase de l'image de P: {image_basis_proj}")
dim_im_p = len(image_basis_proj)
print(f"Dimension de l'image de P: {dim_im_p}")
assert dim_im_p == 1, f"Erreur: La dimension de l'image devrait être 1, mais est {dim_im_p}"
# Vérifions que le vecteur de base trouvé est bien dans l'image (est une colonne de P)
assert image_basis_proj == [[1, 0]], f"Erreur: Base de l'image inattendue: {image_basis_proj}"


# 4. Vérifier la somme directe E = ker P + Im P
# La somme des dimensions doit être égale à dim(E)
dim_E_proj = len(projector_matrix[0]) # Dimension de l'espace de départ (R^2)
print(f"\nDimension de l'espace de départ E (R^2): {dim_E_proj}")
print(f"Somme des dimensions: dim(ker P) + dim(Im P) = {dim_ker_p} + {dim_im_p} = {dim_ker_p + dim_im_p}")

# Assertion pour la somme des dimensions
assert (dim_ker_p + dim_im_p) == dim_E_proj, "Erreur: La somme des dimensions du noyau et de l'image ne correspond pas à la dimension de l'espace de départ."

# L'intersection triviale est implicite si les bases sont correctement calculées
# et que la somme des dimensions est égale à dim(E).
# On peut aussi vérifier le théorème du rang pour un projecteur
is_rank_theorem_valid_proj = check_rank_theorem(projector_matrix)
print(f"Théorème du rang vérifié pour P: {is_rank_theorem_valid_proj} (dim(E) = {dim_E_proj}, dim(ker P) = {dim_ker_p}, rg(P) = {dim_im_p})")
assert is_rank_theorem_valid_proj == True, "Erreur: Le théorème du rang n'est pas vérifié pour le projecteur."
```
