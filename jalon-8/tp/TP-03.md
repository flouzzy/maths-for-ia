## Implémentation Python (From Scratch)

```python
import math

# Une petite tolérance pour les comparaisons de flottants à zéro afin de gérer les imprécisions
EPSILON = 1e-9

def multiply_matrix_vector(matrix, vector):
    """
    Multiplie une matrice par un vecteur.
    Cette fonction représente l'application d'une transformation linéaire.

    Args:
        matrix (list of list of float): La matrice (m lignes, n colonnes) de l'application linéaire.
        vector (list of float): Le vecteur (n éléments) de l'espace de départ.
    Returns:
        list of float: Le vecteur résultant (m éléments) de l'espace d'arrivée.
    Raises:
        ValueError: Si le nombre de colonnes de la matrice ne correspond pas à la dimension du vecteur.
    """
    if not matrix or not matrix[0]:
        # Gérer le cas d'une matrice vide ou de dimension 0xN
        return [] if vector else [] # Retourne vide si le vecteur est aussi vide, sinon erreur de dimension
    if len(matrix[0]) != len(vector):
        raise ValueError("Le nombre de colonnes de la matrice doit être égal à la dimension du vecteur.")

    num_rows = len(matrix)
    num_cols = len(matrix[0])

    result = [0.0] * num_rows
    for i in range(num_rows):
        for j in range(num_cols):
            result[i] += matrix[i][j] * vector[j]
    return result

def _swap_rows(matrix, r1, r2):
    """Fonction utilitaire pour échanger deux lignes dans une matrice."""
    matrix[r1], matrix[r2] = matrix[r2], matrix[r1]

def _scale_row(matrix, r, scalar):
    """Fonction utilitaire pour multiplier une ligne par un scalaire."""
    for j in range(len(matrix[0])):
        matrix[r][j] *= scalar

def _add_scaled_row(matrix, r_target, r_source, scalar):
    """Fonction utilitaire pour ajouter une ligne source multipliée par un scalaire à une ligne cible."""
    for j in range(len(matrix[0])):
        matrix[r_target][j] += scalar * matrix[r_source][j]

def gaussian_elimination(matrix):
    """
    Met une matrice sous forme échelonnée réduite par l'élimination de Gauss-Jordan.
    Cette fonction est cruciale pour déterminer le rang et le noyau d'une application linéaire.

    Args:
        matrix (list of list of float): La matrice d'entrée.
    Returns:
        tuple: (matrice_echelonnee_reduite, indices_colonnes_pivots)
    """
    if not matrix or not matrix[0]:
        return [], []

    num_rows = len(matrix)
    num_cols = len(matrix[0])

    # Créer une copie profonde de la matrice pour ne pas modifier l'originale
    m = [row[:] for row in matrix]

    pivot_cols = []
    lead = 0 # Indice de la colonne où l'on cherche le prochain pivot

    for r in range(num_rows): # r est l'indice de la ligne actuelle que nous traitons
        if lead >= num_cols:
            break

        # Trouver un pivot non nul dans la colonne 'lead' à partir de la ligne 'r'
        i = r
        while i < num_rows and abs(m[i][lead]) < EPSILON:
            i += 1

        if i < num_rows: # Un pivot a été trouvé à la ligne 'i' dans la colonne 'lead'
            _swap_rows(m, i, r) # Amener la ligne avec le pivot en position 'r'
            _scale_row(m, r, 1.0 / m[r][lead]) # Normaliser le pivot à 1

            # Éliminer les autres éléments de la colonne 'lead' (rendre les autres éléments de la colonne 0)
            for i_other in range(num_rows):
                if i_other != r:
                    _add_scaled_row(m, i_other, r, -m[i_other][lead])

            pivot_cols.append(lead)
            lead += 1 # Passer à la colonne suivante pour le prochain pivot
        else: # Pas de pivot dans cette colonne 'lead' pour la ligne 'r' ou en dessous
            # Cette colonne 'lead' n'est pas une colonne pivot.
            # Nous devons chercher un pivot pour la même ligne 'r' dans la colonne suivante.
            lead += 1
            r -= 1 # Décrémenter r pour que la boucle 'for r' le réincrémente et traite la même ligne 'r' avec le nouveau 'lead'

    return m, pivot_cols

def get_rank(echelon_matrix, pivot_cols):
    """
    Calcule le rang d'une application linéaire (ou d'une matrice).
    Le rang est la dimension de l'image, qui est égale au nombre de pivots dans la forme échelonnée réduite.
    """
    return len(pivot_cols)

def get_nullity(matrix_dim, rank):
    """
    Calcule la dimension du noyau (nullité) à partir de la dimension de l'espace de départ et du rang.
    Ceci est une application directe du théorème du rang.
    """
    return matrix_dim - rank

def find_null_space_basis(echelon_matrix, pivot_cols):
    """
    Trouve une base du noyau à partir de la matrice échelonnée réduite.
    Pour chaque variable libre, un vecteur de base est construit en résolvant le système homogène.

    Args:
        echelon_matrix (list of list of float): La matrice sous forme échelonnée réduite.
        pivot_cols (list of int): Les indices des colonnes pivots.
    Returns:
        list of list of float: Une liste de vecteurs formant une base du noyau.
    """
    if not echelon_matrix or not echelon_matrix[0]:
        return []

    num_rows = len(echelon_matrix)
    num_cols = len(echelon_matrix[0])

    # Identifier les colonnes libres (variables non-pivots)
    free_cols = [j for j in range(num_cols) if j not in pivot_cols]

    basis_vectors = []

    for free_var_col in free_cols:
        # Créer un vecteur où la variable libre courante prend la valeur 1,
        # et les autres variables libres prennent la valeur 0.
        null_vector = [0.0] * num_cols
        null_vector[free_var_col] = 1.0

        # Résoudre pour les variables pivots en utilisant les équations de la matrice échelonnée réduite.
        # Pour chaque ligne pivot (r_idx) et sa colonne pivot (pivot_col_idx):
        # L'équation est de la forme: x[pivot_col_idx] + sum(echelon_matrix[r_idx][j] * x[j]) = 0
        # où j sont les indices des variables libres.
        # Donc, x[pivot_col_idx] = - sum(echelon_matrix[r_idx][j] * x[j])
        for r_idx, pivot_col_idx in enumerate(pivot_cols):
            val = 0.0
            for j in free_cols:
                val += echelon_matrix[r_idx][j] * null_vector[j]
            null_vector[pivot_col_idx] = -val

        basis_vectors.append(null_vector)

    return basis_vectors

def find_image_basis(original_matrix, pivot_cols):
    """
    Trouve une base de l'image en utilisant les colonnes de la matrice originale
    qui correspondent aux colonnes pivots de la forme échelonnée réduite.
    Ces colonnes sont linéairement indépendantes et engendrent l'image.

    Args:
        original_matrix (list of list of float): La matrice originale de l'application linéaire.
        pivot_cols (list of int): Les indices des colonnes pivots.
    Returns:
        list of list of float: Une liste de vecteurs formant une base de l'image.
    """
    if not original_matrix or not original_matrix[0]:
        return []

    image_basis = []
    for col_idx in pivot_cols:
        # Extraire la colonne de la matrice originale
        column_vector = [original_matrix[i][col_idx] for i in range(len(original_matrix))]
        image_basis.append(column_vector)

    return image_basis

def check_rank_theorem(matrix):
    """
    Vérifie le théorème du rang pour une application linéaire représentée par une matrice.
    Le théorème du rang stipule que dim(E) = dim(ker f) + dim(Im f).

    Args:
        matrix (list of list of float): La matrice de l'application linéaire.
    Returns:
        tuple: (dimension_espace_depart, dimension_noyau, dimension_image, theoreme_valide)
    """
    if not matrix or not matrix[0]:
        # Cas trivial: matrice vide ou 0xN. dim(E)=0, dim(ker)=0, dim(Im)=0.
        return 0, 0, 0, True

    num_rows = len(matrix)
    num_cols = len(matrix[0]) # Dimension de l'espace de départ E

    # 1. Calculer la forme échelonnée réduite et les pivots
    echelon_m, pivot_cols = gaussian_elimination(matrix)

    # 2. Calculer le rang (dimension de l'image)
    rank_f = get_rank(echelon_m, pivot_cols)

    # 3. Calculer la dimension du noyau (nullité)
    nullity_f = get_nullity(num_cols, rank_f)

    # Pour une validation plus robuste, on calcule aussi les bases et vérifions leurs dimensions
    ker_basis = find_null_space_basis(echelon_m, pivot_cols)
    dim_ker_from_basis = len(ker_basis)

    im_basis = find_image_basis(matrix, pivot_cols)
    dim_im_from_basis = len(im_basis)

    # Vérifier la cohérence des dimensions calculées.
    # Ces assertions garantissent que nos fonctions de base du noyau et de l'image sont correctes.
    assert nullity_f == dim_ker_from_basis, \
        f"Incohérence: nullité calculée ({nullity_f}) != dimension de la base du noyau ({dim_ker_from_basis})"
    assert rank_f == dim_im_from_basis, \
        f"Incohérence: rang calculé ({rank_f}) != dimension de la base de l'image ({dim_im_from_basis})"

    # 4. Vérifier le théorème du rang
    is_theorem_valid = (num_cols == nullity_f + rank_f)

    return num_cols, nullity_f, rank_f, is_theorem_valid

```

## Validation Mathématique

Nous allons valider notre implémentation avec des exemples concrets, en nous basant sur les définitions et théorèmes du cours.

### Exemple 1 : Application Linéaire $f: \mathbb{R}^3 \to \mathbb{R}^2$
Soit l'application linéaire $f(x,y,z) = (x+y, y+z)$.
La matrice associée à cette application dans les bases canoniques est :
$A = \begin{pmatrix} 1 & 1 & 0 \\ 0 & 1 & 1 \end{pmatrix}$

**1. Application de la transformation :**
Appliquons $f$ au vecteur $(1, 2, 3)$.
$f(1,2,3) = (1+2, 2+3) = (3,5)$.
Avec notre fonction `multiply_matrix_vector`:

```python
matrix_A = [[1.0, 1.0, 0.0],
            [0.0, 1.0, 1.0]]
vector_x = [1.0, 2.0, 3.0]
result_vector = multiply_matrix_vector(matrix_A, vector_x)
print(f"f({vector_x}) = {result_vector}")
assert all(abs(val - expected_val) < EPSILON for val, expected_val in zip(result_vector, [3.0, 5.0]))
```

**2. Détermination du Noyau ($\ker f$) et de sa dimension :**
Le noyau est l'ensemble des vecteurs $(x,y,z)$ tels que $f(x,y,z) = (0,0)$.
Cela correspond à la résolution du système :
$x+y = 0 \implies x = -y$
$y+z = 0 \implies z = -y$
Donc $\ker f = \{(-y, y, -y) \mid y \in \mathbb{R}\} = \text{Vect}((-1, 1, -1))$.
La dimension du noyau est 1.

Avec notre code :

```python
matrix_A = [[1.0, 1.0, 0.0],
            [0.0, 1.0, 1.0]]

echelon_m_A, pivot_cols_A = gaussian_elimination(matrix_A)
print(f"Matrice A échelonnée réduite:\n{echelon_m_A}")
print(f"Colonnes pivots pour A: {pivot_cols_A}")

ker_basis_A = find_null_space_basis(echelon_m_A, pivot_cols_A)
print(f"Base du noyau de A: {ker_basis_A}")
dim_ker_A = len(ker_basis_A)
print(f"Dimension du noyau de A: {dim_ker_A}")

# Le vecteur (-1, 1, -1) est un multiple de (1, -1, 1), donc c'est une base valide.
assert dim_ker_A == 1
if ker_basis_A:
    # Normalisons le vecteur trouvé pour une comparaison plus facile
    # On peut diviser par le dernier élément si non nul, ou par le premier, etc.
    # Ici, on s'attend à un vecteur de la forme [k, -k, k] ou [-k, k, -k]
    # Si ker_basis_A[0] est [1.0, -1.0, 1.0], c'est un multiple de [-1.0, 1.0, -1.0]
    expected_vec_1 = [1.0, -1.0, 1.0]
    expected_vec_2 = [-1.0, 1.0, -1.0]
    is_multiple_1 = all(abs(val - expected_val) < EPSILON for val, expected_val in zip(ker_basis_A[0], expected_vec_1))
    is_multiple_2 = all(abs(val - expected_val) < EPSILON for val, expected_val in zip(ker_basis_A[0], expected_vec_2))
    assert is_multiple_1 or is_multiple_2
```

**3. Détermination de l'Image ($\text{Im } f$) et de sa dimension (Rang) :**
L'image est engendrée par les vecteurs colonnes de la matrice $A$.
Les colonnes pivots de la matrice échelonnée réduite sont les colonnes 0 et 1.
Les colonnes correspondantes de la matrice originale $A$ sont $\begin{pmatrix} 1 \\ 0 \end{pmatrix}$ et $\begin{pmatrix} 1 \\ 1 \end{pmatrix}$.
Ces deux vecteurs sont linéairement indépendants et appartiennent à $\mathbb{R}^2$. Ils forment donc une base de $\mathbb{R}^2$.
L'image est $\mathbb{R}^2$, et sa dimension (le rang) est 2.

Avec notre code :

```python
matrix_A = [[1.0, 1.0, 0.0],
            [0.0, 1.0, 1.0]]

# Ré-exécuté gaussian_elimination pour clarté, mais les résultats sont déjà disponibles
echelon_m_A, pivot_cols_A = gaussian_elimination(matrix_A)
im_basis_A = find_image_basis(matrix_A, pivot_cols_A)
print(f"Base de l'image de A: {im_basis_A}")
dim_im_A = len(im_basis_A)
print(f"Dimension de l'image de A (Rang): {dim_im_A}")

assert dim_im_A == 2
assert im_basis_A == [[1.0, 0.0], [1.0, 1.0]]
```

**4. Vérification du Théorème du Rang :**
Le théorème du rang stipule que $\dim E = \dim(\ker f) + \text{rg}(f)$.
Pour notre exemple, $\dim E = \dim \mathbb{R}^3 = 3$.
Nous avons trouvé $\dim(\ker f) = 1$ et $\text{rg}(f) = 2$.
Donc $3 = 1 + 2$, ce qui est vrai.

Avec notre code :

```python
matrix_A = [[1.0, 1.0, 0.0],
            [0.0, 1.0, 1.0]]

dim_E_A, dim_ker_A_check, dim_Im_A_check, is_valid_A = check_rank_theorem(matrix_A)
print(f"Pour la matrice A:")
print(f"  Dimension de l'espace de départ (E): {dim_E_A}")
print(f"  Dimension du noyau (ker f): {dim_ker_A_check}")
print(f"  Dimension de l'image (Im f) / Rang: {dim_Im_A_check}")
print(f"  Théorème du rang valide: {is_valid_A}")

assert dim_E_A == 3
assert dim_ker_A_check == 1
assert dim_Im_A_check == 2
assert is_valid_A is True
```

### Exemple 2 : Matrice avec un noyau plus grand
Soit la matrice $B = \begin{pmatrix} 1 & 2 & 3 \\ 2 & 4 & 6 \\ 0 & 1 & 1 \end{pmatrix}$.
$\dim E = \dim \mathbb{R}^3 = 3$.

```python
matrix_B = [[1.0, 2.0, 3.0],
            [2.0, 4.0, 6.0],
            [0.0, 1.0, 1.0]]

dim_E_B, dim_ker_B, dim_Im_B, is_valid_B = check_rank_theorem(matrix_B)
print(f"\nPour la matrice B:")
print(f"  Dimension de l'espace de départ (E): {dim_E_B}")
print(f"  Dimension du noyau (ker f): {dim_ker_B}")
print(f"  Dimension de l'image (Im f) / Rang: {dim_Im_B}")
print(f"  Théorème du rang valide: {is_valid_B}")

assert dim_E_B == 3
assert dim_ker_B == 1
assert dim_Im_B == 2
assert is_valid_B is True

# Vérifions la base du noyau pour B
echelon_m_B, pivot_cols_B = gaussian_elimination(matrix_B)
ker_basis_B = find_null_space_basis(echelon_m_B, pivot_cols_B)
print(f"Base du noyau de B: {ker_basis_B}")
# Pour B, le système homogène est:
# x + 2y + 3z = 0
# 2x + 4y + 6z = 0 (redondante)
# y + z = 0 => y = -z
# Substituant y dans la première: x + 2(-z) + 3z = 0 => x + z = 0 => x = -z
# Donc ker f = {(-z, -z, z) | z in R} = Vect((-1, -1, 1))
assert len(ker_basis_B) == 1
if ker_basis_B:
    # Le vecteur calculé par le code devrait être un multiple de (-1, -1, 1)
    # Par exemple, si z=1, on a [-1, -1, 1]
    # Si le code donne [1, 1, -1], c'est aussi correct.
    # Vérifions la proportionnalité
    first_vec = ker_basis_B[0]
    # On cherche un scalaire k tel que first_vec = k * [-1, -1, 1]
    # Si first_vec[2] est 0, on peut avoir un problème de division par zéro.
    # On cherche le premier élément non nul pour normaliser.
    if abs(first_vec[2]) > EPSILON:
        k = first_vec[2] / 1.0
    elif abs(first_vec[1]) > EPSILON:
        k = first_vec[1] / -1.0
    elif abs(first_vec[0]) > EPSILON:
        k = first_vec[0] / -1.0
    else: # Vecteur nul, ce qui ne devrait pas arriver pour une base
        k = 0.0

    expected_normalized_vec = [-1.0, -1.0, 1.0]
    actual_normalized_vec = [v / k if abs(k) > EPSILON else v for v in first_vec] # Gérer k=0

    assert all(abs(val - expected_val) < EPSILON for val, expected_val in zip(actual_normalized_vec, expected_normalized_vec))


# Vérifions la base de l'image pour B
im_basis_B = find_image_basis(matrix_B, pivot_cols_B)
print(f"Base de l'image de B: {im_basis_B}")
# La forme échelonnée réduite de B est:
# [[1, 0, 1],
#  [0, 1, 1],
#  [0, 0, 0]]
# Les pivots sont en colonnes 0 et 1.
# Donc les colonnes de la matrice originale sont: [1, 2, 0] et [2, 4, 1]
assert len(im_basis_B) == 2
assert im_basis_B == [[1.0, 2.0, 0.0], [2.0, 4.0, 1.0]]
```

Ces exemples démontrent que notre implémentation Python est capable de calculer correctement le noyau, l'image et de vérifier le théorème du rang, en respectant les principes mathématiques de l'algèbre linéaire. Les assertions garantissent la conformité des résultats avec la théorie.
