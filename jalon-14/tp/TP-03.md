# TP 3 - Suite de Fibonacci et complexité algorithmique

## 1. Objectif du TP
L'objectif est d'implémenter le calcul de la suite de Fibonacci via différentes méthodes (récursive, itérative, matricielle) et d'analyser la complexité.

## 2. Implémentation

```python
def fibonacci_iteratif(n):
    assert isinstance(n, int) and n >= 0
    if n == 0: return 0
    if n == 1: return 1
    a, b = 0, 1
    for _ in range(2, n + 1):
        a, b = b, a + b
    return b

def puissance_matrice(M, n):
    assert n >= 1
    if n == 1: return M
    if n % 2 == 0:
        moitie = puissance_matrice(M, n // 2)
        return [[moitie[0][0]*moitie[0][0] + moitie[0][1]*moitie[1][0], moitie[0][0]*moitie[0][1] + moitie[0][1]*moitie[1][1]],
                [moitie[1][0]*moitie[0][0] + moitie[1][1]*moitie[1][0], moitie[1][0]*moitie[0][1] + moitie[1][1]*moitie[1][1]]]
    else:
        moitie = puissance_matrice(M, (n-1))
        return [[M[0][0]*moitie[0][0] + M[0][1]*moitie[1][0], M[0][0]*moitie[0][1] + M[0][1]*moitie[1][1]],
                [M[1][0]*moitie[0][0] + M[1][1]*moitie[1][0], M[1][0]*moitie[0][1] + M[1][1]*moitie[1][1]]]

def fibonacci_matriciel(n):
    assert isinstance(n, int) and n >= 0
    if n == 0: return 0
    if n == 1: return 1
    M = [[1, 1], [1, 0]]
    Mn = puissance_matrice(M, n - 1)
    return Mn[0][0]

assert fibonacci_iteratif(10) == 55
assert fibonacci_matriciel(10) == 55
assert fibonacci_iteratif(50) == fibonacci_matriciel(50)
print("Validation réussie.")
```
