# TP 3 : La Jauge de Minkowski sur un Polytope

On calcule la jauge de Minkowski pour un polytope (intersection de demi-plans) contenant l'origine, qui est la forme classique pour la programmation linéaire.

```python
def dot_product(v1, v2):
    return sum(x*y for x, y in zip(v1, v2))

class Polytope:
    # Defini par A x <= b
    # A liste de vecteurs normaux, b liste de scalaires
    def __init__(self, A, b):
        self.A = A
        self.b = b
        # Verifier que l'origine est dans l'interieur
        for bi in self.b:
            if bi <= 0:
                raise ValueError("L'origine n'est pas dans l'interieur strictement du polytope.")

    def gauge(self, x):
        # j(x) = inf { t > 0 | x/t in Polytope }
        # x/t in Polytope => A(x/t) <= b => A x <= t b
        # Comme b > 0, t >= (A_i x) / b_i
        # Le plus petit t qui marche pour TOUTES les contraintes est le max des ratios.
        max_ratio = 0.0
        for ai, bi in zip(self.A, self.b):
            val = dot_product(ai, x)
            if val > 0:
                ratio = val / bi
                if ratio > max_ratio:
                    max_ratio = ratio
        return max_ratio

def run_test():
    # Carre de cote 2 centre en 0. x<=1, -x<=1, y<=1, -y<=1
    A = [[1, 0], [-1, 0], [0, 1], [0, -1]]
    b = [1.0, 1.0, 1.0, 1.0]

    P = Polytope(A, b)

    # Point (0.5, 0.5) est dans P. Jauge doit etre max(0.5, 0.5) = 0.5
    assert abs(P.gauge([0.5, 0.5]) - 0.5) < 1e-9

    # Point (2, 0) est hors de P. Jauge doit etre max(2, 0) = 2.0
    assert abs(P.gauge([2.0, 0.0]) - 2.0) < 1e-9

    # Jauge satisfait p(x+y) <= p(x) + p(y)
    p_x = P.gauge([0.5, 0.8])
    p_y = P.gauge([0.6, -0.2])
    p_xy = P.gauge([1.1, 0.6])
    assert p_xy <= p_x + p_y + 1e-9

if __name__ == "__main__":
    run_test()
    print("TP3: Propriétés de la jauge sur polytope validées.")
```
