# TP 3 : Algorithme du Perceptron (Séparation de points)

Implémentation de la recherche itérative d'un hyperplan séparateur pour deux ensembles finis.

```python
def dot(v1, v2): return sum(x*y for x,y in zip(v1,v2))
def add(v1, v2): return [x+y for x,y in zip(v1,v2)]
def scale(c, v): return [c*x for x in v]

def perceptron(X, y, max_iter=1000):
    n = len(X[0])
    w = [0.0] * n
    b = 0.0
    for _ in range(max_iter):
        errors = 0
        for i in range(len(X)):
            if y[i] * (dot(w, X[i]) + b) <= 0:
                w = add(w, scale(y[i], X[i]))
                b += y[i]
                errors += 1
        if errors == 0:
            break
    return w, b

def test_perceptron():
    X = [[1, 2], [2, 1], [-1, -1], [-2, -2]]
    y = [1, 1, -1, -1]
    w, b = perceptron(X, y)

    for xi, yi in zip(X, y):
        assert yi * (dot(w, xi) + b) > 0
    print("TP 3 : Validé")

if __name__ == "__main__":
    test_perceptron()
```
