# TP 4 : Construction itérative de la projection sur un convexe (Algo de Halpern)

Le théorème de projection sur un convexe fermé (lié à Hahn-Banach 2) peut être implémenté via des méthodes itératives dans des espaces de Hilbert.

```python
def dot(x, y):
    return sum(a*b for a, b in zip(x, y))

def dist_sq(x, y):
    return sum((a-b)**2 for a, b in zip(x, y))

def proj_demi_plan(x, a, b_val):
    # Projette x sur { z | a.z <= b_val }
    val = dot(a, x)
    if val <= b_val:
        return x

    norm_a_sq = dot(a, a)
    factor = (val - b_val) / norm_a_sq
    return [xi - factor * ai for xi, ai in zip(x, a)]

def run_test():
    # On projette le point (2,2) sur le convexe C = demi-plan x + y <= 1
    x0 = [2.0, 2.0]
    a = [1.0, 1.0]
    b_val = 1.0

    p = proj_demi_plan(x0, a, b_val)

    # Le projete theorique: la droite perpendiculaire passant par (2,2) est x-y=0.
    # L'intersection avec x+y=1 est (0.5, 0.5).
    assert abs(p[0] - 0.5) < 1e-5
    assert abs(p[1] - 0.5) < 1e-5

    # Validation du theoreme de l'angle obtus
    # (x0 - p) dot (y - p) <= 0 pour tout y dans C
    y = [1.0, -5.0] # Point dans C car 1 - 5 = -4 <= 1
    diff_x0_p = [x0[i] - p[i] for i in range(2)]
    diff_y_p = [y[i] - p[i] for i in range(2)]
    assert dot(diff_x0_p, diff_y_p) <= 1e-9

if __name__ == "__main__":
    run_test()
    print("TP4: Caractérisation variationnelle de la projection validée.")
```
