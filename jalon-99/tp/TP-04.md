# TP 4 : Marge maximale (Hard Margin SVM abstrait)

Recherche du séparateur optimal maximisant la distance entre les deux ensembles via descente de gradient proximale.

```python
def dot(v1, v2): return sum(x*y for x,y in zip(v1,v2))

def svm_gradient_descent(X, y, lr=0.01, epochs=1000):
    n = len(X[0])
    w = [0.0] * n
    b = 0.0

    for _ in range(epochs):
        for i in range(len(X)):
            condition = y[i] * (dot(w, X[i]) + b) >= 1
            if not condition:
                w = [w[j] - lr * (-y[i] * X[i][j]) for j in range(n)]
                b = b - lr * (-y[i])
    return w, b

def test_svm():
    X = [[2, 2], [2, 3], [0, -1], [-1, 0]]
    y = [1, 1, -1, -1]
    w, b = svm_gradient_descent(X, y)

    for xi, yi in zip(X, y):
        assert yi * (dot(w, xi) + b) > 0
    print("TP 4 : Validé")

if __name__ == "__main__":
    test_svm()
```
