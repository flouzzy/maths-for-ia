# TP 5 : Évaluation d'une jauge de Minkowski

Implémentation de la fonctionnelle sous-linéaire (jauge) pour un convexe géométrique simple.

```python
import math

def gauge(x, y):
    # Jauge pour le convexe [-1, 1] x [-1, 1] (norme infinie)
    # C'est l'inf des lambda > 0 tels que (x,y) in lambda C
    # -> max(|x|, |y|)
    return max(abs(x), abs(y))

def test_gauge():
    assert gauge(0.5, -0.5) == 0.5
    assert gauge(2.0, 1.0) == 2.0

    # Propriété de sous-additivité : p(u+v) <= p(u) + p(v)
    u = (1.5, 0.5)
    v = (-0.5, 2.0)
    pu = gauge(*u)
    pv = gauge(*v)
    puv = gauge(u[0]+v[0], u[1]+v[1])
    assert puv <= pu + pv

    # Propriété d'homogénéité : p(lambda u) = lambda p(u) pour lambda > 0
    lam = 3.0
    plamu = gauge(lam*u[0], lam*u[1])
    assert abs(plamu - lam*pu) < 1e-7

    print("TP 5 : Validé")

if __name__ == "__main__":
    test_gauge()
```
