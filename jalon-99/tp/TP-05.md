# TP 5 : SVM à Marge Dure en Dimension Finie

Nous résolvons le problème dual du SVM, qui est la formulation analytique exacte de la séparation stricte de Hahn-Banach pour les polytopes. Nous implémentons une méthode de montée de coordonnées (Coordinate Ascent) sur le problème dual.

```python
def dot(x, y):
    return sum(a*b for a, b in zip(x, y))

def svm_hard_margin_dual(X, y_labels, epochs=1000, lr=0.01):
    N = len(X)
    alpha = [0.0] * N

    # Matrice de Gram
    K = [[0.0 for _ in range(N)] for _ in range(N)]
    for i in range(N):
        for j in range(N):
            K[i][j] = dot(X[i], X[j])

    for _ in range(epochs):
        for i in range(N):
            # Gradient de la fonction objectif duale par rapport a alpha_i
            # F(alpha) = sum(alpha) - 0.5 * sum_i sum_j alpha_i alpha_j y_i y_j K_ij
            grad = 1.0
            for j in range(N):
                grad -= alpha[j] * y_labels[i] * y_labels[j] * K[i][j]

            alpha[i] += lr * grad
            if alpha[i] < 0: # Contrainte de positivite du dual de Lagrange
                alpha[i] = 0.0

    # Reconstruction du vecteur primal w
    dim = len(X[0])
    w = [0.0] * dim
    for i in range(N):
        for d in range(dim):
            w[d] += alpha[i] * y_labels[i] * X[i][d]

    return w

def run_test():
    X = [[2, 2], [2.5, 2.5], [3, 2],
         [-2, -2], [-2.5, -2.5], [-3, -2]]
    y = [1, 1, 1, -1, -1, -1]

    w = svm_hard_margin_dual(X, y)

    # La norme de w definit la marge = 2 / ||w||
    # Verifions la stricte separation avec marge
    for i in range(len(X)):
        val = dot(w, X[i])
        # Le produit scalaire doit etre de meme signe que le label
        # (et decale avec un b=0 grace a la symetrie du dataset autour de 0)
        assert val * y[i] > 0.5 # Marge stricte trouvee

if __name__ == "__main__":
    run_test()
    print("TP5: Marge stricte de séparation SVM optimisée par dualité.")
```
