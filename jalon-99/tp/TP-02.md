# TP 2 : Construction de l'hyperplan séparateur

Implémentation de la méthode géométrique : à partir d'un point extérieur et de son projeté, construire l'équation de l'hyperplan.

```python
import math

def dot(v1, v2):
    return sum(x*y for x, y in zip(v1, v2))

def sub(v1, v2):
    return [x-y for x, y in zip(v1, v2)]

def add(v1, v2):
    return [x+y for x, y in zip(v1, v2)]

def scale(c, v):
    return [c*x for x in v]

def separator_hyperplane(x0, y0):
    # x0 is outside, y0 is its projection on the convex set
    # Returns normal vector w and bias b
    w = sub(x0, y0)
    # the hyperplane passes through the midpoint (x0+y0)/2
    midpoint = scale(0.5, add(x0, y0))
    b = -dot(w, midpoint)
    return w, b

def test_separator():
    x0 = [2.0, 2.0]
    y0 = [1.0, 1.0] # say the convex set is the unit ball (L1 or L2)
    w, b = separator_hyperplane(x0, y0)

    # f(x) = dot(w,x) + b
    f_x0 = dot(w, x0) + b
    f_y0 = dot(w, y0) + b

    assert f_x0 > 0
    assert f_y0 < 0
    print("TP 2 : Validé")

if __name__ == "__main__":
    test_separator()
```
