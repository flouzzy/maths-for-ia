# TP 2 : Séparation Stricte de 2D points (Perceptron Linéaire Basique)

Le théorème d'existence de Hahn-Banach trouve sa traduction algorithmique dans l'algorithme du perceptron : si deux nuages de points (fini) sont strictement séparables, une descente de gradient basique trouve l'hyperplan.

```python
def dot_product(v1, v2):
    return sum(x*y for x, y in zip(v1, v2))

def add_vectors(v1, v2):
    return [x+y for x, y in zip(v1, v2)]

def scale_vector(v, s):
    return [x*s for x in v]

def perceptron(class_1, class_2, max_iter=1000):
    # Les donnees sont de dimension N
    dim = len(class_1[0])
    w = [0.0] * dim
    b = 0.0

    # Dataset total avec labels (+1 pour class 1, -1 pour class 2)
    dataset = [(x, 1) for x in class_1] + [(x, -1) for x in class_2]

    for _ in range(max_iter):
        errors = 0
        for x, y in dataset:
            # Prediction
            activation = dot_product(w, x) + b
            if y * activation <= 0:
                # Mise a jour
                w = add_vectors(w, scale_vector(x, y))
                b += y
                errors += 1
        if errors == 0:
            break

    return w, b

def run_test():
    # Classe 1: Nuage autour de (2,2)
    c1 = [[2,2], [3,2], [2,3], [3,3]]
    # Classe 2: Nuage autour de (-2,-2)
    c2 = [[-2,-2], [-3,-2], [-2,-3], [-3,-3]]

    w, b = perceptron(c1, c2)

    # Validation mathematique stricte : l'hyperplan doit separer
    for x in c1:
        assert dot_product(w, x) + b > 0
    for x in c2:
        assert dot_product(w, x) + b < 0

if __name__ == "__main__":
    run_test()
    print("TP2: Hyperplan séparateur trouvé (Perceptron).")
```
