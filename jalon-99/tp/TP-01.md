# TP 1 : Implémentation de la fonction de Jauge (Minkowski)

Ce TP vise à programmer "from scratch" la fonctionnelle de Minkowski pour un convexe simple en 2D, qui est la base de la démonstration analytique et géométrique de Hahn-Banach.

```python
import math

class ConvexShape:
    # On définit un disque ouvert translaté, qui est convexe.
    # C = { (x,y) | (x-cx)**2 + (y-cy)**2 < r**2 }
    def __init__(self, cx, cy, r):
        self.cx = cx
        self.cy = cy
        self.r = r

    def contains(self, x, y):
        return (x - self.cx)**2 + (y - self.cy)**2 < self.r**2

def minkowski_gauge(shape, point_x, point_y, origin_x=0.0, origin_y=0.0):
    # j_C(x) = inf { t > 0 | x/t in C }
    # Ici on centre autour d'un 'origin' qui doit être dans la forme.
    if not shape.contains(origin_x, origin_y):
        raise ValueError("L'origine choisie n'est pas dans le convexe ouvert.")

    # Pour un cercle, l'intersection du rayon partant de l'origine
    # avec le bord du cercle peut être calculée analytiquement, mais
    # on utilise une recherche dichotomique pour être générique.

    dx = point_x - origin_x
    dy = point_y - origin_y

    if dx == 0 and dy == 0:
        return 0.0

    # Si le point est dans la forme, la jauge est < 1
    # S'il est en dehors, la jauge est >= 1
    # On cherche t tel que (origin + point/t) est sur le bord.
    # Posons s = 1/t. On cherche le sup des s > 0 tels que (origin + s*point) in C.

    low = 0.0
    high = 1000.0 # Borne heuristique

    # Recherche d'une borne supérieure valide
    while shape.contains(origin_x + high * dx, origin_y + high * dy):
        high *= 2

    for _ in range(50):
        mid = (low + high) / 2
        if shape.contains(origin_x + mid * dx, origin_y + mid * dy):
            low = mid
        else:
            high = mid

    # s_max est 'low'. La jauge est t = 1/s_max
    if low < 1e-9:
        return float('inf') # Devrait pas arriver car origin est int(C)

    return 1.0 / low

def run_test():
    C = ConvexShape(1.0, 1.0, 2.0)
    # L'origine (1,1) est le centre, rayon 2.
    assert C.contains(1.0, 1.0)

    # Point (3,1), à distance 2 du centre. Il est sur le bord.
    # Jauge doit être 1.
    j = minkowski_gauge(C, 3.0, 1.0, origin_x=1.0, origin_y=1.0)
    assert abs(j - 1.0) < 1e-4

    # Point (2,1), à distance 1.
    # x/t dans bord -> 1/t = 2 -> t = 0.5. Jauge doit être 0.5
    j2 = minkowski_gauge(C, 2.0, 1.0, origin_x=1.0, origin_y=1.0)
    assert abs(j2 - 0.5) < 1e-4

if __name__ == "__main__":
    run_test()
    print("TP1: Jauge de Minkowski validée mathématiquement.")
```
