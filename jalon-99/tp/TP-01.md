# TP 1 : Projection orthogonale sur un polyèdre convexe

L'objectif de ce TP est de calculer explicitement le projeté d'un point sur un convexe, étape clé pour construire l'hyperplan séparateur.

```python
import math

def dot(v1, v2):
    return sum(x*y for x, y in zip(v1, v2))

def sub(v1, v2):
    return [x-y for x, y in zip(v1, v2)]

def proj_on_segment(p, a, b):
    # Projection of p on segment [a, b]
    ab = sub(b, a)
    ap = sub(p, a)
    l2 = dot(ab, ab)
    if l2 == 0:
        return a
    t = max(0, min(1, dot(ap, ab) / l2))
    return [a[i] + t * ab[i] for i in range(len(a))]

def test_proj():
    a = [0.0, 0.0]
    b = [2.0, 0.0]
    p = [1.0, 1.0]
    proj = proj_on_segment(p, a, b)
    assert abs(proj[0] - 1.0) < 1e-7
    assert abs(proj[1] - 0.0) < 1e-7
    print("TP 1 : Validé")

if __name__ == "__main__":
    test_proj()
```
